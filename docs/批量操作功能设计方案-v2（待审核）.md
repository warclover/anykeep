## 一、竞品调研总结（简要）

### 1.1 高频批量操作排序

| 排名 | 操作                 | 竞品覆盖 |   AnyKeep 现状   |
| :--: | -------------------- | :------: | :--------------: |
|  🥇  | 批量删除             |   6/6    |    ✅ 已实现     |
|  🥈  | 批量移动             |   6/6    |    ✅ 已实现     |
|  🥉  | **批量添加标签**     |   5/6    |     ❌ 缺失      |
|  4   | **批量删除特定标签** |   0/6    | ❌ 差异化机会 ⭐ |

|  6   | 批量导出             |   4/6    |     ❌ 缺失      |

### 1.2 交互设计最佳实践

- **三种选择方式**：Ctrl 单选、Shift 范围选、全选
- **工具栏原则**：选中 ≥1 项时显示，明确显示数量，Esc 退出
- **操作确认**：危险操作需二次确认

## 三、功能解耦设计

### 3.1 设计原则

1. **模块独立**：新功能作为独立模块，不修改现有核心逻辑
2. **接口复用**：复用现有 Hook 接口，通过扩展而非修改
3. **渐进增强**：新功能失败不影响现有功能
4. **特性开关**：支持通过配置开关启用/禁用
5. **统一 Shift 选择**：扩展 `useSelectionMode` 而非新建独立 Hook

## 五、详细技术设计

### 5.1 数据库层设计

**无需新建表**，直接操作现有 `bookmarks.user_tags` 字段。

**数据库 RPC 函数** (合并到单个迁移文件):

```sql
-- 迁移文件: supabase/migrations/20251219000001_add_batch_tags_functions.sql

-- ============================================
-- 批量添加标签函数
-- ============================================
CREATE OR REPLACE FUNCTION batch_add_tags(
  p_user_id UUID,
  p_bookmark_ids INT[],
  p_tags TEXT[]
)
RETURNS JSON AS $$
DECLARE
  v_affected_ids INT[];
  v_affected_count INT;
BEGIN
  -- 更新书签，添加新标签（去重）
  WITH updated AS (
    UPDATE bookmarks
    SET
      user_tags = (
        SELECT ARRAY(
          SELECT DISTINCT unnest(
            array_cat(COALESCE(user_tags, ARRAY[]::TEXT[]), p_tags)
          )
        )
      ),
      updated_at = NOW()
    WHERE id = ANY(p_bookmark_ids)
      AND user_id = p_user_id
      AND deleted_at IS NULL
    RETURNING id
  )
  SELECT ARRAY_AGG(id), COUNT(*)
  INTO v_affected_ids, v_affected_count
  FROM updated;

  RETURN json_build_object(
    'affected_ids', COALESCE(v_affected_ids, ARRAY[]::INT[]),
    'affected_count', COALESCE(v_affected_count, 0)
  );
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- ============================================
-- 批量移除标签函数
-- ============================================
CREATE OR REPLACE FUNCTION batch_remove_tags(
  p_user_id UUID,
  p_bookmark_ids INT[],
  p_tags TEXT[]
)
RETURNS JSON AS $$
DECLARE
  v_affected_ids INT[];
  v_affected_count INT;
BEGIN
  -- 更新书签，移除指定标签
  WITH updated AS (
    UPDATE bookmarks
    SET
      user_tags = (
        SELECT ARRAY(
          SELECT unnest(user_tags)
          EXCEPT
          SELECT unnest(p_tags)
        )
      ),
      updated_at = NOW()
    WHERE id = ANY(p_bookmark_ids)
      AND user_id = p_user_id
      AND deleted_at IS NULL
      AND user_tags && p_tags  -- 只更新包含这些标签的书签
    RETURNING id
  )
  SELECT ARRAY_AGG(id), COUNT(*)
  INTO v_affected_ids, v_affected_count
  FROM updated;

  RETURN json_build_object(
    'affected_ids', COALESCE(v_affected_ids, ARRAY[]::INT[]),
    'affected_count', COALESCE(v_affected_count, 0)
  );
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- ============================================
-- 授权
-- ============================================
GRANT EXECUTE ON FUNCTION batch_add_tags TO authenticated;
GRANT EXECUTE ON FUNCTION batch_remove_tags TO authenticated;

-- ============================================
-- 注释
-- ============================================
COMMENT ON FUNCTION batch_add_tags IS '批量为书签添加标签，自动去重';
COMMENT ON FUNCTION batch_remove_tags IS '批量从书签移除指定标签';
```

### 5.2 API 层设计

#### POST /api/bookmarks/batch-tags

**文件位置**: `src/app/api/bookmarks/batch-tags/route.ts`

```typescript
import { NextRequest } from "next/server";
import { createClient } from "@/lib/supabase/server";
import { batchTagsSchema } from "@/lib/batch/validation";
import {
  successResponse,
  badRequestError,
  unauthorizedError,
  databaseError,
} from "@/lib/utils/api-response";
import { logger } from "@/lib/logger";

export async function POST(request: NextRequest) {
  const startTime = Date.now();

  try {
    // 1. 身份验证
    const supabase = await createClient();
    const {
      data: { user },
      error: authError,
    } = await supabase.auth.getUser();

    if (authError || !user) {
      return unauthorizedError("请先登录");
    }

    // 2. 请求验证
    const body = await request.json();
    const parseResult = batchTagsSchema.safeParse(body);

    if (!parseResult.success) {
      return badRequestError(parseResult.error.errors[0].message);
    }

    const { action, bookmarkIds, tags } = parseResult.data;

    // 3. 去重标签
    const uniqueTags = [...new Set(tags.map((t) => t.trim()))];

    // 4. 执行批量操作
    let result;

    if (action === "add") {
      const { data, error } = await supabase.rpc("batch_add_tags", {
        p_user_id: user.id,
        p_bookmark_ids: bookmarkIds,
        p_tags: uniqueTags,
      });
      if (error) throw error;
      result = data;
    } else {
      const { data, error } = await supabase.rpc("batch_remove_tags", {
        p_user_id: user.id,
        p_bookmark_ids: bookmarkIds,
        p_tags: uniqueTags,
      });
      if (error) throw error;
      result = data;
    }

    // 5. 记录日志
    logger.info({
      operation: `batch_tags_${action}`,
      userId: user.id,
      bookmarkCount: bookmarkIds.length,
      tagCount: uniqueTags.length,
      affectedCount: result.affected_count,
      duration: Date.now() - startTime,
    });

    // 6. 返回响应
    return successResponse({
      action,
      affectedCount: result.affected_count,
      affectedIds: result.affected_ids,
      tags: uniqueTags,
      message:
        action === "add"
          ? `已为 ${result.affected_count} 个书签添加 ${uniqueTags.length} 个标签`
          : `已从 ${result.affected_count} 个书签移除 ${uniqueTags.length} 个标签`,
    });
  } catch (error) {
    logger.error({
      operation: "batch_tags",
      error: error instanceof Error ? error.message : "Unknown error",
    });

    return databaseError("批量标签操作失败，请稍后重试");
  }
}
```

const handleItemClick = useCallback(
  (id: number, event: React.MouseEvent) => {
    if (!isSelectionMode) return;

    if (event.shiftKey && lastSelectedId !== null && allIds) {
      // Shift + 点击：范围选择
      const lastIndex = allIds.indexOf(lastSelectedId);
      const currentIndex = allIds.indexOf(id);

      if (lastIndex !== -1 && currentIndex !== -1) {
        const start = Math.min(lastIndex, currentIndex);
        const end = Math.max(lastIndex, currentIndex);
        const rangeIds = allIds.slice(start, end + 1);

        setSelectedIds((prev) => {
          const newSet = new Set(prev);
          rangeIds.forEach((rid) => newSet.add(rid));
          return newSet;
        });
      }
    } else {
      // 普通点击：切换单项并记录锚点
      toggleSelect(id);
      setLastSelectedId(id);
    }
  },
  [isSelectionMode, lastSelectedId, allIds, toggleSelect],

### 5.4 缓存同步与回滚

**核心原则**: 使用 `useBookmarksSWR` 的 `createSnapshot` 和 `rollback` 实现乐观更新和失败回滚。

```typescript
// 在 useBatchTags 或页面组件中

const handleBatchTagSuccess = useCallback(
  async (bookmarkIds: number[], tags: string[], action: "add" | "remove") => {
    // 1. 创建快照（用于回滚）
    const snapshot = createSnapshot();

    // 2. 乐观更新
    bookmarkIds.forEach((id) => {
      updateCacheItem(id, (bookmark) => {
        if (action === "add") {
          const newTags = [
            ...new Set([...(bookmark.user_tags || []), ...tags]),
          ];
          return { ...bookmark, user_tags: newTags };
        } else {
          const newTags = (bookmark.user_tags || []).filter(
            (t) => !tags.includes(t),
          );
          return { ...bookmark, user_tags: newTags };
        }
      });
    });

    // 3. 调用 API
    try {
      const result = await batchTagsAPI(action, bookmarkIds, tags);
      toast.success(result.message);
    } catch (error) {
      // 4. 失败时回滚
      rollback(snapshot);
      toast.error("操作失败，已恢复原状态");
    }
  },
  [createSnapshot, updateCacheItem, rollback],
);
```
