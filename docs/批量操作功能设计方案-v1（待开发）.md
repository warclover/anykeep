## 三、功能解耦设计

### 3.1 设计原则

1. **模块独立**：新功能作为独立模块，不修改现有核心逻辑

```sql
CREATE OR REPLACE FUNCTION array_remove_multiple(
  arr TEXT[],
  items_to_remove TEXT[]
)
RETURNS TEXT[] AS $$
  SELECT ARRAY(
    SELECT unnest(arr)
    EXCEPT
    SELECT unnest(items_to_remove)
  )
$$ LANGUAGE SQL IMMUTABLE;

COMMENT ON FUNCTION array_remove_multiple IS
  '从数组中批量移除多个元素，用于批量标签删除功能';
```

**完整实现**:

```typescript
// src/app/api/bookmarks/batch-tags/route.ts

import { NextRequest } from "next/server";
import { createClient } from "@/lib/supabase/server";
import { batchTagsSchema } from "@/lib/batch/validation";
import {
  successResponse,
  badRequestError,
  unauthorizedError,
  databaseError,
} from "@/lib/utils/api-response";
import { logger } from "@/lib/logger";

export async function POST(request: NextRequest) {
  const startTime = Date.now();

  try {
    // 1. 身份验证
    const supabase = await createClient();
    const {
      data: { user },
      error: authError,
    } = await supabase.auth.getUser();

    if (authError || !user) {
      return unauthorizedError("请先登录");
    }

    // 2. 请求验证
    const body = await request.json();
    const parseResult = batchTagsSchema.safeParse(body);

    if (!parseResult.success) {
      return badRequestError(parseResult.error.errors[0].message);
    }

    const { action, bookmarkIds, tags } = parseResult.data;

    // 3. 去重标签（忽略大小写）
    const uniqueTags = [...new Set(tags.map((t) => t.trim()))];

    // 4. 执行批量操作
    let result;

    if (action === "add") {
      // 批量添加标签
      const { data, error } = await supabase.rpc("batch_add_tags", {
        p_user_id: user.id,
        p_bookmark_ids: bookmarkIds,
        p_tags: uniqueTags,
      });

      if (error) throw error;
      result = data;
    } else {
      // 批量移除标签
      const { data, error } = await supabase.rpc("batch_remove_tags", {
        p_user_id: user.id,
        p_bookmark_ids: bookmarkIds,
        p_tags: uniqueTags,
      });

      if (error) throw error;
      result = data;
    }

    // 5. 记录日志
    logger.info({
      operation: `batch_tags_${action}`,
      userId: user.id,
      bookmarkCount: bookmarkIds.length,
      tagCount: uniqueTags.length,
      affectedCount: result.affected_count,
      duration: Date.now() - startTime,
    });

    // 6. 返回响应
    return successResponse({
      action,
      affectedCount: result.affected_count,
      affectedIds: result.affected_ids,
      tags: uniqueTags,
      message:
        action === "add"
          ? `已为 ${result.affected_count} 个书签添加 ${uniqueTags.length} 个标签`
          : `已从 ${result.affected_count} 个书签移除 ${uniqueTags.length} 个标签`,
    });
  } catch (error) {
    logger.error({
      operation: "batch_tags",
      error: error instanceof Error ? error.message : "Unknown error",
    });

    return databaseError("批量标签操作失败，请稍后重试");
  }
}
```

```sql
-- 迁移文件: 20251219000002_add_batch_tags_functions.sql

-- 批量添加标签函数
CREATE OR REPLACE FUNCTION batch_add_tags(
  p_user_id UUID,
  p_bookmark_ids INT[],
  p_tags TEXT[]
)
RETURNS JSON AS $$
DECLARE
  v_affected_ids INT[];
  v_affected_count INT;
BEGIN
  -- 更新书签，添加新标签（去重）
  WITH updated AS (
    UPDATE bookmarks
    SET
      user_tags = (
        SELECT ARRAY(
          SELECT DISTINCT unnest(
            array_cat(COALESCE(user_tags, ARRAY[]::TEXT[]), p_tags)
          )
        )
      ),
      updated_at = NOW()
    WHERE id = ANY(p_bookmark_ids)
      AND user_id = p_user_id
      AND deleted_at IS NULL
    RETURNING id
  )
  SELECT ARRAY_AGG(id), COUNT(*)
  INTO v_affected_ids, v_affected_count
  FROM updated;

  RETURN json_build_object(
    'affected_ids', COALESCE(v_affected_ids, ARRAY[]::INT[]),
    'affected_count', COALESCE(v_affected_count, 0)
  );
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 批量移除标签函数
CREATE OR REPLACE FUNCTION batch_remove_tags(
  p_user_id UUID,
  p_bookmark_ids INT[],
  p_tags TEXT[]
)
RETURNS JSON AS $$
DECLARE
  v_affected_ids INT[];
  v_affected_count INT;
BEGIN
  -- 更新书签，移除指定标签
  WITH updated AS (
    UPDATE bookmarks
    SET
      user_tags = (
        SELECT ARRAY(
          SELECT unnest(user_tags)
          EXCEPT
          SELECT unnest(p_tags)
        )
      ),
      updated_at = NOW()
    WHERE id = ANY(p_bookmark_ids)
      AND user_id = p_user_id
      AND deleted_at IS NULL
      AND user_tags && p_tags  -- 只更新包含这些标签的书签
    RETURNING id
  )
  SELECT ARRAY_AGG(id), COUNT(*)
  INTO v_affected_ids, v_affected_count
  FROM updated;

  RETURN json_build_object(
    'affected_ids', COALESCE(v_affected_ids, ARRAY[]::INT[]),
    'affected_count', COALESCE(v_affected_count, 0)
  );
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 授权
GRANT EXECUTE ON FUNCTION batch_add_tags TO authenticated;
GRANT EXECUTE ON FUNCTION batch_remove_tags TO authenticated;
```

```typescript
import { useState, useCallback } from "react";
import { toast } from "sonner";
import { useAbortableFetch } from "@/hooks/use-abortable-fetch";

interface UseBatchTagsOptions {
  onSuccess?: (result: BatchTagsResult) => void;
  onError?: (error: Error) => void;
}

interface BatchTagsResult {
  action: "add" | "remove";
  affectedCount: number;
  affectedIds: number[];
  tags: string[];
}

export function useBatchTags(options: UseBatchTagsOptions = {}) {
  const [isProcessing, setIsProcessing] = useState(false);
  const { fetchWithAbort } = useAbortableFetch();

  const batchAddTags = useCallback(
    async (
      bookmarkIds: number[],
      tags: string[],
    ): Promise<BatchTagsResult | null> => {
      if (bookmarkIds.length === 0 || tags.length === 0) {
        toast.error("请选择书签和标签");
        return null;
      }

      setIsProcessing(true);

      try {
        const response = await fetchWithAbort("/api/bookmarks/batch-tags", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            action: "add",
            bookmarkIds,
            tags,
          }),
        });

        if (!response.ok) {
          const error = await response.json();
          throw new Error(error.message || "操作失败");
        }

        const { data } = await response.json();
        toast.success(data.message);
        options.onSuccess?.(data);
        return data;
      } catch (error) {
        if (error instanceof Error && error.name === "AbortError") {
          return null;
        }
        const message = error instanceof Error ? error.message : "操作失败";
        toast.error(message);
        options.onError?.(error as Error);
        return null;
      } finally {
        setIsProcessing(false);
      }
    },
    [fetchWithAbort, options],
  );

  const batchRemoveTags = useCallback(
    async (
      bookmarkIds: number[],
      tags: string[],
    ): Promise<BatchTagsResult | null> => {
      if (bookmarkIds.length === 0 || tags.length === 0) {
        toast.error("请选择书签和标签");
        return null;
      }

      setIsProcessing(true);

      try {
        const response = await fetchWithAbort("/api/bookmarks/batch-tags", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            action: "remove",
            bookmarkIds,
            tags,
          }),
        });

        if (!response.ok) {
          const error = await response.json();
          throw new Error(error.message || "操作失败");
        }

        const { data } = await response.json();
        toast.success(data.message);
        options.onSuccess?.(data);
        return data;
      } catch (error) {
        if (error instanceof Error && error.name === "AbortError") {
          return null;
        }
        const message = error instanceof Error ? error.message : "操作失败";
        toast.error(message);
        options.onError?.(error as Error);
        return null;
      } finally {
        setIsProcessing(false);
      }
    },
    [fetchWithAbort, options],
  );

  return {
    isProcessing,
    batchAddTags,
    batchRemoveTags,
  };
}
```

#### useShiftSelect Hook

**文件位置**: `src/hooks/batch/use-shift-select.ts`

```typescript
import { useCallback, useRef } from "react";

interface UseShiftSelectOptions {
  selectedIds: Set<number>;
  onSelect: (ids: number[]) => void;
  getAllIds: () => number[]; // 获取当前列表所有 ID
}

export function useShiftSelect({
  selectedIds,
  onSelect,
  getAllIds,
}: UseShiftSelectOptions) {
  const lastSelectedIdRef = useRef<number | null>(null);

  const handleClick = useCallback(
    (id: number, event: React.MouseEvent) => {
      const allIds = getAllIds();

      if (event.shiftKey && lastSelectedIdRef.current !== null) {
        // Shift + 点击：范围选择
        const lastIndex = allIds.indexOf(lastSelectedIdRef.current);
        const currentIndex = allIds.indexOf(id);

        if (lastIndex !== -1 && currentIndex !== -1) {
          const start = Math.min(lastIndex, currentIndex);
          const end = Math.max(lastIndex, currentIndex);
          const rangeIds = allIds.slice(start, end + 1);

          // 合并现有选择和新范围
          const newSelection = new Set([...selectedIds, ...rangeIds]);
          onSelect(Array.from(newSelection));
        }
      } else {
        // 普通点击：切换单项
        lastSelectedIdRef.current = id;
      }
    },
    [selectedIds, onSelect, getAllIds],
  );

  const resetLastSelected = useCallback(() => {
    lastSelectedIdRef.current = null;
  }, []);

  return {
    handleShiftClick: handleClick,
    resetLastSelected,
  };
}
```

```typescript
        : [...prev, tag]
    );
  };

  const addCustomTag = () => {
    const tag = inputValue.trim();
    if (tag && !selectedTags.includes(tag)) {
      setSelectedTags(prev => [...prev, tag]);
      setInputValue('');
    }
  };

  const handleSubmit = async () => {
    if (selectedTags.length === 0) return;
    await batchAddTags(selectedIds, selectedTags);
  };

  return (
    <div className="space-y-4">
      {/* 自定义输入 */}
      <div className="flex gap-2">
        <Input
          placeholder="输入自定义标签..."
          value={inputValue}
          onChange={(e) => setInputValue(e.target.value)}
          onKeyDown={(e) => {
            if (e.key === 'Enter') {
              e.preventDefault();
              addCustomTag();
            }
          }}
          className="flex-1"
        />
        <Button
          variant="outline"
          size="icon"
          onClick={addCustomTag}
          disabled={!inputValue.trim()}
        >
          <Plus className="h-4 w-4" />
        </Button>
      </div>

      {/* 用途标签快捷选择 */}
      <div>
        <p className="text-sm text-muted-foreground mb-2">用途标签</p>
        <div className="flex flex-wrap gap-2">
          {allUsageTags.map(tag => (
            <Badge
              key={tag}
              variant={selectedTags.includes(tag) ? 'default' : 'outline'}
              className="cursor-pointer"
              onClick={() => toggleTag(tag)}
            >
              {tag}
            </Badge>
          ))}
        </div>
      </div>

      {/* 已选择标签 */}
      {selectedTags.length > 0 && (
        <div>
          <p className="text-sm text-muted-foreground mb-2">
            将添加 {selectedTags.length} 个标签:
          </p>
          <div className="flex flex-wrap gap-2">
            {selectedTags.map(tag => (
              <Badge key={tag} variant="secondary" className="gap-1">
                {tag}
                <X
                  className="h-3 w-3 cursor-pointer"
                  onClick={() => toggleTag(tag)}
                />
              </Badge>
            ))}
          </div>
        </div>
      )}

      {/* 操作按钮 */}
      <div className="flex justify-end gap-2 pt-4">
        <Button variant="outline" onClick={onCancel} disabled={isProcessing}>
          取消
        </Button>
        <Button
          onClick={handleSubmit}
          disabled={selectedTags.length === 0 || isProcessing}
        >
          {isProcessing ? (
            <>
              <Loader2 className="h-4 w-4 mr-2 animate-spin" />
              处理中...
            </>
          ) : (
            `添加到 ${selectedIds.length} 个书签`
          )}
        </Button>
      </div>
    </div>
  );
}
```

### 5.1 useSelectionMode 扩展

**修改位置**: `src/hooks/core/use-selection-mode.ts`

```typescript
// 新增状态
interface SelectionModeState {
  isSelectionMode: boolean;
  selectedIds: Set<number>;
  isBatchDeleting: boolean;
  lastSelectedId: number | null; // 🆕 新增
}

// 新增方法
interface SelectionModeActions {
  // ... 现有方法
  handleItemClick: (
    id: number,
    event: React.MouseEvent,
    allIds: number[],
  ) => void; // 🆕
}

// 实现
const handleItemClick = useCallback(
  (id: number, event: React.MouseEvent, allIds: number[]) => {
    if (!isSelectionMode) return;

    if (event.shiftKey && lastSelectedId !== null) {
      // Shift + 点击：范围选择
      const lastIndex = allIds.indexOf(lastSelectedId);
      const currentIndex = allIds.indexOf(id);

      if (lastIndex !== -1 && currentIndex !== -1) {
        const start = Math.min(lastIndex, currentIndex);
        const end = Math.max(lastIndex, currentIndex);
        const rangeIds = allIds.slice(start, end + 1);

        setSelectedIds((prev) => {
          const newSet = new Set(prev);
          rangeIds.forEach((rid) => newSet.add(rid));
          return newSet;
        });
      }
    } else {
      // 普通点击：切换单项并记录
      toggleSelect(id);
      setLastSelectedId(id);
    }
  },
  [isSelectionMode, lastSelectedId, toggleSelect],
);
```
