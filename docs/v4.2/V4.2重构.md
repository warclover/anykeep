## 二、问题诊断

### 2.1 类型系统问题

**根本原因**: 双源类型定义导致循环依赖

```
┌─────────────────┐     冲突      ┌─────────────────┐
│  v42-types.ts   │ ◄──────────► │   supabase.ts   │
│  (手动维护)      │              │  (自动生成)      │
└─────────────────┘              └─────────────────┘
        │                                │
        │  手动同步（易出错）              │  CLI 重新生成（覆盖修改）
        ▼                                ▼
   业务代码使用 any 绕过 ────────► 类型安全性丧失
```

**具体表现**:

1. `operation_snapshots.source` 字段类型不一致
2. `search_folders_unified` 的 `p_embedding` 参数类型：`string` vs `number[]`
3. `job_queue.status` 类型过于宽松
4. `librarian_analysis_history` 表缺少类型定义

### 2.2 架构耦合问题

**当前状态**: 分类逻辑与文件夹管理深度耦合

```
classification.service.ts
    ├── 向量匹配 (分类核心)
    ├── AI 推荐 (分类核心)
    ├── 文件夹画像计算 (自组织功能) ← 耦合
    ├── 热度更新 (自组织功能) ← 耦合
    └── 合并/拆分建议 (自组织功能) ← 耦合
```

import type { Database } from "@/types/supabase";

  ClassificationResult,
  BookmarkInput,
  MatcherContext,
} from "../types/domain-types";
import type { IMatcher, IScoreAggregator } from "../strategies/interfaces";
import { EventBus } from "@/lib/shared/event-bus";
import { ClassificationEvents } from "../events/classification-events";

export class ClassificationService {
  constructor(
    private supabase: SupabaseClient<Database>,
    private matchers: IMatcher[],
    private aggregator: IScoreAggregator,
    private eventBus: EventBus,
  ) {}

  async classify(
    userId: string,
    bookmark: BookmarkInput,
  ): Promise<ClassificationResult> {
    // 1. 准备上下文
    const context = await this.prepareContext(userId, bookmark);

    // 2. 运行匹配器
    const matchResults = await this.runMatchers(context);

    const result = this.aggregator.aggregate(matchResults);

    // 4. 发布事件（通知自组织模块）
    this.eventBus.emit(ClassificationEvents.BOOKMARK_CLASSIFIED, {
      userId,
      bookmarkId: bookmark.id,
      folderId: result.folderId,
      confidence: result.confidence,
    });

    return result;
  }

### Phase 2：事件总线实现（Day 4）

#### 2.1 事件总线核心

```typescript
// src/lib/shared/event-bus/event-bus.ts

import type { EventTypes, EventPayload } from "./event-types";

type EventHandler<T> = (payload: T) => void | Promise<void>;

export class EventBus {
  private handlers = new Map<string, Set<EventHandler<unknown>>>();

  on<K extends keyof EventTypes>(
    event: K,
    handler: EventHandler<EventTypes[K]>,
  ): () => void {
    if (!this.handlers.has(event)) {
      this.handlers.set(event, new Set());
    }
    this.handlers.get(event)!.add(handler as EventHandler<unknown>);

    // 返回取消订阅函数
    return () => this.off(event, handler);
  }

  off<K extends keyof EventTypes>(
    event: K,
    handler: EventHandler<EventTypes[K]>,
  ): void {
    this.handlers.get(event)?.delete(handler as EventHandler<unknown>);
  }

  emit<K extends keyof EventTypes>(event: K, payload: EventTypes[K]): void {
    const eventHandlers = this.handlers.get(event);
    if (eventHandlers) {
      eventHandlers.forEach((handler) => {
        try {
          handler(payload);
        } catch (error) {
          console.error(`Event handler error for ${event}:`, error);
        }
      });
    }
  }
}

// 单例导出
export const eventBus = new EventBus();
```

export interface DBSCANOptions {
  epsilon: number; // 邻域半径
  minPoints: number; // 最小点数
  distanceMetric: "cosine" | "euclidean";
}

export interface ClusterResult {
  clusterId: number;
  points: number[];
  centroid: number[];
  size: number;
}

/**
 * DBSCAN 聚类算法
 * 从 ai-librarian.service.ts 提取的核心算法
 */
export function dbscanCluster(
  embeddings: Map<number, number[]>,
  options: DBSCANOptions,
): ClusterResult[] {
  const { epsilon, minPoints, distanceMetric } = options;
  const visited = new Set<number>();
  const clusters: ClusterResult[] = [];
  let clusterId = 0;

  const distance =
    distanceMetric === "cosine" ? cosineDistance : euclideanDistance;

  for (const [pointId, embedding] of embeddings) {
    if (visited.has(pointId)) continue;
    visited.add(pointId);

    const neighbors = getNeighbors(pointId, embeddings, epsilon, distance);

    if (neighbors.length < minPoints) {
      // 噪声点
      continue;
    }

    // 扩展集群
    const cluster = expandCluster(
      pointId,
      neighbors,
      embeddings,
      visited,
      epsilon,
      minPoints,
      distance,
    );

    clusters.push({
      clusterId: clusterId++,
      points: cluster,
      centroid: calculateCentroid(cluster, embeddings),
      size: cluster.length,
    });
  }

  return clusters;
}
