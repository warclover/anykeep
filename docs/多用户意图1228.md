### 1.2 目标用户群体

**核心定位**：需要**大量阅读和学习**的知识工作者

| 类别       | 用户群体                     | 核心特征                         | 阅读量级     |
| ---------- | ---------------------------- | -------------------------------- | ------------ |
| **学术型** | 本科生、研究生、博士生       | 论文阅读、知识体系构建、复习备考 | 50-200篇/月  |
| **创作型** | 科技博主、知识博主、技术博主 | 素材收集、选题分析、内容输出     | 100-500条/月 |
| **技术型** | 独立开发者、AI发烧友         | 技术调研、解决方案、项目积累     | 50-150篇/月  |

// 主画像类型
export type PersonaType =

// 技术领域（技术型专用）
export type TechDomain =
  | "ai_ml" // AI/机器学习
  | "frontend" // 前端
  | "backend" // 后端

  | "tech" // 科技

  | "other";

export interface UserPersonaV3 {

    domain: TechDomain;
    languages?: string[]; // 编程语言
    frameworks?: string[]; // 技术框架

    responseStyle: "concise" | "detailed" | "auto";

  };

  // 元数据
  createdAt: string;
  updatedAt: string;
  source: "registration" | "settings" | "inferred";
