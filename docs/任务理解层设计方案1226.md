## 一、设计目标

### 1.1 核心目标

将 AnyKeep AI 助手从"工具调用器"升级为"任务理解者"，实现：

1. **深度理解用户意图** - 不只是识别关键词，而是理解用户的真正目标
2. **主动确认理解** - 在执行前确认理解正确，避免误操作
3. **展示思考过程** - 让用户知道 AI 在做什么，提升信任感
4. **渐进式对话** - 不确定时主动澄清，而不是猜测

### 1.2 成功标准

| 指标           | 当前值 | 目标值 | 衡量方式              |
| -------------- | ------ | ------ | --------------------- |
| 意图理解准确率 | ~60%   | >85%   | 用户反馈 + 任务完成率 |
| 误操作率       | ~20%   | <5%    | 用户取消/撤销次数     |
| 用户满意度     | 3.2/5  | 4.5/5  | NPS 调查              |
| 多轮对话成功率 | ~40%   | >80%   | 任务完成率            |

### 4.1 TaskAnalyzer（LLM 深度理解）

```typescript
// src/lib/ai-agent/task-understanding/analyzer/task-analyzer.ts

import { generateObject } from "ai";
import { z } from "zod";
import type { TaskUnderstanding, TaskContext } from "../types";

/**
 * 任务分析器
 *
 * 使用 LLM 深度理解用户的真正目标
 */
export class TaskAnalyzer {
  private model: LanguageModel;

  constructor(config: TaskAnalyzerConfig) {
    this.model = config.model;
  }

  /**
   * 分析用户任务
   */
  async analyze(
    input: string,
    context: TaskContext,
  ): Promise<TaskUnderstanding> {
    const startTime = performance.now();

    const prompt = this.buildAnalysisPrompt(input, context);

    const { object: analysis } = await generateObject({
      model: this.model,
      schema: TaskAnalysisSchema,
      prompt,
      temperature: 0.3, // 低温度，更稳定
    });

    const understanding = this.buildTaskUnderstanding(
      input,
      analysis,
      performance.now() - startTime,
    );

    return understanding;
  }

  /**
   * 构建分析提示词
   */
  private buildAnalysisPrompt(input: string, context: TaskContext): string {
    return `你是一个任务理解专家。用户说了这句话，请深度分析他的真正需求。

## 用户输入
"${input}"

## 用户上下文
- 收藏总数: ${context.stats.total}
- 待整理数: ${context.stats.inbox}
- 最近活跃平台: ${context.recentPlatforms.join(", ") || "无"}
- 常用标签: ${context.frequentTags.slice(0, 10).join(", ") || "无"}
- 文件夹列表: ${context.folders.map((f) => f.name).join(", ") || "无"}
- 上次对话主题: ${context.lastTopic || "无"}

## 分析要求

### 1. 用户的真正目标是什么？
- 不要只看表面意思
- 思考：如果任务完成，用户期望看到什么结果？
- 用一句话描述用户的真正目标

### 2. 用户可能的隐含假设？
- 用户没说但可能期望的
- 例如：不删除数据、保留原有结构、执行前先确认等
- 列出 2-5 个隐含假设

### 3. 成功标准是什么？
- 如何判断任务完成得好？
- 用一句话描述

### 4. 有什么需要澄清的？
- 只列出影响执行的关键问题（0-3 个）
- 每个问题提供 2-4 个选项
- 标记推荐选项
- 标记影响程度 (high/medium/low)
- 如果意图很明确，可以不需要澄清

### 5. 执行计划？
- 列出 2-5 个关键步骤
- 说明每步的目的
- 标记需要用户确认的步骤

### 6. 置信度？
- 0-1 之间
- 考虑因素：意图是否明确、信息是否充足、是否有歧义
- > 0.8: 非常确定
- 0.6-0.8: 比较确定，可能需要确认
- < 0.6: 需要澄清

### 7. 风险级别？
- low: 只读操作，无副作用
- medium: 可能修改数据，但可撤销
- high: 不可撤销的修改（删除等）`;
  }

  /**
   * 构建任务理解结果
   */
  private buildTaskUnderstanding(
    rawInput: string,
    analysis: TaskAnalysis,
    processingTimeMs: number,
  ): TaskUnderstanding {
    const confidence = analysis.confidence;
    const riskLevel = analysis.riskLevel;

    // 决定是否需要确认
    const requiresConfirmation = this.shouldRequireConfirmation(
      confidence,
      riskLevel,
      analysis.clarificationsNeeded.length,
    );

    return {
      id: generateId(),
      rawInput,
      goal: analysis.goal,
      successCriteria: analysis.successCriteria,
      implicitAssumptions: analysis.implicitAssumptions,
      clarificationsNeeded: analysis.clarificationsNeeded,
      executionPlan: analysis.executionPlan,
      confidence,
      requiresConfirmation,
      confirmationReason: requiresConfirmation
        ? this.getConfirmationReason(confidence, riskLevel)
        : undefined,
      riskLevel,
      metadata: {
        createdAt: new Date(),
        source: "llm",
        processingTimeMs,
      },
    };
  }

  /**
   * 判断是否需要用户确认
   */
  private shouldRequireConfirmation(
    confidence: number,
    riskLevel: "low" | "medium" | "high",
    clarificationCount: number,
  ): boolean {
    // 高风险操作总是需要确认
    if (riskLevel === "high") return true;

    // 置信度低需要确认
    if (confidence < 0.7) return true;

    // 有澄清问题需要确认
    if (clarificationCount > 0) return true;

    // 中风险 + 中等置信度需要确认
    if (riskLevel === "medium" && confidence < 0.85) return true;

    return false;
  }

  /**
   * 获取确认原因
   */
  private getConfirmationReason(
    confidence: number,
    riskLevel: "low" | "medium" | "high",
  ): string {
    if (riskLevel === "high") {
      return "这个操作可能会修改或删除数据，需要你确认";
    }
    if (confidence < 0.7) {
      return "我想确认一下我的理解是否正确";
    }
    return "在执行前，我想确认一下计划是否符合你的预期";
  }
}

/**
 * 任务分析 Schema（用于 LLM 结构化输出）
 */
const TaskAnalysisSchema = z.object({
  goal: z.string().describe("用户的真正目标（一句话）"),
  successCriteria: z.string().describe("成功标准"),
  implicitAssumptions: z.array(z.string()).describe("隐含假设"),
  clarificationsNeeded: z
    .array(
      z.object({
        id: z.string(),
        question: z.string(),
        options: z
          .array(
            z.object({
              value: z.string(),
              label: z.string(),
              recommended: z.boolean().optional(),
            }),
          )
          .optional(),
        impact: z.enum(["high", "medium", "low"]),
        required: z.boolean(),
      }),
    )
    .describe("需要澄清的问题"),
  executionPlan: z
    .array(
      z.object({
        step: z.number(),
        description: z.string(),
        action: z.string().optional(),
        requiresConfirmation: z.boolean().optional(),
      }),
    )
    .describe("执行计划"),
  confidence: z.number().min(0).max(1).describe("置信度"),
  riskLevel: z.enum(["low", "medium", "high"]).describe("风险级别"),
});
```

### 4.2 TemplateEngine（任务模板引擎）

```typescript
// src/lib/ai-agent/task-understanding/templates/template-engine.ts

import type {
  TaskTemplate,
  TaskUnderstanding,
  IntentClassification,
  TaskContext,
} from "../types";
import { TASK_TEMPLATES } from "./builtin-templates";

/**
 * 任务模板引擎
 *
 * 匹配和应用预定义的任务模板
 */
export class TemplateEngine {
  private templates: TaskTemplate[];

  constructor() {
    this.templates = TASK_TEMPLATES;
  }

  /**
   * 匹配任务模板
   */
  matchTemplate(
    classification: IntentClassification,
    context: TaskContext,
  ): TaskTemplate | null {
    for (const template of this.templates) {
      if (this.matchesTriggers(template.triggers, classification, context)) {
        return template;
      }
    }
    return null;
  }

  /**
   * 从模板生成任务理解
   */
  generateFromTemplate(
    template: TaskTemplate,
    input: string,
    classification: IntentClassification,
    context: TaskContext,
  ): TaskUnderstanding {
    const startTime = performance.now();

    // 个性化目标描述
    const goal = this.personalizeGoal(
      template.goalTemplate,
      classification,
      context,
    );

    // 筛选相关的澄清问题
    const clarifications = this.filterClarifications(
      template.commonClarifications,
      context,
    );

    // 适配执行计划
    const plan = this.adaptPlan(template.standardPlan, context);

    // 计算置信度（模板匹配通常有较高置信度）
    const confidence = 0.85;

    return {
      id: generateId(),
      rawInput: input,
      goal,
      successCriteria: template.defaultSuccessCriteria,
      implicitAssumptions: this.inferAssumptions(template, context),
      clarificationsNeeded: clarifications,
      executionPlan: plan,
      confidence,
      requiresConfirmation: template.requiresConfirmation,
      confirmationReason: template.requiresConfirmation
        ? "在执行前，我想确认一下计划是否符合你的预期"
        : undefined,
      riskLevel: template.riskLevel,
      matchedTemplate: template.id,
      metadata: {
        createdAt: new Date(),
        source: "template",
        processingTimeMs: performance.now() - startTime,
      },
    };
  }

  /**
   * 检查是否匹配触发条件
   */
  private matchesTriggers(
    triggers: TaskTriggers,
    classification: IntentClassification,
    context: TaskContext,
  ): boolean {
    // 检查意图类型
    if (!triggers.intents.includes(classification.type)) {
      return false;
    }

    // 检查必须的实体
    if (triggers.requiredEntities) {
      for (const entityCond of triggers.requiredEntities) {
        const entity = classification.entities.find(
          (e) => e.type === entityCond.type,
        );
        if (!entity) return false;
        if (
          entityCond.allowedValues &&
          !entityCond.allowedValues.includes(entity.value)
        ) {
          return false;
        }
      }
    }

    // 检查上下文条件
    if (triggers.contextConditions) {
      for (const cond of triggers.contextConditions) {
        if (!this.checkContextCondition(cond, context)) {
          return false;
        }
      }
    }

    return true;
  }

  /**
   * 个性化目标描述
   */
  private personalizeGoal(
    template: string,
    classification: IntentClassification,
    context: TaskContext,
  ): string {
    let goal = template;

    // 替换占位符
    const platformEntity = classification.entities.find(
      (e) => e.type === "platform",
    );
    if (platformEntity) {
      goal = goal.replace(
        "{platform}",
        this.formatPlatform(platformEntity.value),
      );
    }

    const quantityEntity = classification.entities.find(
      (e) => e.type === "quantity",
    );
    if (quantityEntity) {
      goal = goal.replace("{quantity}", quantityEntity.value);
    }

    const topicEntity = classification.entities.find((e) => e.type === "topic");
    if (topicEntity) {
      goal = goal.replace("{topic}", topicEntity.value);
    }

    return goal;
  }

  /**
   * 筛选相关的澄清问题
   */
  private filterClarifications(
    questions: ClarificationQuestion[],
    context: TaskContext,
  ): ClarificationQuestion[] {
    return questions.filter((q) => {
      // 如果上下文已经有答案，跳过
      // 例如：如果用户只有一个文件夹，不需要问"移动到哪个文件夹"
      if (q.id === "target_folder" && context.folders.length <= 1) {
        return false;
      }

      // 如果数量很少，不需要问"要处理多少条"
      if (q.id === "quantity_scope" && context.stats.total < 20) {
        return false;
      }

      return true;
    });
  }

  /**
   * 适配执行计划
   */
  private adaptPlan(
    standardPlan: ExecutionStep[],
    context: TaskContext,
  ): ExecutionStep[] {
    return standardPlan.map((step) => {
      // 根据上下文调整步骤描述
      let description = step.description;

      // 例如：如果数量很多，添加提示
      if (context.stats.total > 100 && step.action === "list_bookmarks") {
        description += "（数量较多，可能需要一点时间）";
      }

      return { ...step, description };
    });
  }

  /**
   * 添加自定义模板
   */
  addTemplate(template: TaskTemplate): void {
    this.templates.unshift(template);
  }
}
```

### 4.4 DialogueFlow（对话流程控制）

```typescript
// src/lib/ai-agent/task-understanding/dialogue/dialogue-flow.ts

import type {
  TaskUnderstanding,
  DialogueState,
  DialogueAction,
  DialoguePhase,
  ClarificationAnswer,
} from "../types";

/**
 * 对话流程控制器
 *
 * 管理渐进式对话的状态和流程
 */
export class DialogueFlow {
  private state: DialogueState;

  constructor() {
    this.state = this.createInitialState();
  }

  /**
   * 处理新的任务理解
   */
  processUnderstanding(understanding: TaskUnderstanding): DialogueAction[] {
    this.state.understanding = understanding;
    const actions: DialogueAction[] = [];

    // 1. 总是先展示理解
    actions.push({
      type: "show_understanding",
      understanding,
    });

    // 2. 如果需要澄清，进入澄清阶段
    if (understanding.clarificationsNeeded.length > 0) {
      this.state.phase = "clarification";
      actions.push({
        type: "ask_clarification",
        questions: understanding.clarificationsNeeded.filter(
          (q) => q.impact === "high",
        ),
      });
      return actions;
    }

    // 3. 如果需要确认，进入确认阶段
    if (understanding.requiresConfirmation) {
      this.state.phase = "confirmation";
      actions.push({
        type: "show_plan",
        plan: understanding.executionPlan,
      });
      actions.push({
        type: "request_confirmation",
        reason: understanding.confirmationReason || "请确认是否执行",
      });
      return actions;
    }

    // 4. 直接执行
    this.state.phase = "execution";
    actions.push({
      type: "execute",
      steps: understanding.executionPlan,
    });

    return actions;
  }

  /**
   * 处理用户的澄清回答
   */
  processClarificationAnswer(answer: ClarificationAnswer): DialogueAction[] {
    this.state.clarificationAnswers.set(answer.questionId, answer);

    // 检查是否还有未回答的高影响问题
    const unansweredHighImpact =
      this.state.understanding!.clarificationsNeeded.filter(
        (q) =>
          q.impact === "high" && !this.state.clarificationAnswers.has(q.id),
      );

    if (unansweredHighImpact.length > 0) {
      // 继续澄清
      return [
        {
          type: "ask_clarification",
          questions: unansweredHighImpact,
        },
      ];
    }

    // 澄清完成，更新理解并进入下一阶段
    this.updateUnderstandingWithAnswers();

    if (this.state.understanding!.requiresConfirmation) {
      this.state.phase = "confirmation";
      return [
        {
          type: "show_plan",
          plan: this.state.understanding!.executionPlan,
        },
        {
          type: "request_confirmation",
          reason: "根据你的回答，我更新了计划。请确认是否执行",
        },
      ];
    }

    // 直接执行
    this.state.phase = "execution";
    return [
      {
        type: "execute",
        steps: this.state.understanding!.executionPlan,
      },
    ];
  }

  /**
   * 处理用户确认
   */
  processConfirmation(confirmed: boolean): DialogueAction[] {
    if (!confirmed) {
      // 用户拒绝，询问原因
      return [
        {
          type: "ask_clarification",
          questions: [
            {
              id: "rejection_reason",
              question: "好的，请告诉我你想怎么调整？",
              type: "free_text",
              impact: "high",
              required: true,
            },
          ],
        },
      ];
    }

    // 用户确认，开始执行
    this.state.phase = "execution";
    this.state.planConfirmed = true;

    return [
      {
        type: "execute",
        steps: this.state.understanding!.executionPlan,
      },
    ];
  }

  /**
   * 格式化理解展示
   */
  formatUnderstanding(understanding: TaskUnderstanding): string {
    let text = `我理解你想要：**${understanding.goal}**\n\n`;

    if (understanding.implicitAssumptions.length > 0) {
      text += "我会注意：\n";
      understanding.implicitAssumptions.forEach((a) => {
        text += `- ${a}\n`;
      });
      text += "\n";
    }

    return text;
  }

  /**
   * 格式化计划展示
   */
  formatPlan(steps: ExecutionStep[]): string {
    let text = "我的计划是：\n\n";

    steps.forEach((step, i) => {
      const marker = step.requiresConfirmation ? "🔔" : `${i + 1}.`;
      text += `${marker} ${step.description}\n`;
    });

    return text;
  }

  /**
   * 重置对话状态
   */
  reset(): void {
    this.state = this.createInitialState();
  }

  /**
   * 获取当前状态
   */
  getState(): DialogueState {
    return { ...this.state };
  }

  private createInitialState(): DialogueState {
    return {
      phase: "understanding",
      clarificationAnswers: new Map(),
      history: [],
    };
  }
```

```typescript
import { TemplateEngine } from "@/lib/ai-agent/task-understanding/templates";

// ... existing code ...

async function* executeWithTaskUnderstanding(
  input: string,
  context: UnifiedContext,
  options: UnifiedOptions,
): AsyncGenerator<UnifiedStreamChunk> {
  const taskAnalyzer = new TaskAnalyzer({ model: context.model });
  const templateEngine = new TemplateEngine();
  const dialogueFlow = new DialogueFlow();

  // 1. 意图分类（现有逻辑）
  const classification = await intentRouter.classify(input);

  // 2. 尝试模板匹配
  const template = templateEngine.matchTemplate(
    classification,
    context.taskContext,
  );

  // 3. 生成任务理解
  let understanding: TaskUnderstanding;

  if (template) {
    // 使用模板快速生成
    understanding = templateEngine.generateFromTemplate(
      template,
      input,
      classification,
      context.taskContext,
    );

    yield {
      type: "routing",
      data: {
        mode: "template",
        template: template.id,
        reason: `匹配任务模板: ${template.name}`,
      },
    };
  } else {
    // 使用 LLM 深度分析
    yield {
      type: "progress",
      data: { step: "analyzing", percent: 10, message: "正在理解你的需求..." },
    };

    understanding = await taskAnalyzer.analyze(input, context.taskContext);

    yield {
      type: "routing",
      data: {
        mode: "llm_analysis",
        confidence: understanding.confidence,
        reason: "LLM 深度分析",
      },
    };
  }

  // 4. 获取对话动作
  const actions = dialogueFlow.processUnderstanding(understanding);

  // 5. 执行对话动作
  for (const action of actions) {
    switch (action.type) {
      case "show_understanding":
        yield {
          type: "understanding",
          data: {
            goal: action.understanding.goal,
            assumptions: action.understanding.implicitAssumptions,
            confidence: action.understanding.confidence,
            requiresConfirmation: action.understanding.requiresConfirmation,
          },
        };
        break;

      case "ask_clarification":
        yield {
          type: "clarification",
          data: {
            questions: action.questions,
          },
        };
        // 需要等待用户回答，暂停生成
        return;

      case "show_plan":
        yield {
          type: "plan",
          data: {
            steps: action.plan,
          },
        };
        break;

      case "request_confirmation":
        yield {
          type: "confirmation_needed",
          data: {
            reason: action.reason,
            understanding,
          },
        };
        // 需要等待用户确认，暂停生成
        return;

      case "execute":
        // 继续执行
        yield* executeSteps(action.steps, understanding, context);
        break;
    }
  }
}
```

## 十一、设计理念修订（第二次审核反馈）

### 11.0 核心理念约束

> **AnyKeep 设计理念：用户零接管**
>
> - ❌ 不要总是弹选项让用户做选择题
> - ✅ 简洁、智能、迅速
> - ✅ AI 主动决策，用户事后纠错

### 11.0.1 原方案问题

原方案中以下设计**违背了零接管理念**：

| 原设计                            | 问题             | 调整方向                     |
| --------------------------------- | ---------------- | ---------------------------- |
| `clarificationsNeeded` 澄清问题   | 让用户做选择题   | **删除**，改用智能默认值     |
| `requiresConfirmation` 执行前确认 | 打断用户流程     | **大幅收紧**，仅删除操作确认 |
| `渐进式对话`                      | 多轮交互增加摩擦 | **改为静默理解**             |
| `UnderstandingConfirmation` 组件  | 弹窗确认         | **改为思考气泡**（非阻断）   |

### 11.0.2 新设计原则

```
原则 1: 静默优先
  AI 在后台完成理解，不打扰用户

原则 2: 智能默认
  不确定时用智能默认值，而不是问用户

原则 3: 事后纠错
  先执行 → 告诉结果 → 用户不满意再调整（一键撤销）

原则 4: 极端才确认
  只有不可逆操作（删除）才弹确认，且是一键式

原则 5: 偏好学习
  记住用户习惯，同类任务下次直接执行
```

### 11.0.3 修订后的交互流程

```
【原流程 - 多轮确认】
用户输入 → AI 理解 → 展示理解 → 用户确认 → 澄清问题 → 用户回答 → 执行

【新流程 - 零接管】
用户输入 → AI 静默理解 → 直接执行 → 展示结果（含思考过程）→ 一键撤销（如需要）
                ↓
            （仅删除操作）→ 快速确认弹窗 → 执行
```

### 12.1 智能默认值系统

**核心思想**：不问用户，AI 自己决定，记录决策供用户事后查看和纠正。

```typescript
// src/lib/ai-agent/task-understanding/smart-defaults.ts

/**
 * 智能默认值服务
 *
 * 决策优先级：用户偏好 > 上下文推断 > 系统默认
 */
class SmartDefaultsService {
  constructor(
    private preferencesService: PreferencesService,
    private contextService: ContextService,
  ) {}

  /**
   * 为任务获取智能默认值
   */
  async resolveDefaults(
    taskType: string,
    context: TaskContext,
  ): Promise<SmartDefault[]> {
    const defaults: SmartDefault[] = [];

    // 1. 整理方式
    if (taskType === "organize") {
      const method = await this.resolveOrganizeMethod(context);
      defaults.push(method);
    }

    // 2. 目标文件夹
    if (taskType === "move" || taskType === "organize") {
      const folder = await this.resolveTargetFolder(context);
      defaults.push(folder);
    }

    // 3. 处理范围
    const scope = await this.resolveScope(context);
    defaults.push(scope);

    return defaults;
  }

  /**
   * 解析整理方式
   */
  private async resolveOrganizeMethod(
    context: TaskContext,
  ): Promise<SmartDefault> {
    // 1. 检查用户偏好
    const userPref = await this.preferencesService.get(context.userId);
    if (userPref.preferredOrganizeMethod) {
      return {
        field: "organizeMethod",
        value: userPref.preferredOrganizeMethod,
        source: "user_preference",
        confidence: 0.95,
      };
    }

    // 2. 上下文推断
    // 如果用户说"按时间整理"，从输入中提取
    const inputHint = this.extractMethodHint(context.input);
    if (inputHint) {
      return {
        field: "organizeMethod",
        value: inputHint,
        source: "context_infer",
        confidence: 0.85,
      };
    }

    // 3. 根据内容特征推断
    if (context.stats.platformDistribution.BILIBILI > 50) {
      // B站内容多，按 UP主 分类可能更合适
      return {
        field: "organizeMethod",
        value: "author",
        source: "context_infer",
        confidence: 0.7,
      };
    }

    // 4. 系统默认
    return {
      field: "organizeMethod",
      value: "topic", // 默认按主题
      source: "system_default",
      confidence: 0.6,
    };
  }

  /**
   * 解析目标文件夹
   */
  private async resolveTargetFolder(
    context: TaskContext,
  ): Promise<SmartDefault> {
    const userPref = await this.preferencesService.get(context.userId);

    // 1. 用户有常用文件夹
    if (userPref.frequentFolders?.length > 0) {
      // 根据内容主题匹配最相关的文件夹
      const bestMatch = this.matchFolderByContent(
        context.targetBookmarks,
        userPref.frequentFolders,
      );
      if (bestMatch) {
        return {
          field: "targetFolder",
          value: bestMatch.id,
          source: "user_preference",
          confidence: 0.8,
        };
      }
    }

    // 2. 根据平台推断
    const platform = context.targetBookmarks[0]?.platform;
    const platformFolder = context.folders.find((f) =>
      f.name.toLowerCase().includes(platform?.toLowerCase() || ""),
    );
    if (platformFolder) {
      return {
        field: "targetFolder",
        value: platformFolder.id,
        source: "context_infer",
        confidence: 0.75,
      };
    }

    // 3. 默认"待整理"
    const inboxFolder = context.folders.find((f) => f.name === "待整理");
    return {
      field: "targetFolder",
      value: inboxFolder?.id || "inbox",
      source: "system_default",
      confidence: 0.5,
    };
  }
}
```

### 12.4 偏好学习强化

**核心思想**：从用户的撤销和调整中学习，下次自动应用。

```typescript
// src/lib/ai-agent/preferences/learning-service-v2.ts

/**
 * 偏好学习服务 v2 - 基于事后反馈
 */
class PreferenceLearningServiceV2 {
  /**
   * 从执行结果学习
   */
  async learnFromExecution(
    userId: string,
    understanding: TaskUnderstanding,
    result: TaskExecutionResult,
    userAction: "accepted" | "undone" | "adjusted",
  ): Promise<void> {
    // 用户接受了结果 → 强化当前默认值
    if (userAction === "accepted") {
      for (const defaultVal of understanding.appliedDefaults) {
        await this.reinforceDefault(userId, defaultVal);
      }
      return;
    }

    // 用户撤销了 → 降低当前默认值的权重
    if (userAction === "undone") {
      for (const defaultVal of understanding.appliedDefaults) {
        await this.weakenDefault(userId, defaultVal);
      }
      return;
    }

    // 用户调整了 → 学习新的偏好
    if (userAction === "adjusted" && result.adjustment) {
      await this.learnNewPreference(userId, result.adjustment);
    }
  }

  /**
   * 强化默认值
   */
  private async reinforceDefault(
    userId: string,
    defaultVal: SmartDefault,
  ): Promise<void> {
    const pref = await this.preferencesService.get(userId);

    // 增加使用计数
    pref.defaultUsage = pref.defaultUsage || {};
    pref.defaultUsage[defaultVal.field] =
      pref.defaultUsage[defaultVal.field] || {};
    pref.defaultUsage[defaultVal.field][defaultVal.value] =
      (pref.defaultUsage[defaultVal.field][defaultVal.value] || 0) + 1;

    // 如果某个值使用次数超过 3 次，设为用户偏好
    if (pref.defaultUsage[defaultVal.field][defaultVal.value] >= 3) {
      pref[`preferred${capitalize(defaultVal.field)}`] = defaultVal.value;
    }

    await this.preferencesService.update(userId, pref);
  }

  /**
   * 弱化默认值
   */
  private async weakenDefault(
    userId: string,
    defaultVal: SmartDefault,
  ): Promise<void> {
    const pref = await this.preferencesService.get(userId);

    pref.defaultUsage = pref.defaultUsage || {};
    pref.defaultUsage[defaultVal.field] =
      pref.defaultUsage[defaultVal.field] || {};
    pref.defaultUsage[defaultVal.field][defaultVal.value] = Math.max(
      0,
      (pref.defaultUsage[defaultVal.field][defaultVal.value] || 0) - 2,
    );

    await this.preferencesService.update(userId, pref);
  }
}
```
