### Phase 1: 创建 BookmarkRepository（1小时）

**目标**：建立统一的书签数据访问层

#### 1.1 创建仓储基类

**文件**：`src/lib/repositories/bookmark.repository.ts`

```typescript
/**
 * 书签仓储 - 统一的数据访问层
 * 所有书签查询必须通过此类，确保过滤条件一致
 */

import type { SupabaseClient } from "@supabase/supabase-js";
import type { Bookmark, Platform } from "@/types/supabase";

// 浏览器平台常量（单一定义点）
export const BROWSER_PLATFORMS: Platform[] = [
  "BROWSER_CHROME",
  "BROWSER_EDGE",
  "BROWSER_FIREFOX",
  "BROWSER_SAFARI",
];

export function isBrowserPlatform(platform: Platform | null): boolean {
  if (!platform) return false;
  return BROWSER_PLATFORMS.includes(platform);
}

export type BookmarkScope = "active" | "trashed" | "all";

export interface BookmarkQueryOptions {
  scope?: BookmarkScope;
  excludeBrowser?: boolean;
  platform?: string;
  folderId?: number;
  limit?: number;
  offset?: number;
}

export class BookmarkRepository {
  constructor(
    private supabase: SupabaseClient,
    private userId: string,
  ) {}

  /**
   * 获取所有书签（支持分页以突破 1000 行限制）
   */
  async getAll(options: BookmarkQueryOptions = {}): Promise<Bookmark[]> {
    const { scope = "active", limit, offset } = options;
    const allBookmarks: Bookmark[] = [];
    const pageSize = limit || 1000;
    let currentOffset = offset || 0;
    let hasMore = true;

    while (hasMore) {
      let query = this.supabase
        .from("bookmarks")
        .select("*")
        .eq("user_id", this.userId)
        .is("deleted_at", null);

      // 应用范围过滤
      query = this.applyScope(query, scope);

      // 分页
      query = query
        .order("created_at", { ascending: false })
        .range(currentOffset, currentOffset + pageSize - 1);

      const { data, error } = await query;

      if (error) {
        throw new Error(`BookmarkRepository.getAll failed: ${error.message}`);
      }

      if (data && data.length > 0) {
        allBookmarks.push(...(data as Bookmark[]));
        currentOffset += data.length;
        // 如果指定了 limit，不继续分页
        hasMore = !limit && data.length === pageSize;
      } else {
        hasMore = false;
      }
    }

    return allBookmarks;
  }

  /**
   * 获取内容书签（排除浏览器书签）
   */
  async getContentBookmarks(): Promise<Bookmark[]> {
    const all = await this.getAll({ scope: "active" });
    return all.filter((b) => !isBrowserPlatform(b.platform as Platform));
  }

  /**
   * 获取浏览器书签
   */
  async getBrowserBookmarks(): Promise<Bookmark[]> {
    const all = await this.getAll({ scope: "active" });
    return all.filter((b) => isBrowserPlatform(b.platform as Platform));
  }

  /**
   * 获取书签数量
   */
  async count(options: BookmarkQueryOptions = {}): Promise<number> {
    const { scope = "active" } = options;

    let query = this.supabase
      .from("bookmarks")
      .select("id", { count: "exact", head: true })
      .eq("user_id", this.userId)
      .is("deleted_at", null);

    query = this.applyScope(query, scope);

    const { count, error } = await query;

    if (error) {
      throw new Error(`BookmarkRepository.count failed: ${error.message}`);
    }

    return count || 0;
  }

  private applyScope(query: any, scope: BookmarkScope) {
    switch (scope) {
      case "active":
        // 活跃书签：排除垃圾箱（兼容 NULL 值）
        return query.or("is_in_trash.eq.false,is_in_trash.is.null");
      case "trashed":
        return query.eq("is_in_trash", true);
      case "all":
        return query;
      default:
        return query.or("is_in_trash.eq.false,is_in_trash.is.null");
    }
  }
}

/**
 * 工厂函数
 */
export function createBookmarkRepository(
  supabase: SupabaseClient,
  userId: string,
): BookmarkRepository {
  return new BookmarkRepository(supabase, userId);
}
```

### Phase 2: 重构 AnalyticsService（1小时）

**目标**：使用 BookmarkRepository，简化统计逻辑

#### 2.1 重构后的 AnalyticsService

```typescript
// src/lib/analytics/analytics.service.ts

import {
  createBookmarkRepository,
  isBrowserPlatform,
} from "@/lib/repositories/bookmark.repository";

export class AnalyticsService {
  private repository: BookmarkRepository;

  constructor(
    supabase: SupabaseClient,
    userId: string,
    config: Partial<AnalyticsConfig> = {},
  ) {
    this.supabase = supabase;
    this.userId = userId;
    this.config = { ...DEFAULT_ANALYTICS_CONFIG, ...config };
    this.repository = createBookmarkRepository(supabase, userId);
  }

  async getFullAnalytics(): Promise<AnalyticsData> {
    // 使用仓储获取数据
    const allBookmarks = await this.repository.getAll({ scope: "active" });

    // 在内存中分离（确保使用同一数据源）
    const contentBookmarks = allBookmarks.filter(
      (b) => !isBrowserPlatform(b.platform as Platform),
    );
    const browserBookmarks = allBookmarks.filter((b) =>
      isBrowserPlatform(b.platform as Platform),
    );

    // 计算统计
    const stats = this.calculateStats(
      allBookmarks,
      contentBookmarks,
      browserBookmarks,
    );

    // ... 其他计算
  }

  private calculateStats(
    allBookmarks: Bookmark[],
    contentBookmarks: Bookmark[],
    browserBookmarks: Bookmark[],
  ): AnalyticsStats {
    const now = new Date();
    const todayStart = this.getDateStart(now, 0);
    const weekStart = this.getDateStart(now, 7);
    const monthStart = this.getDateStart(now, 30);

    // 所有时间统计都基于 contentBookmarks
    const todayCount = this.countByDateRange(contentBookmarks, todayStart);
    const weekCount = this.countByDateRange(contentBookmarks, weekStart);
    const last30DaysCount = this.countByDateRange(contentBookmarks, monthStart);

    return {
      totalBookmarks: allBookmarks.length,
      contentBookmarks: contentBookmarks.length,
      browserBookmarks: browserBookmarks.length,
      todayCount,
      weekCount,
      last30DaysCount,
      unclassifiedCount: contentBookmarks.filter((b) => !b.folder_id).length,
      classificationRate: this.calculateRate(contentBookmarks),
    };
  }

  private countByDateRange(bookmarks: Bookmark[], startDate: Date): number {
    return bookmarks.filter((b) => {
      if (!b.created_at) return false;
      return new Date(b.created_at) >= startDate;
    }).length;
  }

  private getDateStart(base: Date, daysAgo: number): Date {
    const date = new Date(base);
    date.setDate(date.getDate() - daysAgo);
    date.setHours(0, 0, 0, 0);
    return date;
  }
}
```

### Phase 3: 更新数据库视图（30分钟）

**目标**：在数据库层建立防护

#### 3.1 迁移脚本

```sql
-- supabase/migrations/20251220200000_update_active_bookmarks_view.sql

-- 更新视图：兼容 is_in_trash = NULL
CREATE OR REPLACE VIEW v_active_bookmarks AS
SELECT *
FROM bookmarks
WHERE deleted_at IS NULL
  AND (is_in_trash = FALSE OR is_in_trash IS NULL);

-- 创建内容书签视图（排除浏览器）
CREATE OR REPLACE VIEW v_content_bookmarks AS
SELECT *
FROM bookmarks
WHERE deleted_at IS NULL
  AND (is_in_trash = FALSE OR is_in_trash IS NULL)
  AND platform NOT IN ('BROWSER_CHROME', 'BROWSER_EDGE', 'BROWSER_FIREFOX', 'BROWSER_SAFARI');

-- 创建浏览器书签视图
CREATE OR REPLACE VIEW v_browser_bookmarks AS
SELECT *
FROM bookmarks
WHERE deleted_at IS NULL
  AND (is_in_trash = FALSE OR is_in_trash IS NULL)
  AND platform IN ('BROWSER_CHROME', 'BROWSER_EDGE', 'BROWSER_FIREFOX', 'BROWSER_SAFARI');

COMMENT ON VIEW v_content_bookmarks IS '内容书签视图：排除浏览器书签，用于统计分析';
COMMENT ON VIEW v_browser_bookmarks IS '浏览器书签视图：仅浏览器导入的书签';
```

```typescript
describe("Analytics 数据一致性", () => {
```
