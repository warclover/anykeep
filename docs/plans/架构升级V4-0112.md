```typescript
// src/lib/ai-skills/engine/execution-coordinator.ts（修改 L512-552）

private buildAgentContext(context: EngineContext): AgentContext {
  // 构建 System Prompt（包含附件上下文和情景记忆）
  let systemPrompt = context.attachmentContext || "";

  // 注入情景记忆
  if (
    FeatureFlags.ENABLE_EPISODIC_MEMORY &&
    FeatureFlags.ENABLE_MEMORY_PROMPT &&
    context.episodicMemory &&
    !isEmptyMemoryContext(context.episodicMemory)
  ) {
    const memoryPrompt = buildMemoryPrompt(context.episodicMemory);
    systemPrompt = systemPrompt
      ? `${systemPrompt}\n\n${memoryPrompt}`
      : memoryPrompt;
  }

  // 🆕 注入置信度分层信息到 SystemPrompt
  if (context.confidenceLevel && context.responseStrategy) {
    const confidencePrompt = this.buildConfidencePrompt(context);
    systemPrompt = systemPrompt
      ? `${systemPrompt}\n\n${confidencePrompt}`
      : confidencePrompt;
  }

  // 🆕 注入数据上下文到 SystemPrompt
  const dataContextPrompt = this.buildDataContextPrompt(context);
  if (dataContextPrompt) {
    systemPrompt = systemPrompt
      ? `${systemPrompt}\n\n${dataContextPrompt}`
      : dataContextPrompt;
  }

  return {
    // === 现有映射（保持不变） ===
    userId: context.userId,
    conversationId: context.conversationId,
    bookmarkIds: context.bookmarkIds,
    messageHistory: context.messageHistory,
    traceId: context.traceId,
    systemPrompt: systemPrompt || undefined,
    onSearchResults: context.onSearchResults,
    intentType: context.intentType,
    intentSubType: context.intentSubType,
    intentConfidence: context.intentConfidence,
    slots: context.slots,

    // === 🆕 新增映射：上下文参数 ===
    folderId: context.folderId,
    totalBookmarks: context.totalBookmarks,
    totalFolders: context.totalFolders,
    folders: context.folders,
    tags: context.tags,

    // === 🆕 新增映射：置信度分层 ===
    responseStrategy: context.responseStrategy as AgentContext["responseStrategy"],
    confidenceLevel: context.confidenceLevel as AgentContext["confidenceLevel"],
    enhancedConfidence: context.enhancedConfidence,
  };
}

/**
 * 🆕 构建置信度 Prompt
 */
private buildConfidencePrompt(context: EngineContext): string {
  const levelDesc = {
    high: "高（≥85%）- 用户意图明确，可直接执行",
    medium: "中（60-84%）- 意图基本明确，执行后简要解释",
    low: "低（<60%）- 意图不明确，需要澄清",
  };

  const strategyDesc = {
    silent_execute: "静默执行，完成后简短汇报结果",
    with_explanation: "执行 + 简要解释为什么这样做",
    clarify: "轻量级澄清（选项式，非开放问）",
  };

  const level = context.confidenceLevel || "medium";
  const strategy = context.responseStrategy || "with_explanation";

  return `【AI 执行策略】
- 意图置信度: ${levelDesc[level]}
- 建议响应方式: ${strategyDesc[strategy]}
- 置信度分数: ${(context.enhancedConfidence ?? 0.7).toFixed(2)}`;
}

/**
 * 🆕 构建数据上下文 Prompt
 */
private buildDataContextPrompt(context: EngineContext): string | null {
  const parts: string[] = [];

  if (context.folderId !== undefined) {
    const folderName = context.folders?.find(f => f.id === context.folderId)?.name;
    parts.push(`当前文件夹: ${folderName || `ID:${context.folderId}`}`);
  }

  if (context.totalBookmarks !== undefined) {
    parts.push(`书签总数: ${context.totalBookmarks}`);
  }

  if (context.totalFolders !== undefined) {
    parts.push(`文件夹数: ${context.totalFolders}`);
  }

  if (parts.length === 0) return null;

  return `【用户数据上下文】\n${parts.join(" | ")}`;
}
```

### 1.3 胶囊注册表

```typescript
// src/lib/ai-agent/tools/capsules/capsule-registry.ts

import { STATIC_CAPSULES, StaticCapsule } from "./static-capsules";

class StaticCapsuleRegistry {
  private capsuleMap: Map<string, StaticCapsule>;
  private intentMap: Map<IntentType, StaticCapsule[]>;

  constructor() {
    this.capsuleMap = new Map();
    this.intentMap = new Map();
    this.initialize();
  }

  private initialize(): void {
    for (const capsule of STATIC_CAPSULES) {
      this.capsuleMap.set(capsule.id, capsule);

      for (const intent of capsule.intentTypes) {
        const existing = this.intentMap.get(intent) || [];
        existing.push(capsule);
        this.intentMap.set(intent, existing);
      }
    }
  }

  /**
   * 根据意图获取胶囊（核心方法）
   * 简单直接，无 LLM 调用
   */
  getCapsulesForIntent(intent: IntentType): StaticCapsule[] {
    return this.intentMap.get(intent) || [];
  }

  /**
   * 获取胶囊内的工具 ID 列表
   */
  getToolIdsForIntent(intent: IntentType): string[] {
    const capsules = this.getCapsulesForIntent(intent);
    const toolIds = new Set<string>();

    for (const capsule of capsules) {
      for (const toolId of capsule.toolIds) {
        toolIds.add(toolId);
      }
    }

    return Array.from(toolIds);
  }
}

export const capsuleRegistry = new StaticCapsuleRegistry();
```

```typescript
async function buildToolsForIntent(intent: IntentType): Promise<Tool[]> {
  // 获取该意图需要的工具 ID
  const toolIds = capsuleRegistry.getToolIdsForIntent(intent);

  // 从完整工具注册表中筛选

  });

}
```

```typescript
/**
 * L0 规则定义 - 极简版
 *
 * 设计原则：
 * 1. 只处理 100% 确定的场景

export interface L0Rule {

  name: string;
  patterns: RegExp[];           // 匹配模式

  toolId: string;               // 直接调用的工具

export const L0_RULES: L0Rule[] = [
```

```typescript
interface L0MatchResult {
  matched: boolean;
  rule?: L0Rule;
  params?: Record<string, unknown>;
  confidence: number;
}

class L0Router {
  /**
   * 尝试 L0 匹配
   * 返回 null 表示需要走 LLM 路由
   */
  tryMatch(input: string): L0MatchResult {
    const normalizedInput = input.trim();

    // 长度逃逸：超过 50 字符的复杂请求交给 LLM
    if (normalizedInput.length > 50) {
      return { matched: false, confidence: 0 };
    }

    for (const rule of L0_RULES) {
      for (const pattern of rule.patterns) {
        const match = normalizedInput.match(pattern);
        if (match) {
          return {
            matched: true,
            rule,
            params: rule.paramExtractor(match, normalizedInput),
            confidence: 0.95, // L0 规则置信度固定为 95%
          };
        }
      }
    }

    return { matched: false, confidence: 0 };
  }
}

export const l0Router = new L0Router();
```

### 3.1 设计目标

- **仅支持可逆操作**

### 3.2 撤销管理器

```typescript
// src/lib/ai-agent/undo/simple-undo-manager.ts

/**
 * 简单撤销管理器 - 极简版
 *
 * 设计原则：
 * 1. 内存存储，不持久化（重启后清空）
 * 2. 仅记录可逆操作
 * 3. 最多保留 10 条
 */

export interface UndoableOperation {
  id: string;
  toolId: string;
  description: string;           // 操作描述
  params: Record<string, unknown>;
  undoFn: () => Promise<void>;   // 撤销函数
  createdAt: Date;
}

class SimpleUndoManager {
  private stack: UndoableOperation[] = [];
  private readonly MAX_SIZE = 10;

  /**
   * 记录可撤销操作
   */
  record(operation: Omit<UndoableOperation, "id" | "createdAt">): void {
    const record: UndoableOperation = {
      ...operation,
      id: crypto.randomUUID(),
      createdAt: new Date(),
    };

    this.stack.push(record);

    // 超出限制时移除最旧的
    if (this.stack.length > this.MAX_SIZE) {
      this.stack.shift();
    }
  }

  /**
   * 撤销最近一条操作
   */
  async undoLast(): Promise<{ success: boolean; message: string }> {
    const last = this.stack.pop();

    if (!last) {
      return { success: false, message: "没有可撤销的操作" };
    }

    try {
      await last.undoFn();
      return {
        success: true,
        message: `已撤销: ${last.description}`
      };
    } catch (error) {
      // 撤销失败，重新放回栈中
      this.stack.push(last);
      return {
        success: false,
        message: `撤销失败: ${error.message}`
      };
    }
  }

  /**
   * 获取可撤销操作列表
   */
  getUndoableList(): Array<{ id: string; description: string; createdAt: Date }> {
    return this.stack.map(op => ({
      id: op.id,
      description: op.description,
      createdAt: op.createdAt,
    })).reverse(); // 最新的在前
  }

  /**
   * 清空撤销栈
   */
  clear(): void {
    this.stack = [];
  }
}

export const undoManager = new SimpleUndoManager();
```

```typescript
// src/lib/ai-agent/cache/exact-match-cache.ts

/**
 * 精确匹配缓存 - 极简版
 *
 * 不做语义相似度，仅缓存完全相同的请求
 */

class ExactMatchCache {
  private cache: Map<string, { result: unknown; expireAt: number }> = new Map();
  private readonly TTL = 5 * 60 * 1000; // 5 分钟

  private normalize(input: string): string {
    return input.trim().toLowerCase();
  }

  get(input: string): unknown | null {
    const key = this.normalize(input);
    const entry = this.cache.get(key);

    if (!entry) return null;
    if (Date.now() > entry.expireAt) {
      this.cache.delete(key);
      return null;
    }

    return entry.result;
  }

  set(input: string, result: unknown): void {
    const key = this.normalize(input);
    this.cache.set(key, {
      result,
      expireAt: Date.now() + this.TTL,
    });

    // 简单的 LRU：超过 1000 条时清理最旧的
    if (this.cache.size > 1000) {
      const firstKey = this.cache.keys().next().value;
      this.cache.delete(firstKey);
    }
  }
}

export const queryCache = new ExactMatchCache();
```

```typescript
const SLOT_SCHEMAS: Record<string, SlotSchema[]> = {
  move_bookmark: [
    { name: "bookmarkId", required: true, prompt: "请指定要移动的书签" },
    { name: "targetFolder", required: true, prompt: "请指定目标文件夹" },
  ],
  search: [
    { name: "query", required: true, prompt: "请输入搜索关键词" },
  ],
  export: [
    { name: "format", required: true, prompt: "请指定导出格式（CSV/Markdown/JSON）" },
  ],
};

export function checkMissingSlots(
  toolId: string,
  params: Record<string, unknown>
): string | null {
  const schema = SLOT_SCHEMAS[toolId];
  if (!schema) return null;

  const missing = schema
    .filter(slot => slot.required && !params[slot.name])
    .map(slot => slot.prompt);

  if (missing.length === 0) return null;

  return missing.join("；");
}
```
