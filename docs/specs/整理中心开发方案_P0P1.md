### 2.2 模板渲染器

```typescript
// src/lib/ai-skills/skills/batch-organize/template-renderer.ts

import Mustache from "mustache";
import { ORGANIZE_EXPERT_TEMPLATES } from "./templates/expert-response";

interface DiagnosisData {
  stats: CollectionStats;
  issues: OrganizeIssue[];
  options: OrganizeOption[];
}

interface OrganizeOption {
  id: string;
  name: string;
  label: '推荐' | '快速' | '自定义';
  actions: string;
  duration: string;
}

/**
 * 渲染诊断报告
 */
export function renderDiagnosisReport(data: DiagnosisData): string {
  const { stats, issues, options } = data;

  const viewData = {
    total: stats.total.toLocaleString(),
    issues: issues.map((issue) => ({
      severityIcon: SEVERITY_ICONS[issue.severity],
      severityLabel: SEVERITY_LABELS[issue.severity],
      title: issue.title,
      percentage: issue.percentage.toFixed(1),
    })),
    optionCount: options.length,
    options: options.map((opt, index) => ({
      index: index + 1,
      ...opt,
    })),
  };

  return Mustache.render(ORGANIZE_EXPERT_TEMPLATES.diagnosis, viewData);
}

const SEVERITY_ICONS: Record<string, string> = {
  critical: '🔴',
  warning: '🟡',
  info: '🟢',
};

const SEVERITY_LABELS: Record<string, string> = {
  critical: '严重',
  warning: '中等',
  info: '建议',
};
```

```tsx
// src/components/organize/organize-page-client.tsx

"use client";

import { useState } from "react";
import { useUnifiedAI } from "@/hooks/domain/ai/use-unified-ai";
import { QuickActionGrid } from "@/components/organize/quick-action-grid";
import { OrganizeProgress } from "@/components/ai/organize/organize-progress";
import { OrganizeComparison } from "@/components/organize/organize-comparison";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Sparkles } from "lucide-react";

type OrganizeState = 'idle' | 'analyzing' | 'confirming' | 'executing' | 'complete';

interface OrganizeSnapshot {
  uncategorized: number;
  duplicates: number;
  untagged: number;
  stale: number;
  total: number;
}

export function OrganizePageClient() {
  const [state, setState] = useState<OrganizeState>('idle');
  const [beforeSnapshot, setBeforeSnapshot] = useState<OrganizeSnapshot | null>(null);
  const [afterSnapshot, setAfterSnapshot] = useState<OrganizeSnapshot | null>(null);

  const { sendMessage, isLoading: aiLoading } = useUnifiedAI();

  // 处理整理操作
  const handleOrganizeAction = async (actionType: string) => {
    setState('analyzing');

    try {
      // 获取整理前快照
      const beforeStats = await fetchCollectionStats();
      setBeforeSnapshot(beforeStats);

      // 调用 AI 整理
      const result = await sendMessage({
        message: getMessageForAction(actionType),
        context: {
          explicitSkill: getSkillForAction(actionType),
        },
      });

      // 获取整理后快照
      const afterStats = await fetchCollectionStats();
      setAfterSnapshot(afterStats);

      setState('complete');
    } catch (error) {
      setState('idle');
      // 错误处理
    }
  };

  return (
    <div className="container max-w-4xl py-6 space-y-6">
      {/* 页面标题 */}
      <div>
        <h1 className="text-2xl font-bold flex items-center gap-2">
          <Sparkles className="h-6 w-6 text-primary" />
          整理中心
        </h1>
        <p className="text-muted-foreground">
          AI 帮你智能整理收藏，保持知识库整洁有序
        </p>
      </div>

      {/* 主内容区 */}
      {state === 'idle' && (
        <QuickActionGrid onAction={handleOrganizeAction} />
      )}

      {(state === 'analyzing' || state === 'executing') && (
        <OrganizeProgress
          status={state}
          message={state === 'analyzing' ? '正在分析您的收藏...' : '正在执行整理操作...'}
        />
      )}

      {state === 'complete' && beforeSnapshot && afterSnapshot && (
        <OrganizeComparison
          before={beforeSnapshot}
          after={afterSnapshot}
          onReset={() => {
            setState('idle');
            setBeforeSnapshot(null);
            setAfterSnapshot(null);
          }}
        />
      )}
    </div>
  );
}

function getMessageForAction(actionType: string): string {
  const messages: Record<string, string> = {
    'one-click': '帮我一键整理收藏',
    'clean-duplicates': '清理重复的收藏',
    'smart-categorize': '智能分类未分类的收藏',
    'auto-tag': '为没有标签的收藏自动打标签',
    'archive-stale': '归档长期未读的内容',
  };
  return messages[actionType] || '帮我整理收藏';
}

function getSkillForAction(actionType: string): string {
  const skills: Record<string, string> = {
    'one-click': 'batch_organize',
    'clean-duplicates': 'duplicate_cleanup',
    'smart-categorize': 'batch_organize',
    'auto-tag': 'smart_tagging',
    'archive-stale': 'batch_organize',
  };
  return skills[actionType] || 'batch_organize';
}

async function fetchCollectionStats(): Promise<OrganizeSnapshot> {
  const response = await fetch('/api/collection-stats');
  return response.json();
}
```

### 4.3 对比组件实现

```tsx
// src/components/organize/organize-comparison.tsx

"use client";

import { OrganizeSnapshot } from "@/lib/organize/types";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Button } from "@/components/ui/button";
import { Progress } from "@/components/ui/progress";
import { ArrowRight, TrendingUp, TrendingDown, RotateCcw, Share2 } from "lucide-react";
import { PieChart, Pie, Cell, ResponsiveContainer, Legend, Tooltip } from "recharts";

interface OrganizeComparisonProps {
  before: OrganizeSnapshot;
  after: OrganizeSnapshot;
  onReset?: () => void;
  onShare?: () => void;
  className?: string;
}

const CHART_COLORS = {
  categorized: '#22c55e',   // 绿色 - 已分类
  uncategorized: '#ef4444', // 红色 - 未分类
  tagged: '#3b82f6',        // 蓝色 - 已标签
  untagged: '#f59e0b',      // 橙色 - 无标签
};

export function OrganizeComparison({
  before,
  after,
  onReset,
  onShare,
  className,
}: OrganizeComparisonProps) {
  // 计算改进指标
  const improvements = {
    categorized: before.uncategorized - after.uncategorized,
    cleaned: before.duplicates - after.duplicates,
    tagged: before.untagged - after.untagged,
    archived: before.stale - after.stale,
  };

  const totalImproved = Object.values(improvements).reduce((a, b) => Math.max(0, a) + Math.max(0, b), 0);
  const improvementPercent = before.total > 0
    ? Math.round((totalImproved / before.total) * 100)
    : 0;

  // 准备饼图数据
  const beforeChartData = [
    { name: '已分类', value: before.total - before.uncategorized, color: CHART_COLORS.categorized },
    { name: '未分类', value: before.uncategorized, color: CHART_COLORS.uncategorized },
  ].filter(d => d.value > 0);

  const afterChartData = [
    { name: '已分类', value: after.total - after.uncategorized, color: CHART_COLORS.categorized },
    { name: '未分类', value: after.uncategorized, color: CHART_COLORS.uncategorized },
  ].filter(d => d.value > 0);

  return (
    <Card className={className}>
      <CardHeader>
        <div className="flex items-center justify-between">
          <CardTitle className="flex items-center gap-2">
            ✨ 整理完成！
            {improvementPercent > 0 && (
              <span className="text-green-600 text-lg">
                优化了 {improvementPercent}%
              </span>
            )}
          </CardTitle>
          <div className="flex gap-2">
            {onShare && (
              <Button variant="outline" size="sm" onClick={onShare}>
                <Share2 className="h-4 w-4 mr-1" />
                分享成果
              </Button>
            )}
            {onReset && (
              <Button variant="outline" size="sm" onClick={onReset}>
                <RotateCcw className="h-4 w-4 mr-1" />
                继续整理
              </Button>
            )}
          </div>
        </div>
      </CardHeader>
      <CardContent className="space-y-6">
        {/* 关键指标变化 */}
        <div className="grid grid-cols-2 gap-4">
          <MetricCard
            label="未分类"
            before={before.uncategorized}
            after={after.uncategorized}
            isLowerBetter
          />
          <MetricCard
            label="重复项"
            before={before.duplicates}
            after={after.duplicates}
            isLowerBetter
          />
          <MetricCard
            label="无标签"
            before={before.untagged}
            after={after.untagged}
            isLowerBetter
          />
          <MetricCard
            label="长期未读"
            before={before.stale}
            after={after.stale}
            isLowerBetter
          />
        </div>

        {/* 分布对比图 */}
        <div className="grid grid-cols-2 gap-4 pt-4 border-t">
          <div>
            <p className="text-sm font-medium text-center mb-2 text-muted-foreground">
              整理前
            </p>
            <div className="h-40">
              <ResponsiveContainer width="100%" height="100%">
                <PieChart>
                  <Pie
                    data={beforeChartData}
                    dataKey="value"
                    nameKey="name"
                    cx="50%"
                    cy="50%"
                    outerRadius={55}
                    strokeWidth={2}
                  >
                    {beforeChartData.map((entry, index) => (
                      <Cell key={`cell-before-${index}`} fill={entry.color} />
                    ))}
                  </Pie>
                  <Tooltip />
                  <Legend />
                </PieChart>
              </ResponsiveContainer>
            </div>
          </div>
          <div>
            <p className="text-sm font-medium text-center mb-2 text-primary">
              整理后 ✨
            </p>
            <div className="h-40">
              <ResponsiveContainer width="100%" height="100%">
                <PieChart>
                  <Pie
                    data={afterChartData}
                    dataKey="value"
                    nameKey="name"
                    cx="50%"
                    cy="50%"
                    outerRadius={55}
                    strokeWidth={2}
                  >
                    {afterChartData.map((entry, index) => (
                      <Cell key={`cell-after-${index}`} fill={entry.color} />
                    ))}
                  </Pie>
                  <Tooltip />
                  <Legend />
                </PieChart>
              </ResponsiveContainer>
            </div>
          </div>
        </div>

        {/* 操作汇总 */}
        {totalImproved > 0 && (
          <div className="p-4 bg-green-50 dark:bg-green-950/30 rounded-lg">
            <p className="text-sm font-medium text-green-800 dark:text-green-200 mb-2">
              🎉 本次整理成果

              {improvements.categorized > 0 && (
                <span>📁 分类了 {improvements.categorized} 条</span>
              )}
              {improvements.cleaned > 0 && (
                <span>🧹 清理了 {improvements.cleaned} 条重复</span>
              )}
              {improvements.tagged > 0 && (
                <span>🏷️ 标记了 {improvements.tagged} 条</span>

              {improvements.archived > 0 && (

/**
 * 指标变化卡片
 */
function MetricCard({
  label,
  before,
  after,
  isLowerBetter,
}: {
  label: string;
  before: number;
  after: number;
  isLowerBetter?: boolean;
}) {
  const change = after - before;
  const isPositive = isLowerBetter ? change < 0 : change > 0;
  const absChange = Math.abs(change);

  return (
    <div className="p-3 bg-muted/50 rounded-lg">
      <p className="text-sm text-muted-foreground mb-1">{label}</p>
      <div className="flex items-center gap-2">
        <span className="text-lg font-medium text-muted-foreground">
          {before}
        </span>
        <ArrowRight className="h-4 w-4 text-muted-foreground" />
        <span className={`text-lg font-bold ${isPositive ? 'text-green-600' : after === before ? '' : 'text-red-600'}`}>
          {after}
        </span>
        {change !== 0 && (
          <span className={`text-sm flex items-center ${isPositive ? 'text-green-600' : 'text-red-600'}`}>
            {isPositive ? (
              <TrendingDown className="h-3 w-3 mr-0.5" />
            ) : (
              <TrendingUp className="h-3 w-3 mr-0.5" />
            )}
            {absChange}
          </span>
        )}
      </div>
    </div>
  );
}
```

```typescript
// src/hooks/domain/use-organize-comparison.ts

import { useState, useCallback } from "react";
import { OrganizeSnapshot } from "@/lib/organize/types";

export function useOrganizeComparison() {
  const [beforeSnapshot, setBeforeSnapshot] = useState<OrganizeSnapshot | null>(null);
  const [afterSnapshot, setAfterSnapshot] = useState<OrganizeSnapshot | null>(null);
  const [isLoading, setIsLoading] = useState(false);

  // 整理前获取快照
  const captureBeforeSnapshot = useCallback(async () => {
    setIsLoading(true);
    try {
      const response = await fetch('/api/collection-stats');
      const data = await response.json();
      setBeforeSnapshot(data);
      return data;
    } finally {
      setIsLoading(false);
    }
  }, []);

  // 整理后获取快照
  const captureAfterSnapshot = useCallback(async () => {
    setIsLoading(true);
    try {
      const response = await fetch('/api/collection-stats');
      const data = await response.json();
      setAfterSnapshot(data);
      return data;
    } finally {
      setIsLoading(false);
    }
  }, []);

  // 重置
  const reset = useCallback(() => {
    setBeforeSnapshot(null);
    setAfterSnapshot(null);
  }, []);

  return {
    beforeSnapshot,
    afterSnapshot,
    isLoading,
    captureBeforeSnapshot,
    captureAfterSnapshot,
    reset,
    hasComparison: beforeSnapshot !== null && afterSnapshot !== null,
  };
}
```

```tsx
// src/components/ai/organize/organize-progress.tsx

interface OrganizeProgressProps {
  status: 'analyzing' | 'executing' | 'complete';
  message?: string;
  steps?: OrganizeStep[];
  currentStep?: number;
  progress?: number;
}

interface OrganizeStep {
  id: string;
  label: string;
  status: 'pending' | 'running' | 'complete';
}

const DEFAULT_STEPS: OrganizeStep[] = [
  { id: 'scan', label: '扫描收藏库', status: 'pending' },
  { id: 'analyze', label: '分析问题', status: 'pending' },
  { id: 'plan', label: '生成方案', status: 'pending' },
  { id: 'execute', label: '执行整理', status: 'pending' },
  { id: 'verify', label: '验证结果', status: 'pending' },
];

export function OrganizeProgress({
  status,
  message,
  steps = DEFAULT_STEPS,
  currentStep = 0,
  progress = 0,
}: OrganizeProgressProps) {
  const [displaySteps, setDisplaySteps] = useState(steps);

  // 根据状态更新步骤
  useEffect(() => {
    const updated = steps.map((step, index) => ({
      ...step,
      status:
        index < currentStep ? 'complete' :
        index === currentStep ? 'running' :
        'pending',
    }));
    setDisplaySteps(updated as OrganizeStep[]);
  }, [steps, currentStep]);

  return (
    <Card className="border-primary/30">
      <CardContent className="p-6">
        <div className="space-y-6">
          {/* 主进度条 */}
          <div className="space-y-2">
            <div className="flex items-center justify-between text-sm">
              <span className="text-muted-foreground">
                {message || '正在整理...'}
              </span>
              <span className="font-medium">{progress}%</span>
            </div>
            <Progress value={progress} className="h-2" />
          </div>

          {/* 步骤列表 */}
          <div className="space-y-3">
            {displaySteps.map((step, index) => (
              <div
                key={step.id}
                className={cn(
                  "flex items-center gap-3 p-2 rounded-md transition-colors",
                  step.status === 'running' && "bg-primary/5",
                  step.status === 'complete' && "text-muted-foreground"
                )}
              >
                {step.status === 'complete' && (
                  <CheckCircle className="h-5 w-5 text-green-500" />
                )}
                {step.status === 'running' && (
                  <Loader2 className="h-5 w-5 text-primary animate-spin" />
                )}
                {step.status === 'pending' && (
                  <Circle className="h-5 w-5 text-muted-foreground/50" />
                )}
                <span className={cn(
                  "text-sm",
                  step.status === 'running' && "font-medium text-primary"
                )}>
                  {step.label}
                </span>
              </div>
            ))}
          </div>
        </div>
      </CardContent>
    </Card>
  );
}
```
