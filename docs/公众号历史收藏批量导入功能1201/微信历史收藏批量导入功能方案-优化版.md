## 一、问题分析

### 1.1 原方案存在的问题

❌ **用户操作门槛高**：

- 需要下载并安装 PyWxDump 工具
- 需要了解命令行操作
- 需要手动导出 JSON 文件再上传
- **预计流失率 > 80%**

❌ **技术术语复杂**：

- 用户不理解"解密数据库"、"JSON 导出"等概念
- 操作步骤多达 4-5 步

### 1.2 其他方案评估

经过深入调研，我发现以下替代方案都存在问题：

| 方案                    | 可行性          | 问题                         |
| ----------------------- | --------------- | ---------------------------- |
| 微信官方 API            | ❌ 不可行       | 微信未提供收藏导出 API       |
| 逐条手动分享            | ❌ 不符合需求   | 无法批量导入历史收藏         |
| 自动化脚本（模拟点击）  | ❌ 不推荐       | 速度慢、不稳定、只能获取链接 |
| 微信 Hook（DLL 注入）   | ❌ 不推荐       | 法律风险高、可能被封号       |
| 网页版 + 上传数据库文件 | ❌ 不推荐       | 隐私和安全问题               |
| **官方一键工具**        | ✅ **最佳方案** | 用户体验最佳                 |

## 二、优化方案设计

### 2.1 核心思路

**将技术复杂度隐藏在工具内部，让用户只需"下载 → 运行 → 等待"三步完成**

```
原方案（4-5步）：                  优化方案（3步）：
下载 PyWxDump                      下载 AnyKeep 导入助手
↓                                  ↓
运行工具，选择导出收藏              双击运行，登录 AnyKeep 账号
↓                                  ↓
等待导出完成                        点击"一键导入"按钮
↓                                  ↓
登录 AnyKeep，上传 JSON 文件        自动完成（无需用户操作）
↓
等待导入完成
```

### 2.2 两阶段实施策略

#### 🚀 **阶段 1：快速 MVP（推荐优先实施）**

**目标**：用最小成本验证需求，快速上线

**方案**：

- 将 PyWxDump 核心功能打包成**独立的 .exe 文件**
- 集成自动上传功能，直接连接 AnyKeep API
- 用户只需：下载 → 双击 → 输入 AnyKeep 账号 → 等待完成

**用户流程**：

```
1. 访问 AnyKeep 官网

3. 双击运行
4. 输入 AnyKeep 账号密码（或扫码登录）
5. 点击"开始导入"按钮
6. 等待完成（自动检测微信 → 解密 → 提取 → 上传）
```

**技术实现**：

- 使用 PyInstaller 将 Python 脚本打包成 .exe
- 集成 PyWxDump 的核心代码
- 添加简单的控制台界面（进度条）
- 直接调用 AnyKeep API 上传数据

**开发周期**：1-2 周

### 3.2 核心代码结构

**主程序**: `anykeep_import_tool.py`

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
AnyKeep 收藏导入助手
一键导入微信历史收藏到 AnyKeep
"""

import sys
import json
import requests
from pywxdump import get_wechat_db_key, decrypt_database, extract_favorites

# 配置
ANYKEEP_API = "https://anykeep.com/api/wechat-favorite/batch-import"


def main():
    print("=" * 60)
    print("      AnyKeep 收藏导入助手 v1.0")
    print("=" * 60)
    print()

    # 步骤 1: 检测微信
    print("[1/6] 检测微信客户端...")
    if not check_wechat_running():
        print("❌ 错误：请先启动 PC 微信客户端")
        sys.exit(1)
    print("✅ 微信检测成功")
    print()

    # 步骤 2: 获取数据库密钥
    print("[2/6] 获取数据库密钥...")
    try:
        db_key = get_wechat_db_key()
        print(f"✅ 密钥获取成功")
    except Exception as e:
        print(f"❌ 密钥获取失败：{e}")
        sys.exit(1)
    print()

    # 步骤 3: 解密数据库
    print("[3/6] 解密收藏数据库...")
    try:
        db_path = find_favorite_db()
        decrypt_database(db_path, db_key, "./decrypted_favorite.db")
        print("✅ 数据库解密成功")
    except Exception as e:
        print(f"❌ 解密失败：{e}")
        sys.exit(1)
    print()

    # 步骤 4: 提取收藏
    print("[4/6] 提取收藏内容...")
    try:
        favorites = extract_favorites("./decrypted_favorite.db")
        print(f"✅ 提取成功，共 {len(favorites)} 条收藏")
    except Exception as e:
        print(f"❌ 提取失败：{e}")
        sys.exit(1)
    print()

    # 步骤 5: 登录 AnyKeep
    print("[5/6] 连接 AnyKeep...")
    print("请输入 AnyKeep 账号信息：")
    email = input("  邮箱: ")
    password = input("  密码: ")

    try:
        token = login_anykeep(email, password)
        print("✅ 登录成功")
    except Exception as e:
        print(f"❌ 登录失败：{e}")
        sys.exit(1)
    print()

    # 步骤 6: 上传到 AnyKeep
    print("[6/6] 上传收藏到 AnyKeep...")
    try:
        result = upload_to_anykeep(token, favorites)
        print("✅ 导入完成！")
        print()
        print("=" * 60)
        print(f"  ✅ 成功: {result['succeeded']} 篇")
        print(f"  ❌ 失败: {result['failed']} 篇")
        print(f"  🔄 重复: {result['duplicates']} 篇")
        print("=" * 60)
    except Exception as e:
        print(f"❌ 上传失败：{e}")
        sys.exit(1)

    print()
    input("按 Enter 键退出...")


def check_wechat_running():
    """检测微信是否运行"""
    import psutil
    for proc in psutil.process_iter(['name']):
        if 'wechat' in proc.info['name'].lower():
            return True
    return False


def find_favorite_db():
    """查找 Favorite.db 路径"""
    import os
    import glob

    user_dir = os.path.expanduser("~")
    pattern = os.path.join(
        user_dir,
        "Documents/WeChat Files/*/Msg/Favorite.db"
    )

    results = glob.glob(pattern)
    if not results:
        raise FileNotFoundError("未找到微信收藏数据库")

    return results[0]


def login_anykeep(email, password):
    """登录 AnyKeep 并获取 token"""
    response = requests.post(
        "https://anykeep.com/api/auth/login",
        json={"email": email, "password": password}
    )

    if response.status_code != 200:
        raise Exception("登录失败，请检查账号密码")

    return response.json()['token']


def upload_to_anykeep(token, favorites):
    """上传收藏到 AnyKeep"""
    response = requests.post(
        ANYKEEP_API,
        headers={"Authorization": f"Bearer {token}"},
        json={"favorites": favorites}
    )

    if response.status_code != 200:
        raise Exception("上传失败")

    return response.json()['data']


if __name__ == "__main__":
    main()
```

### 3.5 API 端点设计

**位置**: `src/app/api/wechat-favorite/batch-import-direct/route.ts`

```typescript
import { NextRequest, NextResponse } from "next/server";
import { createServerClient } from "@/lib/supabase/server";

/**
 * 接收工具直接上传的收藏数据
 * POST /api/wechat-favorite/batch-import-direct
 */
export async function POST(request: NextRequest) {
  try {
    // 1. 验证 Authorization token
    const authHeader = request.headers.get("Authorization");
    if (!authHeader?.startsWith("Bearer ")) {
      return NextResponse.json({ error: "未授权" }, { status: 401 });
    }

    const token = authHeader.substring(7);

    // 2. 验证 token 并获取用户
    const supabase = createServerClient();
    const {
      data: { user },
      error: authError,
    } = await supabase.auth.getUser(token);

    if (authError || !user) {
      return NextResponse.json({ error: "无效的 token" }, { status: 401 });
    }

    // 3. 获取收藏数据
    const { favorites } = await request.json();

    if (!favorites || !Array.isArray(favorites)) {
      return NextResponse.json({ error: "数据格式错误" }, { status: 400 });
    }

    // 4. 批量导入（复用现有逻辑）
    const importer = new WechatFavoriteBatchImporter();
    const result = await importer.importFavorites(favorites, user.id);

    return NextResponse.json({
      success: true,
      data: result,
    });
  } catch (error: any) {
    console.error("批量导入失败", error);
    return NextResponse.json(
      { success: false, error: error.message },
      { status: 500 },
    );
  }
}
```
