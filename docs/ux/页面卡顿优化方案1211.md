```typescript
  if (folder.parent_id) return await buildFolderPath(folder.parent_id);
};

// ✅ 优化：一次性查询所有文件夹，客户端构建路径
const { data: allFolders } = await supabase
  .from("folders")
  .select("id, name, parent_id")
  .eq("user_id", user.id);

// 内存中构建路径树
function buildFolderPathFromCache(folderId: number, folders: Folder[]): Folder[] {
  const path: Folder[] = [];
  let currentId: number | null = folderId;

  while (currentId) {
    const folder = folders.find(f => f.id === currentId);
    if (!folder) break;
    path.unshift(folder);
    currentId = folder.parent_id;
  }

  return path;
}
```

**预期收益**：

- 查询次数：N+1 → 1
- 响应时间：300-800ms → 100-200ms

#### 3.1 NavigationContext 拆分

**文件**：`src/contexts/navigation-context.tsx`

```typescript

  new URLSearchParams(),
```

**说明**：

- 将 NavigationContext 拆分为 3 个独立的 Context
- 组件只订阅需要的状态，减少不必要的重渲染
- 风险较高，建议分步实施

#### 3.2 使用 useSyncExternalStore

对于高频更新的状态（如滚动位置），使用 `useSyncExternalStore` 避免触发 React 重渲染：

```typescript
// 新建 src/stores/scroll-store.ts
let scrollPosition = 0;
const listeners = new Set<() => void>();

export const scrollStore = {
  getSnapshot: () => scrollPosition,
  subscribe: (listener: () => void) => {
    listeners.add(listener);
    return () => listeners.delete(listener);
  },
  setScrollPosition: (pos: number) => {
    scrollPosition = pos;
    listeners.forEach((l) => l());
  },
};

// 使用
function useScrollPosition() {
  return useSyncExternalStore(scrollStore.subscribe, scrollStore.getSnapshot);
}
```

**优化方案**：创建共享的 viewport hook

```typescript
// 新建 src/hooks/use-viewport.ts
import { useSyncExternalStore } from "react";

let isMobile = typeof window !== "undefined" && window.innerWidth < 768;
const listeners = new Set<() => void>();

if (typeof window !== "undefined") {
  window.addEventListener("resize", () => {
    const newIsMobile = window.innerWidth < 768;
    if (newIsMobile !== isMobile) {
      isMobile = newIsMobile;
      listeners.forEach((l) => l());
    }
  });
}

export function useIsMobile() {
  return useSyncExternalStore(
    (callback) => {
      listeners.add(callback);
      return () => listeners.delete(callback);
    },
    () => isMobile,
    () => false, // SSR fallback
  );
}
```

```typescript
  // ...
```

## 四、注意事项

### 4.1 缓存策略变更风险

- ISR 可能导致数据短暂不一致（最长 60 秒）
- 解决方案：关键操作后调用 `revalidatePath()` 强制刷新
- 用户侧可通过下拉刷新获取最新数据

### 4.2 memo 使用注意

- 确保传递给 memo 组件的 props 引用稳定
- 对象/数组 props 使用 useMemo 包装
- 函数 props 使用 useCallback 包装

```typescript
// ✅ 正确
const handleEdit = useCallback((id: number) => {
  // ...
}, []);

<BookmarkCard onEdit={handleEdit} />

// ❌ 错误：每次渲染创建新函数，memo 失效
<BookmarkCard onEdit={(id) => { /* ... */ }} />
```

### 4.3 Context 拆分风险

- 涉及较大重构，影响多个组件
- 建议分步实施：先拆分，再迁移
- 保留旧 Context 作为兼容层
