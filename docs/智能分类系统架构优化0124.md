```typescript
// src/lib/classification/facade.ts
import { StrategyResolver } from "./strategy-resolver";
import type { ClassificationRequest, ClassificationResult } from "./core/types";

export class ClassificationFacade {
  private resolver: StrategyResolver;

  constructor(config?: ClassificationFacadeConfig) {
    this.resolver = new StrategyResolver(config);
  }

  /**
   * 统一分类入口
   * 替代原 classifyBookmarkV2()
   */
  async classify(
    request: ClassificationRequest,
  ): Promise<ClassificationResult> {
    const strategy = await this.resolver.resolve(request.userId);
    return strategy.classify(request);
  }

  /**
   * 批量分类
   */
  async classifyBatch(
    requests: ClassificationRequest[],
  ): Promise<ClassificationResult[]> {
    // 按用户分组，每组使用相同策略
    const grouped = this.groupByUser(requests);
    const results: ClassificationResult[] = [];

    for (const [userId, userRequests] of grouped) {
      const strategy = await this.resolver.resolve(userId);
      const batchResults = await strategy.classifyBatch(userRequests);
      results.push(...batchResults);
    }

    return results;
  }

  /**
   * 仅 T0（用于 Optimistic UI）
   */
  async classifyQuick(
    request: ClassificationRequest,
  ): Promise<ClassificationResult> {
    const strategy = await this.resolver.resolve(request.userId);
    if ("classifyT0Only" in strategy) {
      return strategy.classifyT0Only(request);
    }
    return strategy.classify(request);
  }
}

// 单例便捷函数
let facadeInstance: ClassificationFacade | null = null;

export function getClassificationFacade(): ClassificationFacade {
  if (!facadeInstance) {
    facadeInstance = new ClassificationFacade();
  }
  return facadeInstance;
}
```

#### 2.4 策略解析器实现

```typescript
// src/lib/classification/strategy-resolver.ts
import { FeatureFlagService } from "./infrastructure/feature-flags";
import { ClassicStrategy } from "./strategies/classic";
import { ModernStrategy } from "./strategies/modern";
import type { IClassificationStrategy } from "./core/interfaces";

export class StrategyResolver {
  private featureFlags: FeatureFlagService;
  private classicStrategy: ClassicStrategy | null = null;
  private modernStrategy: ModernStrategy | null = null;

  constructor(config?: StrategyResolverConfig) {
    this.featureFlags = new FeatureFlagService(config?.featureFlags);
  }

  async resolve(userId: string): Promise<IClassificationStrategy> {
    // 1. 检查环境变量强制指定
    const envVersion = process.env.CLASSIFICATION_VERSION;
    if (envVersion === "v4.1") {
      return this.getClassicStrategy();
    }

    // 2. 检查 V5 灰度发布
    if (!this.featureFlags.isEnabledForUser(userId)) {
      return this.getClassicStrategy();
    }

    // 3. 默认使用 Modern 策略
    return this.getModernStrategy();
  }

  private getClassicStrategy(): ClassicStrategy {
    if (!this.classicStrategy) {
      this.classicStrategy = new ClassicStrategy();
    }
    return this.classicStrategy;
  }

  private getModernStrategy(): ModernStrategy {
    if (!this.modernStrategy) {
      this.modernStrategy = new ModernStrategy();
    }
    return this.modernStrategy;
  }
}
```

```typescript
// src/lib/classification/infrastructure/feature-flags.ts

// 合并原有两套 Feature Flags
export interface ClassificationFeatureFlags {
  // === 策略选择 ===
  defaultStrategy: "classic" | "modern";

  // === 灰度发布 ===
  rolloutStage: RolloutStage;
  userWhitelist: string[];
  userBlacklist: string[];

  // === Modern 策略配置 ===
  enableL0Rules: boolean; // L0 规则匹配
  enableT1Correction: boolean; // T1 LLM 纠正
  enableCrystalGrowth: boolean; // 晶体生长（原 V5.5）

  // === Classic 策略配置 ===
  enableAISelection: boolean; // AI 选择题模式
  enableAICreate: boolean; // AI 创建文件夹

  // === 置信度阈值 ===
  autoClassifyThreshold: number; // 默认 80
  suggestThreshold: number; // 默认 60
  t1TriggerThreshold: number; // 默认 70

  // === 性能 ===
  enablePipelineCache: boolean;
  pipelineCacheTtlMs: number;
}

export enum RolloutStage {
  DISABLED = 0,
  CANARY = 5,
  EARLY = 20,
  MAJORITY = 50,
  FULL = 100,
}

export class FeatureFlagService {
  private config: ClassificationFeatureFlags;

  constructor(overrides?: Partial<ClassificationFeatureFlags>) {
    this.config = {
      // 从环境变量读取
      defaultStrategy: this.readEnv(
        "CLASSIFICATION_DEFAULT_STRATEGY",
        "modern",
      ),
      rolloutStage: this.readEnvNumber("CLASSIFICATION_ROLLOUT_STAGE", 100),
      userWhitelist: this.readEnvArray("CLASSIFICATION_WHITELIST"),
      userBlacklist: this.readEnvArray("CLASSIFICATION_BLACKLIST"),

      enableL0Rules: this.readEnvBool("CLASSIFICATION_ENABLE_L0_RULES", true),
      enableT1Correction: this.readEnvBool("CLASSIFICATION_ENABLE_T1", true),
      enableCrystalGrowth: this.readEnvBool(
        "CLASSIFICATION_ENABLE_CRYSTAL",
        false,
      ),

      enableAISelection: this.readEnvBool(
        "CLASSIFICATION_ENABLE_AI_SELECT",
        true,
      ),
      enableAICreate: this.readEnvBool("CLASSIFICATION_ENABLE_AI_CREATE", true),

      autoClassifyThreshold: this.readEnvNumber(
        "CLASSIFICATION_AUTO_THRESHOLD",
        80,
      ),
      suggestThreshold: this.readEnvNumber(
        "CLASSIFICATION_SUGGEST_THRESHOLD",
        60,
      ),
      t1TriggerThreshold: this.readEnvNumber("CLASSIFICATION_T1_THRESHOLD", 70),

      enablePipelineCache: this.readEnvBool(
        "CLASSIFICATION_ENABLE_CACHE",
        true,
      ),
      pipelineCacheTtlMs: this.readEnvNumber(
        "CLASSIFICATION_CACHE_TTL_MS",
        600000,
      ),

      ...overrides,
    };
  }

  isEnabledForUser(userId: string): boolean {
    // 黑名单优先
    if (this.config.userBlacklist.includes(userId)) {
      return false;
    }
    // 白名单直接通过
    if (this.config.userWhitelist.includes(userId)) {
      return true;
    }
    // 按比例灰度
    const bucket = this.getUserBucket(userId);
    return bucket < this.config.rolloutStage;
  }

  private getUserBucket(userId: string): number {
    // MD5 hash → 0-99
    const hash = crypto.createHash("md5").update(userId).digest("hex");
    return parseInt(hash.slice(0, 8), 16) % 100;
  }

  // ... 其他辅助方法
}
```

```typescript
export function createContainer(
  supabase: SupabaseClient,
  overrides?: Partial<ContainerConfig>,
): Container {
  const container = new Container();

  // 基础设施
  container.bind(TYPES.SupabaseClient).toConstantValue(supabase);
  container.bind(TYPES.Logger).to(LoggerService).inSingletonScope();

  // 共享服务
  container
    .bind(TYPES.EmbeddingService)
    .to(DashScopeEmbeddingAdapter)
    .inSingletonScope();
  container.bind(TYPES.LLMService).to(LLMService).inSingletonScope();
  container
    .bind(TYPES.VectorSearchService)
    .to(VectorSearchService)
    .inSingletonScope();

  // 分类服务
  container
    .bind(TYPES.ClassificationFacade)
    .to(ClassificationFacade)
    .inSingletonScope();
  container
    .bind(TYPES.StrategyResolver)
    .to(StrategyResolver)
    .inSingletonScope();
  container.bind(TYPES.ClassicStrategy).to(ClassicStrategy).inSingletonScope();
  container.bind(TYPES.ModernStrategy).to(ModernStrategy).inSingletonScope();

  // 允许覆盖（测试用）
  if (overrides) {
    for (const [key, impl] of Object.entries(overrides)) {
      container.rebind(TYPES[key]).toConstantValue(impl);
    }
  }

  return container;
}

// 全局容器
let globalContainer: Container | null = null;

export function getContainer(): Container {
  if (!globalContainer) {
    throw new Error(
      "Container not initialized. Call initializeContainer() first.",
    );
  }
  return globalContainer;
}

export function initializeContainer(supabase: SupabaseClient): Container {
  globalContainer = createContainer(supabase);
  return globalContainer;
}

// 测试辅助
export function resetContainer(): void {
  globalContainer = null;
}
```

#### 5.2 集成方式

```typescript
// strategies/modern/pipeline.service.ts
export class ModernPipelineService implements IClassificationStrategy {
  private crystalGrowth?: CrystalGrowthService;

  constructor(
    private l0: L0RuleMatcher,
    private t0: T0Generator,
    private t1: T1Corrector,
    private featureFlags: FeatureFlagService,
  ) {
    // 按需初始化晶体生长
    if (this.featureFlags.get("enableCrystalGrowth")) {
      this.crystalGrowth = new CrystalGrowthService();
    }
  }

  async classify(
    request: ClassificationRequest,
  ): Promise<ClassificationResult> {
    // L0 规则匹配
    const l0Result = await this.l0.match(request);
    if (l0Result.matched) {
      return l0Result.result;
    }

    // T0 快速生成
    const t0Result = await this.t0.generate(request);

    // 决定是否需要 T1
    if (t0Result.confidence >= this.featureFlags.get("t1TriggerThreshold")) {
      return t0Result;
    }

    // T1 纠正
    const t1Result = await this.t1.correct(request, t0Result);

    // 晶体生长增强（如果启用）
    if (this.crystalGrowth && t1Result.confidence < 60) {
      return this.crystalGrowth.enhance(request, t1Result);
    }

    return t1Result;
  }
}
```
