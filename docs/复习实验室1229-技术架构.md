### 1.2 技术决策摘要

| 决策点     | 选择                       | 理由                                          |
| ---------- | -------------------------- | --------------------------------------------- |
| Skill 架构 | 新增 `ActiveRecallSkill`   | 增强现有 `KnowledgeReviewSkill`，共享算法模块 |
| 数据存储   | 新增 `flashcards` 表       | 现有 `review_records` 不支持问答对存储        |
| API 设计   | RESTful + 复用统一入口     | 专用端点处理 CRUD，统一入口处理 AI 对话       |
| 通知机制   | 复用 `NotificationService` | 现有服务完全满足需求                          |
| 调度系统   | Edge Functions + Cron      | 无服务器，成本低，Supabase 原生支持           |
| 触发机制   | 标签变更事件               | 通过 API 中间件拦截，无需 Database Trigger    |

## 三、数据库架构设计

### 3.1 新增表结构

#### 表 1: flashcards（核心）

```sql
-- 文件: supabase/migrations/2025XXXX_001_create_flashcards_table.sql

CREATE TABLE flashcards (
  id BIGSERIAL PRIMARY KEY,
  user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
  bookmark_id BIGINT NOT NULL REFERENCES bookmarks(id) ON DELETE CASCADE,

  -- 问答内容
  question TEXT NOT NULL,
  answer TEXT NOT NULL,
  source_paragraph TEXT,  -- 原文核心段落

  -- SM-2 算法参数
  status VARCHAR(20) DEFAULT 'pending'
    CHECK (status IN ('pending', 'learning', 'mastered')),
  ease_factor DECIMAL(4,2) DEFAULT 2.50
    CHECK (ease_factor >= 1.30 AND ease_factor <= 3.00),
  interval_days INT DEFAULT 1
    CHECK (interval_days >= 1 AND interval_days <= 365),
  repetitions INT DEFAULT 0
    CHECK (repetitions >= 0),

  -- 调度时间
  next_review_at TIMESTAMPTZ,
  last_reviewed_at TIMESTAMPTZ,

  -- 元数据
  difficulty VARCHAR(10) DEFAULT 'medium'
    CHECK (difficulty IN ('easy', 'medium', 'hard')),
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now(),

  -- 唯一约束：同一书签不重复问题
  CONSTRAINT flashcards_unique_question
    UNIQUE (user_id, bookmark_id, question)
);

-- 性能索引
CREATE INDEX idx_flashcards_user_status
  ON flashcards(user_id, status)
  WHERE status != 'mastered';

CREATE INDEX idx_flashcards_next_review
  ON flashcards(user_id, next_review_at)
  WHERE status != 'mastered' AND next_review_at IS NOT NULL;

CREATE INDEX idx_flashcards_bookmark
  ON flashcards(bookmark_id);

-- 更新时间触发器
CREATE TRIGGER update_flashcards_updated_at
  BEFORE UPDATE ON flashcards
  FOR EACH ROW
  EXECUTE FUNCTION update_updated_at_column();
```

### 3.3 数据库函数

```sql
-- 文件: supabase/migrations/2025XXXX_006_create_review_functions.sql

-- 函数 1: 获取待复习的 Flashcards（优先级排序）
CREATE OR REPLACE FUNCTION get_pending_flashcards(
  p_user_id UUID,
  p_limit INT DEFAULT 10,
  p_included_folders BIGINT[] DEFAULT NULL,
  p_excluded_folders BIGINT[] DEFAULT NULL
)
RETURNS TABLE (
  flashcard_id BIGINT,
  bookmark_id BIGINT,
  bookmark_title TEXT,
  bookmark_url TEXT,
  question TEXT,
  answer TEXT,
  source_paragraph TEXT,
  status VARCHAR(20),
  next_review_at TIMESTAMPTZ,
  days_overdue INT,
  priority_score DECIMAL(5,2)
)
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
BEGIN
  RETURN QUERY
  SELECT
    f.id AS flashcard_id,
    f.bookmark_id,
    b.title AS bookmark_title,
    b.url AS bookmark_url,
    f.question,
    f.answer,
    f.source_paragraph,
    f.status,
    f.next_review_at,
    EXTRACT(DAY FROM (now() - f.next_review_at))::INT AS days_overdue,
    -- 优先级计算: 逾期天数权重 + 难度权重 + 间隔权重
    (
      COALESCE(EXTRACT(DAY FROM (now() - f.next_review_at)), 0) * 0.4 +
      (3.0 - f.ease_factor) * 10 * 0.3 +
      (1.0 / NULLIF(f.interval_days, 0)) * 100 * 0.3
    )::DECIMAL(5,2) AS priority_score
  FROM flashcards f
  JOIN bookmarks b ON f.bookmark_id = b.id
  WHERE f.user_id = p_user_id
    AND f.status != 'mastered'
    AND (f.next_review_at IS NULL OR f.next_review_at <= now() + INTERVAL '3 days')
    -- 文件夹筛选
    AND (p_included_folders IS NULL OR b.folder_id = ANY(p_included_folders))
    AND (p_excluded_folders IS NULL OR b.folder_id != ALL(p_excluded_folders))
  ORDER BY priority_score DESC
  LIMIT p_limit;
END;
$$;

-- 函数 2: 更新 Flashcard 状态（SM-2 算法结果）
CREATE OR REPLACE FUNCTION update_flashcard_review(
  p_flashcard_id BIGINT,
  p_user_id UUID,
  p_response VARCHAR(20),
  p_new_interval INT,
  p_new_ease DECIMAL(4,2),
  p_new_repetitions INT,
  p_response_time_ms INT DEFAULT NULL
)
RETURNS JSON
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
  v_old_interval INT;
  v_old_ease DECIMAL(4,2);
  v_new_status VARCHAR(20);
  v_next_review TIMESTAMPTZ;
BEGIN
  -- 获取当前值
  SELECT interval_days, ease_factor INTO v_old_interval, v_old_ease
  FROM flashcards WHERE id = p_flashcard_id AND user_id = p_user_id;

  IF NOT FOUND THEN
    RETURN json_build_object('success', false, 'error', 'Flashcard not found');
  END IF;

  -- 计算新状态
  v_new_status := CASE
    WHEN p_response = 'mastered' AND p_new_repetitions >= 3 THEN 'mastered'
    ELSE 'learning'
  END;

  -- 计算下次复习时间
  v_next_review := CASE
    WHEN v_new_status = 'mastered' THEN NULL
    ELSE now() + (p_new_interval || ' days')::INTERVAL
  END;

  -- 更新 Flashcard
  UPDATE flashcards SET
    status = v_new_status,
    ease_factor = p_new_ease,
    interval_days = p_new_interval,
    repetitions = p_new_repetitions,
    next_review_at = v_next_review,
    last_reviewed_at = now(),
    updated_at = now()
  WHERE id = p_flashcard_id AND user_id = p_user_id;

  -- 记录复习会话
  INSERT INTO review_sessions (
    user_id, flashcard_id, response, response_time_ms,
    prev_interval, new_interval, prev_ease, new_ease
  ) VALUES (
    p_user_id, p_flashcard_id, p_response, p_response_time_ms,
    v_old_interval, p_new_interval, v_old_ease, p_new_ease
  );

  RETURN json_build_object(
    'success', true,
    'new_status', v_new_status,
    'next_review_at', v_next_review
  );
END;
$$;

CREATE OR REPLACE FUNCTION get_review_stats(
  p_user_id UUID,
  p_time_range VARCHAR(10) DEFAULT 'month'
)
RETURNS JSON
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
  v_start_date TIMESTAMPTZ;
  v_total_flashcards INT;
  v_mastered_count INT;
  v_learning_count INT;
  v_pending_count INT;
  v_sessions_count INT;
  v_mastered_rate DECIMAL(5,2);
  v_avg_response_time INT;
BEGIN
  -- 计算时间范围
  v_start_date := CASE p_time_range
    WHEN 'week' THEN now() - INTERVAL '7 days'
    WHEN 'month' THEN now() - INTERVAL '30 days'
    ELSE '1970-01-01'::TIMESTAMPTZ
  END;

  -- Flashcard 统计
  SELECT
    COUNT(*),
    COUNT(*) FILTER (WHERE status = 'mastered'),
    COUNT(*) FILTER (WHERE status = 'learning'),
    COUNT(*) FILTER (WHERE status = 'pending')
  INTO v_total_flashcards, v_mastered_count, v_learning_count, v_pending_count
  FROM flashcards
  WHERE user_id = p_user_id;

  -- 复习会话统计
  SELECT
    COUNT(*),
    AVG(response_time_ms)::INT
  INTO v_sessions_count, v_avg_response_time
  FROM review_sessions
  WHERE user_id = p_user_id AND created_at >= v_start_date;

  -- 计算掌握率
  v_mastered_rate := CASE
    WHEN v_total_flashcards > 0
    THEN (v_mastered_count::DECIMAL / v_total_flashcards * 100)
    ELSE 0
  END;

  RETURN json_build_object(
    'total_flashcards', v_total_flashcards,
    'mastered_count', v_mastered_count,
    'learning_count', v_learning_count,
    'pending_count', v_pending_count,
    'mastered_rate', v_mastered_rate,
    'sessions_in_period', v_sessions_count,
    'avg_response_time_ms', v_avg_response_time,
    'time_range', p_time_range
  );
END;
$$;
```

```typescript
export async function GET(request: NextRequest) {
  // 速率限制
  const rateLimitResult = await rateLimiter.check(request, "brain-sync-queue");
  if (!rateLimitResult.success) {
    return NextResponse.json({ error: "Too many requests" }, { status: 429 });
  }

  const supabase = await createServerClient();

  // 获取当前用户
  const {
    data: { user },
    error: authError,
  } = await supabase.auth.getUser();

  const { data: preferences } = await supabase
    .from("review_preferences")
    .select("*")
    .eq("user_id", user.id)
    .single();

  if (!preferences?.enabled) {
    return NextResponse.json({
      queue: [],
      stats: { message: "主动复习功能未启用" },
    });
  }

  // 调用数据库函数获取队列
  const { data: queue, error } = await supabase.rpc("get_pending_flashcards", {

    p_excluded_folders: preferences.excluded_folder_ids,
```

### 5.2 SM-2 算法实现

```typescript
// 文件: src/lib/ai-skills/shared/review/sm2-algorithm.ts

/**
 * SuperMemo SM-2 间隔重复算法
 *
 * 核心公式:
 * - EF' = EF + (0.1 - (5-q) * (0.08 + (5-q) * 0.02))
 * - I(1) = 1, I(2) = 3, I(n) = I(n-1) * EF  (for n > 2, q >= 3)
 *
 * 其中:
 * - EF: Easiness Factor (难度因子), 范围 1.3-2.5
 * - q: Quality of response (回忆质量), 范围 0-5
 * - I(n): 第 n 次复习的间隔天数
 */

export interface SM2Input {
  quality: number; // 回忆质量 0-5
  repetitions: number; // 当前连续正确次数
  easeFactor: number; // 当前难度因子
  interval: number; // 当前间隔（天）
}

export interface SM2Output {
  interval: number; // 新间隔（天）
  repetitions: number; // 新连续正确次数
  easeFactor: number; // 新难度因子
}

export class SM2Algorithm {
  private static readonly MIN_EASE_FACTOR = 1.3;
  private static readonly MAX_EASE_FACTOR = 2.5;
  private static readonly MAX_INTERVAL = 90; // AnyKeep 限制最大90天

  /**
   * 计算下次复习参数
   */
  static calculate(input: SM2Input): SM2Output {
    const { quality, repetitions, easeFactor, interval } = input;

    // 验证输入
    if (quality < 0 || quality > 5) {
      throw new Error(`Invalid quality: ${quality}. Must be 0-5.`);
    }

    // 质量 < 3 表示遗忘，重置复习进度
    if (quality < 3) {
      return {
        interval: 1,
        repetitions: 0,
        easeFactor: Math.max(this.MIN_EASE_FACTOR, easeFactor - 0.2),
      };
    }

    // 计算新的难度因子
    const efDelta = 0.1 - (5 - quality) * (0.08 + (5 - quality) * 0.02);
    const newEaseFactor = Math.max(
      this.MIN_EASE_FACTOR,
      Math.min(this.MAX_EASE_FACTOR, easeFactor + efDelta),
    );

    // 计算新间隔
    let newInterval: number;
    if (repetitions === 0) {
      newInterval = 1; // 第一次：1天
    } else if (repetitions === 1) {
      newInterval = 3; // 第二次：3天
    } else if (repetitions === 2) {
      newInterval = 7; // 第三次：7天（符合1-3-7）
    } else {
      newInterval = Math.round(interval * newEaseFactor);
    }

    // 限制最大间隔
    newInterval = Math.min(newInterval, this.MAX_INTERVAL);

    return {
      interval: newInterval,
      repetitions: repetitions + 1,
      easeFactor: newEaseFactor,
    };
  }

  /**
   * 将用户反馈转换为 SM-2 质量分数
   */
  static responseToQuality(
    response: "mastered" | "fuzzy" | "forgotten",
  ): number {
    const mapping = {
      mastered: 5, // 完美回忆
      fuzzy: 3, // 勉强回忆
      forgotten: 0, // 完全遗忘
    };
    return mapping[response];
  }

  /**
   * 预测下次复习日期
   */
  static predictNextReview(interval: number): Date {
    const next = new Date();
    next.setDate(next.getDate() + interval);
    return next;
  }
}
```

### 5.3 Flashcard 生成器

```typescript
// 文件: src/lib/ai-skills/shared/review/flashcard-generator.ts

import { generateText } from "ai";
import { selectModel } from "@/lib/ai-agent/config/model-selector";
import { logger } from "@/lib/logger";

export interface FlashcardGenerationInput {
  title: string;
  summary?: string;
  content: string;
  count: number;
  difficulty: "easy" | "medium" | "hard";
}

export interface GeneratedFlashcard {
  question: string;
  answer: string;
  sourceParagraph: string;
}

const GENERATION_PROMPT = `
你是一位教育学专家，擅长将复杂知识转化为可记忆的知识点。

根据以下内容，生成 {count} 个高质量的复习问答对（Flashcards）。

## 要求
1. 问题必须是开放式的（如"为什么..."、"如何..."、"什么是..."）
2. 避免是非题或选择题
3. 答案简洁但完整，通常 2-4 句话
4. 每个问题聚焦一个核心知识点
5. 难度等级：{difficulty}
6. 提取与问题相关的原文片段作为 sourceParagraph（不超过100字）

## 内容
标题：{title}
摘要：{summary}
原文：{content}

## 输出格式（严格 JSON 数组）
\`\`\`json
[
  {
    "question": "问题内容",
    "answer": "参考答案",
    "sourceParagraph": "原文相关片段"
  }
]
\`\`\`

请只输出 JSON 数组，不要有其他内容。
`;

export async function generateFlashcards(
  input: FlashcardGenerationInput,
): Promise<GeneratedFlashcard[]> {
  const { title, summary, content, count, difficulty } = input;

  // 选择模型（Flashcard 生成使用 standard 层级）
  const model = selectModel("standard");

  const prompt = GENERATION_PROMPT.replace("{count}", count.toString())
    .replace("{difficulty}", difficulty)
    .replace("{title}", title)
    .replace("{summary}", summary || "无摘要")
    .replace("{content}", content.slice(0, 4000)); // 限制内容长度

  try {
    const { text } = await generateText({
      model,
      prompt,
      maxTokens: 2000,
      temperature: 0.7,
    });

    // 解析 JSON
    const jsonMatch = text.match(/\[[\s\S]*\]/);
    if (!jsonMatch) {
      logger.warn("Flashcard generation returned non-JSON response", { text });
      return [];
    }

    const flashcards = JSON.parse(jsonMatch[0]) as GeneratedFlashcard[];

    // 验证结构
    return flashcards.filter(
      (fc) =>
        fc.question &&
        fc.answer &&
        typeof fc.question === "string" &&
        typeof fc.answer === "string",
    );
  } catch (error) {
    logger.error("Flashcard generation failed", { error, title });
    throw error;
  }
}
```

```typescript
      );

      // Step 3-5: 复习循环
      let reviewedCount = 0;
      let masteredCount = 0;

      for (let i = 0; i < queue.length; i++) {
        if (this.isCancelled(context)) {
          yield* this.emitText("\n\n⏸️ 复习已暂停，下次继续。");
          break;
        }

        const item = queue[i];
        const progress = 20 + Math.round((i / queue.length) * 60);

        yield* this.reportProgress(
          "review_loop",
          progress,
          `复习中 (${i + 1}/${queue.length})`,
        );

        // 展示问题
        yield* this.emitText(
          `\n\n🧠 **知识同步 ${i + 1}/${queue.length}**\n\n` +
            `📖 来源：「${item.bookmarkTitle}」\n\n` +
            `**问题**：${item.question}\n\n` +
            `_(想好答案后，点击「查看答案」或告诉我你的理解)_`,
        );

        // 请求用户确认查看答案
        const viewAnswer = await this.requestConfirmation(
          {
            title: "查看答案",
            message: "准备好查看参考答案了吗？",
            options: [
              { label: "💭 查看答案", value: "show" },
              { label: "⏭️ 跳过", value: "skip" },
            ],
          },
          context,
        );

        if (viewAnswer === "skip") {
          continue;
        }

        // 展示答案
        yield* this.emitText(
          `\n**参考答案**：${item.answer}\n\n` +
            (item.sourceParagraph
              ? `> 📖 原文片段：${item.sourceParagraph}\n\n`
              : "") +
            `这个知识点你掌握了吗？`,
        );

        // 收集反馈
        const response = (await this.requestConfirmation(
          {
            title: "复习反馈",
            message: "评估你对这个知识点的掌握程度",
            options: [
              {
                label: "✅ 已内化",
                value: "mastered",
                description: "完全记得",
              },
              { label: "🤔 模糊", value: "fuzzy", description: "记得一点" },
              {
                label: "❌ 遗忘",
                value: "forgotten",
                description: "完全不记得",
              },
            ],
          },
          context,
        )) as "mastered" | "fuzzy" | "forgotten";

        // 处理反馈
        const sm2Result = SM2Algorithm.calculate({
          quality: SM2Algorithm.responseToQuality(response),
          repetitions: item.repetitions,
          easeFactor: item.easeFactor,
          interval: item.intervalDays,
        });

        await this.updateFlashcardReview(
          item.flashcardId,
          userId,
          response,
          sm2Result,
        );

        reviewedCount++;
        if (response === "mastered") {
          masteredCount++;
          yield* this.emitText(`\n🎉 太棒了！这个知识点已刻入脑回路！`);
        } else if (response === "fuzzy") {
          yield* this.emitText(
            `\n📖 没关系，${sm2Result.interval} 天后我们再来巩固。`,
          );
        } else {
          yield* this.emitText(
            `\n🔄 知识需要反复锤炼，${sm2Result.interval} 天后重新开始！`,
          );
        }
      }

      // Step 6: 生成总结
      yield* this.reportProgress("generate_summary", 95, "生成复习总结...");

      const masteryRate =
        reviewedCount > 0
          ? Math.round((masteredCount / reviewedCount) * 100)
          : 0;

      const summary =
        `\n\n---\n\n` +
        `📊 **本次复习总结**\n\n` +
        `- 复习卡片：${reviewedCount} 张\n` +
        `- 已内化：${masteredCount} 张\n` +
        `- 内化率：${masteryRate}%\n\n` +
        `继续加油！知识需要反复锤炼才能真正内化 💪`;

      yield* this.emitText(summary);

      yield* this.emitDeliverable(
        this.createDeliverable(
          "review_summary",
          "复习总结",
          { reviewedCount, masteredCount, masteryRate },
          "json",
        ),
      );

      yield* this.emitDone(true, summary);
    } catch (error) {
      logger.error("ActiveRecallSkill execution failed", { error, userId });
      yield* this.emitError(
        "复习过程中出现错误，请稍后重试",
        "EXECUTION_ERROR",
        true,
      );
    }
  }
```

```typescript
// 文件: src/lib/ai-agent/tools/active-recall/generate-flashcards.tool.ts

import { z } from "zod";
import { Tool, ToolContext } from "@/lib/ai-agent/types";
import { generateFlashcards } from "@/lib/ai-skills/shared/review/flashcard-generator";
import { REVIEW_TRIGGER_TAGS } from "@/lib/ai-skills/shared/review/review-trigger-tags";
import { createServerClient } from "@/lib/supabase/server";
import { logger } from "@/lib/logger";

const paramsSchema = z.object({
  bookmarkId: z.number().int().positive().describe("书签 ID"),
  count: z.number().int().min(3).max(5).default(3).describe("生成数量 (3-5)"),
  difficulty: z
    .enum(["easy", "medium", "hard"])
    .default("medium")
    .describe("难度等级"),
});

type GenerateFlashcardsParams = z.infer<typeof paramsSchema>;

interface GenerateFlashcardsResult {
  success: boolean;
  bookmarkId: number;
  flashcardsCount: number;
  flashcards: Array<{
    id: number;
    question: string;
    answer: string;
  }>;
  message: string;
}

export const generateFlashcardsTool: Tool<
  GenerateFlashcardsParams,
  GenerateFlashcardsResult
> = {
  definition: {
    name: "generate_flashcards",
    description:
      "为指定书签生成 3-5 个知识复习问答对（Flashcards）。" +
      "只对包含「值得复读」标签的书签生效。",
    parameters: paramsSchema,
    category: "action",
    requiresConfirmation: false,
    estimatedDurationMs: 3000,
  },

  async execute(
    params: GenerateFlashcardsParams,
    context: ToolContext,
  ): Promise<GenerateFlashcardsResult> {
    const { bookmarkId, count, difficulty } = params;
    const userId = context.userId;

    const supabase = await createServerClient();

    // 1. 获取书签
    const { data: bookmark, error: bookmarkError } = await supabase
      .from("bookmarks")
      .select("id, title, url, ai_summary, content, ai_tags, user_tags")
      .eq("id", bookmarkId)
      .eq("user_id", userId)
      .single();

    if (bookmarkError || !bookmark) {
      return {
        success: false,
        bookmarkId,
        flashcardsCount: 0,
        flashcards: [],
        message: `书签 ${bookmarkId} 不存在或无权访问`,
      };
    }

    // 2. 检查是否有触发标签
    const allTags = [
      ...(bookmark.ai_tags || []),
      ...(bookmark.user_tags || []),
    ];
    const hasTriggerTag = allTags.some((tag) =>
      REVIEW_TRIGGER_TAGS.some((trigger) =>
        tag.toLowerCase().includes(trigger.toLowerCase()),
      ),
    );

    if (!hasTriggerTag) {
      return {
        success: false,
        bookmarkId,
        flashcardsCount: 0,
        flashcards: [],
        message:
          "该书签未标记「值得复读」，无法生成 Flashcards。请先添加相关标签。",
      };
    }

    // 3. 检查现有 Flashcards 数量
    const { count: existingCount } = await supabase
      .from("flashcards")
      .select("*", { count: "exact", head: true })
      .eq("bookmark_id", bookmarkId)
      .eq("user_id", userId);

    if ((existingCount || 0) >= count) {
      return {
        success: true,
        bookmarkId,
        flashcardsCount: existingCount || 0,
        flashcards: [],
        message: `该书签已有 ${existingCount} 个 Flashcards，无需重复生成`,
      };
    }

    // 4. 生成 Flashcards
    const toGenerate = count - (existingCount || 0);
    const generated = await generateFlashcards({
      title: bookmark.title,
      summary: bookmark.ai_summary,
      content: bookmark.content || "",
      count: toGenerate,
      difficulty,
    });

    // 5. 存储到数据库
    const flashcardsToInsert = generated.map((fc) => ({
      user_id: userId,
      bookmark_id: bookmarkId,
      question: fc.question,
      answer: fc.answer,
      source_paragraph: fc.sourceParagraph,
      status: "pending",
      difficulty,
      next_review_at: new Date(Date.now() + 24 * 60 * 60 * 1000).toISOString(), // D1
    }));

    const { data: inserted, error: insertError } = await supabase
      .from("flashcards")
      .insert(flashcardsToInsert)
      .select("id, question, answer");

    if (insertError) {
      logger.error("Failed to insert flashcards", { error: insertError });
      return {
        success: false,
        bookmarkId,
        flashcardsCount: 0,
        flashcards: [],
        message: "保存 Flashcards 失败，请稍后重试",
      };
    }

    return {
      success: true,
      bookmarkId,
      flashcardsCount: inserted?.length || 0,
      flashcards: inserted || [],
      message: `已为「${bookmark.title}」生成 ${inserted?.length || 0} 个 Flashcards`,
    };
  },
};
```

```typescript
// 文件: supabase/functions/generate-weekly-review-queue/index.ts

import { serve } from "https://deno.land/std@0.168.0/http/server.ts";
import { createClient } from "https://esm.sh/@supabase/supabase-js@2";

const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers":
    "authorization, x-client-info, apikey, content-type",
};

serve(async (req) => {
  // CORS 预检
  if (req.method === "OPTIONS") {
    return new Response("ok", { headers: corsHeaders });
  }

  const supabase = createClient(
    Deno.env.get("SUPABASE_URL")!,
    Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!,
  );

  const weekStart = getWeekStart(new Date());
  const results: Array<{ userId: string; queueSize: number }> = [];

  try {
    // 1. 获取所有启用复习功能的用户
    const { data: users, error: usersError } = await supabase
      .from("review_preferences")
      .select("user_id, weekly_limit, included_folder_ids, excluded_folder_ids")
      .eq("enabled", true);

    if (usersError) throw usersError;

    for (const user of users || []) {
      // 2. 获取待复习的 Flashcards
      const { data: flashcards, error: fcError } = await supabase.rpc(
        "get_pending_flashcards",
        {
          p_user_id: user.user_id,
          p_limit: user.weekly_limit || 10,
          p_included_folders: user.included_folder_ids,
          p_excluded_folders: user.excluded_folder_ids,
        },
      );

      if (fcError || !flashcards?.length) continue;

      const flashcardIds = flashcards.map((fc: any) => fc.flashcard_id);

      // 3. 创建/更新每周队列
      const { error: upsertError } = await supabase
        .from("weekly_review_queue")
        .upsert(
          {
            user_id: user.user_id,
            week_start: weekStart,
            flashcard_ids: flashcardIds,
            completed_ids: [],
          },
          {
            onConflict: "user_id,week_start",
          },
        );

      if (!upsertError) {
        results.push({
          userId: user.user_id,
          queueSize: flashcardIds.length,
        });
      }
    }

    return new Response(
      JSON.stringify({
        success: true,
        weekStart,
        generatedQueues: results.length,
        results,
      }),
      {
        headers: { ...corsHeaders, "Content-Type": "application/json" },
      },
    );
  } catch (error) {
    return new Response(JSON.stringify({ error: error.message }), {
      status: 500,
      headers: { ...corsHeaders, "Content-Type": "application/json" },
    });
  }
});

function getWeekStart(date: Date): string {
  const d = new Date(date);
  const day = d.getDay();
  const diff = d.getDate() - day + (day === 0 ? -6 : 1); // 周一为起始
  d.setDate(diff);
  d.setHours(0, 0, 0, 0);
  return d.toISOString().split("T")[0];
}
```

```typescript
// 文件: supabase/functions/send-review-notifications/index.ts

import { serve } from "https://deno.land/std@0.168.0/http/server.ts";
import { createClient } from "https://esm.sh/@supabase/supabase-js@2";

serve(async (req) => {
  const supabase = createClient(
    Deno.env.get("SUPABASE_URL")!,
    Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!,
  );

  const now = new Date();
  const currentDay = getDayName(now);
  const notified: string[] = [];

  try {
    // 1. 查询当前时段需要通知的用户
    const { data: users } = await supabase
      .from("review_preferences")
      .select("user_id, preferred_time, timezone, notification_channel")
      .eq("enabled", true)
      .eq("preferred_day", currentDay)
      .neq("notification_channel", "none");

    for (const user of users || []) {
      // 2. 检查时间窗口
      const userHour = getHourInTimezone(now, user.timezone);
      const preferredHour = parseInt(user.preferred_time.split(":")[0]);

      if (userHour !== preferredHour) continue;

      // 3. 检查本周是否已通知
      const weekStart = getWeekStart(now);
      const { data: existingNotification } = await supabase
        .from("messages")
        .select("id")
        .eq("user_id", user.user_id)
        .eq("type", "review_reminder")
        .gte("created_at", weekStart)
        .limit(1);

      if (existingNotification?.length) continue;

      // 4. 获取待复习数量
      const { data: queue } = await supabase
        .from("weekly_review_queue")
        .select("flashcard_ids, completed_ids")
        .eq("user_id", user.user_id)
        .eq("week_start", weekStart)
        .single();

      const pendingCount =
        (queue?.flashcard_ids?.length || 0) -
        (queue?.completed_ids?.length || 0);

      if (pendingCount === 0) continue;

      // 5. 发送通知
      await supabase.from("messages").insert({
        user_id: user.user_id,
        type: "review_reminder",
        level: "normal",
        title: "🧠 本周知识同步",
        content: `你有 ${pendingCount} 个知识点等待复习，开始同步你的脑回路吧！`,
        metadata: { pendingCount, weekStart },
      });

      notified.push(user.user_id);
    }

    return new Response(
      JSON.stringify({
        success: true,
        notifiedCount: notified.length,
        timestamp: now.toISOString(),
      }),
      { headers: { "Content-Type": "application/json" } },
    );
  } catch (error) {
    return new Response(JSON.stringify({ error: error.message }), {
      status: 500,
      headers: { "Content-Type": "application/json" },
    });
  }
});

// 工具函数
function getDayName(date: Date): string {
  const days = [
    "sunday",
    "monday",
    "tuesday",
    "wednesday",
    "thursday",
    "friday",
    "saturday",
  ];
  return days[date.getUTCDay()];
}

function getHourInTimezone(date: Date, timezone: string): number {
  return parseInt(
    date.toLocaleString("en-US", {
      hour: "numeric",
      hour12: false,
      timeZone: timezone,
    }),
  );
}

function getWeekStart(date: Date): string {
  const d = new Date(date);
  const day = d.getUTCDay();
  const diff = d.getUTCDate() - day + (day === 0 ? -6 : 1);
  d.setUTCDate(diff);
  return d.toISOString().split("T")[0];
}
```

```typescript
// 文件: src/hooks/brain-sync/use-review-session.ts

import { useState, useCallback } from "react";
import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";

interface ReviewSessionState {
  currentIndex: number;
  reviewedCount: number;
  masteredCount: number;
  isShowingAnswer: boolean;
}

export function useReviewSession() {
  const queryClient = useQueryClient();
  const [state, setState] = useState<ReviewSessionState>({
    currentIndex: 0,
    reviewedCount: 0,
    masteredCount: 0,
    isShowingAnswer: false,
  });

  // 获取队列
  const { data: queue, isLoading } = useQuery({
    queryKey: ["brain-sync", "queue"],
    queryFn: async () => {
      const res = await fetch("/api/brain-sync/queue");
      return res.json();
    },
  });

  // 提交反馈
  const submitResponse = useMutation({
    mutationFn: async (params: {
      flashcardId: number;
      response: "mastered" | "fuzzy" | "forgotten";
      responseTimeMs?: number;
    }) => {
      const res = await fetch("/api/brain-sync/response", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(params),
      });
      return res.json();
    },
    onSuccess: (data, variables) => {
      setState((prev) => ({
        ...prev,
        reviewedCount: prev.reviewedCount + 1,
        masteredCount:
          variables.response === "mastered"
            ? prev.masteredCount + 1
            : prev.masteredCount,
        currentIndex: prev.currentIndex + 1,
        isShowingAnswer: false,
      }));
      queryClient.invalidateQueries({ queryKey: ["brain-sync"] });
    },
  });

  const showAnswer = useCallback(() => {
    setState((prev) => ({ ...prev, isShowingAnswer: true }));
  }, []);

  const skipCard = useCallback(() => {
    setState((prev) => ({
      ...prev,
      currentIndex: prev.currentIndex + 1,
      isShowingAnswer: false,
    }));
  }, []);

  return {
    queue: queue?.queue || [],
    stats: queue?.stats,
    isLoading,
    currentCard: queue?.queue?.[state.currentIndex],
    state,
    showAnswer,
    skipCard,
    submitResponse: submitResponse.mutate,
    isSubmitting: submitResponse.isPending,
  };
}
```

```typescript
describe("SM2Algorithm", () => {
  describe("calculate", () => {

        interval: 0,

      expect(result.interval).toBe(1);
      expect(result.repetitions).toBe(1);
    });

      const result = SM2Algorithm.calculate({
        quality: 4,
        repetitions: 1,
        easeFactor: 2.5,
        interval: 1,
      });
      expect(result.interval).toBe(3);
      expect(result.repetitions).toBe(2);
    });

    it("should return interval 7 for third review", () => {
      const result = SM2Algorithm.calculate({
        quality: 4,
        repetitions: 2,
        easeFactor: 2.5,
        interval: 3,
      });
      expect(result.interval).toBe(7);
      expect(result.repetitions).toBe(3);
    });

    it("should reset on quality < 3", () => {
      const result = SM2Algorithm.calculate({
        quality: 2,
        repetitions: 5,
        easeFactor: 2.5,
        interval: 30,
      });
      expect(result.interval).toBe(1);
      expect(result.repetitions).toBe(0);
      expect(result.easeFactor).toBeLessThan(2.5);
    });

    it("should cap interval at 90 days", () => {
      const result = SM2Algorithm.calculate({
        quality: 5,
        repetitions: 10,
        easeFactor: 2.5,
        interval: 60,
      });
      expect(result.interval).toBeLessThanOrEqual(90);
    });

    it("should maintain easeFactor minimum of 1.3", () => {
      let result = { easeFactor: 2.5, repetitions: 5, interval: 30 };
      // 连续遗忘多次
      for (let i = 0; i < 10; i++) {
        result = SM2Algorithm.calculate({
          quality: 0,
          ...result,
        });
      }
      expect(result.easeFactor).toBeGreaterThanOrEqual(1.3);
    });
  });

  describe("responseToQuality", () => {
    it("should map mastered to 5", () => {
      expect(SM2Algorithm.responseToQuality("mastered")).toBe(5);
    });

    it("should map fuzzy to 3", () => {
      expect(SM2Algorithm.responseToQuality("fuzzy")).toBe(3);
    });

    it("should map forgotten to 0", () => {
      expect(SM2Algorithm.responseToQuality("forgotten")).toBe(0);
    });
  });
});
```

```typescript
export const REVIEW_TRIGGER_TAGS = [
  // 中文
  "值得复读",
  "值得复习",
  "需要记住",
  "重要知识",
  "学习笔记",
  "深度阅读",
  "核心概念",

  // 英文
  "worth-review",
  "to-memorize",
  "important",
  "must-remember",
  "key-concept",
  "deep-read",
];

export function hasReviewTriggerTag(tags: string[]): boolean {
  return tags.some((tag) =>
    REVIEW_TRIGGER_TAGS.some((trigger) =>
      tag.toLowerCase().includes(trigger.toLowerCase()),
    ),
  );
}
```
