## 二、功能分级规划

### 2.1 分级策略原则

```
┌─────────────────────────────────────────────────────────────────────┐
│                       功能分级策略                                   │
├─────────────────────────────────────────────────────────────────────┤
│                                                                     │
│   🎯 核心原则：                                                      │
│                                                                     │
│   1. 免费版要够用                                                    │
│      - 核心功能完整，能满足轻度用户日常需求                           │
│      - 不能让用户觉得"残废版"                                        │
│                                                                     │
│   2. Pro 版要有感知价值                                              │
│      - 针对重度用户和专业场景                                        │
│      - 功能差异明显，一眼能看出区别                                   │
│                                                                     │
│   3. 升级门槛要自然                                                  │
│      - 用户使用到一定程度自然遇到限制                                 │
│      - 不是强制广告，而是使用需求驱动                                 │
│                                                                     │
│   4. 降级保护                                                       │
│      - Pro 用户降级后，已有数据保留                                  │
│      - 仅限制新增，不删除历史                                        │
│                                                                     │
└─────────────────────────────────────────────────────────────────────┘
```

## 五、数据库设计

### 5.1 新增表结构

```sql
-- ============================================
-- 订阅表：记录用户订阅状态
-- ============================================
CREATE TABLE public.subscriptions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,

  -- 订阅信息
  plan_type VARCHAR(20) NOT NULL DEFAULT 'pro',  -- 'pro', 'pro_plus' (预留)
  status VARCHAR(20) NOT NULL DEFAULT 'active',   -- 'trialing', 'active', 'cancelled', 'expired', 'past_due'

  -- 时间周期
  trial_start_at TIMESTAMP WITH TIME ZONE,
  trial_end_at TIMESTAMP WITH TIME ZONE,
  current_period_start TIMESTAMP WITH TIME ZONE NOT NULL,
  current_period_end TIMESTAMP WITH TIME ZONE NOT NULL,

  -- 续费设置
  cancel_at_period_end BOOLEAN DEFAULT FALSE,     -- 是否在周期结束时取消
  cancelled_at TIMESTAMP WITH TIME ZONE,          -- 取消时间

  -- 支付信息
  payment_method VARCHAR(50),                      -- 'wechat', 'alipay', 'stripe'
  external_subscription_id VARCHAR(255),           -- 外部订阅 ID

  -- 来源追踪
  upgrade_source VARCHAR(100),                     -- 'pricing_page', 'feature_md', 'feature_snapshot'

  -- 时间戳
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),

  -- 约束
  CONSTRAINT valid_status CHECK (status IN ('trialing', 'active', 'cancelled', 'expired', 'past_due')),
  CONSTRAINT valid_plan CHECK (plan_type IN ('pro', 'pro_plus'))
);

-- 索引
CREATE INDEX idx_subscriptions_user_id ON public.subscriptions(user_id);
CREATE INDEX idx_subscriptions_status ON public.subscriptions(status);
CREATE INDEX idx_subscriptions_period_end ON public.subscriptions(current_period_end);

-- ============================================
-- 账单历史表：记录所有支付记录
-- ============================================
CREATE TABLE public.billing_history (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
  subscription_id UUID REFERENCES public.subscriptions(id),

  -- 账单信息
  amount DECIMAL(10, 2) NOT NULL,
  currency VARCHAR(3) NOT NULL DEFAULT 'CNY',     -- 'CNY', 'USD'
  status VARCHAR(20) NOT NULL DEFAULT 'pending',   -- 'pending', 'completed', 'failed', 'refunded'

  -- 账单类型
  billing_type VARCHAR(20) NOT NULL,               -- 'subscription', 'one_time', 'refund'
  billing_period_start TIMESTAMP WITH TIME ZONE,
  billing_period_end TIMESTAMP WITH TIME ZONE,

  -- 支付信息
  payment_method VARCHAR(50),
  external_payment_id VARCHAR(255),                -- 外部支付单号
  external_invoice_id VARCHAR(255),                -- 外部发票号

  -- 发票
  invoice_url TEXT,
  invoice_requested BOOLEAN DEFAULT FALSE,
```

```sql
-- 检查用户订阅状态

CREATE OR REPLACE FUNCTION check_subscription_status(p_user_id UUID)
RETURNS TABLE (
  is_pro BOOLEAN,
  is_trialing BOOLEAN,
  days_remaining INT,
  plan_type VARCHAR,
  status VARCHAR
) AS $$
BEGIN
  RETURN QUERY
  SELECT
    s.status IN ('active', 'trialing') AS is_pro,
    s.status = 'trialing' AS is_trialing,
    GREATEST(0, EXTRACT(DAY FROM s.current_period_end - NOW())::INT) AS days_remaining,
    s.plan_type,
    s.status
  FROM public.subscriptions s
  WHERE s.user_id = p_user_id
    AND s.status IN ('active', 'trialing', 'past_due')
  ORDER BY s.created_at DESC
  LIMIT 1;

  -- 如果没有订阅记录，返回默认值
  IF NOT FOUND THEN
    RETURN QUERY SELECT FALSE, FALSE, 0, 'none'::VARCHAR, 'none'::VARCHAR;
  END IF;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- ============================================
-- 升级用户到 Pro

CREATE OR REPLACE FUNCTION upgrade_to_pro(
  p_user_id UUID,
  p_plan_type VARCHAR DEFAULT 'pro',
  p_period_months INT DEFAULT 1,
  p_payment_method VARCHAR DEFAULT 'wechat',
  p_external_id VARCHAR DEFAULT NULL,
  p_upgrade_source VARCHAR DEFAULT 'pricing_page'
)
RETURNS UUID AS $$
DECLARE
  v_subscription_id UUID;
  v_period_start TIMESTAMP WITH TIME ZONE := NOW();
  v_period_end TIMESTAMP WITH TIME ZONE := NOW() + (p_period_months || ' months')::INTERVAL;
BEGIN
  -- 创建或更新订阅
  INSERT INTO public.subscriptions (
    user_id,
    plan_type,
    status,
    current_period_start,
    current_period_end,
    payment_method,
    external_subscription_id,
    upgrade_source
  ) VALUES (
    p_user_id,
    p_plan_type,
    'active',
    v_period_start,
    v_period_end,
    p_payment_method,
    p_external_id,
    p_upgrade_source
  )
  ON CONFLICT (user_id)
  DO UPDATE SET
    plan_type = EXCLUDED.plan_type,
    status = 'active',
    current_period_start = EXCLUDED.current_period_start,
    current_period_end = EXCLUDED.current_period_end,
    payment_method = EXCLUDED.payment_method,
    external_subscription_id = EXCLUDED.external_subscription_id,
    updated_at = NOW()
  RETURNING id INTO v_subscription_id;

  -- 更新用户 profile
  UPDATE public.profiles
  SET
    membership_type = 'PRO',
    subscription_status = 'active',
    storage_quota = -1,  -- 无限
    updated_at = NOW()
  WHERE id = p_user_id;

  RETURN v_subscription_id;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- ============================================
-- 开始试用
-- ============================================
CREATE OR REPLACE FUNCTION start_trial(
  p_user_id UUID,
  p_trial_days INT DEFAULT 7
)
RETURNS UUID AS $$
DECLARE
  v_subscription_id UUID;
  v_already_used BOOLEAN;
BEGIN
  -- 检查是否已使用过试用
  SELECT trial_used INTO v_already_used
  FROM public.profiles
  WHERE id = p_user_id;

  IF v_already_used THEN
    RAISE EXCEPTION 'Trial already used';
  END IF;

  -- 创建试用订阅
  INSERT INTO public.subscriptions (
    user_id,
    plan_type,
    status,
    trial_start_at,
    trial_end_at,
    current_period_start,
    current_period_end,
    upgrade_source
  ) VALUES (
    p_user_id,
    'pro',
    'trialing',
    NOW(),
    NOW() + (p_trial_days || ' days')::INTERVAL,
    NOW(),
    NOW() + (p_trial_days || ' days')::INTERVAL,
    'trial'
  )
  RETURNING id INTO v_subscription_id;

  -- 更新用户 profile
  UPDATE public.profiles
  SET
    membership_type = 'PRO',
    subscription_status = 'trialing',
    trial_used = TRUE,
    storage_quota = -1,
    updated_at = NOW()
  WHERE id = p_user_id;

  RETURN v_subscription_id;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- ============================================
-- 取消订阅
-- ============================================
CREATE OR REPLACE FUNCTION cancel_subscription(
  p_user_id UUID,
  p_immediate BOOLEAN DEFAULT FALSE
)
RETURNS VOID AS $$
BEGIN
  IF p_immediate THEN
    -- 立即取消
    UPDATE public.subscriptions
    SET
      status = 'cancelled',
      cancelled_at = NOW(),
      updated_at = NOW()
    WHERE user_id = p_user_id AND status IN ('active', 'trialing');

    -- 降级用户
    UPDATE public.profiles
    SET
      membership_type = 'FREE',
      subscription_status = 'none',
      storage_quota = 500,
      updated_at = NOW()
    WHERE id = p_user_id;
  ELSE
    -- 周期结束时取消
    UPDATE public.subscriptions
    SET
      cancel_at_period_end = TRUE,
      cancelled_at = NOW(),
      updated_at = NOW()
    WHERE user_id = p_user_id AND status IN ('active', 'trialing');
  END IF;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- ============================================
-- 定时任务：检查过期订阅
-- ============================================
CREATE OR REPLACE FUNCTION check_expired_subscriptions()
RETURNS INT AS $$
DECLARE
  v_count INT := 0;
BEGIN
  -- 更新已过期的订阅
  WITH expired AS (
    UPDATE public.subscriptions
    SET
      status = 'expired',
      updated_at = NOW()
    WHERE status IN ('active', 'trialing')
      AND current_period_end < NOW()
      AND cancel_at_period_end = TRUE
    RETURNING user_id
  )
  -- 降级用户
  UPDATE public.profiles p
  SET
    membership_type = 'FREE',
    subscription_status = 'expired',
    storage_quota = 500,
    updated_at = NOW()
  FROM expired e
  WHERE p.id = e.user_id;

  GET DIAGNOSTICS v_count = ROW_COUNT;
  RETURN v_count;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
```

### 8.3 微信支付集成示例

```typescript
// src/lib/payment/wechat.ts

import crypto from "crypto";

export class WechatPayService {
  private appId: string;
  private mchId: string;
  private apiKey: string;
  private notifyUrl: string;

  constructor() {
    this.appId = process.env.WECHAT_APP_ID!;
    this.mchId = process.env.WECHAT_MCH_ID!;
    this.apiKey = process.env.WECHAT_API_KEY!;
    this.notifyUrl = process.env.WECHAT_NOTIFY_URL!;
  }

  /**
   * 创建支付订单（Native 支付，返回二维码）
   */
  async createNativeOrder(params: {
    orderId: string;
    amount: number;
    description: string;
  }): Promise<{ qrCodeUrl: string; prepayId: string }> {
    const requestData = {
      appid: this.appId,
      mchid: this.mchId,
      description: params.description,
      out_trade_no: params.orderId,
      notify_url: this.notifyUrl,
      amount: {
        total: Math.round(params.amount * 100), // 分
        currency: "CNY",
      },
    };

    const response = await this.request(
      "/v3/pay/transactions/native",
      requestData,
    );

    return {
      qrCodeUrl: response.code_url,
      prepayId: response.prepay_id,
    };
  }

  /**
   * 验证回调签名
   */
  verifySignature(
    timestamp: string,
    nonce: string,
    body: string,
    signature: string,
  ): boolean {
    const message = `${timestamp}\n${nonce}\n${body}\n`;
    const expectedSignature = crypto
      .createHmac("sha256", this.apiKey)
      .update(message)
      .digest("base64");

    return signature === expectedSignature;
  }

  /**
   * 解密回调数据
   */
  decryptResource(resource: {
    ciphertext: string;
    nonce: string;
    associated_data: string;
  }): any {
    // AES-GCM 解密
    const decipher = crypto.createDecipheriv(
      "aes-256-gcm",
      this.apiKey,
      resource.nonce,
    );
    decipher.setAAD(Buffer.from(resource.associated_data));

    let decrypted = decipher.update(resource.ciphertext, "base64", "utf8");
    decrypted += decipher.final("utf8");

    return JSON.parse(decrypted);
  }
}
```
