## 二、功能分级规划

### 2.1 分级策略原则

```
┌─────────────────────────────────────────────────────────────────────┐
│                       功能分级策略                                   │
├─────────────────────────────────────────────────────────────────────┤
│                                                                     │
│   🎯 核心原则：                                                      │
│                                                                     │
│   1. 免费版要够用                                                    │
│      - 核心功能完整，能满足轻度用户日常需求                           │
│      - 不能让用户觉得"残废版"                                        │
│                                                                     │
│   2. Pro 版要有感知价值                                              │
│      - 针对重度用户和专业场景                                        │
│      - 功能差异明显，一眼能看出区别                                   │
│                                                                     │
│   3. 升级门槛要自然                                                  │
│      - 用户使用到一定程度自然遇到限制                                 │
│      - 不是强制广告，而是使用需求驱动                                 │
│                                                                     │
│   4. 降级保护                                                       │
│      - Pro 用户降级后，已有数据保留                                  │
│      - 仅限制新增，不删除历史                                        │
│                                                                     │
└─────────────────────────────────────────────────────────────────────┘
```

```typescript
// src/features/subscription/lib/subscription-bridge.ts
// 负责将 subscriptions 表状态同步到 profiles 表

import { createServerClient } from "@/lib/supabase/server";

export class SubscriptionBridge {
  /**
   * 将订阅状态同步到 profiles 表
   * 这是唯一修改 profiles.membership_type 的入口
   */
  static async syncToProfile(
    userId: string,
    status: "active" | "trialing" | "cancelled" | "expired",
  ): Promise<void> {
    const supabase = await createServerClient();

    const membershipType =
      status === "active" || status === "trialing" ? "PRO" : "FREE";
    const storageQuota = membershipType === "PRO" ? -1 : 500;

    await supabase
      .from("profiles")
      .update({
        membership_type: membershipType,
        subscription_status: status,
        storage_quota: storageQuota,
        updated_at: new Date().toISOString(),
      })
      .eq("id", userId);
  }

  /**
   * 批量检查过期订阅并降级
   * 由 Cron Job 调用
   */
  static async processExpiredSubscriptions(): Promise<number> {
    const supabase = await createServerClient();

    const { data, count } = await supabase.rpc("check_expired_subscriptions");

    return count || 0;
  }
}
```

```sql
-- ============================================
-- 检查用户订阅状态

CREATE OR REPLACE FUNCTION check_subscription_status(p_user_id UUID)
RETURNS TABLE (
  is_pro BOOLEAN,
  is_trialing BOOLEAN,
  days_remaining INT,
  plan_type VARCHAR,
  status VARCHAR
) AS $$
BEGIN
  RETURN QUERY
  SELECT
    s.status IN ('active', 'trialing') AS is_pro,
    s.status = 'trialing' AS is_trialing,
    GREATEST(0, EXTRACT(DAY FROM s.current_period_end - NOW())::INT) AS days_remaining,
    s.plan_type,
    s.status
  FROM public.subscriptions s
  WHERE s.user_id = p_user_id
    AND s.status IN ('active', 'trialing', 'past_due')
  ORDER BY s.created_at DESC
  LIMIT 1;

  -- 如果没有订阅记录，返回默认值
  IF NOT FOUND THEN
    RETURN QUERY SELECT FALSE, FALSE, 0, 'none'::VARCHAR, 'none'::VARCHAR;
  END IF;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- ============================================
-- 升级用户到 Pro（原子操作）
-- ============================================
CREATE OR REPLACE FUNCTION upgrade_to_pro(
  p_user_id UUID,
  p_plan_type VARCHAR DEFAULT 'pro',
  p_period_months INT DEFAULT 1,
  p_payment_method VARCHAR DEFAULT 'wechat',
  p_external_id VARCHAR DEFAULT NULL,
  p_upgrade_source VARCHAR DEFAULT 'pricing_page'
)
RETURNS UUID AS $$
DECLARE
  v_subscription_id UUID;
  v_period_start TIMESTAMP WITH TIME ZONE := NOW();
  v_period_end TIMESTAMP WITH TIME ZONE := NOW() + (p_period_months || ' months')::INTERVAL;
BEGIN
  -- 1. 取消现有订阅（如有）
  UPDATE public.subscriptions
  SET status = 'cancelled', cancelled_at = NOW(), updated_at = NOW()
  WHERE user_id = p_user_id AND status IN ('active', 'trialing');

  -- 2. 创建新订阅
  INSERT INTO public.subscriptions (
    user_id,
    plan_type,
    status,
    current_period_start,
    current_period_end,
    payment_method,
    external_subscription_id,
    upgrade_source
  ) VALUES (
    p_user_id,
    p_plan_type,
    'active',
    v_period_start,
    v_period_end,
    p_payment_method,
    p_external_id,
    p_upgrade_source
  )
  RETURNING id INTO v_subscription_id;

  -- 3. 更新用户 profile（桥接同步）
  UPDATE public.profiles
  SET
    membership_type = 'PRO',
    subscription_status = 'active',
    storage_quota = -1,  -- 无限
    updated_at = NOW()
  WHERE id = p_user_id;

  RETURN v_subscription_id;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- ============================================
-- 开始试用
-- ============================================
CREATE OR REPLACE FUNCTION start_trial(
  p_user_id UUID,
  p_trial_days INT DEFAULT 7
)
RETURNS UUID AS $$
DECLARE
  v_subscription_id UUID;
  v_already_used BOOLEAN;
BEGIN
  -- 检查是否已使用过试用
  SELECT trial_used INTO v_already_used
  FROM public.profiles
  WHERE id = p_user_id;

  IF v_already_used THEN
    RAISE EXCEPTION 'Trial already used';
  END IF;

  -- 创建试用订阅
  INSERT INTO public.subscriptions (
    user_id,
    plan_type,
    status,
    trial_start_at,
    trial_end_at,
    current_period_start,
    current_period_end,
    upgrade_source
  ) VALUES (
    p_user_id,
    'pro',
    'trialing',
    NOW(),
    NOW() + (p_trial_days || ' days')::INTERVAL,
    NOW(),
    NOW() + (p_trial_days || ' days')::INTERVAL,
    'trial'
  )
  RETURNING id INTO v_subscription_id;

  -- 更新用户 profile
  UPDATE public.profiles
  SET
    membership_type = 'PRO',
    subscription_status = 'trialing',
    trial_used = TRUE,
    storage_quota = -1,
    updated_at = NOW()
  WHERE id = p_user_id;

  RETURN v_subscription_id;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- ============================================
-- 取消订阅
-- ============================================
CREATE OR REPLACE FUNCTION cancel_subscription(
  p_user_id UUID,
  p_immediate BOOLEAN DEFAULT FALSE
)
RETURNS VOID AS $$
BEGIN
  IF p_immediate THEN
    -- 立即取消
    UPDATE public.subscriptions
    SET
      status = 'cancelled',
      cancelled_at = NOW(),
      updated_at = NOW()
    WHERE user_id = p_user_id AND status IN ('active', 'trialing');

    -- 降级用户
    UPDATE public.profiles
    SET
      membership_type = 'FREE',
      subscription_status = 'none',
      storage_quota = 500,
      updated_at = NOW()
    WHERE id = p_user_id;
  ELSE
    -- 周期结束时取消
    UPDATE public.subscriptions
    SET
      cancel_at_period_end = TRUE,
      cancelled_at = NOW(),
      updated_at = NOW()
    WHERE user_id = p_user_id AND status IN ('active', 'trialing');
  END IF;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- ============================================
-- 定时任务：检查过期订阅
-- ============================================
CREATE OR REPLACE FUNCTION check_expired_subscriptions()
RETURNS INT AS $$
DECLARE
  v_count INT := 0;
BEGIN
  -- 更新已过期的订阅
  WITH expired AS (
    UPDATE public.subscriptions
    SET
      status = 'expired',
      updated_at = NOW()
    WHERE status IN ('active', 'trialing')
      AND current_period_end < NOW()
      AND (cancel_at_period_end = TRUE OR status = 'trialing')
    RETURNING user_id
  )
  -- 降级用户
  UPDATE public.profiles p
  SET
    membership_type = 'FREE',
    subscription_status = 'expired',
    storage_quota = 500,
    updated_at = NOW()
  FROM expired e
  WHERE p.id = e.user_id;

  GET DIAGNOSTICS v_count = ROW_COUNT;
  RETURN v_count;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
```

```typescript
// src/lib/payment/lemonsqueezy.ts
// 仅需 ~100 行代码

import crypto from "crypto";

const LEMON_API_KEY = process.env.LEMON_SQUEEZY_API_KEY!;
const LEMON_STORE_ID = process.env.LEMON_SQUEEZY_STORE_ID!;
const LEMON_WEBHOOK_SECRET = process.env.LEMON_SQUEEZY_WEBHOOK_SECRET!;

/**
 * 创建 Checkout URL（用户点击升级后跳转）
 */
export async function createCheckoutUrl(params: {
  variantId: string; // Pro 月付或年付的 variant ID
  userId: string;
  userEmail: string;
}): Promise<string> {
  const response = await fetch("https://api.lemonsqueezy.com/v1/checkouts", {
    method: "POST",
    headers: {
      Authorization: `Bearer ${LEMON_API_KEY}`,
      "Content-Type": "application/vnd.api+json",
    },
    body: JSON.stringify({
      data: {
        type: "checkouts",
        attributes: {
          checkout_data: {
            custom: {
              user_id: params.userId, // 关联用户
            },
            email: params.userEmail,
          },
          product_options: {
            redirect_url: `${process.env.NEXT_PUBLIC_APP_URL}/settings/subscription?success=true`,
          },
        },
        relationships: {
          store: { data: { type: "stores", id: LEMON_STORE_ID } },
          variant: { data: { type: "variants", id: params.variantId } },
        },
      },
    }),
  });

  const data = await response.json();
  return data.data.attributes.url;
}

/**
 * 验证 Webhook 签名
 */
export function verifyWebhookSignature(
  payload: string,
  signature: string,
): boolean {
  const hmac = crypto.createHmac("sha256", LEMON_WEBHOOK_SECRET);
  const digest = hmac.update(payload).digest("hex");
  return crypto.timingSafeEqual(Buffer.from(signature), Buffer.from(digest));
}

/**
 * 处理 Webhook 事件
 */
export async function handleWebhookEvent(event: LemonWebhookEvent) {
  const { event_name, data } = event;
  const userId = data.attributes.custom_data?.user_id;

  if (!userId) {
    console.error("Missing user_id in webhook");
    return;
  }

  switch (event_name) {
    case "subscription_created":
    case "subscription_updated":
      await updateSubscription(userId, {
        status: data.attributes.status === "active" ? "active" : "cancelled",
        lemonSubscriptionId: data.id,
        currentPeriodEnd: new Date(data.attributes.renews_at),
      });
      break;

    case "subscription_cancelled":
      await updateSubscription(userId, {
        status: "cancelled",
        cancelAtPeriodEnd: true,
      });
      break;

    case "subscription_expired":
      await downgradeUser(userId);
      break;
  }
}
```

```typescript
// src/app/api/webhooks/lemonsqueezy/route.ts
// Webhook 处理路由

import { NextRequest, NextResponse } from "next/server";
import {
  verifyWebhookSignature,
  handleWebhookEvent,
} from "@/lib/payment/lemonsqueezy";

export async function POST(request: NextRequest) {
  const signature = request.headers.get("x-signature") || "";
  const payload = await request.text();

  // 验证签名
  if (!verifyWebhookSignature(payload, signature)) {
    return NextResponse.json({ error: "Invalid signature" }, { status: 401 });
  }

  // 处理事件
  const event = JSON.parse(payload);
  await handleWebhookEvent(event);

  return NextResponse.json({ received: true });
}
```
