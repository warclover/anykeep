## 六、推断规则设计详解

### 6.1 推断条件类型

```typescript
// src/lib/ai-agent/inference/types.ts

export type ConditionType =
  | "tag_pattern" // 标签正则匹配
  | "platform_ratio" // 平台占比
  | "content_type" // 内容类型分布
  | "time_pattern" // 时间模式（如工作日/周末）
  | "folder_structure" // 文件夹命名模式
  | "bookmark_frequency" // 收藏频率
  | "domain_pattern"; // 域名模式

export interface ConditionMatcher {
  // 平台占比匹配器
  platforms?: string[];
  minRatio?: number;
  maxRatio?: number;

  // 内容类型匹配器
  contentTypes?: string[];
  minCount?: number;

  // 时间模式匹配器
  timeRange?: "workday" | "weekend" | "evening";
  minPercentage?: number;

  // 域名模式匹配器
  domains?: (string | RegExp)[];
  minRatio?: number;
}
```
