## 二、智能能力提升建议

### 2.1 🧠 增强意图理解（高价值）

**问题**：当前 `IntentClassifier` 依赖 37 条硬编码规则 + LLM 兜底，复杂语义理解有限。

**建议：引入轻量级本地意图嵌入模型**

```
当前流程:
Input → 37 Rules Matching → (未命中) → LLM Classification → Intent

优化后:
Input → Local Embedding (WASM) → Cosine Similarity → Top-K Intents → Confidence Routing
                                                           ↓
                                           (低置信度) → LLM Disambiguation
```

**推荐方案：@xenova/transformers (WASM)**

| 特性         | 说明                       |
| ------------ | -------------------------- |
| **延迟**     | <50ms                      |
| **成本**     | $0（本地运行）             |
| **准确性**   | 高（Sentence Transformer） |
| **模型大小** | ~30MB（首次加载后缓存）    |
| **离线支持** | ✅ 支持                    |

**实现架构**：

```typescript
// src/lib/ai-skills/router/intent-embedder.ts
import { pipeline } from "@xenova/transformers";

class IntentEmbedder {
  private embedder: Pipeline;
  private intentEmbeddings: Map<Intent, Float32Array>; // 预计算的意图向量

  async initialize(): Promise<void> {
    // 加载轻量级模型 (all-MiniLM-L6-v2)
    this.embedder = await pipeline(
      "feature-extraction",
      "Xenova/all-MiniLM-L6-v2",
    );

    // 预计算所有意图的嵌入向量
    await this.precomputeIntentEmbeddings();
  }

  async classify(input: string): Promise<IntentScore[]> {
    const inputEmbed = await this.embedder(input, {
      pooling: "mean",
      normalize: true,
    });

    return Array.from(this.intentEmbeddings.entries())
      .map(([intent, embed]) => ({
        intent,
        score: this.cosineSimilarity(inputEmbed.data, embed),
      }))
      .sort((a, b) => b.score - a.score);
  }

  private cosineSimilarity(a: Float32Array, b: Float32Array): number {
    let dot = 0;
    for (let i = 0; i < a.length; i++) {
      dot += a[i] * b[i];
    }
    return dot; // 已归一化，点积即余弦相似度
  }
}
```

### 2.2 🔄 多轮对话增强

**问题**：当前指代解析（`ReferenceResolver`）仅支持简单模式（"它"、"上面的"），复杂代词链容易出错。

**建议：引入实体追踪 + 对话状态机**

```typescript
// src/lib/ai-memory/entity-tracker.ts
interface EntityMention {
  id: string;
  type: "bookmark" | "folder" | "tag" | "search_result" | "action_target";
  value: string | string[]; // 实体值或 ID 列表
  mentionedAt: number; // 时间戳
  context: string; // 提及上下文
  confidence: number; // 置信度
}

interface ConversationState {
  entities: EntityMention[]; // 实体追踪栈（LIFO）
  topicStack: Topic[]; // 话题栈（支持话题切换）
  pendingActions: PendingAction[]; // 待确认操作
  userPreferences: InferredPref[]; // 推断的偏好
  lastToolCall: ToolCallRecord; // 最后的工具调用
}

class EntityTracker {
  private state: ConversationState;

  // 记录新实体
  trackEntity(entity: EntityMention): void {
    this.state.entities.unshift(entity);
    // 保持栈大小限制
    if (this.state.entities.length > 20) {
      this.state.entities.pop();
    }
  }

  // 解析指代词
  resolveReference(reference: string): EntityMention | null {
    const patterns = [
      { regex: /^(它|这个|那个)$/, resolver: () => this.state.entities[0] },
      { regex: /^(这些|它们|那些)$/, resolver: () => this.findPluralEntity() },
      {
        regex: /^第(\d+)个$/,
        resolver: (m) => this.findNthEntity(parseInt(m[1])),
      },
      {
        regex: /^(上面的|刚才的|之前的)/,
        resolver: () => this.state.entities[0],
      },
      {
        regex: /^(搜索结果|结果)$/,
        resolver: () => this.findByType("search_result"),
      },
    ];

    for (const { regex, resolver } of patterns) {
      const match = reference.match(regex);
      if (match) {
        return resolver(match);
      }
    }
    return null;
  }

  private findPluralEntity(): EntityMention | null {
    return this.state.entities.find((e) => Array.isArray(e.value));
  }

  private findNthEntity(n: number): EntityMention | null {
    const searchResult = this.findByType("search_result");
    if (searchResult && Array.isArray(searchResult.value)) {
      return {
        ...searchResult,
        value: searchResult.value[n - 1],
        type: "bookmark",
      };
    }
    return null;
  }

  private findByType(type: EntityMention["type"]): EntityMention | null {
    return this.state.entities.find((e) => e.type === type) || null;
  }
}
```

```typescript
class ProactiveTriggerEngine {
  private triggers: ProactiveTrigger[];
  private cooldowns: Map<string, number>; // triggerId -> lastTriggeredAt

  async evaluate(context: UserContext): Promise<ProactiveAction[]> {
    const now = Date.now();
    const actions: ProactiveAction[] = [];

    for (const trigger of this.triggers) {
      // 检查冷却时间
      const lastTriggered = this.cooldowns.get(trigger.id) || 0;
      if (now - lastTriggered < trigger.cooldown) {
        continue;
      }

      // 评估条件
      if (trigger.condition(context)) {
        actions.push(trigger.action(context));
        this.cooldowns.set(trigger.id, now);
      }
    }

    // 按优先级排序，最多返回 2 条
    return actions
      .sort((a, b) => this.priorityScore(b) - this.priorityScore(a))
      .slice(0, 2);
  }

  private priorityScore(action: ProactiveAction): number {
    const scores = { high: 3, medium: 2, low: 1 };
    return scores[action.priority] || 0;
  }
}
```

## 三、响应速度优化建议

### 3.1 ⚡ 工具并行执行（高优先级）

**问题**：当前工具调用是串行的，每轮需等待 LLM 决策。

**现状分析**：

```
用户: "搜索技术文章并统计标签分布"
                     ↓
Round 1: LLM → 决定调用 search-bookmarks → 执行 → 返回结果 (1.5s)
                     ↓
Round 2: LLM → 决定调用 get-tag-stats → 执行 → 返回结果 (1.2s)
                     ↓
Round 3: LLM → 生成最终响应 (0.8s)
                     ↓
总时间: 3.5s+
```

**建议：引入工具依赖图 + 并行执行**

```typescript
// src/lib/ai-agent/execution/parallel-tool-executor.ts
interface ToolDependency {
  toolId: string;
  dependsOn: string[]; // 依赖的其他工具 ID
}

interface ToolExecutionPlan {
  levels: ToolCall[][]; // 按层级组织，同层可并行
}

class ParallelToolExecutor {
  // 构建依赖图
  private buildDependencyGraph(tools: ToolCall[]): Map<string, string[]> {
    const graph = new Map<string, string[]>();

    for (const tool of tools) {
      const deps = this.analyzeDependencies(tool, tools);
      graph.set(tool.id, deps);
    }

    return graph;
  }

  // 分析工具间的依赖关系
  private analyzeDependencies(tool: ToolCall, allTools: ToolCall[]): string[] {
    const deps: string[] = [];

    // 检查参数引用
    for (const [key, value] of Object.entries(tool.args)) {
      if (typeof value === "string" && value.startsWith("$result.")) {
        const refToolId = value.split(".")[1];
        if (allTools.some((t) => t.id === refToolId)) {
          deps.push(refToolId);
        }
      }
    }

    return deps;
  }

  // 拓扑排序，生成执行计划
  private createExecutionPlan(tools: ToolCall[]): ToolExecutionPlan {
    const graph = this.buildDependencyGraph(tools);
    const levels: ToolCall[][] = [];
    const executed = new Set<string>();

    while (executed.size < tools.length) {
      const currentLevel: ToolCall[] = [];

      for (const tool of tools) {
        if (executed.has(tool.id)) continue;

        const deps = graph.get(tool.id) || [];
        if (deps.every((dep) => executed.has(dep))) {
          currentLevel.push(tool);
        }
      }

      if (currentLevel.length === 0) {
        throw new Error("Circular dependency detected");
      }

      levels.push(currentLevel);
      currentLevel.forEach((t) => executed.add(t.id));
    }

    return { levels };
  }

  // 并行执行
  async execute(
    tools: ToolCall[],
    context: ExecutionContext,
  ): Promise<Map<string, ToolResult>> {
    const plan = this.createExecutionPlan(tools);
    const results = new Map<string, ToolResult>();

    for (const level of plan.levels) {
      // 同一层级的工具并行执行
      const levelResults = await Promise.all(
        level.map(async (tool) => {
          // 替换参数中的结果引用
          const resolvedArgs = this.resolveArguments(tool.args, results);
          const result = await this.executeTool(
            tool.name,
            resolvedArgs,
            context,
          );
          return { toolId: tool.id, result };
        }),
      );

      for (const { toolId, result } of levelResults) {
        results.set(toolId, result);
      }
    }

    return results;
  }

  private resolveArguments(
    args: Record<string, unknown>,
    results: Map<string, ToolResult>,
  ): Record<string, unknown> {
    const resolved: Record<string, unknown> = {};

    for (const [key, value] of Object.entries(args)) {
      if (typeof value === "string" && value.startsWith("$result.")) {
        const [, toolId, ...path] = value.split(".");
        const toolResult = results.get(toolId);
        resolved[key] = path.reduce((obj, p) => obj?.[p], toolResult?.data);
      } else {
        resolved[key] = value;
      }
    }

    return resolved;
  }
}
```

```typescript
    const channel = this.supabase
      .channel(`user-context:${userId}`)
      .on(
        "postgres_changes",
        {
          event: "*",
          schema: "public",
          table: "bookmarks",
          filter: `user_id=eq.${userId}`,
        },
        (payload) => this.handleBookmarkChange(userId, payload),
      )
      .on(
        "postgres_changes",
        {
          event: "*",
          schema: "public",
          table: "folders",
          filter: `user_id=eq.${userId}`,
        },
        (payload) => this.handleFolderChange(userId, payload),
      )
      .on(
        "postgres_changes",
        {
          event: "*",
          schema: "public",
          table: "bookmark_tags",
          filter: `user_id=eq.${userId}`,
        },
        (payload) => this.handleTagChange(userId, payload),
      )
      .subscribe();

    this.subscriptions.set(userId, channel);
  }

  // 获取上下文（优先从缓存）
  async getContext(userId: string): Promise<CachedUserContext> {
    let context = this.cache.get(userId);

    if (!context) {
      // 缓存未命中，全量加载并订阅
      await this.initializeUser(userId);
      context = this.cache.get(userId)!;
    }

    return context;
  }

  // 处理书签变更（增量更新）
  private handleBookmarkChange(
    userId: string,
    payload: RealtimePostgresChangesPayload<Bookmark>,
  ): void {
    const context = this.cache.get(userId);
    if (!context) return;

    switch (payload.eventType) {
      case "INSERT":
        context.stats.total++;
        context.stats.thisWeek++;
        context.stats.today++;
        break;
      case "DELETE":
        context.stats.total--;
        // 需要更精确的统计逻辑
        break;
      case "UPDATE":
        // 标签或文件夹变更可能影响统计
        break;
    }

    context.lastUpdated = Date.now();
    this.cache.set(userId, context);
  }

  // 处理文件夹变更
  private handleFolderChange(
    userId: string,
    payload: RealtimePostgresChangesPayload<Folder>,
  ): void {
    const context = this.cache.get(userId);
    if (!context) return;

    // 重建文件夹树（文件夹变更不频繁，可以全量刷新）
    this.refreshFolderTree(userId).then((tree) => {
      context.folderTree = tree;
      context.lastUpdated = Date.now();
      this.cache.set(userId, context);
    });
  }

  // 处理标签变更
  private handleTagChange(
    userId: string,
    payload: RealtimePostgresChangesPayload<BookmarkTag>,
  ): void {
    const context = this.cache.get(userId);
    if (!context) return;

    // 更新最近使用的标签
    if (payload.eventType === "INSERT") {
      const tagName = payload.new.tag_name;
      context.recentTags = [
        tagName,
        ...context.recentTags.filter((t) => t !== tagName),
      ].slice(0, 20);
    }

    context.lastUpdated = Date.now();
    this.cache.set(userId, context);
  }

  cleanup(userId: string): void {
    const channel = this.subscriptions.get(userId);
    if (channel) {
      this.supabase.removeChannel(channel);
      this.subscriptions.delete(userId);
    }
    this.cache.delete(userId);
  }

    const [stats, recentTags, folderTree, settings] = await Promise.all([
      this.loadStats(userId),
      this.loadRecentTags(userId),
      this.loadFolderTree(userId),
```

### 3.3 🚀 流式工具调用 + 投机执行

**问题**：用户需要等待完整响应才能看到结果。

**建议：边生成边执行 + 投机性预执行**

```typescript
// src/lib/ai-agent/execution/speculative-executor.ts
interface ToolIntent {
  toolName: string;
  confidence: number;
  partialArgs: Record<string, unknown>;
}

interface SpeculativeResult {
  toolId: string;
  result: ToolResult;
  isValid: boolean; // 最终验证后是否有效
}

class SpeculativeToolExecutor {
  private predictions: Map<string, Promise<SpeculativeResult>> = new Map();

  // 工具检测模式
  private readonly TOOL_PATTERNS = [
    {
      pattern: /搜索|查找|找一下/,
      tools: ["search-bookmarks", "semantic-search"],
    },
    {
      pattern: /统计|分析|分布/,
      tools: ["get-bookmark-stats", "get-tag-stats"],
    },
    {
      pattern: /整理|分类|清理/,
      tools: ["batch-organize", "auto-classify-inbox"],
    },
    { pattern: /标签|打标/, tools: ["add-tags", "batch-tags"] },
  ];

  async *executeWithSpeculation(
    input: string,
    stream: AsyncGenerator<LLMChunk>,
    context: ExecutionContext,
  ): AsyncGenerator<ExecutionChunk> {
    // 1. 基于用户输入预测可能的工具
    const predictedTools = this.predictTools(input);

    // 2. 对只读工具进行投机执行
    for (const tool of predictedTools) {
      if (this.isReadOnlyTool(tool.toolName)) {
        const prediction = this.speculativeExecute(tool, context);
        this.predictions.set(tool.toolName, prediction);
      }
    }

    // 3. 处理 LLM 流式输出
    const buffer: string[] = [];
    let confirmedTools: ToolCall[] = [];

    for await (const chunk of stream) {
      buffer.push(chunk.text);
      yield { type: "text", data: chunk.text };

      // 检测工具调用确认
      const accumulated = buffer.join("");
      const detectedCalls = this.detectToolCalls(accumulated);

      for (const call of detectedCalls) {
        if (!confirmedTools.some((c) => c.id === call.id)) {
          confirmedTools.push(call);

          // 检查是否有投机执行结果
          const speculation = this.predictions.get(call.name);
          if (speculation) {
            const result = await speculation;
            if (this.validateSpeculation(result, call)) {
              // 投机执行结果有效，直接使用
              yield {
                type: "tool_result",
                data: { ...result.result, speculative: true },
              };
              continue;
            }
          }

          // 没有投机结果或验证失败，正常执行
          const result = await this.executeTool(call, context);
          yield { type: "tool_result", data: result };
        }
      }
    }
  }

  // 预测可能使用的工具
  private predictTools(input: string): ToolIntent[] {
    const predictions: ToolIntent[] = [];

    for (const { pattern, tools } of this.TOOL_PATTERNS) {
      if (pattern.test(input)) {
        for (const toolName of tools) {
          predictions.push({
            toolName,
            confidence: 0.7,
            partialArgs: this.extractPartialArgs(input, toolName),
          });
        }
      }
    }

    return predictions.sort((a, b) => b.confidence - a.confidence).slice(0, 2);
  }

  // 投机执行（仅只读工具）
  private async speculativeExecute(
    intent: ToolIntent,
    context: ExecutionContext,
  ): Promise<SpeculativeResult> {
    try {
      const result = await this.executeTool(
        {
          id: `speculative-${intent.toolName}`,
          name: intent.toolName,
          args: intent.partialArgs,
        },
        context,
      );

      return { toolId: intent.toolName, result, isValid: true };
    } catch (error) {
      return { toolId: intent.toolName, result: null, isValid: false };
    }
  }

  // 验证投机结果是否与实际调用匹配
  private validateSpeculation(
    speculation: SpeculativeResult,
    actualCall: ToolCall,
  ): boolean {
    if (!speculation.isValid) return false;

    // 检查参数是否匹配
    const specArgs = JSON.stringify(speculation.partialArgs);
    const actualArgs = JSON.stringify(actualCall.args);

    // 允许一定的参数差异（如分页参数）
    return (
      specArgs === actualArgs ||
      this.argsCompatible(speculation.partialArgs, actualCall.args)
    );
  }

  // 只读工具白名单
  private isReadOnlyTool(toolName: string): boolean {
    const readOnlyTools = [
      "search-bookmarks",
      "semantic-search",
      "list-bookmarks",
      "get-bookmark",
      "get-bookmark-stats",
      "get-tag-stats",
      "get-folder-tree",
      "count-bookmarks",
    ];
    return readOnlyTools.includes(toolName);
  }
}
```

```typescript
// src/lib/ai-agent/model/adaptive-model-selector.ts
interface ModelMetrics {
  latencyP50: number; // 50分位延迟
  latencyP95: number; // 95分位延迟
  latencyP99: number; // 99分位延迟
  errorRate: number; // 错误率 (0-1)
  qualityScore: number; // 质量评分 (0-1，基于用户反馈)
  costPerToken: number; // 每 token 成本
  currentLoad: number; // 当前负载 (0-1)
  availability: number; // 可用性 (0-1)
}

interface SelectionConstraints {
  maxLatency: number; // 最大可接受延迟 (ms)
  maxCost: number; // 最大成本预算
  minQuality: number; // 最低质量要求

  // 权重 (总和 = 1)
  latencyWeight: number;
  costWeight: number;
  qualityWeight: number;
}

class AdaptiveModelSelector {
  private metrics: Map<string, ModelMetrics> = new Map();
  private metricsWindow: number = 5 * 60 * 1000; // 5分钟滑动窗口

  // 模型池配置
  private readonly MODEL_POOL = {
    fast: ["deepseek-chat", "gpt-4o-mini", "claude-3-haiku"],
    standard: ["gpt-4o", "claude-3-5-sonnet"],
    deep: ["gpt-4-turbo", "claude-3-opus"],
  };

  // 选择最优模型
  selectModel(
    tier: "fast" | "standard" | "deep",
    constraints: SelectionConstraints,
  ): string {
    const candidates = this.MODEL_POOL[tier];

    // 过滤不满足硬约束的模型
    const eligible = candidates.filter((model) => {
      const m = this.getMetrics(model);
      return (
        m.latencyP95 <= constraints.maxLatency &&
        m.errorRate < 0.05 &&
        m.availability > 0.95
      );
    });

    if (eligible.length === 0) {
      // 降级到下一层
      return this.fallbackModel(tier);
    }

    // 多目标优化评分
    return eligible
      .map((model) => ({
        model,
        score: this.calculateScore(model, constraints),
      }))
      .sort((a, b) => b.score - a.score)[0].model;
  }

  // 计算综合评分
  private calculateScore(
    model: string,
    constraints: SelectionConstraints,
  ): number {
    const m = this.getMetrics(model);

    // 归一化各指标到 0-1
    const normalizedLatency = 1 - m.latencyP50 / constraints.maxLatency;
    const normalizedCost = 1 - m.costPerToken / constraints.maxCost;
    const normalizedQuality = m.qualityScore;

    return (
      constraints.latencyWeight * normalizedLatency +
      constraints.costWeight * normalizedCost +
      constraints.qualityWeight * normalizedQuality
    );
  }

  // 降级策略
  private fallbackModel(tier: "fast" | "standard" | "deep"): string {
    const fallbackChain = {
      deep: "standard",
      standard: "fast",
      fast: "fast",
    };

    const fallbackTier = fallbackChain[tier];
    if (fallbackTier === tier) {
      // 已经是最低层，返回默认模型
      return "deepseek-chat";
    }

    return this.selectModel(fallbackTier as any, {
      maxLatency: 5000,
      maxCost: 0.01,
      minQuality: 0.7,
      latencyWeight: 0.5,
      costWeight: 0.3,
      qualityWeight: 0.2,
    });
  }

  // 记录指标
  recordMetric(
    model: string,
    latency: number,
    success: boolean,
    userFeedback?: "positive" | "negative",
  ): void {
    const current = this.metrics.get(model) || this.defaultMetrics();

    // 使用指数移动平均更新
    const alpha = 0.1;
    current.latencyP50 = current.latencyP50 * (1 - alpha) + latency * alpha;
    current.errorRate =
      current.errorRate * (1 - alpha) + (success ? 0 : 1) * alpha;

    if (userFeedback) {
      const feedbackValue = userFeedback === "positive" ? 1 : 0;
      current.qualityScore =
        current.qualityScore * (1 - alpha) + feedbackValue * alpha;
    }

    this.metrics.set(model, current);
  }

  private getMetrics(model: string): ModelMetrics {
    return this.metrics.get(model) || this.defaultMetrics();
  }

  private defaultMetrics(): ModelMetrics {
    return {
      latencyP50: 1000,
      latencyP95: 2000,
      latencyP99: 3000,
      errorRate: 0.01,
      qualityScore: 0.8,
      costPerToken: 0.001,
      currentLoad: 0.5,
      availability: 0.99,
    };
  }
}
```
