```typescript
// 图片持久化策略
interface ImagePersistenceStrategy {
  // 1. 优先尝试上传到 Supabase Storage
```

#### 实现流程

```
微信文章正文
     ↓

     ↓

     ↓

     ↓

     ↓
存储映射关系到 images JSONB 字段
```

### 3.3 正文抓取方案

#### 微信文章 HTML 结构

```html
<!-- 正文容器 -->
<div id="js_content" class="rich_media_content">
  <p>段落内容...</p>
  <section>
    <p>更多内容...</p>
    <img src="https://mmbiz.qpic.cn/..." />
  </section>
</div>
```

#### 抓取逻辑

```typescript
// src/lib/utils/wechat-content-extractor.ts

interface WechatArticleContent {
  html: string; // 原始 HTML
  text: string; // 纯文本
  markdown: string; // Markdown
  images: ImageMeta[];
}

async function extractWechatContent(
  html: string,
): Promise<WechatArticleContent> {
  // 1. 定位正文容器
  const contentMatch = html.match(
    /<div[^>]*id="js_content"[^>]*>([\s\S]*?)<\/div>/,
  );
  if (!contentMatch) {
    throw new Error("无法找到正文内容");
  }

  const contentHtml = contentMatch[1];

  // 2. 提取所有图片
  const images = extractImages(contentHtml);

  // 3. 清理 HTML（移除脚本、样式、广告）
  const cleanHtml = cleanWechatHtml(contentHtml);

  // 4. 转换为纯文本
  const text = htmlToText(cleanHtml);

  // 5. 转换为 Markdown
  const markdown = htmlToMarkdown(cleanHtml);

  return { html: cleanHtml, text, markdown, images };
}

function extractImages(html: string): ImageMeta[] {
  const imgRegex = /<img[^>]+src="([^"]+)"[^>]*>/g;
  const images: ImageMeta[] = [];

  let match;
  while ((match = imgRegex.exec(html)) !== null) {
    const src = match[1];

    // 过滤无效图片
    if (isValidImage(src)) {
      const alt = extractAlt(match[0]);
      images.push({ original_url: src, alt });
    }
  }

  return images;
}

function isValidImage(src: string): boolean {
  // 排除条件
  const excludePatterns = [
    /wx_fmt=gif/, // 表情包
    /tp=webp.*wxfrom=5/, // 追踪像素
    /res\.wx\.qq\.com/, // 微信资源
    /mmbiz\.qlogo\.cn/, // 头像
  ];

  // 尺寸检查（通过 URL 参数）
  const widthMatch = src.match(/wx_fmt=\w+.*?(\d+)/);
  if (widthMatch && parseInt(widthMatch[1]) < 100) {
    return false;
  }

  return !excludePatterns.some((p) => p.test(src));
}
```

### 3.4 重新获取功能

#### API 设计

```typescript
// POST /api/bookmarks/[id]/refetch

interface RefetchRequest {
  bookmarkId: number;
  options?: {
    forceUpdate?: boolean; // 强制更新（即使内容未变）
    updateImages?: boolean; // 重新下载图片
  };
}

interface RefetchResponse {
  success: boolean;
  changes: {
    titleChanged: boolean;
    contentChanged: boolean;
    imagesUpdated: number;
  };
  newVersion: number;
}
```

#### 实现逻辑

```typescript
async function refetchBookmark(
  id: number,
  options: RefetchOptions,
): Promise<RefetchResponse> {
  // 1. 获取现有记录
  const bookmark = await getBookmark(id);

  // 2. 重新抓取内容
  const newContent = await crawlWechatArticle(bookmark.url);

  // 3. 对比变化
  const changes = compareContent(bookmark, newContent);

  // 4. 更新数据库
  if (changes.hasChanges || options.forceUpdate) {
    await updateBookmark(id, {
      ...newContent,
      content_version: bookmark.content_version + 1,
      crawled_at: new Date(),
    });
  }

  // 5. 重新处理图片（如果需要）
  if (options.updateImages) {
    await persistImages(id, newContent.images);
  }

  return {
    success: true,
    changes,
    newVersion: bookmark.content_version + 1,
  };
}
```

## 五、技术细节

### 5.1 HTML 清理规则

```typescript
function cleanWechatHtml(html: string): string {
  return (
    html
      // 移除脚本
      .replace(/<script[^>]*>[\s\S]*?<\/script>/gi, "")
      // 移除样式
      .replace(/<style[^>]*>[\s\S]*?<\/style>/gi, "")
      // 移除微信特定广告
      .replace(/<div[^>]*class="[^"]*ad[^"]*"[^>]*>[\s\S]*?<\/div>/gi, "")
      // 移除空段落
      .replace(/<p[^>]*>\s*<\/p>/gi, "")
      // 移除微信赞赏按钮
      .replace(/<mp-common-product[^>]*>[\s\S]*?<\/mp-common-product>/gi, "")
      // 标准化图片标签
      .replace(/data-src=/g, "src=")
  );
}
```

### 5.3 图片并发下载

```typescript
async function persistImages(
  images: ImageMeta[],
  options: { concurrency: number; timeout: number },
): Promise<PersistedImage[]> {
  const results: PersistedImage[] = [];

  // 使用 p-limit 控制并发
  const limit = pLimit(options.concurrency);

  const tasks = images.map((img, index) =>
    limit(async () => {
      try {
        // 1. 下载图片
        const response = await fetch(img.original_url, {
          signal: AbortSignal.timeout(options.timeout),
        });

        if (!response.ok) throw new Error(`HTTP ${response.status}`);

        const buffer = await response.arrayBuffer();

        // 2. 上传到 Supabase Storage
        const fileName = `wechat/${Date.now()}_${index}.${getExtension(img.original_url)}`;
        const { data, error } = await supabase.storage
          .from("article-images")
          .upload(fileName, buffer, {
            contentType: response.headers.get("content-type") || "image/jpeg",
          });

        if (error) throw error;

        // 3. 获取公开 URL
        const {
          data: { publicUrl },
        } = supabase.storage.from("article-images").getPublicUrl(fileName);

        return {
          original_url: img.original_url,
          persisted_url: publicUrl,
          alt: img.alt,
          status: "success",
        };
      } catch (error) {
        // 降级到代理服务
        return {
          original_url: img.original_url,
          persisted_url: `https://images.weserv.nl/?url=${encodeURIComponent(img.original_url)}`,
          alt: img.alt,
          status: "fallback",
          error: error.message,
        };
      }
    }),
  );

  return Promise.all(tasks);
}
```
