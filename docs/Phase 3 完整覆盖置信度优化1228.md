## 三、置信度优化系统

### 3.1 多信号融合算法

```typescript
// src/lib/ai-agent/inference/confidence-calculator.ts

export interface ConfidenceCalculator {
  /**
   * 计算综合置信度
   *
   * 公式：C = Σ(Si × Wi × Di) / Σ(Wi)
   *
   * Si = 信号强度 (0-1)
   * Wi = 信号权重 (0-1)
   * Di = 时序衰减因子 (0-1)
   */
  calculate(signals: InferenceSignal[]): ConfidenceResult;
}

export interface InferenceSignal {
  type: SignalType;
  value: number; // 原始强度 0-1
  weight: number; // 信号权重 0-1
  timestamp: Date; // 信号时间
  source: SignalSource; // 信号来源
}

export type SignalType =
  | "keyword_match" // 关键词匹配
  | "platform_affinity" // 平台偏好
  | "tag_pattern" // 标签模式
  | "content_type" // 内容类型
  | "behavior_frequency" // 行为频率
  | "temporal_pattern" // 时间模式
  | "explicit_feedback"; // 显式反馈

export interface ConfidenceResult {
  persona: PersonaType;
  confidence: number; // 综合置信度 0-1
  breakdown: {
    signal: SignalType;
    contribution: number; // 该信号贡献度
  }[];
  reliability: "high" | "medium" | "low"; // 可靠性评估
}
```

### 3.2 时序衰减模型

```typescript
// src/lib/ai-agent/inference/temporal-decay.ts

/**
 * 时序衰减函数
 *
 * 使用指数衰减模型：D(t) = e^(-λt)
 *
 * 其中：
 * - t = 距离当前的天数
 * - λ = 衰减常数（默认 0.05，半衰期约 14 天）
 */
export class TemporalDecayCalculator {
  private readonly decayConstant: number;
  private readonly minDecay: number;

  constructor(config: DecayConfig = {}) {
    this.decayConstant = config.lambda ?? 0.05;
    this.minDecay = config.minDecay ?? 0.1; // 最低保留 10%
  }

  /**
   * 计算衰减因子
   */
  calculate(timestamp: Date): number {
    const now = new Date();
    const daysDiff =
      (now.getTime() - timestamp.getTime()) / (1000 * 60 * 60 * 24);
    const decay = Math.exp(-this.decayConstant * daysDiff);
    return Math.max(decay, this.minDecay);
  }

  /**
   * 批量计算带权重的信号
   */
  applyToSignals(signals: InferenceSignal[]): WeightedSignal[] {
    return signals.map((signal) => ({
      ...signal,
      decayFactor: this.calculate(signal.timestamp),
      effectiveWeight: signal.weight * this.calculate(signal.timestamp),
    }));
  }
}

// 衰减曲线参考
// Day 0:  100%
// Day 7:  70%
// Day 14: 50%
// Day 30: 22%
// Day 60: 5%
```

### 3.3 反馈学习机制

```typescript
// src/lib/ai-agent/inference/feedback-learner.ts

export interface FeedbackLearner {
  /**
   * 记录用户反馈
   */
  recordFeedback(feedback: UserFeedback): Promise<void>;

  /**
   * 基于反馈调整权重
   */
  adjustWeights(personaId: PersonaType): Promise<WeightAdjustment>;
}

export interface UserFeedback {
  type: "confirm" | "reject" | "switch"; // 确认/拒绝/切换
  inferredPersona: PersonaType; // 推断的画像
  actualPersona?: PersonaType; // 用户选择的画像（switch 时）
  context: {
    triggeredBy: SignalType[]; // 触发推断的信号
    confidence: number; // 推断时的置信度
  };
  timestamp: Date;
}

export interface WeightAdjustment {
  personaId: PersonaType;
  adjustments: {
    signal: SignalType;
    oldWeight: number;
    newWeight: number;
    reason: string;
  }[];
}

/**
 * 权重调整策略
 *
 * - confirm: 小幅增加触发信号权重 (+5%)
 * - reject: 降低触发信号权重 (-10%)
 * - switch: 降低原画像信号权重，增加目标画像信号权重
 *
 * 调整幅度受置信度影响：
 * - 高置信度误判 → 大幅调整
 * - 低置信度误判 → 小幅调整
 */
```

describe('ConfidenceCalculator', () => {
  describe('多信号融合', () => {
    it('should correctly weight multiple signals', () => {
      const signals = [
        { type: 'keyword_match', value: 0.9, weight: 1.0, ... },
        { type: 'platform_affinity', value: 0.7, weight: 0.8, ... },
        { type: 'behavior_frequency', value: 0.5, weight: 0.6, ... },
      ];
      const result = calculator.calculate(signals);
      expect(result.confidence).toBeCloseTo(0.73, 2);
    });

    it('should apply temporal decay correctly', () => {
      // 14 天前的信号应衰减约 50%
      const oldSignal = { ..., timestamp: daysAgo(14) };
      const result = calculator.calculate([oldSignal]);
      expect(result.confidence).toBeLessThan(oldSignal.value * 0.6);
    });
  });

  describe('画像推断', () => {
    it.each([
      ['security_researcher', securityKeywords, 0.85],
      ['architect', architectKeywords, 0.80],
      ['language_learner', languageKeywords, 0.75],
    ])('should infer %s persona with confidence >= %f',
      (persona, keywords, minConfidence) => {
        const result = inferrer.infer(keywords);
        expect(result.persona).toBe(persona);
        expect(result.confidence).toBeGreaterThanOrEqual(minConfidence);
      }
    );
  });
});
