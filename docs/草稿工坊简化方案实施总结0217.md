**变更前**:

- 页面 1: `/knowledge-studio/v2` - 选择蓝图和素材
- 页面 2: `/knowledge-studio/v2/[blueprintId]` - Briefing、大纲、生成、结果

### SSE 流式生成（保留原有逻辑）

```typescript
const startGeneration = useCallback(
  async (outlineData?: BlueprintOutline) => {
    // ... 准备工作 ...

    const response = await fetch("/api/knowledge-studio/v2/generate", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        blueprintId: selectedBlueprintId,
        materials: selectedMaterials,
        briefingContext, // 从补充说明构建
        outline: outlineData,
        targetLength: "medium",
        platformStyle,
      }),
      signal: abortControllerRef.current.signal,
    });

    const reader = response.body?.getReader();
    const decoder = new TextDecoder();
    let contentBuffer = "";

    while (true) {
      const { done, value } = await reader.read();
      if (done) break;

      const text = decoder.decode(value);
      const lines = text.split("\n");

      for (const line of lines) {
        if (line.startsWith("data: ")) {
          const data = JSON.parse(line.slice(6));

          switch (data.type) {
            case "progress":
              setProgress({ step: data.data.step, percent: data.data.percent });
              break;
            case "content":
              contentBuffer += data.data.text;
              setGeneratedContent(contentBuffer);
              break;
            case "done":
              setReportId(data.data.reportId);
              setCurrentStep("result");
              break;
          }
        }
      }
    }
  },
  [selectedBlueprintId, selectedMaterials, platformStyle],
);
```
