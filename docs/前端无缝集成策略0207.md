```typescript
// use-notification-count.ts
export function useNotificationCount() {
  const supabase = createBrowserSupabaseClient();

  return useQuery({
    queryKey: ["notification-count"],
    queryFn: async () => {
      const { data } = await supabase.rpc("get_unread_notification_count");
      return data ?? 0;
    },
    refetchInterval: 30000, // 30秒轮询
  });
}
```

```tsx
// folder-heat-indicator.tsx 新建
"use client";

import { Flame } from "lucide-react";
import {
  Tooltip,
  TooltipContent,
  TooltipTrigger,
} from "@/components/ui/tooltip";
import { cn } from "@/lib/utils";

interface FolderHeatIndicatorProps {
  heat: number; // 0-100
  size?: "sm" | "md" | "lg";
  showTooltip?: boolean;
}

function getHeatColor(heat: number): string {
  if (heat < 20) return "text-blue-400";
  if (heat < 40) return "text-cyan-400";
  if (heat < 60) return "text-yellow-400";
  if (heat < 80) return "text-orange-400";
  return "text-red-400";
}

function getHeatLabel(heat: number): string {
  if (heat < 20) return "冷门";
  if (heat < 40) return "较少访问";
  if (heat < 60) return "一般";
  if (heat < 80) return "活跃";
  return "热门";
}

export function FolderHeatIndicator({
  heat,
  size = "sm",
  showTooltip = true,
}: FolderHeatIndicatorProps) {
  const sizeClasses = {
    sm: "h-3 w-3",
    md: "h-4 w-4",
    lg: "h-5 w-5",
  };

  const indicator = (
    <Flame className={cn(sizeClasses[size], getHeatColor(heat))} />
  );

  if (!showTooltip) return indicator;

  return (
    <Tooltip>
      <TooltipTrigger asChild>{indicator}</TooltipTrigger>
      <TooltipContent>
        <p>
          热度: {heat}% ({getHeatLabel(heat)})
        </p>
        <p className="text-xs text-muted-foreground">基于近 30 天访问频率</p>
      </TooltipContent>
    </Tooltip>
  );
}
```

```typescript
// use-ai-cost-stats.ts 新建
"use client";

import { useQuery } from "@tanstack/react-query";
import { createBrowserSupabaseClient } from "@/lib/supabase/client";

export function useAiCostStats() {
  const supabase = createBrowserSupabaseClient();

  return useQuery({
    queryKey: ["ai-cost-stats"],
    queryFn: async () => {
      const [dailyCost, chatStats, quotaStats, entitlement] = await Promise.all(
        [
          supabase.rpc("get_user_daily_llm_cost", { p_days: 30 }),
          supabase.rpc("get_user_chat_stats"),
          supabase.rpc("get_unified_quota_stats"),
          supabase.rpc("get_user_entitlement"),
        ],
      );

      return {
        dailyCost: dailyCost.data?.today ?? 0,
        monthlyCost: dailyCost.data?.month ?? 0,
        chatStats: chatStats.data ?? { today: 0, month: 0 },
        quotaStats: quotaStats.data ?? { used: 0, total: 100000 },
        entitlement: entitlement.data,
      };
    },
  });
}
```

```tsx
"use client";

import { useKnowledgeGraph } from "@/hooks/domain/knowledge/use-knowledge-graph";
import { GraphCanvas } from "./graph-canvas";
import { GraphControls } from "./graph-controls";

interface KnowledgeGraphPanelProps {
  selectedMaterials: number[];
  onNodeClick: (nodeId: number) => void;
  maxNodes?: number;
}

export function KnowledgeGraphPanel({
  selectedMaterials,
  onNodeClick,
  maxNodes = 100,
}: KnowledgeGraphPanelProps) {
  const {
    nodes,
    edges,
    isLoading,
    focusNode,
    setFocusNode,
    zoomLevel,
    setZoomLevel,
  } = useKnowledgeGraph({ maxNodes });

  if (isLoading) {
    return (
      <div className="h-[400px] flex items-center justify-center">
        <Loader className="h-8 w-8 animate-spin" />
      </div>
    );
  }

  if (nodes.length === 0) {

      <div className="h-[400px] flex items-center justify-center text-muted-foreground">
        <p>暂无足够数据生成知识图谱</p>
      </div>
    );
  }

  return (
    <div className="relative h-[400px] border rounded-lg overflow-hidden">
      <GraphControls
        zoomLevel={zoomLevel}
        onZoomChange={setZoomLevel}
        onReset={() => setFocusNode(null)}
      />

      <GraphCanvas
        nodes={nodes}
        edges={edges}
        selectedNodes={selectedMaterials}
        focusNode={focusNode}
        zoomLevel={zoomLevel}
        onNodeClick={onNodeClick}
        onNodeFocus={setFocusNode}
      />

      {nodes.length >= maxNodes && (
        <div className="absolute bottom-2 left-2 text-xs text-muted-foreground bg-background/80 px-2 py-1 rounded">
```
