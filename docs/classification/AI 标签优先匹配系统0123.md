**问题**：书签的 `ai_tags` 字段已传入管道，但 **没有任何匹配器** 将其与文件夹名称/标签进行直接匹配。

## 二、设计目标

| 目标           | 说明                                          |
| -------------- | --------------------------------------------- |
| **最高优先级** | AI 标签匹配放在 L0 最前端，优先于规则匹配     |
| **高置信度**   | 精确匹配时置信度 ≥ 0.95，直接分类无需后续流程 |
| **零延迟**     | 纯内存计算，<10ms 完成匹配                    |
| **自学习**     | 基于文件夹内书签自动聚合标签画像              |
| **降级兼容**   | 未匹配时无缝传入 T0 向量检索                  |

```typescript
export interface FolderTagProfile {

  folderName: string;
```

### 4.3 AI 标签匹配器 (新增核心模块)

```typescript
// ai-tag-matcher.ts (新增���件)

export class AiTagMatcher {
  private config: TagMatcherConfig;
  private folderProfiles: Map<string, FolderTagProfile[]>; // userId -> profiles

  constructor(
    private logger: LoggerService,
    private supabase: SupabaseClient,
    config?: Partial<TagMatcherConfig>,
  ) {
    this.config = {
      exactMatchConfidence: config?.exactMatchConfidence ?? 0.98,
      coreTagConfidence: config?.coreTagConfidence ?? 0.92,
      relatedTagConfidence: config?.relatedTagConfidence ?? 0.85,
      minMatchingTags: config?.minMatchingTags ?? 1,
      minCoreTagHits: config?.minCoreTagHits ?? 2,
    };
    this.folderProfiles = new Map();
  }

  /**
   * 主匹配方法 - 在 L0 最前端调用
   */
  async match(request: ClassificationRequest): Promise<TagMatchResult | null> {
    if (!request.tags || request.tags.length === 0) {
      return null; // 无标签，跳过
    }

    const profiles = await this.getFolderProfiles(request.userId);
    if (profiles.length === 0) {
      return null; // 无文件夹画像，跳过
    }

    const normalizedTags = request.tags.map((t) => t.toLowerCase().trim());

    // 策略 1：标签 ↔ 文件夹名精确匹配（最高优先级）
    const exactMatch = this.matchExactFolderName(normalizedTags, profiles);
    if (exactMatch) {
      this.logger.info(
        {
          bookmarkId: request.bookmarkId,
          matchType: "exact_name",
          ...exactMatch,
        },
        "AI标签精确匹配成功",
      );
      return exactMatch;
    }

    // 策略 2：标签 ↔ 文件夹核心标签匹配
    const coreMatch = this.matchCoreTags(normalizedTags, profiles);
    if (coreMatch) {
      this.logger.info(
        {
          bookmarkId: request.bookmarkId,
          matchType: "core_tag",
          ...coreMatch,
        },
        "AI标签核心标签匹配成功",
      );
      return coreMatch;
    }

    // 策略 3：标签 ↔ 文件夹相关标签匹配
    const relatedMatch = this.matchRelatedTags(normalizedTags, profiles);
    if (relatedMatch) {
      this.logger.info(
        {
          bookmarkId: request.bookmarkId,
          matchType: "related_tag",
          ...relatedMatch,
        },
        "AI标签相关标签匹配成功",
      );
      return relatedMatch;
    }

    return null; // 无匹配，继续后续流程
  }

  /**
   * 策略 1：标签与文件夹名精确匹配
   * - AI 标签 "独立开发与产品策略" 匹配文件夹 "独立开发与产品策略"
   * - AI 标签 "前端开发" 匹配文件夹 "前端开发"
   */
  private matchExactFolderName(
    tags: string[],
    profiles: FolderTagProfile[],
  ): TagMatchResult | null {
    for (const profile of profiles) {
      const folderNameLower = profile.folderName.toLowerCase();

      for (const tag of tags) {
        // 完全匹配
        if (tag === folderNameLower) {
          return {
            matched: true,
            folderId: profile.folderId.toString(),
            folderName: profile.folderName,
            confidence: this.config.exactMatchConfidence,
            matchType: "exact_name",
            matchedTags: [tag],
            reason: `标签 "${tag}" 与文件夹名完全匹配`,
          };
        }

        // 包含匹配（标签包含文件夹名，或文件夹名包含标签）
        if (tag.length >= 4 && folderNameLower.length >= 4) {
          if (tag.includes(folderNameLower) || folderNameLower.includes(tag)) {
            return {
              matched: true,
              folderId: profile.folderId.toString(),
              folderName: profile.folderName,
              confidence: this.config.exactMatchConfidence - 0.03,
              matchType: "exact_name",
              matchedTags: [tag],
              reason: `标签 "${tag}" 与文件夹名 "${profile.folderName}" 高度相关`,
            };
          }
        }
      }
    }
    return null;
  }

  /**
   * 策略 2：匹配文件夹核心标签
   */
  private matchCoreTags(
    tags: string[],
    profiles: FolderTagProfile[],
  ): TagMatchResult | null {
    let bestMatch: TagMatchResult | null = null;
    let bestScore = 0;

    for (const profile of profiles) {
      if (profile.coreTags.length === 0) continue;

      const matchedTags = tags.filter((t) =>
        profile.coreTags.some(
          (ct) =>
            ct.toLowerCase() === t ||
            ct.toLowerCase().includes(t) ||
            t.includes(ct.toLowerCase()),
        ),
      );

      if (matchedTags.length >= this.config.minMatchingTags) {
        // 计算匹配分数：匹配数 / 总核心标签数
        const score = matchedTags.length / Math.max(profile.coreTags.length, 1);

        if (score > bestScore) {
          bestScore = score;
          const confidence =
            matchedTags.length >= this.config.minCoreTagHits
              ? this.config.coreTagConfidence
              : this.config.coreTagConfidence - 0.05;

          bestMatch = {
            matched: true,
            folderId: profile.folderId.toString(),
            folderName: profile.folderName,
            confidence,
            matchType: "core_tag",
            matchedTags,
            reason: `匹配 ${matchedTags.length} 个核心标签: ${matchedTags.join(", ")}`,
          };
        }
      }
    }
    return bestMatch;
  }

  /**
   * 策略 3：匹配文件夹相关标签
   */
  private matchRelatedTags(
    tags: string[],
    profiles: FolderTagProfile[],
  ): TagMatchResult | null {
    let bestMatch: TagMatchResult | null = null;
    let bestScore = 0;

    for (const profile of profiles) {
      if (profile.relatedTags.length === 0) continue;

      const matchedTags = tags.filter((t) =>
        profile.relatedTags.some(
          (rt) =>
            rt.toLowerCase() === t ||
            rt.toLowerCase().includes(t) ||
            t.includes(rt.toLowerCase()),
        ),
      );

      // 相关标签需要更多命中数
      if (matchedTags.length >= 3) {
        const score =
          matchedTags.length / Math.max(profile.relatedTags.length, 1);

        if (score > bestScore) {
          bestScore = score;
          bestMatch = {
            matched: true,
            folderId: profile.folderId.toString(),
            folderName: profile.folderName,
            confidence: this.config.relatedTagConfidence,
            matchType: "related_tag",
            matchedTags,
            reason: `匹配 ${matchedTags.length} 个相关标签: ${matchedTags.join(", ")}`,
          };
        }
      }
    }
    return bestMatch;
  }

  /**
   * 获取用户的文件夹标签画像（带缓存）
   */
  async getFolderProfiles(userId: string): Promise<FolderTagProfile[]> {
    // 缓存检查
    if (this.folderProfiles.has(userId)) {
      return this.folderProfiles.get(userId)!;
    }

    // 从数据库加载
    const { data, error } = await this.supabase
      .from("folder_tag_profiles")
      .select(
        `
        folder_id,
        core_tags,
        related_tags,
        folder_name_tokens,
        bookmark_count,
        folders!inner(name)
      `,
      )
      .eq("user_id", userId);

    if (error || !data) {
      this.logger.warn({ error }, "加载文件夹标签画像失败");
      return [];
    }

    const profiles: FolderTagProfile[] = data.map((row) => ({
      folderId: row.folder_id,
      folderName: row.folders.name,
      coreTags: row.core_tags || [],
      relatedTags: row.related_tags || [],
      folderNameTokens: row.folder_name_tokens || [],
      bookmarkCount: row.bookmark_count,
    }));

    this.folderProfiles.set(userId, profiles);
    return profiles;
  }

  /**
   * 清除用户缓存（文件夹变更时调用）
   */
  clearCache(userId: string): void {
    this.folderProfiles.delete(userId);
  }
}
```

### 4.4 文件夹标签画像生成器 (新增)

```typescript
// folder-tag-profiler.service.ts (新增文件)

export class FolderTagProfilerService {
  constructor(
    private logger: LoggerService,
    private supabase: SupabaseClient,
  ) {}

  /**
   * 为用户所有文件夹生成/更新标签画像
   */
  async updateAllProfiles(userId: string): Promise<void> {
    const folders = await this.getUserFolders(userId);

    for (const folder of folders) {
      await this.updateFolderProfile(userId, folder.id, folder.name);
    }
  }

  /**
   * 获取用户所有文件夹
   */
  private async getUserFolders(
    userId: string,
  ): Promise<Array<{ id: number; name: string }>> {
    const { data, error } = await this.supabase
      .from("folders")
      .select("id, name")
      .eq("user_id", userId);

    if (error) {
      this.logger.error({ error }, "获取用户文件夹失败");
      return [];
    }

    return data || [];
  }

  /**
   * 更新单个文件夹的标签画像
   */
  async updateFolderProfile(
    userId: string,
    folderId: number,
    folderName: string,
  ): Promise<void> {
    // 1. 获取文件夹内所有书签的标签
    const { data: bookmarks } = await this.supabase
      .from("bookmarks")
      .select("ai_tags")
      .eq("user_id", userId)
      .eq("folder_id", folderId)
      .not("ai_tags", "is", null);

    if (!bookmarks || bookmarks.length === 0) {
      // 空文件夹：使用文件夹名分词作为核心标签
      await this.upsertProfile(userId, folderId, {
        coreTags: this.tokenizeFolderName(folderName),
        relatedTags: [],
        tagFrequency: {},
        folderNameTokens: this.tokenizeFolderName(folderName),
        bookmarkCount: 0,
      });
      return;
    }

    // 2. 统计标签频率
    const tagFrequency: Record<string, number> = {};
    for (const bookmark of bookmarks) {
      for (const tag of bookmark.ai_tags || []) {
        const normalizedTag = tag.toLowerCase().trim();
        tagFrequency[normalizedTag] = (tagFrequency[normalizedTag] || 0) + 1;
      }
    }

    // 3. 按频率排序，分为核心标签和相关标签
    const totalBookmarks = bookmarks.length;
    const sortedTags = Object.entries(tagFrequency).sort((a, b) => b[1] - a[1]);

    const coreTags: string[] = [];
    const relatedTags: string[] = [];

    for (const [tag, count] of sortedTags) {
      const frequency = count / totalBookmarks;
      if (frequency >= 0.3) {
        coreTags.push(tag);
      } else if (frequency >= 0.1) {
        relatedTags.push(tag);
      }
    }

    // 4. 将文件夹名分词加入核心标签（确保匹配）
    const folderTokens = this.tokenizeFolderName(folderName);
    for (const token of folderTokens) {
      if (!coreTags.includes(token)) {
        coreTags.unshift(token); // 添加到最前面
      }
    }

    // 5. 更新数据库
    await this.upsertProfile(userId, folderId, {
      coreTags: coreTags.slice(0, 20), // 最多 20 个核心标签
      relatedTags: relatedTags.slice(0, 30), // 最多 30 个相关标签
      tagFrequency,
      folderNameTokens: folderTokens,
      bookmarkCount: bookmarks.length,
    });
  }

  /**
   * 插入或更新文件夹标签画像
   */
  private async upsertProfile(
    userId: string,
    folderId: number,
    data: {
      coreTags: string[];
      relatedTags: string[];
      tagFrequency: Record<string, number>;
      folderNameTokens: string[];
      bookmarkCount: number;
    },
  ): Promise<void> {
    const { error } = await this.supabase.from("folder_tag_profiles").upsert(
      {
        user_id: userId,
        folder_id: folderId,
        core_tags: data.coreTags,
        related_tags: data.relatedTags,
        tag_frequency: data.tagFrequency,
        folder_name_tokens: data.folderNameTokens,
        bookmark_count: data.bookmarkCount,
        last_updated_at: new Date().toISOString(),
      },
      {
        onConflict: "user_id,folder_id",
      },
    );

    if (error) {
      this.logger.error({ error, folderId }, "更新文件夹标签画像失败");
    }
  }

  /**
   * 文件夹名分词
   * "独立开发与产品策略" → ["独立开发", "产品策略", "独立开发与产品策略"]
   */
  private tokenizeFolderName(name: string): string[] {
    const tokens: string[] = [name.toLowerCase()];

    // 按常见分隔符拆分
    const separators = ["与", "和", "/", "|", "-", "_", "·"];
    for (const sep of separators) {
      if (name.includes(sep)) {
        tokens.push(...name.split(sep).map((t) => t.trim().toLowerCase()));
      }
    }

    return [...new Set(tokens)].filter((t) => t.length >= 2);
  }
}
```

### 4.5 L0 RuleMatcher 集成

```typescript
// l0-rule-matcher.ts 修改

export class L0RuleMatcher {
  private aiTagMatcher: AiTagMatcher;

  constructor(logger: LoggerService, rules: Rule[], supabase?: SupabaseClient) {
    // ... 现有逻辑

    // 新增：初始化 AI 标签匹配器
    if (supabase) {
      this.aiTagMatcher = new AiTagMatcher(logger, supabase);
    }
  }

  /**
   * 主匹配方法（修改）
   */
  async match(request: ClassificationRequest): Promise<L0MatchResult | null> {
    // 【新增】Phase 0: AI 标签匹配（最高优先级）
    if (this.aiTagMatcher && request.tags?.length) {
      const tagMatch = await this.aiTagMatcher.match(request);
      if (tagMatch) {
        return {
          folderId: tagMatch.folderId,
          confidence: tagMatch.confidence,
          matchType: "ai_tag",
          reason: tagMatch.reason,
        };
      }
    }

    // Phase 1: 用户自定义规则匹配（现有逻辑）
    for (const rule of this.rules) {
      if (this.matchRule(request, rule)) {
        return {
          folderId: rule.folderId,
          confidence: rule.confidence,
          matchType: "user_rule",
          reason: `命中规则: ${rule.name}`,
        };
      }
    }

    return null;
  }
}
```
