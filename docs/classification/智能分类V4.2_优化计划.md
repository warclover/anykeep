```typescript
// src/lib/classification/services/operation-log.service.ts

interface OperationSnapshot {
  id: string;
  userId: string;
  action: "create" | "split" | "merge" | "rename" | "move";
  folderId?: number;
  folderIds?: number[];
  bookmarkIds: number[];
  beforeState: {
    folderName?: string;
    folderNames?: string[];
    profileEmbedding?: number[];
    bookmarkFolderMap?: Record<number, number>; // bookmarkId -> folderId
  };
  afterState: {
    folderName?: string;
    folderNames?: string[];
    newFolderIds?: number[];
  };
  reason: string;
  createdAt: Date;
  expiresAt: Date;
}

class OperationLogService {
  // 创建操作快照（在执行任何演进操作前）
  async createSnapshot(
    userId: string,
    action: OperationSnapshot["action"],
    details: Omit<
      OperationSnapshot,
      "id" | "userId" | "createdAt" | "expiresAt"
    >,
  ): Promise<string> {
    const { data, error } = await this.supabase
      .from("operation_snapshots")
      .insert({
        user_id: userId,
        action,
        folder_id: details.folderId,
        folder_ids: details.folderIds,
        bookmark_ids: details.bookmarkIds,
        before_state: details.beforeState,
        after_state: details.afterState,
        reason: details.reason,
      })
      .select("id")
      .single();

    if (error) throw error;
    return data.id;
  }

  // 撤销操作
  async undoOperation(snapshotId: string, userId: string): Promise<void> {
    const snapshot = await this.getSnapshot(snapshotId, userId);

    if (!snapshot || snapshot.is_undone) {
      throw new Error("快照不存在或已撤销");
    }
    if (new Date(snapshot.expires_at) < new Date()) {
      throw new Error("快照已过期（超过 2 天）");
    }

    // 根据 action 类型执行回滚
    switch (snapshot.action) {
      case "create":
        await this.rollbackCreate(snapshot);
        break;
      case "split":
        await this.rollbackSplit(snapshot);
        break;
      case "merge":
        await this.rollbackMerge(snapshot);
        break;
      case "rename":
        await this.rollbackRename(snapshot);
        break;
    }

    // 标记为已撤销
    await this.supabase
      .from("operation_snapshots")
      .update({ is_undone: true })
      .eq("id", snapshotId);
  }
}
```

```typescript
// src/lib/classification/services/job-queue.service.ts

class JobQueueService {
  // 入队任务
  async enqueue(
    userId: string,
    type: string,
    payload: Record<string, unknown>,
    priority = 0,
  ): Promise<string> {
    const { data } = await this.supabase
      .from("job_queue")
      .insert({ user_id: userId, type, payload, priority })
      .select("id")
      .single();
    return data.id;
  }

  // Worker 处理（Edge Function 或定时任务）
  async processNextJob(): Promise<void> {
    // 获取下一个待处理任务（带锁）
    const { data: job } = await this.supabase.rpc("claim_next_job"); // 原子操作

    if (!job) return;

    try {
      await this.executeJob(job);
      await this.markCompleted(job.id);
    } catch (error) {
      await this.markFailed(job.id, error.message);
    }
  }

  private async executeJob(job: Job): Promise<void> {
    switch (job.type) {
      case "process_inbox":
        await this.orchestrator.processInboxCache(job.user_id);
        break;
      case "check_split":
        await this.orchestrator.checkSplitCandidates(job.user_id);
        break;
      case "check_merge":
        await this.orchestrator.checkMergeCandidates(job.user_id);
        break;
      case "check_rename":
        await this.orchestrator.checkRenamingCandidates(job.user_id);
        break;
      case "update_profile":
        await this.profileService.batchUpdateProfile(job.payload.folderId);
        break;
    }
  }
}
```

```typescript
// src/lib/classification/services/smart-sorting.service.ts

interface InboxClusterAnalysis {
  clusters: Array<{
    bookmarkIds: number[];
    centroid: number[];
    avgSimilarity: number;
    suggestedName: string;
    topTags: string[];
    sampleTitles: string[];
  }>;
  noiseBookmarks: number[]; // 无法聚类的书签
}

const CLUSTER_CONFIG = {
  minInboxSize: 5, // 待整理 ≥5 条才分析
  minClusterSize: 3, // ≥3 条才建议创建文件夹
  similarityThreshold: 0.6, // DBSCAN eps 参数
  cronSchedule: "0 * * * *", // 每小时执行
  maxDailyCreations: 6, // 每日创建上限
  maxTotalFolders: 50, // 文件夹总数上限
};

class SmartSortingService {
  // 定期任务：处理缓存池
  async processInboxCache(userId: string): Promise<void> {
    // 1. 获取待分拣书签
    const pendingBookmarks = await this.getPendingBookmarks(userId);
    if (pendingBookmarks.length < CLUSTER_CONFIG.minInboxSize) {
      return; // 数量不足，跳过
    }

    // 2. 检查创建限制
    const todayCreations = await this.getTodayCreationCount(userId);
    if (todayCreations >= CLUSTER_CONFIG.maxDailyCreations) {
      logger.info({ userId }, "Daily folder creation limit reached");
      return;
    }

    // 3. 先做数学聚类，避免无效 LLM 调用
    const clusters = densityClustering(
      pendingBookmarks.map((b) => ({
        id: b.id,
        embedding: b.embedding,
        title: b.title,
      })),
      CLUSTER_CONFIG.similarityThreshold,
      CLUSTER_CONFIG.minClusterSize,
    );

    if (clusters.length === 0) {
      logger.debug(
        { userId, pendingCount: pendingBookmarks.length },
        "No valid clusters found",
      );
      return;
    }

    // 4. 为每个聚类创建文件夹
    for (const cluster of clusters) {
      if (todayCreations >= CLUSTER_CONFIG.maxDailyCreations) break;

      // 检查是否与现有文件夹重复
      const isDuplicate = await this.checkDuplicateFolder(
        userId,
        cluster.centroid,
      );
      if (isDuplicate) continue;

      // 生成文件夹名称（先查缓存，再调 LLM）
      const folderName = await this.generateFolderName(cluster);

      // 创建快照（支持撤销）
      const snapshotId = await this.operationLog.createSnapshot(
        userId,
        "create",
        {
          bookmarkIds: cluster.bookmarkIds,
          beforeState: {
            bookmarkFolderMap: this.getBookmarkFolderMap(cluster.bookmarkIds),
          },
          afterState: { folderName },
          reason: `聚类创建: ${cluster.bookmarkIds.length} 条相似内容`,
        },
      );

      // 执行创建
      const newFolder = await this.createFolderWithProfile(
        userId,
        folderName,
        cluster,
      );

      // 移动书签 + 清除 pending 标记
      await this.moveAndClearPending(cluster.bookmarkIds, newFolder.id);

      // 记录日志 + 发送通知
      await this.notifyUser(userId, "folder_created", {
        folderName,
        bookmarkCount: cluster.bookmarkIds.length,
        snapshotId, // 用于撤销
      });

      todayCreations++;
    }
  }

  // 本地缓存映射（减少 LLM 调用）
  private async generateFolderName(cluster: BookmarkCluster): Promise<string> {
    const topTags = this.extractTopTags(cluster.bookmarkIds);
    const tagKey = topTags.sort().join(",");

    // 常见组合缓存
    const cache: Record<string, string> = {
      "React,Vue,Frontend,前端": "前端开发",
      "Python,AI,机器学习,深度学习": "AI 学习",
      "Docker,K8s,云原生,DevOps": "云原生",
      "TypeScript,Node.js,后端": "后端开发",
      // ... 更多常见组合
    };

    // 模糊匹配缓存
    for (const [key, value] of Object.entries(cache)) {
      const keyTags = key.split(",");
      const overlap = topTags.filter((t) =>
        keyTags.some(
          (k) =>
            t.toLowerCase().includes(k.toLowerCase()) ||
            k.toLowerCase().includes(t.toLowerCase()),
        ),
      );
      if (overlap.length >= 2) {
        return value;
      }
    }

    // 缓存未命中，调用 LLM
    return this.llmGenerateName(cluster);
  }
}
```

```typescript
// src/lib/classification/services/profile-maintenance.service.ts

const HOTNESS_CONFIG = {
  activeThresholdHours: 24, // 24小时内更新过视为活跃
  activeAdditionThreshold: 5, // 新增 ≥5 条视为活跃
  decayHalfLifeDays: 30, // 30天半衰期
};

class ProfileMaintenanceService {
  // 书签添加时调用：热度分级更新
  async updateProfileIfActive(
    folderId: number,
    newBookmarkEmbedding: number[],
  ): Promise<void> {
    const profile = await this.getProfile(folderId);
    const hoursSinceUpdate = this.getHoursSince(profile.last_updated_at);

    // 判断是否活跃文件夹
    const isActive =
      hoursSinceUpdate < HOTNESS_CONFIG.activeThresholdHours ||
      profile.recent_additions >= HOTNESS_CONFIG.activeAdditionThreshold;

    if (isActive) {
      // 活跃文件夹：实时微调
      const alpha = 0.1; // 新内容权重 10%
      const newEmbedding = this.interpolateEmbeddings(
        profile.profile_embedding,
        newBookmarkEmbedding,
        alpha,
      );

      await this.supabase
        .from("folder_profiles")
        .update({
          profile_embedding: newEmbedding,
          last_updated_at: new Date().toISOString(),
          recent_additions: profile.recent_additions + 1,
        })
        .eq("folder_id", folderId);
    } else {
      // 非活跃文件夹：入队批量更新
      await this.jobQueue.enqueue(profile.user_id, "update_profile", {
        folderId,
      });

      // 仅更新计数器
      await this.supabase
        .from("folder_profiles")
        .update({ recent_additions: profile.recent_additions + 1 })
        .eq("folder_id", folderId);
    }
  }

  // 批量更新（异步任务）
  async batchUpdateProfile(folderId: number): Promise<void> {
    const bookmarks = await this.getBookmarksWithEmbeddings(folderId);

    // 时间衰减加权
    const weightedEmbedding = this.calculateTimeWeightedProfile(bookmarks);

    // 更新内聚度
    const internalSimilarity = this.calculateInternalSimilarity(bookmarks);

    await this.supabase
      .from("folder_profiles")
      .update({
        profile_embedding: weightedEmbedding,
        internal_similarity: internalSimilarity,
        last_updated_at: new Date().toISOString(),
        recent_additions: 0, // 重置计数器
      })
      .eq("folder_id", folderId);
  }

  // 时间衰减加权计算
  private calculateTimeWeightedProfile(
    bookmarks: BookmarkWithEmbedding[],
  ): number[] {
    const now = Date.now();
    const halfLife = HOTNESS_CONFIG.decayHalfLifeDays * 24 * 60 * 60 * 1000;

    let weightedSum = new Array(1024).fill(0);
    let totalWeight = 0;

    for (const bookmark of bookmarks) {
      const age = now - new Date(bookmark.created_at).getTime();
      const weight = Math.pow(0.5, age / halfLife); // 时间衰减

      for (let i = 0; i < 1024; i++) {
        weightedSum[i] += bookmark.embedding[i] * weight;
      }
      totalWeight += weight;
    }

    return weightedSum.map((v) => v / totalWeight);
  }
}
```

#### 3.2 智能重命名 (2天)

```typescript
// src/lib/classification/services/rename.service.ts

const RENAME_CONFIG = {
  driftThreshold: 0.3, // 名称与内容相似度 < 0.7 触发检测
  checkInterval: "0 0 * * 0", // 每周日凌晨检查
};

class RenameService {
  // 周期任务：检查重命名候选
  async checkRenamingCandidates(userId: string): Promise<void> {
    const folders = await this.getFoldersForRenameCheck(userId);

    for (const folder of folders) {
      // 跳过锁定的文件夹
      if (!folder.is_ai_renamable || folder.is_system_locked) continue;

      // 计算当前名称 Embedding 与 Profile Embedding 的距离
      const nameEmbedding = await generateEmbedding(folder.name);
      const similarity = cosineSimilarity(
        nameEmbedding,
        folder.profile_embedding,
      );

      if (similarity < 1 - RENAME_CONFIG.driftThreshold) {
        // 内容已漂移，生成新名称建议
        const suggestedName = await this.suggestNewName(folder);

        if (suggestedName && suggestedName !== folder.name) {
          await this.proposeOrExecuteRename(userId, folder, suggestedName);
        }
      }
    }
  }

  private async proposeOrExecuteRename(
    userId: string,
    folder: FolderProfile,
    newName: string,
  ): Promise<void> {
    const userSettings = await this.getUserSettings(userId);

    // 创建快照
    const snapshotId = await this.operationLog.createSnapshot(
      userId,
      "rename",
      {
        folderId: folder.id,
        bookmarkIds: [],
        beforeState: { folderName: folder.name },
        afterState: { folderName: newName },
        reason: `内容漂移检测: 相似度低于阈值`,
      },
    );

    if (userSettings.ai_autonomy_level === 1) {
      // Level 1 全托管：直接执行
      await this.executeRename(folder.id, newName);
      await this.notifyUser(userId, "folder_renamed", {
        from: folder.name,
        to: newName,
        snapshotId,
      });
    } else {
      // Level 2/3：仅建议
      await this.createRenameSuggestion(
        userId,
        folder.id,
        folder.name,
        newName,
        snapshotId,
      );
    }
  }
}
```

```typescript
const SPLIT_CONFIG = {
  minSize: 30, // 至少 30 条才考虑分拆
  maxSimilarity: 0.5, // 内聚度 < 0.5 说明内容分散
  minClusters: 2, // 至少能分成 2 个有效聚类
  checkInterval: "0 0 * * 1", // 每周一凌晨检查
};

class SplitService {
  async checkSplitCandidates(userId: string): Promise<void> {
    // 查询内聚度低的大文件夹
    const candidates = await this.getSplitCandidates(userId);

    for (const folder of candidates) {
      if (folder.is_system_locked) continue;

      // 聚类分析
      const bookmarks = await this.getBookmarksWithEmbeddings(folder.id);
      const clusters = densityClustering(
        bookmarks.map((b) => ({
          id: b.id,
          embedding: b.embedding,
          title: b.title,
        })),
        0.6,
        SPLIT_CONFIG.minClusters,
      );

      if (clusters.length < SPLIT_CONFIG.minClusters) continue;

      // 为每个聚类生成名称
      const splitPlan = await Promise.all(
        clusters.map(async (cluster) => ({
          name: await this.generateClusterName(cluster),
          bookmarkIds: cluster.bookmarkIds,
          sampleTitles: cluster.sampleTitles.slice(0, 3),
        })),
      );

      await this.proposeOrExecuteSplit(userId, folder, splitPlan);
    }
  }

  private async proposeOrExecuteSplit(
    userId: string,
    folder: FolderProfile,
    splitPlan: SplitPlanItem[],
  ): Promise<void> {
    const userSettings = await this.getUserSettings(userId);

    // 创建快照
    const snapshotId = await this.operationLog.createSnapshot(userId, "split", {
      folderId: folder.id,
      bookmarkIds: splitPlan.flatMap((p) => p.bookmarkIds),
      beforeState: {
        folderName: folder.name,
        bookmarkFolderMap: this.getBookmarkFolderMap(
          splitPlan.flatMap((p) => p.bookmarkIds),
        ),
      },
      afterState: { folderNames: splitPlan.map((p) => p.name) },
      reason: `内聚度过低 (${folder.internal_similarity.toFixed(2)})，检测到 ${splitPlan.length} 个子主题`,
    });

    if (userSettings.ai_autonomy_level === 1) {
      // Level 1 全托管：直接执行
      const newFolderIds = await this.executeSplit(userId, folder, splitPlan);
      await this.notifyUser(userId, "folder_split", {
        originalName: folder.name,
        newFolders: splitPlan.map((p) => p.name),
        snapshotId,
      });
    } else {
      // Level 2/3：创建建议
      await this.createSplitSuggestion(userId, folder, splitPlan, snapshotId);
    }
  }
}
```

```typescript
// src/lib/classification/services/merge.service.ts

const MERGE_CONFIG = {
  minSimilarity: 0.92, // 相似度 ≥ 0.92
  tagOverlapRate: 0.8, // 标签重叠率 ≥ 80%
  maxCombinedSize: 100, // 合并后不超过 100 条
  checkInterval: "0 0 * * 3", // 每周三凌晨检查
};

class MergeService {
  async checkMergeCandidates(userId: string): Promise<void> {
    const folders = await this.getAllUserFolders(userId);

    // 两两比较相似度
    const mergeCandidates: MergeCandidate[] = [];

    for (let i = 0; i < folders.length; i++) {
      for (let j = i + 1; j < folders.length; j++) {
        const f1 = folders[i];
        const f2 = folders[j];

        // 跳过锁定的文件夹
        if (f1.is_system_locked || f2.is_system_locked) continue;

        const similarity = cosineSimilarity(
          f1.profile_embedding,
          f2.profile_embedding,
        );
        const tagOverlap = this.calculateTagOverlap(f1.top_tags, f2.top_tags);

        if (
          similarity >= MERGE_CONFIG.minSimilarity &&
          tagOverlap >= MERGE_CONFIG.tagOverlapRate &&
          f1.bookmark_count + f2.bookmark_count <= MERGE_CONFIG.maxCombinedSize
        ) {
          mergeCandidates.push({
            folder1: f1,
            folder2: f2,
            similarity,
            tagOverlap,
          });
        }
      }
    }

    // 处理合并候选
    for (const candidate of mergeCandidates) {
      await this.proposeOrExecuteMerge(userId, candidate);
    }
  }
}
```

```typescript
// src/lib/classification/services/cost-monitor.service.ts

class CostMonitorService {
  private readonly DAILY_LIMIT_CENTS = 1000; // $10/天
  private readonly ALERT_THRESHOLD = 0.8; // 80% 时报警

  async checkDailyCost(): Promise<void> {
    const today = new Date().toISOString().split("T")[0];

    const { data } = await this.supabase
      .from("llm_usage_log")
      .select("cost_cents")
      .gte("created_at", `${today}T00:00:00Z`)
      .lt("created_at", `${today}T23:59:59Z`);

    const totalCost = (data || []).reduce(
      (sum, row) => sum + row.cost_cents,
      0,
    );

    if (totalCost >= this.DAILY_LIMIT_CENTS * this.ALERT_THRESHOLD) {
      await this.sendAlert("LLM 成本接近每日上限", {
        currentCost: totalCost / 100,
        limit: this.DAILY_LIMIT_CENTS / 100,
        percentage: ((totalCost / this.DAILY_LIMIT_CENTS) * 100).toFixed(1),
      });
    }
  }

  // 包装 LLM 调用，自动记录成本
  async callWithLogging<T>(
    userId: string,
    operation: string,
    llmCall: () => Promise<{
      result: T;
      usage: { inputTokens: number; outputTokens: number };
    }>,
  ): Promise<T> {
    const { result, usage } = await llmCall();

    const costCents = this.calculateCost(usage);

    await this.supabase.from("llm_usage_log").insert({
      user_id: userId,
      operation,
      model: "qwen-max",
      input_tokens: usage.inputTokens,
      output_tokens: usage.outputTokens,
      cost_cents: costCents,
    });

    return result;
  }
}
```
