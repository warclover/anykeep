## 🔔 事件总线实现

### 1. 事件总线核心实现

**文件**: `src/lib/events/event-bus.ts`

```typescript
/**
 * 轻量级事件总线实现
 *
 * 特性：
 * - 类型安全
 * - 支持同步/异步订阅
 * - 支持事件过滤
 * - 无外部依赖
 */

type EventHandler<T = any> = (event: T) => void | Promise<void>;
type EventFilter<T = any> = (event: T) => boolean;

interface Subscription {
  id: string;
  handler: EventHandler;
  filter?: EventFilter;
  once?: boolean;
}

export class EventBus {
  private subscriptions: Map<string, Subscription[]> = new Map();
  private subscriptionIdCounter = 0;

  /**
   * 订阅事件
   */
  subscribe<T = any>(
    eventType: string,
    handler: EventHandler<T>,
    filter?: EventFilter<T>,
  ): string {
    const subscriptionId = `sub_${++this.subscriptionIdCounter}`;
    const subscription: Subscription = {
      id: subscriptionId,
      handler,
      filter,
    };

    if (!this.subscriptions.has(eventType)) {
      this.subscriptions.set(eventType, []);
    }

    this.subscriptions.get(eventType)!.push(subscription);

    return subscriptionId;
  }

  /**
   * 订阅事件（仅触发一次）
   */
  subscribeOnce<T = any>(
    eventType: string,
    handler: EventHandler<T>,
    filter?: EventFilter<T>,
  ): string {
    const subscriptionId = this.subscribe(eventType, handler, filter);
    const subscription = this.subscriptions
      .get(eventType)!
      .find((s) => s.id === subscriptionId);
    if (subscription) {
      subscription.once = true;
    }
    return subscriptionId;
  }

  /**
   * 取消订阅
   */
  unsubscribe(eventType: string, subscriptionId: string): boolean {
    const subscriptions = this.subscriptions.get(eventType);
    if (!subscriptions) return false;

    const index = subscriptions.findIndex((s) => s.id === subscriptionId);
    if (index === -1) return false;

    subscriptions.splice(index, 1);
    return true;
  }

  /**
   * 发布事件
   */
  async publish<T = any>(eventType: string, event: T): Promise<void> {
    const subscriptions = this.subscriptions.get(eventType) || [];
    const toRemove: string[] = [];

    for (const subscription of subscriptions) {
      // 应用过滤器
      if (subscription.filter && !subscription.filter(event)) {
        continue;
      }

      try {
        // 执行处理器（支持异步）
        await subscription.handler(event);

        // 如果是 once 订阅，标记为删除
        if (subscription.once) {
          toRemove.push(subscription.id);
        }
      } catch (error) {
        console.error(`[EventBus] Error in handler for ${eventType}:`, error);
        // 继续执行其他处理器，不中断
      }
    }

    // 清理 once 订阅
    for (const id of toRemove) {
      this.unsubscribe(eventType, id);
    }
  }

  /**
   * 清空所有订阅
   */
  clear(eventType?: string): void {
    if (eventType) {
      this.subscriptions.delete(eventType);
    } else {
      this.subscriptions.clear();
    }
  }

  /**
   * 获取订阅数量
   */
  getSubscriptionCount(eventType?: string): number {
    if (eventType) {
      return this.subscriptions.get(eventType)?.length || 0;
    }
    return Array.from(this.subscriptions.values()).reduce(
      (sum, subs) => sum + subs.length,
      0,
    );
  }
}

// 单例实例
export const eventBus = new EventBus();
```

### 1. 文件夹画像只读仓库（分类模块使用）

**文件**: `src/lib/classification/repositories/folder-profile-read.repository.ts`

```typescript
/**
 * 文件夹画像只读仓库
 *
 * 分类模块只能读取 folder_profiles，不能修改
 */
import type { SupabaseClient } from "@supabase/supabase-js";
import type { Database } from "@/types/supabase";

export class FolderProfileReadRepository {
  constructor(private supabase: SupabaseClient<Database>) {}

  /**
   * 统一向量搜索（只读）
   */
  async searchFoldersUnified(
    userId: string,
    embedding: number[],
    limit: number = 5,
  ): Promise<
    Array<{
      folderId: number;
      folderName: string;
      similarity: number;
      matchType: "profile" | "name";
      bookmarkCount: number;
      topTags: string[];
    }>
  > {
    const { data, error } = await this.supabase.rpc("search_folders_unified", {
      p_user_id: userId,
      p_embedding: embedding,
      p_limit: limit,
    });

    if (error) {
      throw new Error(`Failed to search folders: ${error.message}`);
    }

    return data || [];
  }

  /**
   * 获取文件夹画像（只读）
   */
  async getProfile(folderId: number) {
    const { data, error } = await this.supabase
      .from("folder_profiles")
      .select("*")
      .eq("folder_id", folderId)
      .single();

    if (error) {
      throw new Error(`Failed to get profile: ${error.message}`);
    }

    return data;
  }
}
```

### 2. 文件夹画像读写仓库（自组织模块使用）

**文件**: `src/lib/folder-organization/repositories/folder-profile.repository.ts`

```typescript
/**
 * 文件夹画像读写仓库
 *
 * 自组织模块拥有 folder_profiles 的完整读写权限
 */
import type { SupabaseClient } from "@supabase/supabase-js";
import type { Database } from "@/types/supabase";

export class FolderProfileRepository {
  constructor(private supabase: SupabaseClient<Database>) {}

  /**
   * 更新文件夹画像
   */
  async updateProfile(
    folderId: number,
    updates: Partial<{
      profile_embedding: number[];
      name_embedding: number[];
      top_tags: string[];
      internal_similarity: number;
      is_ai_renamable: boolean;
      is_system_locked: boolean;
      last_updated_at: Date;
    }>,
  ): Promise<void> {
    const { error } = await this.supabase
      .from("folder_profiles")
      .update({
        ...updates,
        updated_at: new Date().toISOString(),
      })
      .eq("folder_id", folderId);

    if (error) {
      throw new Error(`Failed to update profile: ${error.message}`);
    }
  }

  /**
   * 计算文件夹画像（调用 RPC）
   */
  async computeProfile(folderId: number): Promise<void> {
    const { error } = await this.supabase.rpc("compute_folder_profile", {
      p_folder_id: folderId,
    });

    if (error) {
      throw new Error(`Failed to compute profile: ${error.message}`);
    }
  }

  /**
   * 创建文件夹画像
   */
  async createProfile(
    folderId: number,
    userId: string,
    profile: {
      profile_embedding?: number[];
      name_embedding?: number[];
      top_tags?: string[];
    },
  ): Promise<void> {
    const { error } = await this.supabase.from("folder_profiles").insert({
      folder_id: folderId,
      user_id: userId,
      ...profile,
      created_at: new Date().toISOString(),
      updated_at: new Date().toISOString(),
    });

    if (error) {
      throw new Error(`Failed to create profile: ${error.message}`);
    }
  }
}

export class FolderEvolutionService {
  constructor(
    private smartSortingService: SmartSortingService,
    private messageService: MessageService,
  ) {
    this.setupEventSubscriptions();
  }

  private setupEventSubscriptions() {
    // 订阅文件夹创建建议事件
    eventBus.subscribe<FolderCreateSuggestedEvent>(
      FOLDER_EVENTS.CREATE_SUGGESTED,
      async (event) => {
        await this.handleCreateSuggestion(event);
      },
    );
  }

  private async handleCreateSuggestion(event: FolderCreateSuggestedEvent) {
    const settings = await this.getUserSettings(event.userId);

    if (settings.autonomyLevel === 1) {
      // 全自动：直接创建
      await this.smartSortingService.createFolderFromSuggestion(event);
    } else if (settings.autonomyLevel === 2) {
      // 半自动：生成确认消息
      await this.messageService.sendCreateConfirmation(event);
    }
    // Level 3: 仅记录，不执行
  }
}
```
