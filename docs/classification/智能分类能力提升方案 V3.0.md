## 一、问题诊断

| 问题层       | 症状                                     | 根因分析                                 |
| ------------ | ---------------------------------------- | ---------------------------------------- |
| **检索层**   | 明明有合适文件夹却没搜到                 | 向量检索依赖标题质量，标题模糊则召回率低 |
| **上下文层** | AI 只能从 3-5 个候选中选，选项全错则必错 | 缺乏"上帝视角"，不知道用户完整文件夹列表 |
| **推理层**   | AI 被表面关键词误导                      | Prompt 是选择题，无思维链，无反思        |
| **冷启动层** | 新用户/新领域内容无法分类                | 无历史数据，无通用分类兜底               |
| **扩展层**   | 用户新兴趣无对应文件夹，只能进待整理     | 系统不敢自动创建新文件夹                 |
| **反馈层**   | 用户纠正后系统没学到                     | 缺乏闭环，权重固定                       |

```typescript
export class UniversalCategoryMatcher {
  async match(context: ClassificationContext): Promise<UniversalMatchResult> {
    // Step 1: AI 将内容映射到通用分类
    const universalCategory = await this.classifyToUniversal(context);

    // Step 2: 查找用户是否有匹配的文件夹
    const userFolders = await this.folderRepo.getAllUserFolders(context.userId);
    const matchedFolder = this.findMatchingUserFolder(
      universalCategory,
      userFolders,
    );

    if (matchedFolder) {
      return {
        type: "matched",
        folderId: matchedFolder.id,
        folderName: matchedFolder.name,
        confidence: universalCategory.confidence,
        universalCategory: universalCategory.id,
      };
    }

    // 未找到匹配 → 返回建议创建信息
    return {
      type: "suggest_create",
      suggestedName: universalCategory.name,
      universalCategory: universalCategory.id,
      confidence: universalCategory.confidence,
      reason: `内容属于「${universalCategory.name}」领域`,
    };
  }
}
```

```typescript
// src/lib/classification/strategies/enhanced-vector-matcher.ts
export class EnhancedVectorMatcher {
  async match(context: ClassificationContext): Promise<MatchResult[]> {
    // Step 1: Query Expansion - 生成扩展关键词
    const expandedKeywords = await this.expandQuery(context);

    // Step 2: 多路召回
    const results = await Promise.all([
      this.vectorSearch(context.embedding, 5),
      ...expandedKeywords
        .slice(0, 3)
        .map((kw) => this.keywordVectorSearch(kw, 3)),
    ]);

    // Step 3: 融合去重 + 重排序
    return this.fuseAndRerank(results, context);
```

### 4.3 文件夹意图指纹系统 ⭐ 核心增强

**问题诊断**：文件夹的 Embedding 是平均值，特征被稀释，导致匹配不精准。

**解决方案**：为每个文件夹维护 Top 10 强特征标签（意图指纹），用于精准碰撞匹配。

```typescript
// src/lib/classification/services/folder-fingerprint.service.ts
export interface FolderFingerprint {
  folderId: number;
  signatureTags: string[]; // Top 10 强特征标签
  tagWeights: Record<string, number>; // 标签权重（用户纠正会调整）
  lastUpdated: Date;
}

export class FolderFingerprintService {
  /**
   * 生成/更新文件夹的意图指纹
   * 触发时机：文件夹创建、书签归入、用户纠正分类
   */
  async updateFingerprint(folderId: number): Promise<FolderFingerprint> {
    // 1. 获取文件夹内所有书签的标签
    const bookmarks = await this.bookmarkRepo.findByFolderId(folderId, {
      limit: 100,
    });
    const allTags = bookmarks.flatMap((b) => b.ai_tags || []);

    // 2. 统计标签频率
    const tagFrequency: Record<string, number> = {};
    for (const tag of allTags) {
      const normalizedTag = tag.toLowerCase();
      tagFrequency[normalizedTag] = (tagFrequency[normalizedTag] || 0) + 1;
    }

    // 3. 结合用户纠正历史调整权重
    const corrections = await this.getCorrectionWeights(folderId);
    const adjustedWeights = this.applyCorrections(tagFrequency, corrections);

    // 4. 提取 Top 10 强特征标签
    const signatureTags = Object.entries(adjustedWeights)
      .sort((a, b) => b[1] - a[1])
      .slice(0, 10)
      .map(([tag]) => tag);

    // 5. 持久化指纹
    const fingerprint: FolderFingerprint = {
      folderId,
      signatureTags,
      tagWeights: adjustedWeights,
      lastUpdated: new Date(),
    };
    await this.fingerprintRepo.upsert(fingerprint);

    return fingerprint;
  }

  /**
   * 用户纠正分类时，更新指纹权重
   */
  async applyUserCorrection(params: {
    bookmarkId: number;
    fromFolderId: number | null;
    toFolderId: number;
    bookmarkTags: string[];
  }): Promise<void> {
    const { bookmarkTags, fromFolderId, toFolderId } = params;

    // 从原文件夹降低这些标签的权重
    if (fromFolderId) {
      await this.adjustTagWeights(fromFolderId, bookmarkTags, -0.2);
    }

    // 在目标文件夹增加这些标签的权重
    await this.adjustTagWeights(toFolderId, bookmarkTags, +0.3);

    // 异步重新计算指纹
    if (fromFolderId) this.updateFingerprint(fromFolderId);
    this.updateFingerprint(toFolderId);
  }
}
```

### 4.4 Tag-to-Tag 精准匹配

**核心理念**：Tag-to-Tag 匹配比 Text-to-Text Embedding 匹配更精准且可解释。

```typescript
// src/lib/classification/strategies/tag-matcher.ts
export class TagMatcher {
  /**
   * 基于标签碰撞的文件夹匹配
   */
  async match(context: ClassificationContext): Promise<MatchResult[]> {
    const bookmarkTags = context.tags || [];
    if (bookmarkTags.length === 0) {
      return [];
    }

    // 1. 获取所有文件夹的意图指纹
    const fingerprints = await this.fingerprintRepo.findByUserId(
      context.userId,
    );

    // 2. 计算标签碰撞得分
    const matchResults: MatchResult[] = [];

    for (const fp of fingerprints) {
      const { score, matchedTags } = this.calculateTagCollision(
        bookmarkTags,
        fp.signatureTags,
        fp.tagWeights,
      );

      if (score > 0) {
        matchResults.push({
          folderId: fp.folderId,
          confidence: Math.min(score * 100, 100),
          matchType: "tag_collision",
          reason: `标签匹配: ${matchedTags.slice(0, 3).join(", ")}`,
          matchedTags,
        });
      }
    }

    // 3. 按得分排序
    return matchResults.sort((a, b) => b.confidence - a.confidence);
  }

  /**
   * 计算标签碰撞得分
   */
  private calculateTagCollision(
    bookmarkTags: string[],
    signatureTags: string[],
    tagWeights: Record<string, number>,
  ): { score: number; matchedTags: string[] } {
    const normalizedBookmarkTags = bookmarkTags.map((t) => t.toLowerCase());
    const matchedTags: string[] = [];
    let totalScore = 0;

    for (const tag of normalizedBookmarkTags) {
      // 精确匹配
      if (signatureTags.includes(tag)) {
        const weight = tagWeights[tag] || 1;
        totalScore += 0.2 * weight;
        matchedTags.push(tag);
        continue;
      }

      // 模糊匹配（包含关系）
      const fuzzyMatch = signatureTags.find(
        (st) => st.includes(tag) || tag.includes(st),
      );
      if (fuzzyMatch) {
        const weight = tagWeights[fuzzyMatch] || 1;
        totalScore += 0.1 * weight;
        matchedTags.push(fuzzyMatch);
      }
    }

    return { score: totalScore, matchedTags };
  }
}
```

### 4.5 k-NN 直接锁定机制 ⭐ 核心增强

**问题诊断**：AI 容易"想太多"，忽略最简单的类比逻辑。

**解决方案**：如果新书签与已有书签相似度 > 0.9，直接锁定目标文件夹，跳过 AI 调用。

```typescript
// src/lib/classification/strategies/knn-direct-matcher.ts
export class KnnDirectMatcher {
  /**
   * 高相似度直接锁定
   * 这是最符合人类直觉的："这篇跟我上次存的那篇是一样的"
   */
  async match(
    context: ClassificationContext,
  ): Promise<DirectMatchResult | null> {
    // 1. 找最相似的书签
    const topSimilar = await this.bookmarkRepo.findMostSimilar(
      context.userId,
      context.embedding,
      { limit: 1, minSimilarity: 0.9 }, // 阈值 0.9
    );

    if (topSimilar.length === 0 || !topSimilar[0].folder_id) {
      return null;
    }

    const match = topSimilar[0];

    // 2. 直接锁定，跳过 AI
    return {
      type: "direct_lock",
      folderId: match.folder_id,
      folderName: match.folder_name,
      confidence: Math.round(match.similarity * 100),
      reason: `与「${match.title.slice(0, 30)}...」高度相似`,
      skipAI: true, // 标记跳过 AI 调用
      referenceBookmarkId: match.id,
    };
  }
}
```

```typescript
// src/lib/classification/services/folder-auto-create.service.ts
export class FolderAutoCreateService {
  /**
   * 评估是否应该自动创建新文件夹
   */
  async evaluateAutoCreate(params: {
    userId: string;
    bookmarkContext: ClassificationContext;
    universalCategory: UniversalMatchResult;
    existingFolders: Folder[];
  }): Promise<AutoCreateDecision> {
    const { userId, universalCategory, existingFolders } = params;

    // 检查 1: 全局开关
    if (!AUTO_CREATE_CONFIG.enabled) {
      return { shouldCreate: false, reason: "global_disabled" };
    }

    // 检查 2: 用户个人设置
    const userSettings = await this.getUserAutoCreateSettings(userId);
    if (!userSettings.autoCreateEnabled) {
      return { shouldCreate: false, reason: "user_disabled" };
    }

    // 检查 3: 文件夹数量上限
    if (existingFolders.length >= AUTO_CREATE_CONFIG.maxFoldersPerUser) {
      return {
        shouldCreate: false,
        reason: "folder_limit_reached",
        suggestAction: "suggest_merge", // 建议合并现有文件夹
      };
    }

    // 检查 4: 每日创建配额
    const todayCreatedCount = await this.getTodayAutoCreateCount(userId);
    if (todayCreatedCount >= AUTO_CREATE_CONFIG.maxAutoCreatePerDay) {
      return {
        shouldCreate: false,
        reason: "daily_limit_reached",
        suggestAction: "suggest_create", // 保存建议，让用户手动创建
      };
    }

    // 检查 5: 通用分类置信度
    if (
      universalCategory.confidence <
      AUTO_CREATE_CONFIG.universalCategoryConfidenceThreshold
    ) {
      return { shouldCreate: false, reason: "low_confidence" };
    }

    // 检查 6: 名称相似度（避免创建重复文件夹）
    const proposedName = universalCategory.suggestedName;
    const similarFolder = await this.findSimilarFolder(
      proposedName,
      existingFolders,
    );
    if (
      similarFolder &&
      similarFolder.similarity >= AUTO_CREATE_CONFIG.nameSimilarityThreshold
    ) {
      return {
        shouldCreate: false,
        reason: "similar_folder_exists",
        existingFolder: similarFolder.folder,
        suggestAction: "use_existing",
      };
    }

    // 全部检查通过 → 允许自动创建
    return {
      shouldCreate: true,
      proposedName: this.standardizeFolderName(proposedName),
      universalCategoryId: universalCategory.universalCategory,
      trigger: AutoCreateTrigger.NO_MATCH_WITH_UNIVERSAL,
    };
  }

  /**
   * 执行自动创建文件夹
   */
  async autoCreateFolder(params: {
    userId: string;
    name: string;
    universalCategoryId: string;
    firstBookmarkId: number;
  }): Promise<AutoCreateResult> {
    const { userId, name, universalCategoryId, firstBookmarkId } = params;

    // 1. 创建文件夹
    const folder = await this.folderRepo.create({
      user_id: userId,
      name: name,
      source: "ai_auto_created",
      metadata: {
        universal_category: universalCategoryId,
        created_reason: "auto_classification",
        first_bookmark_id: firstBookmarkId,
      },
    });

    // 2. 将书签归入新文件夹
    await this.bookmarkRepo.update(firstBookmarkId, { folder_id: folder.id });

    // 3. 记录创建日志（用于配额统计）
    await this.logAutoCreate(userId, folder.id, universalCategoryId);

    // 4. 初始化文件夹画像
    await this.folderProfileService.initProfile(folder.id);

    // 5. 触发回溯分类（异步执行，不阻塞主流程）
    this.triggerBackfillClassification(
      userId,
      folder.id,
      universalCategoryId,
    ).catch((err) =>
      logger.error(
        { err, folderId: folder.id },
        "Backfill classification failed",
      ),
    );

    return {
      success: true,
      folder: folder,
      message: `已自动创建文件夹「${name}」并归类`,
    };
  }

  /**
   * 触发待整理内容的回溯分类
   */
  private async triggerBackfillClassification(
    userId: string,
    newFolderId: number,
    universalCategoryId: string,
  ): Promise<void> {
    const backfillService = new BackfillClassificationService();
    await backfillService.classifyInboxToNewFolder({
      userId,
      targetFolderId: newFolderId,
      universalCategoryId,
    });
  }

  /**
   * 标准化文件夹名称
   */
  private standardizeFolderName(rawName: string): string {
    // 移除多余空格
    let name = rawName.trim().replace(/\s+/g, " ");

    // 长度限制
    if (name.length > 20) {
      name = name.slice(0, 20);
    }

    // 首字母大写（英文）
    name = name.replace(/\b\w/g, (c) => c.toUpperCase());

    return name;
  }

  /**
   * 查找名称相似的现有文件夹
   */
  private async findSimilarFolder(
    proposedName: string,
    existingFolders: Folder[],
  ): Promise<{ folder: Folder; similarity: number } | null> {
    const normalizedProposed = proposedName.toLowerCase();

    for (const folder of existingFolders) {
      const normalizedExisting = folder.name.toLowerCase();

      // 完全包含关系
      if (
        normalizedExisting.includes(normalizedProposed) ||
        normalizedProposed.includes(normalizedExisting)
      ) {
        return { folder, similarity: 0.9 };
      }

      // 计算编辑距离相似度
      const similarity = this.calculateStringSimilarity(
        normalizedProposed,
        normalizedExisting,
      );
      if (similarity >= AUTO_CREATE_CONFIG.nameSimilarityThreshold) {
        return { folder, similarity };
      }
    }

    return null;
  }
}
```

#### 5.6.3 回溯分类服务

```typescript
// src/lib/classification/services/backfill-classification.service.ts
export class BackfillClassificationService {
  /**
   * 将待整理内容回溯分类到新创建的文件夹
   */
  async classifyInboxToNewFolder(params: {
    userId: string;
    targetFolderId: number;
    universalCategoryId: string;
  }): Promise<BackfillResult> {
    const { userId, targetFolderId, universalCategoryId } = params;

    // 1. 延迟执行（给用户撤销时间）
    await this.delay(BACKFILL_CONFIG.delayBeforeStart);

    // 2. 检查文件夹是否仍然存在（可能被撤销）
    const folder = await this.folderRepo.findById(targetFolderId);
    if (!folder) {
      return { success: false, reason: "folder_deleted", classifiedCount: 0 };
    }

    // 3. 获取待整理中的书签
    const inboxBookmarks = await this.bookmarkRepo.findByUserId(userId, {
      folderId: null, // 待整理
      limit: BACKFILL_CONFIG.maxBookmarksPerBatch,
    });

    if (inboxBookmarks.length === 0) {
      return { success: true, classifiedCount: 0 };
    }

    // 4. 获取新文件夹的特征（用于匹配）
    const folderProfile = await this.getFolderMatchProfile(
      targetFolderId,
      universalCategoryId,
    );

    // 5. 计算每条书签的匹配度
    const matchResults = await Promise.all(
      inboxBookmarks.map((bookmark) =>
        this.calculateMatchScore(bookmark, folderProfile),
      ),
    );

    // 6. 筛选符合条件的书签
    const toClassify = matchResults
      .filter((r) => r.confidence >= BACKFILL_CONFIG.confidenceThreshold)
      .sort((a, b) => b.confidence - a.confidence);

    if (toClassify.length === 0) {
      return { success: true, classifiedCount: 0 };
    }

    // 7. 批量归类
    const classifiedBookmarkIds = toClassify.map((r) => r.bookmarkId);
    await this.bookmarkRepo.batchUpdateFolder(
      classifiedBookmarkIds,
      targetFolderId,
    );

    // 8. 记录归类日志
    await this.logBackfillClassification(
      userId,
      targetFolderId,
      classifiedBookmarkIds,
    );

    // 9. 更新文件夹画像
    await this.folderProfileService.updateProfile(targetFolderId);

    // 10. 发送通知
    if (BACKFILL_CONFIG.notifyOnComplete && toClassify.length > 0) {
      await this.notifyBackfillComplete(userId, folder.name, toClassify);
    }

    return {
      success: true,
      classifiedCount: toClassify.length,
      classifiedBookmarks: toClassify.map((r) => ({
        id: r.bookmarkId,
        title: r.bookmarkTitle,
        confidence: r.confidence,
      })),
    };
  }

  /**
   * 计算书签与目标文件夹的匹配度
   */
  private async calculateMatchScore(
    bookmark: Bookmark,
    folderProfile: FolderMatchProfile,
  ): Promise<MatchScoreResult> {
    let totalScore = 0;

    // 1. 通用分类匹配（完全匹配得 100 分）
    if (bookmark.universal_category_id === folderProfile.universalCategoryId) {
      totalScore += 100 * BACKFILL_CONFIG.universalCategoryWeight;
    }

    // 2. 标签匹配（交集比例）
    const bookmarkTags = bookmark.ai_tags || [];
    const folderKeywords = folderProfile.keywords;
    const tagOverlap = this.calculateTagOverlap(bookmarkTags, folderKeywords);
    totalScore += tagOverlap * 100 * BACKFILL_CONFIG.tagMatchWeight;

    // 3. 向量相似度
    if (bookmark.embedding && folderProfile.centroidEmbedding) {
      const similarity = cosineSimilarity(
        bookmark.embedding,
        folderProfile.centroidEmbedding,
      );
      totalScore += similarity * 100 * BACKFILL_CONFIG.vectorSimilarityWeight;
    }

    return {
      bookmarkId: bookmark.id,
      bookmarkTitle: bookmark.title,
      confidence: Math.round(totalScore),
      matchDetails: {
        universalCategory:
          bookmark.universal_category_id === folderProfile.universalCategoryId,
        tagOverlap,
        vectorSimilarity: bookmark.embedding
          ? cosineSimilarity(
              bookmark.embedding,
              folderProfile.centroidEmbedding || [],
            )
          : 0,
      },
    };
  }

  /**
   * 获取文件夹匹配特征
   */
  private async getFolderMatchProfile(
    folderId: number,
    universalCategoryId: string,
  ): Promise<FolderMatchProfile> {
    // 获取通用分类的同义词作为关键词
    const category = UNIVERSAL_CATEGORIES.find(
      (c) => c.id === universalCategoryId,
    );
    const keywords = category?.synonyms || [];

    // 获取文件夹的质心向量（如果有历史内容）
    const centroidEmbedding =
      await this.folderProfileService.getCentroidEmbedding(folderId);

    return {
      folderId,
      universalCategoryId,
      keywords,
      centroidEmbedding,
    };
  }

  /**
   * 计算标签重叠度
   */
  private calculateTagOverlap(
    bookmarkTags: string[],
    folderKeywords: string[],
  ): number {
    if (bookmarkTags.length === 0 || folderKeywords.length === 0) return 0;

    const normalizedBookmarkTags = bookmarkTags.map((t) => t.toLowerCase());
    const normalizedFolderKeywords = folderKeywords.map((k) => k.toLowerCase());

    let matchCount = 0;
    for (const tag of normalizedBookmarkTags) {
      if (
        normalizedFolderKeywords.some(
          (keyword) => tag.includes(keyword) || keyword.includes(tag),
        )
      ) {
        matchCount++;
      }
    }

    return matchCount / Math.max(bookmarkTags.length, 1);
  }

  /**
   * 发送回溯分类完成通知
   */
  private async notifyBackfillComplete(
    userId: string,
    folderName: string,
    classifiedItems: MatchScoreResult[],
  ): Promise<void> {
    // 通过 WebSocket 或消息队列发送通知
    await this.notificationService.send(userId, {
      type: "backfill_classification_complete",
      data: {
        folderName,
        count: classifiedItems.length,
        items: classifiedItems.slice(0, 5).map((item) => ({
          title: item.bookmarkTitle,
          confidence: item.confidence,
        })),
        hasMore: classifiedItems.length > 5,
      },
    });
  }

  private delay(ms: number): Promise<void> {
    return new Promise((resolve) => setTimeout(resolve, ms));
  }
}
```

```typescript
async classify(context: ClassificationContext): Promise<ClassificationResult> {
  // 第一层：快速匹配（无 API 成本）
  const [vectorResult, historyResult, universalResult] = await Promise.all([
    this.vectorMatcher.match(context),
    this.historyMatcher.match(context),
    this.universalMatcher.match(context),
  ]);

  const fastScore = this.aggregator.aggregate([vectorResult, historyResult, universalResult]);

  // 高置信度 → 直接分类，跳过 AI
  if (fastScore.confidence >= 80) {
    return {
      action: 'classify',
      folderId: fastScore.folderId,
      confidence: fastScore.confidence
    };
  }

  // 中置信度 → 调用 AI 确认
  if (fastScore.confidence >= 50) {
    const aiContext = await this.prepareAIContext(context, fastScore);
    const aiResult = await this.aiRecommender.recommend(aiContext);
    return this.aggregator.finalDecision([fastScore], aiResult);
  }

  // 低置信度 → 评估是否自动创建文件夹
  if (universalResult.type === 'suggest_create' && universalResult.confidence >= 75) {
    const createDecision = await this.autoCreateService.evaluateAutoCreate({
      userId: context.userId,
      bookmarkContext: context,
      universalCategory: universalResult,
      existingFolders: await this.folderRepo.getAllUserFolders(context.userId)
    });

    if (createDecision.shouldCreate) {
      return {
        action: 'auto_create',
        proposedFolderName: createDecision.proposedName,
        universalCategory: createDecision.universalCategoryId,
        confidence: universalResult.confidence
      };
    }

    if (createDecision.suggestAction === 'suggest_create') {
      return {
        action: 'suggest_create',
        proposedFolderName: createDecision.proposedName,
        reason: createDecision.reason
      };
    }
  }

  // 兜底 → 送入待整理
  return { action: 'inbox' };
}
```

#### 8.3.2 整理大师服务

```typescript
// src/lib/classification/services/ai-librarian.service.ts
export class AILibrarianService {
  /**
   * 执行全库分析
   */
  async analyze(userId: string): Promise<LibrarianAnalysis> {
    const folders = await this.folderRepo.getAllUserFolders(userId);
    const inboxBookmarks = await this.bookmarkRepo.findByUserId(userId, {
      folderId: null,
    });

    // 并行执行各项分析
    const [duplicates, empty, oversized, poorlyNamed, missing] =
      await Promise.all([
        this.detectDuplicateFolders(folders),
        this.detectEmptyFolders(folders),
        this.detectOversizedFolders(folders),
        this.detectPoorlyNamedFolders(folders),
        this.detectMissingCategories(inboxBookmarks, folders),
      ]);

    // 计算健康评分
    const healthScore = this.calculateHealthScore({
      duplicates,
      empty,
      oversized,
      poorlyNamed,
      missing,
      totalFolders: folders.length,
    });

    return {
      duplicateFolders: duplicates,
      emptyFolders: empty,
      oversizedFolders: oversized,
      poorlyNamedFolders: poorlyNamed,
      missingCategories: missing,
      healthScore,
    };
  }

  /**
   * 检测重复/高度相似的文件夹
   */
  private async detectDuplicateFolders(
    folders: Folder[],
  ): Promise<LibrarianAnalysis["duplicateFolders"]> {
    const duplicates: LibrarianAnalysis["duplicateFolders"] = [];
    const processed = new Set<number>();

    for (let i = 0; i < folders.length; i++) {
      if (processed.has(folders[i].id)) continue;

      const similar: Folder[] = [folders[i]];

      for (let j = i + 1; j < folders.length; j++) {
        if (processed.has(folders[j].id)) continue;

        // 计算名称相似度
        const nameSimilarity = this.calculateStringSimilarity(
          folders[i].name.toLowerCase(),
          folders[j].name.toLowerCase(),
        );

        // 计算指纹标签重叠度
        const tagOverlap = await this.calculateFingerprintOverlap(
          folders[i].id,
          folders[j].id,
        );

        // 综合相似度
        const similarity = nameSimilarity * 0.4 + tagOverlap * 0.6;

        if (similarity > 0.7) {
          similar.push(folders[j]);
          processed.add(folders[j].id);
        }
      }

      if (similar.length > 1) {
        processed.add(folders[i].id);
        duplicates.push({
          folders: similar,
          similarity: 0.7,
          suggestedMerge: this.suggestMergedName(similar),
        });
      }
    }

    return duplicates;
  }

  /**
   * 检测待整理中缺失的关键分类
   */
  private async detectMissingCategories(
    inboxBookmarks: Bookmark[],
    existingFolders: Folder[],
  ): Promise<LibrarianAnalysis["missingCategories"]> {
    if (inboxBookmarks.length === 0) return [];

    // 统计待整理中的通用分类分布
    const categoryCount: Record<string, number> = {};
    for (const bookmark of inboxBookmarks) {
      const category = bookmark.universal_category_id;
      if (category) {
        categoryCount[category] = (categoryCount[category] || 0) + 1;
      }
    }

    // 找出数量较多但用户没有对应文件夹的分类
    const missing: LibrarianAnalysis["missingCategories"] = [];
    const existingFolderNames = existingFolders.map((f) =>
      f.name.toLowerCase(),
    );

    for (const [categoryId, count] of Object.entries(categoryCount)) {
      if (count < 3) continue; // 至少 3 条才建议创建

      const category = UNIVERSAL_CATEGORIES.find((c) => c.id === categoryId);
      if (!category) continue;

      // 检查是否已有相似文件夹
      const hasMatch = existingFolderNames.some(
        (name) =>
          name.includes(category.name.toLowerCase()) ||
          category.synonyms.some((s) => name.includes(s.toLowerCase())),
      );

      if (!hasMatch) {
        missing.push({
          category: category.name,
          bookmarkCount: count,
          reason: `待整理中有 ${count} 条「${category.name}」相关内容`,
        });
      }
    }

    return missing.sort((a, b) => b.bookmarkCount - a.bookmarkCount);
  }

  /**
   * 计算健康评分
   */
  private calculateHealthScore(analysis: {
    duplicates: LibrarianAnalysis["duplicateFolders"];
    empty: Folder[];
    oversized: LibrarianAnalysis["oversizedFolders"];
    poorlyNamed: LibrarianAnalysis["poorlyNamedFolders"];
    missing: LibrarianAnalysis["missingCategories"];
    totalFolders: number;
  }): number {
    let score = 100;

    // 扣分项
    score -= analysis.duplicates.length * 10; // 每组重复扣 10 分
    score -= analysis.empty.length * 5; // 每个空文件夹扣 5 分
    score -= analysis.oversized.length * 8; // 每个过大文件夹扣 8 分
    score -= analysis.poorlyNamed.length * 3; // 每个命名问题扣 3 分
    score -= Math.min(analysis.missing.length * 5, 20); // 缺失分类最多扣 20 分

    return Math.max(0, Math.min(100, score));
  }
}
```

```tsx
          {/* 健康评分 */}
          <div className="flex items-center gap-4">
            <div
              className={cn(
                "text-4xl font-bold",
                analysis.healthScore >= 80
                  ? "text-green-500"
                  : analysis.healthScore >= 60
                    ? "text-yellow-500"
                    : "text-red-500",
              )}
            >
              {analysis.healthScore}
            </div>
            <div className="text-sm text-muted-foreground">
              文件夹结构健康评分
            </div>
          </div>

          {/* 问题列表 */}
          {analysis.duplicateFolders.length > 0 && (
            <LibrarianSection
              title="重复文件夹"
              icon={<Copy className="w-4 h-4" />}
              items={analysis.duplicateFolders.map((d) => ({
                description: `「${d.folders.map((f) => f.name).join("」「")}」内容相似`,
                action: `建议合并为「${d.suggestedMerge}」`,
                onApply: () => mergeFolders(d.folders, d.suggestedMerge),
              }))}
            />
          )}

          {analysis.missingCategories.length > 0 && (
            <LibrarianSection
              title="建议新增的分类"
              icon={<FolderPlus className="w-4 h-4" />}
              items={analysis.missingCategories.map((m) => ({
                description: m.reason,
                action: `创建「${m.category}」文件夹`,
                onApply: () => createFolder(m.category),
              }))}
            />
          )}

          {analysis.emptyFolders.length > 0 && (
            <LibrarianSection
              title="空文件夹"
              icon={<FolderX className="w-4 h-4" />}
              items={analysis.emptyFolders.map((f) => ({
                description: `「${f.name}」没有任何内容`,
                action: "删除",
                onApply: () => deleteFolder(f.id),
              }))}
            />
          )}
        </div>
      )}
    </Card>
  );
}
```

## 十一、风险控制

### 11.1 自动创建风险

| 风险               | 缓解措施                         |
| ------------------ | -------------------------------- |
| 创建过多无用文件夹 | 每日限额 + 用户可关闭 + 撤销功能 |
| 文件夹名称不规范   | 标准化处理 + 基于通用分类        |
| 与现有文件夹重复   | 相似度检测 + 阈值控制            |
| 用户不知情         | 创建后立即通知 + 8秒内可撤销     |

### 11.2 回溯分类风险

| 风险               | 缓解措施                      |
| ------------------ | ----------------------------- |
| 误归类大量内容     | 置信度阈值 70% + 撤销全部功能 |
| 回溯任务阻塞主流程 | 异步执行 + 3秒延迟启动        |
| 用户不知情         | 完成后汇总通知 + 详情列表     |
| 撤销后数据不一致   | 日志追踪 + 批量回滚           |
