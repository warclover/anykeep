### Phase 2: 文件夹 Profile 智能演进（1 周）⭐⭐⭐⭐

**目标**：确保 profile 随收藏变化实时、智能更新

#### 2.1 增强增量更新（2天）

**现有问题**：profile 是简单加权平均，不能反映内容变化趋势

**优化方案**：

```typescript
// 时间衰减加权
function calculateTimeWeightedProfile(bookmarks: Bookmark[]): number[] {
  const now = Date.now();
  const halfLife = 30 * 24 * 60 * 60 * 1000; // 30天半衰期

  let weightedSum = new Array(1024).fill(0);
  let totalWeight = 0;

  for (const bookmark of bookmarks) {
    const age = now - new Date(bookmark.created_at).getTime();
    const weight = Math.pow(0.5, age / halfLife); // 时间衰减

    for (let i = 0; i < 1024; i++) {
      weightedSum[i] += bookmark.embedding[i] * weight;
    }
    totalWeight += weight;
  }

  return weightedSum.map((v) => v / totalWeight);
}
```

#### 2.2 Profile 漂移检测（2天）

**场景**：文件夹内容主题发生显著变化

```typescript
interface ProfileDriftResult {
  hasDrift: boolean;
  driftScore: number; // 0-1，越高漂移越大
  oldCentroid: number[];
  newCentroid: number[];
  suggestedAction: "keep" | "split" | "rename";
}

// 检测逻辑
async function detectProfileDrift(
  folderId: number,
): Promise<ProfileDriftResult> {
  // 对比最近30天 vs 之前的内容向量
  const recentEmbeddings = await getRecentBookmarkEmbeddings(folderId, 30);
  const oldEmbeddings = await getOlderBookmarkEmbeddings(folderId, 30);

  const recentCentroid = calculateCentroid(recentEmbeddings);
  const oldCentroid = calculateCentroid(oldEmbeddings);

  const driftScore = 1 - cosineSimilarity(recentCentroid, oldCentroid);

  return {
    hasDrift: driftScore > 0.3,
    driftScore,
    suggestedAction: driftScore > 0.5 ? "split" : "keep",
  };
}
```

```typescript
async function executeFolderMerge(
  folder1Id: number,
  folder2Id: number,
  targetName: string,
): Promise<MergeResult> {
  // 1. 确定目标文件夹（保留较大的）
  const [f1, f2] = await Promise.all([
    getFolderWithCount(folder1Id),
    getFolderWithCount(folder2Id),
  ]);
  const target = f1.count >= f2.count ? f1 : f2;
  const source = f1.count < f2.count ? f1 : f2;

  // 2. 移动书签
  await moveAllBookmarks(source.id, target.id);

  // 3. 更新文件夹名称（如果需要）
  if (targetName !== target.name) {
    await renameFolder(target.id, targetName);
  }

  // 4. 删除源文件夹
  await deleteFolder(source.id);

  // 5. 重新计算 profile
  await recomputeFolderProfile(target.id);

  return { success: true, targetFolderId: target.id };
}
```

### Phase 5: 相关内容推荐（1 周）⭐⭐⭐

**目标**：借鉴 Recall 知识图谱，展示相关收藏

#### 5.1 相关书签 API

```sql
CREATE FUNCTION find_related_bookmarks(
  p_bookmark_id BIGINT,
  p_user_id UUID,
  p_limit INT DEFAULT 5
)
RETURNS TABLE (
  id BIGINT,
  title TEXT,
  similarity FLOAT,
  folder_name TEXT
)
LANGUAGE sql STABLE
AS $$
  SELECT b.id, b.title,
         (1 - (b.embedding <=> source.embedding))::FLOAT AS similarity,
         f.name AS folder_name
  FROM bookmarks b
  JOIN bookmarks source ON source.id = p_bookmark_id
  LEFT JOIN folders f ON f.id = b.folder_id
  WHERE b.user_id = p_user_id
    AND b.id != p_bookmark_id
    AND b.embedding IS NOT NULL
    AND b.deleted_at IS NULL
  ORDER BY b.embedding <=> source.embedding
  LIMIT p_limit;
$$;
```
