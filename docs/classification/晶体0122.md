### 11.1 设计哲学：流体分类学 (Fluid Taxonomy)

**问题根源**：人类不擅长分类。

用户创建的"临时"、"资料"、"待办"等文件夹，是破坏向量空间的罪魁祸首：

- 语义模糊，无法精准匹配

- 重叠严重，"学习资料" vs "技术文档" 难以区分

- 随意命名，违背 MECE 原则

**解决方案**：从"文件管理器"进化为"自组织知识库"。

### 11.2 核心机制：晶体生长算法 (Crystal Growth Algorithm)

#### 11.2.1 概念模型

| 术语 | 类比 | 定义 |

|------|------|------|

| **混沌池 (Chaos Pool)** | 过饱和溶液 | 所有未分类书签的虚拟空间 |

| **晶核 (Crystal Nucleus)** | 晶种 | 3+ 个语义距离极近的书签 |

| **结晶 (Crystallization)** | 晶体析出 | 晶核成长为正式文件夹 |

| **晶界 (Grain Boundary)** | 晶体边界 | 文件夹之间的语义分界线 |

| **重结晶 (Recrystallization)** | 晶体合并/分裂 | 文件夹动态调整 |

 * 晶体生长算法服务

 * 实现去文件夹化的自组织分类

export class CrystalGrowthService {
  // 结晶��值：至少 3 个书签聚集

  private readonly NUCLEATION_THRESHOLD = 3; // 命中现有文件夹的距离阈值

  private readonly HIT_DISTANCE = 0.15; // 晶核形成的邻居半径

  private readonly NUCLEUS_RADIUS = 0.2; // 语义防撞阈值

  private readonly COLLISION_THRESHOLD = 0.85;

  constructor(private supabase: SupabaseClient) {} /**

   * 处理新书签 - 晶体生长主入口

   */

  async processNewBookmark(
    userId: string,

    bookmark: {
      id: number;
      title: string;
      content?: string;
      embedding: number[];
    },
  ): Promise<CrystalGrowthResult> {
    const startTime = performance.now(); // Step 1: 尝试命中现有文件夹

    const hitResult = await this.tryHitExistingFolder(
      userId,
      bookmark.embedding,
    );

    if (hitResult) {
      await this.assignToFolder(bookmark.id, hitResult.folderId);

      return {
        action: "HIT_EXISTING",

        folderId: hitResult.folderId,

        folderName: hitResult.folderName,

        confidence: hitResult.similarity,

        latencyMs: performance.now() - startTime,
      };
    } // Step 2: 尝试在混沌池中形成晶核

    const nucleationResult = await this.tryNucleation(userId, bookmark);

    if (nucleationResult) {
      return {
        action: "CRYSTALLIZED",

        folderId: nucleationResult.folderId,

        folderName: nucleationResult.folderName,

        clusterSize: nucleationResult.clusterSize,

        latencyMs: performance.now() - startTime,
      };
    } // Step 3: 留在混沌池

    await this.addToChaosPool(userId, bookmark);

    return {
      action: "POOLED",

      message: "书签已保存，等待相关内容聚集后自动分类",

      latencyMs: performance.now() - startTime,
    };
  } /**

   * 尝试命中现有文件夹

   */

  private async tryHitExistingFolder(
    userId: string,

    embedding: number[],
  ): Promise<{
    folderId: number;
    folderName: string;
    similarity: number;
  } | null> {
    const { data: folders } = await this.supabase.rpc("find_nearest_folder", {
      p_user_id: userId,

      p_embedding: embedding,

      p_limit: 1,
    });

    if (folders && folders.length > 0) {
      const nearest = folders[0];

      const distance = 1 - nearest.similarity;

      if (distance < this.HIT_DISTANCE) {
        return {
          folderId: nearest.folder_id,

          folderName: nearest.folder_name,

          similarity: nearest.similarity,
        };
      }
    }

    return null;
  } /**

   * 尝试在混沌池中形成晶核

   */

  private async tryNucleation(
    userId: string,

    bookmark: {
      id: number;
      title: string;
      content?: string;
      embedding: number[];
    },
  ): Promise<{
    folderId: number;
    folderName: string;
    clusterSize: number;
  } | null> {
    // 查找混沌池中的邻居

    const { data: neighbors } = await this.supabase.rpc(
      "find_chaos_pool_neighbors",
      {
        p_user_id: userId,

        p_embedding: bookmark.embedding,

        p_radius: this.NUCLEUS_RADIUS,

        p_limit: 10,
      },
    );

    if (!neighbors || neighbors.length < this.NUCLEATION_THRESHOLD - 1) {
      return null; // 邻居不够，无法结晶
    } // 形成晶核簇

    const cluster = [
      bookmark,
      ...neighbors.slice(0, this.NUCLEATION_THRESHOLD - 1),
    ]; // 获取现有文件夹名称（用于 MECE 保证）

    const { data: existingFolders } = await this.supabase

      .from("folders")

      .select("id, name")

      .eq("user_id", userId)

      .eq("is_deleted", false); // 生成 MECE 文件夹名称

    const folderName = await this.generateMECEFolderName(
      cluster.map((b) => b.title),

      existingFolders?.map((f) => f.name) || [],
    ); // 检查语义防撞

    const collision = await this.checkSemanticCollision(
      userId,

      folderName,

      existingFolders || [],
    );

    if (collision) {
      // 合并到已有文件夹，而不是新建

      const mergedName = await this.generateMergedName(
        collision.existingName,

        folderName,

        cluster.map((b) => b.title),
      ); // 更新现有文件夹名称

      await this.supabase

        .from("folders")

        .update({ name: mergedName })

        .eq("id", collision.existingId); // 移动书签到现有文件夹

      for (const b of cluster) {
        await this.assignToFolder(b.id, collision.existingId);

        await this.removeFromChaosPool(userId, b.id);
      }

      logger.info(
        {
          userId,

          action: "MERGE_COLLISION",

          oldName: collision.existingName,

          newName: mergedName,

          clusterSize: cluster.length,
        },
        "V6.5: Semantic collision resolved by merge",
      );

      return {
        folderId: collision.existingId,

        folderName: mergedName,

        clusterSize: cluster.length,
      };
    } // 创建新文件夹

    const { data: newFolder } = await this.supabase

      .from("folders")

      .insert({
        user_id: userId,

        name: folderName,

        is_ai_generated: true,

        crystal_birth_time: new Date().toISOString(),
      })

      .select("id")

      .single();

    if (!newFolder) throw new Error("Failed to create folder"); // 移动书签到新文件夹

    for (const b of cluster) {
      await this.assignToFolder(b.id, newFolder.id);

      await this.removeFromChaosPool(userId, b.id);
    } // 生成合成锚点

    await this.generateSyntheticAnchor(userId, newFolder.id, folderName);

    logger.info(
      {
        userId,

        folderId: newFolder.id,

        folderName,

        clusterSize: cluster.length,
      },
      "V6.5: New crystal formed",
    );

    return {
      folderId: newFolder.id,

      folderName,

      clusterSize: cluster.length,
    };
  } /**

   * 生成 MECE 保证的文件夹名称

   */

${existingFolderNames.length > 0 ? existingFolderNames.map((n, i) => `${i + 1}. ${n}`).join("\n") : "(暂无)"}

${bookmarkTitles.map((t, i) => `${i + 1}. ${t}`).join("\n")}

    const response = await openaiClient.chat.completions.create({
      model: "gpt-4o-mini",

      messages: [{ role: "user", content: prompt }],

  private async checkSemanticCollision(
    userId: string,

    newName: string,

    existingFolders: Array<{ id: number; name: string }>,

    const newEmbedding = await embeddingService.embed(newName);

    for (const folder of existingFolders) {

    const response = await openaiClient.chat.completions.create({
      model: "gpt-4o-mini",

      temperature: 0.3,

  private cosineSimilarity(a: number[], b: number[]): number {
    let dot = 0,
      normA = 0,
      normB = 0;

    for (let i = 0; i < a.length; i++) {
      dot += a[i] * b[i];

      normA += a[i] * a[i];

      normB += b[i] * b[i];
    }

    return dot / (Math.sqrt(normA) * Math.sqrt(normB));
  }

### 11.3 语义防撞系统 (Semantic Collision Guard)

#### 11.3.1 防撞矩阵

| 场景 | 检测条件 | 处理策略 |

|------|----------|----------|

| **完全重复** | 新名 = 旧名 | 直接合并，不新建 |

| **语义等价** | Sim(新名, 旧名) > 0.90 | 合并并保留旧名 |

| **高度相似** | 0.85 < Sim < 0.90 | 合并并生成新名 |

| **部分重叠** | 0.70 < Sim < 0.85 | 提示用户确认 |

| **语义独立** | Sim < 0.70 | 允许新建 |

### 11.4 动态重结晶 (Dynamic Recrystallization)

文件夹不是静态的，随着内容增长，需要**分裂**或**合并**。

#### 11.4.1 分裂条件 (Folder Splitting)

当文件夹内书签数量 > 50 且**内部方差过大**时触发分裂：

```sql

-- 检测需要分裂的文件夹

SELECT folder_id, folder_name, bookmark_count, internal_variance

FROM (

  SELECT

    f.id AS folder_id,

    f.name AS folder_name,

    COUNT(b.id) AS bookmark_count,

    -- 计算内部方差（书签向量到质心的平均距离）

    AVG(1 - (b.embedding <=> f.content_centroid)) AS internal_variance

  FROM folders f

  JOIN bookmarks b ON b.folder_id = f.id

  WHERE f.user_id = $user_id

  GROUP BY f.id, f.name, f.content_centroid

) stats

WHERE bookmark_count > 50 AND internal_variance < 0.75;

```

**分裂流程**:

1. 对文件夹内书签进行 K-Means 聚类（K=2 或 3）

2. 为每个子簇生成新名称

3. 创建子文件夹，迁移书签

4. 保留父文件夹作为"虚拟分组"

#### 11.4.2 合并条件 (Folder Merging)

当两个文件夹**语义距离过近**且**��签数都较少**时触发合并：

```sql

-- 检测需要合并的文件夹对

SELECT

  f1.id AS folder1_id, f1.name AS folder1_name, f1.bookmark_count AS count1,

  f2.id AS folder2_id, f2.name AS folder2_name, f2.bookmark_count AS count2,

  1 - (f1.content_centroid <=> f2.content_centroid) AS similarity

FROM folders f1

JOIN folders f2 ON f1.user_id = f2.user_id AND f1.id < f2.id

WHERE f1.user_id = $user_id

  AND f1.is_deleted = FALSE AND f2.is_deleted = FALSE

  AND f1.bookmark_count < 20 AND f2.bookmark_count < 20

  AND 1 - (f1.content_centroid <=> f2.content_centroid) > 0.80;

```

#### 11.6.1 混沌池表

```sql

-- 存储未分类书签的向量信息

CREATE TABLE chaos_pool (

  id BIGSERIAL PRIMARY KEY,

  user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,

  bookmark_id BIGINT NOT NULL REFERENCES bookmarks(id) ON DELETE CASCADE,

  embedding vector(1536) NOT NULL,

  created_at TIMESTAMPTZ DEFAULT NOW(),



  UNIQUE(user_id, bookmark_id)

);



-- 向量索引（用于邻居查找）

CREATE INDEX idx_chaos_pool_embedding

  ON chaos_pool USING ivfflat (embedding vector_cosine_ops);



CREATE INDEX idx_chaos_pool_user

  ON chaos_pool (user_id, created_at);

```

#### 11.6.3 核心 RPC 函数

```sql

-- 查找混沌池中的邻居

CREATE OR REPLACE FUNCTION find_chaos_pool_neighbors(

  p_user_id UUID,

  p_embedding vector(1536),

  p_radius FLOAT DEFAULT 0.20,

  p_limit INT DEFAULT 10

)

RETURNS TABLE (

  bookmark_id BIGINT,

  title TEXT,

  distance FLOAT

) AS $$

BEGIN

  RETURN QUERY

  SELECT

    cp.bookmark_id,

    b.title,

    (1 - (cp.embedding <=> p_embedding)) AS distance

  FROM chaos_pool cp

  JOIN bookmarks b ON b.id = cp.bookmark_id

  WHERE cp.user_id = p_user_id

    AND (1 - (cp.embedding <=> p_embedding)) > (1 - p_radius)

  ORDER BY distance DESC

  LIMIT p_limit;

END;

$$ LANGUAGE plpgsql STABLE;

```

### 11.9 核心理念总结

> **"废除空文件夹"**：系统里不允许存在空文件夹。文件夹是内容的**涌现 (Emergence)**，而不是容器。

>

> **"延迟满足"**：单篇书签允许暂时"无家可归"。只有积累了足够的**语义质量 (Semantic Mass)**，才允许坍缩成文件夹。

>

> **"动态重构"**：文件夹的名字不是永恒的。随着内容增加，AI 会自动把"前端"分裂成"React"和"Vue"，或者把"川菜"和"粤菜"合并成"中式烹饪"。

**这种设计将彻底消除"马太效应"和"命名模糊"问题，因为每一个文件夹都是由其内容本身"定义"的，而不是由用户拍脑袋决定的。**
