## 1. 集成总览

### 1.1 集成原则

```
✅ 核心原则：渐进式集成，零破坏性改动

1. 包装器模式：v6.0 包装现有服务，不修改核心代码
```

## 3. 集成点设计

### 3.1 集成点 1: 书签创建流程

**位置**: `src/app/api/bookmarks/route.ts`

**现状代码**:

```typescript

  shouldUseV2Classification,
```

**v6.0 集成方案**:

```typescript
// src/lib/classification/v6/router.ts (新增)
import { classifyBookmarkV2 } from "@/lib/classification/server";
import { V6ClassificationService } from "./v6-classification.service";
import { V6_FEATURE_FLAGS } from "./config";
import { logger } from "@/lib/logger";

/**
 * V6 版本路由器
 * 根据 Feature Flags 和用户 ID 决定使用 v5 或 v6
 */
export async function classifyBookmark(
  userId: string,
  bookmark: {
    title: string;
    summary?: string;
    tags?: string[];
    embedding?: number[];
    url?: string;
  },
) {
  // 1. 检查是否启用 v6
  if (!V6_FEATURE_FLAGS.V6_ENABLE_L1_FASTPASS) {
    logger.debug(
      { userId, version: "v5" },
      "V6: Feature flag disabled, using v5",
    );
    return classifyBookmarkV2(userId, bookmark);
  }

  // 2. 灰度分流：根据用户 ID 哈希决定是否使用 v6
  if (!shouldUseV6(userId)) {
    logger.debug(
      { userId, version: "v5" },
      "V6: User not in rollout, using v5",
    );
    return classifyBookmarkV2(userId, bookmark);
  }

  // 3. 使用 v6 三级漏斗
  logger.info(
    { userId, version: "v6" },
    "V6: Using v6 tri-funnel classification",
  );

  const v6Service = new V6ClassificationService();
  return v6Service.classify(userId, bookmark);
}

/**
 * 灰度分流逻辑
 * 根据用户 ID 哈希 + 配置的百分比决定是否使用 v6
 */
function shouldUseV6(userId: string): boolean {
  const rolloutPercentage = V6_FEATURE_FLAGS.V6_ROLLOUT_PERCENTAGE; // 0-100

  if (rolloutPercentage === 0) return false;
  if (rolloutPercentage === 100) return true;

  // 简单哈希分流
  const hash = userId.split("").reduce((acc, char) => {
    return acc + char.charCodeAt(0);
  }, 0);

  return hash % 100 < rolloutPercentage;
}
```

**v6.0 集成方案**:

```typescript
// src/lib/classification/v6/wrapper.ts (新增)
import type { IClassificationService } from "../services/interfaces";
import { V6ClassificationService } from "./v6-classification.service";
import { V6_FEATURE_FLAGS } from "./config";

/**
 * V6 版本包装器
 * 包装 TrustModeClassificationService，实现版本路由
 */
export function createV6VersionRouter(
  v5Service: IClassificationService,
): IClassificationService {
  const v6Service = new V6ClassificationService();

  return {
    async classify(userId, bookmark) {
      // 版本路由逻辑（同 3.1）
      if (!V6_FEATURE_FLAGS.V6_ENABLE_L1_FASTPASS || !shouldUseV6(userId)) {
        return v5Service.classify(userId, bookmark);
      }

      // 调用 v6
      const v6Result = await v6Service.classify(userId, bookmark);

      // 🔄 转换为 v5 兼容格式（如果需要）
      return normalizeV6Result(v6Result);
    },
  };
}

/**
 * 将 v6 结果转换为 v5 兼容格式
 */
function normalizeV6Result(v6Result: any): ClassificationResult {
  return {
    // v5 标准字段
    folderId: v6Result.folderId,
    folderName: v6Result.folderName,
    confidence: v6Result.confidence,
    action: v6Result.action,
    reason: v6Result.reason,
    scores: v6Result.scores || { vectorScore: 0, historyScore: 0, aiScore: 0 },

    // v6 扩展字段（可选）
    ...(v6Result.layer && { layer: v6Result.layer }), // 'L1', 'L2', 'L3'
    ...(v6Result.governanceSuggestion && {
      governanceSuggestion: v6Result.governanceSuggestion,
    }),
  };
}
```

## 6. 数据库迁移方案

### 6.1 迁移脚本

**位置**: `supabase/migrations/20260121000001_add_v6_tables.sql`

```sql
-- =====================================================
-- V6.0 智能分类三级漏斗架构 - 数据库迁移
-- =====================================================

-- 1. 贝叶斯索引表（可选，也可纯文件存储）
CREATE TABLE IF NOT EXISTS classification_bayesian_indexes (
  user_id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  index_data JSONB NOT NULL,  -- 完整的 BayesianIndex JSON
  version BIGINT NOT NULL,    -- 时间戳版本号
  total_docs INT NOT NULL,
  total_tokens INT NOT NULL,
  size_kb NUMERIC(10,2),
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

COMMENT ON TABLE classification_bayesian_indexes IS 'V6.0 L1 贝叶斯索引存储';
COMMENT ON COLUMN classification_bayesian_indexes.index_data IS '包含 folders 和 tokens 的完整索引';
COMMENT ON COLUMN classification_bayesian_indexes.version IS '索引版本号（时间戳），用于客户端缓存校验';

-- 索引
CREATE INDEX idx_bayesian_indexes_version
  ON classification_bayesian_indexes(version);

-- RLS 策略
ALTER TABLE classification_bayesian_indexes ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Users can view own index"
  ON classification_bayesian_indexes
  FOR SELECT
  USING (auth.uid() = user_id);

CREATE POLICY "System can manage indexes"
  ON classification_bayesian_indexes
  FOR ALL
  USING (auth.uid() = user_id);

-- =====================================================

-- 2. 治理建议表
CREATE TABLE IF NOT EXISTS classification_governance_suggestions (
  id BIGSERIAL PRIMARY KEY,
  user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
  folder_id BIGINT NOT NULL REFERENCES folders(id) ON DELETE CASCADE,
  suggestion_type TEXT NOT NULL CHECK (suggestion_type IN ('rename', 'merge', 'split')),
  current_name TEXT,
  suggested_name TEXT,
  reason TEXT,
  confidence NUMERIC(3,2),
```

```typescript
"use client";

import { useEffect } from "react";
import { toast } from "sonner";

interface GovernanceSuggestion {
  suggestionId: number;
  folderId: number;

  reason: string;

export function showGovernanceToast(suggestion: GovernanceSuggestion) {
  toast.custom(
    (t) => (
      <div className="rounded-lg bg-card p-4 shadow-lg border">
        <div className="flex items-start gap-3">
          <div className="text-2xl">💡</div>

            </p>
          </div>
        </div>

        <div className="flex gap-2 mt-3">
          <button
            onClick={async () => {
              await applyRename(suggestion.suggestionId);
              toast.success(`已重命名为 "${suggestion.renameTo}"`);
              toast.dismiss(t);
            }}
            className="flex-1 px-3 py-1.5 text-sm bg-primary text-primary-foreground rounded hover:bg-primary/90"
          >
            <Check className="w-4 h-4 inline mr-1" />
            确认重命名
          </button>
          <button
            onClick={() => {
              rejectSuggestion(suggestion.suggestionId);
              toast.dismiss(t);
            }}
            className="px-3 py-1.5 text-sm border rounded hover:bg-accent"
          >
            <X className="w-4 h-4 inline mr-1" />
            忽略
          </button>
        </div>
      </div>
    ),
    {
      duration: 10000, // 10 秒
      position: "bottom-right",
    },
  );
}

async function applyRename(suggestionId: number) {
  await fetch("/api/classification/v6/governance/apply", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ suggestionId, action: "accept" }),
  });
}

async function rejectSuggestion(suggestionId: number) {
  await fetch("/api/classification/v6/governance/apply", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ suggestionId, action: "ignore" }),
  });
}
```

- ❌ P90 延迟 > 1000ms
- ❌ 数据库连接池耗尽

**回滚步骤**:

```bash
# 1. 禁用 Feature Flag（无需重启）
curl -X POST https://anykeep.com/api/admin/feature-flags \
  -H "Authorization: Bearer $ADMIN_TOKEN" \
  -d '{"V6_ENABLE_L1_FASTPASS": false}'

# 2. 验证所有请求走 v5
tail -f /var/log/anykeep/classification.log | grep "version.*v5"

# 3. 通知团队
slack-notify "#dev-alerts" "V6 已紧急回滚到 v5，请检查日志"
```

```typescript
// src/lib/classification/v6/config.ts

// 维护一个黑名单（临时措施）
const V6_BLACKLIST_USERS = new Set([

function shouldUseV6(userId: string): boolean {
  // 黑名单用户强制使用 v5
  if (V6_BLACKLIST_USERS.has(userId)) {

  // 正常灰度逻辑

}
```
