## 一、优化背景

基于 v1.0 方案的架构师审核，发现以下 **3 个逻辑盲区** 和 **2 个架构一致性风险**：

| 序号 | 风险类型         | 问题描述                                       | 严重程度 |
| ---- | ---------------- | ---------------------------------------------- | -------- |
| 1    | SEO 垃圾标签     | Meta keywords 可能包含 SEO 堆砌词，导致误分类  | 🔴 高    |
| 2    | 字符串匹配脆弱性 | "Vue.js" vs "VueJS"、"Golang" vs "Go" 无法匹配 | 🔴 高    |
| 3    | 部分匹配危险性   | "Java" 部分匹配 "JavaScript" 导致误判          | 🟡 中    |
| 4    | 黑名单缺失       | 无法排除已知错误的标签-文件夹关联              | 🟡 中    |
| 5    | 前端截胡缺失     | 未利用客户端预匹配减少后端计算                 | 🟢 低    |

### 3.1 标签可信度过滤器 (新增)

```typescript
// tag-trust-filter.ts (新增文件)

/**
 * 标签来源等级
 */
export enum TagSourceLevel {
  /** Level S: 高信 - 用户手动/LLM 生成 */
  S = "S",
  /** Level B: 中信 - OpenGraph article:tag */
  B = "B",
  /** Level C: 低信 - Meta keywords (SEO 重灾区) */
  C = "C",
}

/**
 * 标签来源权重
 */
export const TAG_SOURCE_WEIGHTS: Record<TagSourceLevel, number> = {
  [TagSourceLevel.S]: 1.0,
  [TagSourceLevel.B]: 0.8,
  [TagSourceLevel.C]: 0.3,
};

/**
 * 全局停用词表（高 IDF 万金油词）
 */
export const GLOBAL_STOP_TAGS = new Set([
  // 通用无意义词
  "home",
  "page",
  "article",
  "post",
  "blog",
  "news",
  "content",
  "首页",
  "文章",
  "帖子",
  "博客",
  "新闻",
  "内容",
  "分享",
  // SEO 常见堆砌词
  "best",
  "top",
  "guide",
  "tutorial",
  "how to",
  "tips",
  "最佳",
  "指南",
  "教程",
  "技巧",
  "入门",
  "精选",
  // 过于宽泛的技术词
  "technology",
  "tech",
  "software",
  "internet",
  "web",
  "online",
  "技术",
  "科技",
  "软件",
  "互联网",
  "网络",
]);

/**
 * 带权重的标签
 */
export interface WeightedTag {
  tag: string;
  normalizedTag: string;
  sourceLevel: TagSourceLevel;
  weight: number;
  isFiltered: boolean;
  filterReason?: string;
}

/**
 * 标签可信度过滤器
 */
export class TagTrustFilter {
  private userBlacklist: Map<string, Set<string>>; // userId -> folderId -> blacklistTags

  constructor() {
    this.userBlacklist = new Map();
  }

  /**
   * 过滤并加权标签
   */
  filterAndWeight(
    tags: string[],
    sourceLevel: TagSourceLevel,
    userId?: string,
    targetFolderId?: string,
  ): WeightedTag[] {
    const result: WeightedTag[] = [];

    for (const tag of tags) {
      const normalizedTag = this.normalize(tag);
      const baseWeight = TAG_SOURCE_WEIGHTS[sourceLevel];

      // 检查全局停用词
      if (GLOBAL_STOP_TAGS.has(normalizedTag)) {
        result.push({
          tag,
          normalizedTag,
          sourceLevel,
          weight: 0,
          isFiltered: true,
          filterReason: "global_stop_tag",
        });
        continue;
      }

      // 检查用户黑名单
      if (
        userId &&
        targetFolderId &&
        this.isBlacklisted(userId, targetFolderId, normalizedTag)
      ) {
        result.push({
          tag,
          normalizedTag,
          sourceLevel,
          weight: 0,
          isFiltered: true,
          filterReason: "user_blacklist",
        });
        continue;
      }

      // 标签长度过滤（过短或过长的标签可信度低）
      let lengthPenalty = 1.0;
      if (normalizedTag.length < 2) {
        lengthPenalty = 0.3;
      } else if (normalizedTag.length > 30) {
        lengthPenalty = 0.5;
      }

      result.push({
        tag,
        normalizedTag,
        sourceLevel,
        weight: baseWeight * lengthPenalty,
        isFiltered: false,
      });
    }

    return result;
  }

  /**
   * 激进标准化
   * "Vue.js" → "vuejs"
   * "Next.js" → "nextjs"
   * "C++" → "cpp"
   */
  private normalize(tag: string): string {
    return tag
      .toLowerCase()
      .replace(/[.\-_\s]/g, "") // 去除点、横杠、下划线、空格
      .replace(/\+\+/g, "pp") // C++ → cpp
      .replace(/#/g, "sharp") // C# → csharp
      .trim();
  }

  /**
   * 检查是否在用户黑名单中
   */
  private isBlacklisted(
    userId: string,
    folderId: string,
    tag: string,
  ): boolean {
    const userBlacklist = this.userBlacklist.get(userId);
    if (!userBlacklist) return false;

    const folderBlacklist = userBlacklist.get(folderId);
    if (!folderBlacklist) return false;

    return folderBlacklist.has(tag);
  }

  /**
   * 添加标签到黑名单
   */
  addToBlacklist(userId: string, folderId: string, tag: string): void {
    if (!this.userBlacklist.has(userId)) {
      this.userBlacklist.set(userId, new Map());
    }
    const userMap = this.userBlacklist.get(userId)!;

    if (!userMap.has(folderId)) {
      userMap.set(folderId, new Set());
    }
    userMap.get(folderId)!.add(this.normalize(tag));
  }

  /**
   * 仅返回可用于直接匹配的高信标签（Level S/B）
   */
  getDirectMatchableTags(weightedTags: WeightedTag[]): string[] {
    return weightedTags
      .filter((wt) => !wt.isFiltered && wt.sourceLevel !== TagSourceLevel.C)
      .map((wt) => wt.normalizedTag);
  }
}

export const tagTrustFilter = new TagTrustFilter();
```

### 3.2 增强型标签标准化器 (新增)

```typescript
// tag-normalizer.ts (新增文件)

import { tagSynonymMapper } from "./tag-synonym-mapper";

/**
 * 标准化结果
 */
export interface NormalizationResult {
  original: string;
  normalized: string;
  variants: string[]; // 所有变体形式（同义词 + 标准化变体）
}

/**
 * ���强型标签标准化器
 */
export class TagNormalizer {
  /**
   * 激进标准化
   * 用于匹配比较，去除所有可能导致不匹配的字符
   */
  normalize(tag: string): string {
    return tag
      .toLowerCase()
      .replace(/[\s\-_.·/|]/g, "") // 去除所有分隔符
      .replace(/\+\+/g, "pp") // C++ → cpp
      .replace(/#/g, "sharp") // C# → csharp
      .replace(/[^a-z0-9\u4e00-\u9fa5]/g, "") // 仅保留字母数字和中文
      .trim();
  }

  /**
   * 获取标签的所有可能形式
   */
  getAllVariants(tag: string): NormalizationResult {
    const normalized = this.normalize(tag);
    const synonyms = tagSynonymMapper.getSynonyms(tag);

    const variants = new Set<string>();

    // 添加原始标准化形式
    variants.add(normalized);

    // 添加所有同义词的标准化形式
    for (const synonym of synonyms) {
      variants.add(this.normalize(synonym));
    }

    return {
      original: tag,
      normalized,
      variants: Array.from(variants),
    };
  }

  /**
   * 检查两个标签是否等价（标准化后相等或互为同义词）
   */
  areEquivalent(tag1: string, tag2: string): boolean {
    const norm1 = this.normalize(tag1);
    const norm2 = this.normalize(tag2);

    // 标准化后直接相等
    if (norm1 === norm2) return true;

    // 检查同义词
    return tagSynonymMapper.areSynonyms(tag1, tag2);
  }

  /**
   * 批量标准化并去重
   */
  normalizeAndDedupe(tags: string[]): string[] {
    const seen = new Set<string>();
    const result: string[] = [];

    for (const tag of tags) {
      const normalized = this.normalize(tag);
      if (!seen.has(normalized)) {
        seen.add(normalized);
        result.push(normalized);
      }
    }

    return result;
  }
}

export const tagNormalizer = new TagNormalizer();
```

### 3.3 文件夹别名管理器 (新增)

```typescript
// folder-alias-manager.ts (新增文件)

import { createLogger } from "@/lib/logger";
import type { SupabaseClient } from "@supabase/supabase-js";
import { tagNormalizer } from "./tag-normalizer";

const logger = createLogger("folder-alias-manager");

/**
 * 别名学习记录
 */
interface AliasLearningRecord {
  tag: string;
  folderId: string;
  hitCount: number;
  lastHitAt: Date;
}

/**
 * 文件夹别名管理器
 *
 * 功能：
 * 1. 管理文件夹的手动别名
 * 2. 自动学习用户行为中的隐含别名
 */
export class FolderAliasManager {
  // 别名缓存: userId -> folderId -> aliases[]
  private aliasCache: Map<string, Map<string, string[]>>;
  // 学习中的别名: userId -> AliasLearningRecord[]
  private learningRecords: Map<string, AliasLearningRecord[]>;

  // 自动学习阈值
  private static readonly LEARNING_THRESHOLD = 3; // 相同标签-文件夹组合出现 3 次后自动添加别名

  constructor(private supabase: SupabaseClient) {
    this.aliasCache = new Map();
    this.learningRecords = new Map();
  }

  /**
   * 获取文件夹的所有别名（包括标准化的文件夹名）
   */
  async getAliases(
    userId: string,
    folderId: string,
    folderName: string,
  ): Promise<string[]> {
    // 缓存检查
    const userCache = this.aliasCache.get(userId);
    if (userCache?.has(folderId)) {
      return userCache.get(folderId)!;
    }

    // 从数据库加载
    const { data, error } = await this.supabase
      .from("folder_tag_profiles")
      .select("aliases")
      .eq("user_id", userId)
      .eq("folder_id", parseInt(folderId))
      .single();

    const dbAliases: string[] = data?.aliases || [];

    // 始终包含文件夹名的标准化形式
    const normalizedName = tagNormalizer.normalize(folderName);
    const allAliases = new Set([
      normalizedName,
      ...dbAliases.map((a) => tagNormalizer.normalize(a)),
    ]);

    const result = Array.from(allAliases);

    // 更新缓存
    if (!this.aliasCache.has(userId)) {
      this.aliasCache.set(userId, new Map());
    }
    this.aliasCache.get(userId)!.set(folderId, result);

    return result;
  }

  /**
   * 检查标签是否匹配文件夹别名
   */
  async matchesAlias(
    userId: string,
    folderId: string,
    folderName: string,
    tag: string,
  ): Promise<boolean> {
    const aliases = await this.getAliases(userId, folderId, folderName);
    const normalizedTag = tagNormalizer.normalize(tag);
    return aliases.includes(normalizedTag);
  }

  /**
   * 记录学习事件（用户将带有某标签的书签放入某文件夹）
   */
  recordLearningEvent(userId: string, folderId: string, tag: string): void {
    const normalizedTag = tagNormalizer.normalize(tag);

    if (!this.learningRecords.has(userId)) {
      this.learningRecords.set(userId, []);
    }

    const records = this.learningRecords.get(userId)!;
    const existing = records.find(
      (r) => r.folderId === folderId && r.tag === normalizedTag,
    );

    if (existing) {
      existing.hitCount++;
      existing.lastHitAt = new Date();

      // 达到阈值，自动添加别名
      if (existing.hitCount >= FolderAliasManager.LEARNING_THRESHOLD) {
        this.addAlias(userId, folderId, normalizedTag);
        // 从学习列表移除
        const index = records.indexOf(existing);
        records.splice(index, 1);
      }
    } else {
      records.push({
        tag: normalizedTag,
        folderId,
        hitCount: 1,
        lastHitAt: new Date(),
      });
    }
  }

  /**
   * 手动添加别名
   */
  async addAlias(
    userId: string,
    folderId: string,
    alias: string,
  ): Promise<void> {
    const normalizedAlias = tagNormalizer.normalize(alias);

    // 更新数据库
    const { data: existing } = await this.supabase
      .from("folder_tag_profiles")
      .select("aliases")
      .eq("user_id", userId)
      .eq("folder_id", parseInt(folderId))
      .single();

    const currentAliases: string[] = existing?.aliases || [];
    if (!currentAliases.includes(normalizedAlias)) {
      currentAliases.push(normalizedAlias);

      await this.supabase
        .from("folder_tag_profiles")
        .update({ aliases: currentAliases })
        .eq("user_id", userId)
        .eq("folder_id", parseInt(folderId));
    }

    // 清除缓存
    this.clearCache(userId, folderId);

    logger.info(
      { userId, folderId, alias: normalizedAlias },
      "已添加文件夹别名",
    );
  }

  /**
   * 清除缓存
   */
  clearCache(userId: string, folderId?: string): void {
    if (folderId) {
      this.aliasCache.get(userId)?.delete(folderId);
    } else {
      this.aliasCache.delete(userId);
    }
  }
}
```

### 3.4 增强型 AI 标签匹配器 (修改)

```typescript
// ai-tag-matcher.ts (v1.1 增强版 - ���键修改部分)

import {
  tagTrustFilter,
  TagSourceLevel,
  type WeightedTag,
} from "./tag-trust-filter";
import { tagNormalizer } from "./tag-normalizer";
import { FolderAliasManager } from "./folder-alias-manager";

/**
 * 匹配模式
 */
export type MatchMode =
  | "exact" // 精确匹配 - 直接返回
  | "normalized" // 标准化匹配 - 直接返回
  | "partial" // 部分匹配 - 传递给 T0
  | "core_tag" // 核心标签匹配
  | "related_tag"; // 相关标签匹配

/**
 * 增强型匹配结果
 */
export interface EnhancedTagMatchResult extends TagMatchResult {
  /** 匹配模式 */
  matchMode: MatchMode;
  /** 是否需要 T0 校验 */
  needsT0Validation: boolean;
  /** 标签分数（用于 Rerank） */
  tagScore: number;
}

export class AiTagMatcher {
  private folderAliasManager: FolderAliasManager;

  constructor(
    private supabase: SupabaseClient,
    config?: Partial<TagMatcherConfig>,
  ) {
    this.config = { ...DEFAULT_CONFIG, ...config };
    this.folderProfiles = new Map();
    this.folderAliasManager = new FolderAliasManager(supabase);
  }

  /**
   * 增强型主匹配方法
   */
  async matchEnhanced(
    request: ClassificationRequest,
    tagSourceLevel: TagSourceLevel = TagSourceLevel.S,
  ): Promise<EnhancedTagMatchResult | null> {
    const startTime = Date.now();

    if (!request.tags || request.tags.length === 0) {
      return null;
    }

    // Phase 0.1: 标签可信度过滤
    const weightedTags = tagTrustFilter.filterAndWeight(
      request.tags,
      tagSourceLevel,
      request.userId,
    );

    const usableTags = weightedTags.filter((wt) => !wt.isFiltered);
    if (usableTags.length === 0) {
      logger.debug({ bookmarkId: request.bookmarkId }, "所有标签被过滤");
      return null;
    }

    const profiles = await this.getFolderProfiles(request.userId);
    if (profiles.length === 0) {
      return null;
    }

    // Phase 0.2: 标准化处理
    const normalizedTags = usableTags.map((wt) => ({
      ...wt,
      normalized: tagNormalizer.normalize(wt.tag),
    }));

    // 同义词扩展
    const expandedTags = new Set<string>();
    for (const wt of normalizedTags) {
      const variants = tagNormalizer.getAllVariants(wt.tag);
      variants.variants.forEach((v) => expandedTags.add(v));
    }
    const allTags = Array.from(expandedTags);

    // Phase 0.3: 分层匹配

    // 策略 1: 精确匹配（标准化后完全相等）
    const exactMatch = await this.matchExactNormalized(
      allTags,
      profiles,
      request.userId,
    );
    if (exactMatch) {
      const duration = Date.now() - startTime;
      return {
        ...exactMatch,
        matchMode: "exact",
        needsT0Validation: false,
        tagScore: 1.0,
      };
    }

    // 策略 2: 别名匹配
    const aliasMatch = await this.matchAliases(
      allTags,
      profiles,
      request.userId,
    );
    if (aliasMatch) {
      const duration = Date.now() - startTime;
      return {
        ...aliasMatch,
        matchMode: "normalized",
        needsT0Validation: false,
        tagScore: 0.95,
      };
    }

    // 策略 3: 部分匹配（需要 T0 校验）
    const partialMatch = this.matchPartialSafe(allTags, profiles);
    if (partialMatch) {
      // ⚠️ 部分匹配不直接返回，而是传递给 T0 作为 Feature
      return {
        ...partialMatch,
        matchMode: "partial",
        needsT0Validation: true, // 必须经过 T0 向量校验
        tagScore: 0.6, // 作为 Rerank 的输入分数
      };
    }

    // 策略 4: 核心标签匹配（高信标签才触发）
    if (tagSourceLevel !== TagSourceLevel.C) {
      const coreMatch = this.matchCoreTags(allTags, profiles);
      if (coreMatch) {
        return {
          ...coreMatch,
          matchMode: "core_tag",
          needsT0Validation: coreMatch.confidence < 0.9,
          tagScore: coreMatch.confidence,
        };
      }
    }

    return null;
  }

  /**
   * 策略 1: 标准化后精确匹配
   */
  private async matchExactNormalized(
    tags: string[],
    profiles: FolderTagProfile[],
    userId: string,
  ): Promise<TagMatchResult | null> {
    for (const profile of profiles) {
      const normalizedFolderName = tagNormalizer.normalize(profile.folderName);

      for (const tag of tags) {
        if (tag === normalizedFolderName) {
          return {
            matched: true,
            folderId: profile.folderId.toString(),
            folderName: profile.folderName,
            confidence: this.config.exactMatchConfidence,
            matchType: "exact_name",
            matchedTags: [tag],
            reason: `标签 "${tag}" 与文件夹名完全匹配（标准化）`,
          };
        }
      }
    }
    return null;
  }

  /**
   * 策略 2: 别名匹配
   */
  private async matchAliases(
    tags: string[],
    profiles: FolderTagProfile[],
    userId: string,
  ): Promise<TagMatchResult | null> {
    for (const profile of profiles) {
      for (const tag of tags) {
        const matches = await this.folderAliasManager.matchesAlias(
          userId,
          profile.folderId.toString(),
          profile.folderName,
          tag,
        );

        if (matches) {
          return {
            matched: true,
            folderId: profile.folderId.toString(),
            folderName: profile.folderName,
            confidence: this.config.exactMatchConfidence - 0.03,
            matchType: "exact_name",
            matchedTags: [tag],
            reason: `标签 "${tag}" 匹配文件夹别名`,
          };
        }
      }
    }
    return null;
  }

  /**
   * 策略 3: 安全的部分匹配
   *
   * ⚠️ 防止 "Java" 误匹配 "JavaScript" 的情况
   */
  private matchPartialSafe(
    tags: string[],
    profiles: FolderTagProfile[],
  ): TagMatchResult | null {
    for (const profile of profiles) {
      const normalizedFolderName = tagNormalizer.normalize(profile.folderName);

      for (const tag of tags) {
        // 只有当一方完全包含另一方，且长度比例合理时才匹配
        const isContained =
          normalizedFolderName.includes(tag) ||
          tag.includes(normalizedFolderName);

        if (!isContained) continue;

        // 长度比例检查：避免 "java" 匹配 "javascript"
        const shorter = Math.min(tag.length, normalizedFolderName.length);
        const longer = Math.max(tag.length, normalizedFolderName.length);
        const ratio = shorter / longer;

        // 至少 70% 的重叠才算安全
        if (ratio < 0.7) {
          logger.debug(
            { tag, folderName: profile.folderName, ratio },
            "部分匹配被拒绝：长度比例不足",
          );
          continue;
        }

        return {
          matched: true,
          folderId: profile.folderId.toString(),
          folderName: profile.folderName,
          confidence: 0.6, // 低置信度，需要 T0 校验
          matchType: "exact_name",
          matchedTags: [tag],
          reason: `标签 "${tag}" 与文件夹名 "${profile.folderName}" 部分匹配（需校验）`,
        };
      }
    }
    return null;
  }
}
```

### 3.5 T0 Rerank 机制 (修改)

```typescript
// hybrid-retriever.ts 增强部分

/**
 * 应用标签加成的 Rerank
 *
 * v1.1: 使用加分制而非替换制
 */
applyTagBoostRerank(
  results: RetrievalResult[],
  tagMatchResult: EnhancedTagMatchResult | null,
): RetrievalResult[] {
  if (!tagMatchResult || !tagMatchResult.needsT0Validation) {
    return results;
  }

  // 找到标签匹配的文件夹
  const targetFolderId = tagMatchResult.folderId;
  const tagScore = tagMatchResult.tagScore;

  return results.map(result => {
    if (result.folderId === targetFolderId) {
      // Rerank 公式: FinalScore = VectorScore × 0.7 + TagScore × 0.3
      const vectorScore = result.score;
      const finalScore = vectorScore * 0.7 + tagScore * 0.3;

      logger.debug({
        folderId: result.folderId,
        vectorScore,
        tagScore,
        finalScore,
      }, 'Rerank 应用标���加成');

      return {
        ...result,
        score: finalScore,
        metadata: {
          ...result.metadata,
          tagBoostApplied: true,
          originalVectorScore: vectorScore,
        },
      };
    }
    return result;
  }).sort((a, b) => b.score - a.score);
}
```
