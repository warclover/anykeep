### 1.1 后端任务

#### 1.1.1 操作快照表设计 (2天)

```sql
-- 操作快照表（支持 2 天撤销）
CREATE TABLE operation_snapshots (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES auth.users(id),
  action TEXT NOT NULL CHECK (action IN ('create', 'split', 'merge', 'rename', 'move')),
  folder_id BIGINT,
  folder_ids BIGINT[],
  bookmark_ids BIGINT[],
  before_state JSONB NOT NULL,
  after_state JSONB NOT NULL,
  reason TEXT,
  is_undone BOOLEAN DEFAULT FALSE,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  expires_at TIMESTAMPTZ DEFAULT (NOW() + INTERVAL '2 days'),
  CONSTRAINT valid_expiry CHECK (expires_at > created_at)
);

CREATE INDEX idx_snapshots_user_created ON operation_snapshots(user_id, created_at DESC);
CREATE INDEX idx_snapshots_expires ON operation_snapshots(expires_at) WHERE NOT is_undone;
```
