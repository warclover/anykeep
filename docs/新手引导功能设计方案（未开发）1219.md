## 核心设计原则

### 🔒 三大铁律

| 原则           | 要求                         | 实现方式                                 |
| -------------- | ---------------------------- | ---------------------------------------- |
| **完全解耦**   | 不修改任何已有业务代码       | 独立模块 + Portal 渲染 + data 属性探测   |
| **用户可控**   | 一键跳过、一键关闭、随时退出 | 全局快捷键 + 醒目关闭按钮 + 点击遮罩关闭 |
| **零干扰降级** | 模块故障不影响主功能         | 错误边界 + 懒加载 + 静默失败             |

### 1.2 零侵入实现策略

| 策略              | 说明                                           | 示例                                               |
| ----------------- | ---------------------------------------------- | -------------------------------------------------- |
| **Portal 渲染**   | 所有引导 UI 渲染到 body 末尾，不嵌入业务组件树 | `createPortal(<Overlay/>, document.body)`          |
| **data 属性探测** | 通过 data-tour 属性定位元素，无需业务组件感知  | `document.querySelector('[data-tour="search"]')`   |
| **样式隔离**      | 引导组件使用独立 CSS 类名前缀，避免样式冲突    | `.onboarding-*`                                    |
| **懒加载**        | 引导模块按需加载，不增加首屏 bundle            | `dynamic(() => import('@/components/onboarding'))` |
| **错误边界**      | 引导模块异常不影响主应用                       | `<OnboardingErrorBoundary>`                        |

```typescript
// src/features/onboarding/hooks/use-keyboard.ts

export function useOnboardingKeyboard() {
  useEffect(() => {
    function handleKeyDown(e: KeyboardEvent) {
      // ESC - 关闭当前引导
      if (e.key === "Escape") {
        onboarding.closeCurrentStep();
      }

      // Ctrl/Cmd + Shift + O - 切换引导开关
      if ((e.ctrlKey || e.metaKey) && e.shiftKey && e.key === "O") {
        onboarding.toggleEnabled();
      }
    }

    document.addEventListener("keydown", handleKeyDown);
    return () => document.removeEventListener("keydown", handleKeyDown);
  }, []);
}
```

```typescript
// src/features/onboarding/hooks/use-smart-trigger.ts

/**
 * 智能触发检测 - 确保在合适的时机显示引导
 */
export function useSmartTrigger() {
  const [canShow, setCanShow] = useState(false);

  useEffect(
    () => {
      // 检测条件
      const checks = {
        // 1. 页面已完全加载
        pageLoaded: document.readyState === "complete",

        // 2. 用户已停止操作（空闲检测）
        userIdle: checkUserIdle(SETTINGS.userIdleThreshold),

        // 3. 没有其他模态框打开
        noModalOpen: !document.querySelector('[role="dialog"]'),

        // 4. 不在输入状态
        notTyping:
          document.activeElement?.tagName !== "INPUT" &&
          document.activeElement?.tagName !== "TEXTAREA",

        // 5. 距离上次引导已过最小间隔
        intervalPassed: checkMinInterval(),

        // 6. 本次会话未超过最大显示数
        underSessionLimit: checkSessionLimit(),
      };

      setCanShow(Object.values(checks).every(Boolean));
    },
    [
      /* dependencies */
    ],
  );

  return canShow;
}
```

### 3.3 非阻断式 Tooltip

```tsx
// src/features/onboarding/components/TooltipHint.tsx

/**
 * 轻量级提示组件 - 完全不阻断用户操作
 */
export function TooltipHint({ step }: { step: OnboardingStep }) {
  const targetRect = useTargetElement(step.targetSelector);
  const [visible, setVisible] = useState(true);

  // 自动消失
  useEffect(() => {
    if (SETTINGS.tooltipAutoDismiss > 0) {
      const timer = setTimeout(() => {
        setVisible(false);
        onboarding.completeStep(step.id);
      }, SETTINGS.tooltipAutoDismiss);
      return () => clearTimeout(timer);
    }
  }, []);

  if (!visible || !targetRect) return null;

  return createPortal(
    <div
      className="onboarding-tooltip"
      style={{
        position: "fixed",
        ...calculatePosition(targetRect, step.placement),
        // 关键：不拦截点击事件
        pointerEvents: "none",
      }}
    >
      <div
        className="onboarding-tooltip-content"
        style={{ pointerEvents: "auto" }} // 仅内容区域可交互
      >
        <button
          className="onboarding-tooltip-close"
          onClick={() => {
            setVisible(false);
            onboarding.dismissStep(step.id);
          }}
        >
          ×
        </button>
        <p className="onboarding-tooltip-title">{step.title}</p>
        <p className="onboarding-tooltip-desc">{step.description}</p>
      </div>
    </div>,
    document.body,
  );
}
```

### 4.2 ErrorBoundary 实现

```tsx
// src/features/onboarding/components/OnboardingRoot.tsx

import { Component, ReactNode } from "react";

interface Props {
  children: ReactNode;
}

interface State {
  hasError: boolean;
}

class OnboardingErrorBoundary extends Component<Props, State> {
  state: State = { hasError: false };

  static getDerivedStateFromError(): State {
    return { hasError: true };
  }

  componentDidCatch(error: Error, errorInfo: React.ErrorInfo) {
    // 静默记录，不打扰用户
    console.warn("[Onboarding] Error caught:", error.message);

    // 可选：发送到错误监控服务
    // reportError(error, errorInfo);
  }

  render() {
    if (this.state.hasError) {
      // 返回空，引导失败不影响主应用
      return null;
    }
    return this.props.children;
  }
}

/**
 * 引导模块根组件
 * 使用错误边界包裹，确保故障不影响主应用
 */
export function OnboardingRoot() {
  return (
    <OnboardingErrorBoundary>
      <OnboardingProvider>
        <OnboardingLayer />
      </OnboardingProvider>
    </OnboardingErrorBoundary>
  );
}
```

### 4.3 存储降级策略

```typescript
// src/features/onboarding/store/storage.ts

/**
 * 多级存储策略
 * 优先级：内存 → LocalStorage → 数据库
 * 任何一层失败都不影响功能
 */

// 内存缓存（最快，但会话丢失）
let memoryCache: OnboardingProgress | null = null;

export async function saveProgress(
  progress: OnboardingProgress,
): Promise<void> {
  // 1. 始终保存到内存（最可靠）
  memoryCache = progress;

  // 2. 尝试保存到 LocalStorage（静默失败）
  try {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(progress));
  } catch (e) {
    console.warn("[Onboarding] LocalStorage save failed:", e);
    // 不抛出错误，继续执行
  }

  // 3. 异步同步到数据库（完全后台，不阻塞）
  syncToDatabase(progress).catch((e) => {
    console.warn("[Onboarding] Database sync failed:", e);
    // 下次会话重试
  });
}

export function loadProgress(): OnboardingProgress {
  // 1. 优先从内存读取
  if (memoryCache) return memoryCache;

  // 2. 尝试从 LocalStorage 读取
  try {
    const stored = localStorage.getItem(STORAGE_KEY);
    if (stored) {
      memoryCache = JSON.parse(stored);
      return memoryCache;
    }
  } catch (e) {
    console.warn("[Onboarding] LocalStorage read failed:", e);
  }

  // 3. 返回默认初始状态
  return getInitialProgress();
}
```

### 6.1 状态结构

```typescript
// src/features/onboarding/types.ts

/**
 * 引导进度
 */
export interface OnboardingProgress {
  /** 是否已全局禁用引导 */
  disabled: boolean;

  /** Wizard 是否已完成 */
  wizardCompleted: boolean;
  wizardCompletedAt?: string;

  /** 各步骤状态 */
  steps: Record<string, StepStatus>;

  /** 已 dismiss 的步骤（不再提示） */
  dismissedSteps: string[];

  /** 本次会话已显示的引导数 */
  sessionShownCount: number;

  /** 上次显示引导的时间 */
  lastShownAt?: string;
}

/**
 * 步骤状态
 */
export interface StepStatus {
  stepId: string;
  status: "pending" | "completed" | "skipped" | "dismissed";
  completedAt?: string;
  userChoice?: Record<string, unknown>;
}

/**
 * Context 值
 */
export interface OnboardingContextValue {
  // 状态
  progress: OnboardingProgress;
  currentStep: OnboardingStep | null;
  isWizardOpen: boolean;
  isSpotlightOpen: boolean;
  isEnabled: boolean;

  // 控制方法
  closeAll: () => void; // 关闭所有引导
  skipAll: () => void; // 跳过全部引导
  toggleEnabled: () => void; // 切换全局开关
  completeStep: (id: string) => void;
  dismissStep: (id: string) => void;
  resetProgress: () => void; // 重置进度
}
```

### 6.2 Provider 实现

```typescript
// src/features/onboarding/hooks/use-onboarding.ts

'use client';

import { createContext, useContext, useCallback, useEffect, useState } from 'react';
import { ONBOARDING_SETTINGS } from '../config/settings';
import { ONBOARDING_STEPS } from '../config/steps';
import { loadProgress, saveProgress } from '../store/storage';
import type { OnboardingContextValue, OnboardingProgress, OnboardingStep } from '../types';

const OnboardingContext = createContext<OnboardingContextValue | null>(null);

export function useOnboarding() {
  const context = useContext(OnboardingContext);
  if (!context) {
    throw new Error('useOnboarding must be used within OnboardingProvider');
  }
  return context;
}

export function OnboardingProvider({ children }: { children: React.ReactNode }) {
  const [progress, setProgress] = useState<OnboardingProgress>(loadProgress);
  const [currentStep, setCurrentStep] = useState<OnboardingStep | null>(null);
  const [isWizardOpen, setIsWizardOpen] = useState(false);
  const [isSpotlightOpen, setIsSpotlightOpen] = useState(false);

  // ==========================================
  // 全局控制
  // ==========================================

  /** 关闭所有引导 */
  const closeAll = useCallback(() => {
    setIsWizardOpen(false);
    setIsSpotlightOpen(false);
    setCurrentStep(null);
  }, []);

  /** 跳过全部引导（一键跳过） */
  const skipAll = useCallback(() => {
    const newProgress = {
      ...progress,
      wizardCompleted: true,
      wizardCompletedAt: new Date().toISOString(),
    };
    setProgress(newProgress);
    saveProgress(newProgress);
    closeAll();
  }, [progress, closeAll]);

  /** 切换全局开关 */
  const toggleEnabled = useCallback(() => {
    const newProgress = {
      ...progress,
      disabled: !progress.disabled,
    };
    setProgress(newProgress);
    saveProgress(newProgress);

    if (newProgress.disabled) {
      closeAll();
    }
  }, [progress, closeAll]);

  // ==========================================
  // 步骤控制
  // ==========================================

  const completeStep = useCallback((stepId: string) => {
    const newProgress = {
      ...progress,
      steps: {
        ...progress.steps,
        [stepId]: {
          stepId,
          status: 'completed' as const,
          completedAt: new Date().toISOString(),
        },
      },
      lastShownAt: new Date().toISOString(),
    };
    setProgress(newProgress);
    saveProgress(newProgress);

    // 前进到下一步或关闭
    goToNextStep(stepId);
  }, [progress]);

  const dismissStep = useCallback((stepId: string) => {
    const newProgress = {
      ...progress,
      dismissedSteps: [...progress.dismissedSteps, stepId],
      steps: {
        ...progress.steps,
        [stepId]: {
          stepId,
          status: 'dismissed' as const,
          completedAt: new Date().toISOString(),
        },
      },
    };
    setProgress(newProgress);
    saveProgress(newProgress);
    closeAll();
  }, [progress, closeAll]);

  // ==========================================
  // 快捷键支持
  // ==========================================

  useEffect(() => {
    if (!ONBOARDING_SETTINGS.escToClose) return;

    function handleKeyDown(e: KeyboardEvent) {
      if (e.key === 'Escape') {
        closeAll();
      }
    }

    document.addEventListener('keydown', handleKeyDown);
    return () => document.removeEventListener('keydown', handleKeyDown);
  }, [closeAll]);

  // ==========================================
  // 渲染
  // ==========================================

  const value: OnboardingContextValue = {
    progress,
    currentStep,
    isWizardOpen,
    isSpotlightOpen,
    isEnabled: !progress.disabled,
    closeAll,
    skipAll,
    toggleEnabled,
    completeStep,
    dismissStep,
    resetProgress: () => {
      const initial = getInitialProgress();
      setProgress(initial);
      saveProgress(initial);
    },
  };

  return (
    <OnboardingContext.Provider value={value}>
      {children}
    </OnboardingContext.Provider>
  );
}
```

```tsx
// src/features/onboarding/components/OnboardingLayer.tsx

"use client";

import { useEffect, useState } from "react";
import { createPortal } from "react-dom";
import { useOnboarding } from "../hooks/use-onboarding";
import { WizardDialog } from "./WizardDialog";
import { SpotlightOverlay } from "./SpotlightOverlay";
import { TooltipHint } from "./TooltipHint";

/**
 * 引导渲染层
 * 通过 Portal 渲染到 body，完全独立于主应用组件树
 */
export function OnboardingLayer() {
  const [mounted, setMounted] = useState(false);
  const { isEnabled, isWizardOpen, isSpotlightOpen, currentStep } =
    useOnboarding();

  useEffect(() => {
    setMounted(true);
  }, []);

  // 未挂载或已禁用时不渲染
  if (!mounted || !isEnabled) return null;

  return createPortal(
    <div
      id="onboarding-layer"
      className="onboarding-layer"
      style={{
        position: "fixed",
        inset: 0,
        zIndex: 9999,
        pointerEvents: "none", // 默认不拦截事件
      }}
    >
      {/* Wizard 向导 */}
      {isWizardOpen && currentStep?.type === "wizard" && (
        <WizardDialog step={currentStep} />
      )}

      {/* Spotlight 聚光灯 */}
      {isSpotlightOpen && currentStep?.type === "spotlight" && (
        <SpotlightOverlay step={currentStep} />
      )}

      {/* Tooltip 提示 */}
      {currentStep?.type === "tooltip" && <TooltipHint step={currentStep} />}
    </div>,
    document.body,
  );
}
```

### 7.2 Wizard 对话框

```tsx
// src/features/onboarding/components/WizardDialog.tsx

"use client";

import { useCallback } from "react";
import { motion, AnimatePresence } from "framer-motion";
import { X } from "lucide-react";
import {
  Dialog,
  DialogContent,
  DialogHeader,
  DialogTitle,
} from "@/components/ui/dialog";
import { Button } from "@/components/ui/button";
import { Progress } from "@/components/ui/progress";
import { useOnboarding } from "../hooks/use-onboarding";
import { ONBOARDING_GROUPS } from "../config/steps";
import type { OnboardingStep } from "../types";

interface Props {
  step: OnboardingStep;
}

export function WizardDialog({ step }: Props) {
  const { closeAll, skipAll, completeStep, getGroupProgress } = useOnboarding();

  const group = step.group ? ONBOARDING_GROUPS[step.group] : null;
  const progress = group ? getGroupProgress(step.group!) : null;

  return (
    <Dialog open onOpenChange={(open) => !open && closeAll()}>
      <DialogContent
        className="sm:max-w-lg"
        style={{ pointerEvents: "auto" }} // 恢复事件拦截
      >
        {/* 进度条 */}
        {progress && (
          <div className="absolute top-0 left-0 right-0">
            <Progress
              value={(progress.completed / progress.total) * 100}
              className="h-1 rounded-none"
            />
          </div>
        )}

        {/* 关闭按钮（始终可见） */}
        <button
          className="absolute right-4 top-4 rounded-sm opacity-70 hover:opacity-100"
          onClick={closeAll}
        >
          <X className="h-4 w-4" />
          <span className="sr-only">关闭</span>
        </button>

        <DialogHeader className="pt-6">
          {progress && (
            <div className="flex items-center gap-2 mb-2">
              <span className="text-xs text-muted-foreground">
                步骤 {progress.completed + 1}/{progress.total}
              </span>
              <div className="flex gap-1">
                {Array.from({ length: progress.total }).map((_, i) => (
                  <div
                    key={i}
                    className={`w-2 h-2 rounded-full ${
                      i <= progress.completed ? "bg-primary" : "bg-muted"
                    }`}
                  />
                ))}
              </div>
            </div>
          )}
          <DialogTitle>{step.title}</DialogTitle>
        </DialogHeader>

        <div className="py-4">
          {/* 步骤内容 */}
          {step.image && (
            <img
              src={step.image}
              alt={step.title}
              className="w-full max-w-xs mx-auto mb-4"
            />
          )}
          <p className="text-muted-foreground whitespace-pre-line">
            {step.description}
          </p>
        </div>

        <div className="flex justify-between items-center pt-4 border-t">
          {/* 跳过全部按钮（始终可见） */}
          <Button
            variant="ghost"
            onClick={skipAll}
            className="text-muted-foreground"
          >
            {step.skipAction || "跳过全部引导"}
          </Button>

          {/* 主操作按钮 */}
          <Button onClick={() => completeStep(step.id)}>
            {step.primaryAction || "下一步"}
          </Button>
        </div>
      </DialogContent>
    </Dialog>
  );
}
```
