# 微信快照样式选项升级方案

> **目标**：允许用户选择样式保留级别，在"仅文字效果"和"完整快照"之间切换
> **日期**：2025-12-21
> **状态**：待审核

---

## 一、需求分析

### 1.1 当前实现

当前仅支持"仅文字效果"模式，白名单包含 9 个属性：

```typescript
// wechat-article.config.ts
export const ALLOWED_STYLE_PROPERTIES = [
  "color",
  "font-size",
  "font-weight",
  "font-style",
  "text-align",
  "text-decoration",
  "line-height",
  "letter-spacing",
  "text-indent",
];
```

### 1.2 用户需求

| 模式           | 描述                                 | 适用场景                   |
| -------------- | ------------------------------------ | -------------------------- |
| **仅文字效果** | 保留基础文字样式（颜色、字号、对齐） | 追求简洁阅读，减少视觉干扰 |
| **完整快照**   | 保留完整排版（背景、边框、布局）     | 追求原汁原味，保留设计意图 |

### 1.3 安全考虑

完整快照模式需要额外的安全措施：

- 仍然过滤危险值（`javascript:`, `expression()`, `url()` 等）
- 限制 `position` 为 `relative`/`static`（防止覆盖攻击）
- 禁止 `z-index` 过大值

```typescript
  "justify-content",
  "align-items",
  "gap",
  "grid-gap",
  "transform", // 仅允许安全的 transform（后续过滤）
];

/**
 * 完整快照模式 - 额外的危险值过滤
 */
export const FULL_SNAPSHOT_DANGEROUS_VALUES: RegExp[] = [
  // 继承基础危险值
  ...DANGEROUS_STYLE_VALUES,

  // 额外限制
  /position\s*:\s*(fixed|absolute|sticky)/i, // 仅允许 relative/static
  /z-index\s*:\s*(\d{4,}|[5-9]\d{2,})/i, // 禁止 z-index > 499
  /background\s*:[^;]*url\s*\(/i, // 禁止背景图片 URL
];

/**
 * 根据级别获取样式白名单
 */
export function getStylePropertiesByLevel(level: StylePreserveLevel): string[] {
  return level === "full-snapshot"
    ? FULL_SNAPSHOT_STYLE_PROPERTIES
    : TEXT_ONLY_STYLE_PROPERTIES;
}

/**
 * 根据级别获取危险值黑名单
 */
export function getDangerousValuesByLevel(level: StylePreserveLevel): RegExp[] {
  return level === "full-snapshot"
    ? FULL_SNAPSHOT_DANGEROUS_VALUES
    : DANGEROUS_STYLE_VALUES;
}
```

### 2.2 内容处理器修改

```typescript
// src/lib/wechat-article/processors/content.processor.ts

import {
  getStylePropertiesByLevel,
  getDangerousValuesByLevel,
  type StylePreserveLevel,
} from "../wechat-article.config";

/**
 * 清理单个 style 属性值（支持样式级别）
 */
function sanitizeStyleAttribute(
  styleValue: string,
  level: StylePreserveLevel = "text-only",
): string {
  if (!styleValue) return "";

  const allowedProperties = getStylePropertiesByLevel(level);
  const dangerousValues = getDangerousValuesByLevel(level);

  const styles = styleValue.split(";").filter(Boolean);
  const safeStyles: string[] = [];

  for (const style of styles) {
    const colonIndex = style.indexOf(":");
    if (colonIndex === -1) continue;

    const prop = style.substring(0, colonIndex).trim().toLowerCase();
    const value = style.substring(colonIndex + 1).trim();

    if (!prop || !value) continue;

    // 检查属性白名单
    if (!allowedProperties.includes(prop)) continue;

    // 检查危险值（使用完整样式字符串检查）
    const fullStyle = `${prop}: ${value}`;
    const isDangerous = dangerousValues.some((pattern) =>
      pattern.test(fullStyle),
    );
    if (isDangerous) continue;

    // 特殊处理：background 属性需要额外过滤 url()
    if (prop === "background" && /url\s*\(/i.test(value)) {
      // 仅保留颜色部分，移除 url()
      const colorOnly = value.replace(/url\s*\([^)]*\)/gi, "").trim();
      if (colorOnly) {
        safeStyles.push(`${prop}: ${colorOnly}`);
      }
      continue;
    }

    safeStyles.push(`${prop}: ${value}`);
  }

  return safeStyles.join("; ");
}
```
