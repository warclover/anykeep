```typescript
// src/lib/ai-routing/entry/unified-entry.ts
import { IntentAnalyzer } from "../intent/intent-analyzer";
import { ModeSelector, ExecutionMode } from "../routing/mode-selector";
import { getKnowledgeAgent } from "@/lib/ai-agent";
import { getOrchestrationEngine } from "@/lib/ai-skills";
import { getWorkflowOrchestrator } from "@/lib/workflow";

export interface UnifiedRequest {
  message: string;
  userId: string;
  sessionId?: string;
  context?: RequestContext;
}

export interface UnifiedResponse {
  mode: ExecutionMode;
  result: unknown;
  metadata: ResponseMetadata;
}

export class UnifiedEntry {
  private intentAnalyzer: IntentAnalyzer;
  private modeSelector: ModeSelector;

  constructor() {
    this.intentAnalyzer = new IntentAnalyzer();
    this.modeSelector = new ModeSelector();
  }

  async process(request: UnifiedRequest): Promise<UnifiedResponse> {
    // 1. 分析意图
    const intentResult = await this.intentAnalyzer.analyze(request.message, {
      userId: request.userId,
      context: request.context,
    });

    // 2. 选择执行模式
    const mode = this.modeSelector.select(intentResult);

    // 3. 路由到对应执行器
    const result = await this.route(mode, request, intentResult);

    return {
      mode,
      result,
      metadata: {
        intent: intentResult.intent,
        confidence: intentResult.confidence,
        latencyMs: Date.now() - startTime,
      },
    };
  }

  private async route(
    mode: ExecutionMode,
    request: UnifiedRequest,
    intentResult: IntentResult,
  ): Promise<unknown> {
    switch (mode) {
      case ExecutionMode.ATOMIC:
        return this.executeAtomic(request, intentResult);

      case ExecutionMode.ORCHESTRATION:
        return this.executeOrchestration(request, intentResult);

      case ExecutionMode.WORKFLOW:
        return this.executeWorkflow(request, intentResult);

      default:
        throw new Error(`Unknown execution mode: ${mode}`);
    }
  }

  private async executeAtomic(
    request: UnifiedRequest,
    intentResult: IntentResult,
  ): Promise<unknown> {
    const agent = getKnowledgeAgent();
    return agent.execute({
      message: request.message,
      userId: request.userId,
      intent: intentResult,
    });
  }

  private async executeOrchestration(
    request: UnifiedRequest,
    intentResult: IntentResult,
  ): Promise<unknown> {
    const engine = getOrchestrationEngine();
    return engine.execute({
      skillId: intentResult.matchedSkill!,
      input: request,
      context: request.context,
    });
  }

  private async executeWorkflow(
    request: UnifiedRequest,
    intentResult: IntentResult,
  ): Promise<unknown> {
    const orchestrator = getWorkflowOrchestrator();
    return orchestrator.trigger({
      workflowId: intentResult.matchedWorkflow!,
      input: request,
      userId: request.userId,
    });
  }
}

// 单例
let entryInstance: UnifiedEntry | null = null;

export function getUnifiedEntry(): UnifiedEntry {
  if (!entryInstance) {
    entryInstance = new UnifiedEntry();
  }
  return entryInstance;
}
```

#### 2.3 统一注册器

```typescript
// src/lib/ai-agent/tools/tool-registry.ts
import { bookmarkTools, bookmarkToolsMetadata } from "./bookmark-tools";
import { folderTools, folderToolsMetadata } from "./folder-tools";
import { searchTools, searchToolsMetadata } from "./search-tools";
// ... 导入所有分组

interface ToolGroup {
  groupId: string;
  groupName: string;
  tools: Record<string, ToolDefinition>;
  metadata: ToolGroupMetadata;
}

class ToolRegistry {
  private groups: Map<string, ToolGroup> = new Map();
  private allTools: Map<string, ToolDefinition> = new Map();

  constructor() {
    this.registerGroup("bookmark", bookmarkTools, bookmarkToolsMetadata);
    this.registerGroup("folder", folderTools, folderToolsMetadata);
    this.registerGroup("search", searchTools, searchToolsMetadata);
    // ... 注册所有分组
  }

  private registerGroup(
    groupId: string,
    tools: Record<string, ToolDefinition>,
    metadata: ToolGroupMetadata,
  ): void {
    this.groups.set(groupId, {
      groupId,
      groupName: metadata.groupName,
      tools,
      metadata,
    });

    for (const [, tool] of Object.entries(tools)) {
      if (this.allTools.has(tool.name)) {
        throw new Error(`Duplicate tool name: ${tool.name}`);
      }
      this.allTools.set(tool.name, tool);
    }
  }

  getTool(name: string): ToolDefinition | undefined {
    return this.allTools.get(name);
  }

  getAllTools(): ToolDefinition[] {
    return Array.from(this.allTools.values());
  }

  getToolsByGroup(groupId: string): ToolDefinition[] {
    const group = this.groups.get(groupId);
    return group ? Object.values(group.tools) : [];
  }

  getGroups(): ToolGroup[] {
    return Array.from(this.groups.values());
  }

  getMetadata(): ToolRegistryMetadata {
    return {
      totalTools: this.allTools.size,
      groups: Array.from(this.groups.values()).map((g) => ({
        groupId: g.groupId,
        groupName: g.groupName,
        toolCount: Object.keys(g.tools).length,
      })),
    };
  }
}

// 单例
export const toolRegistry = new ToolRegistry();
```

```typescript
// scripts/check-dependencies.ts
import * as madge from "madge";

const LAYER_RULES = {
  "ai-routing": { canDependOn: ["ai-agent", "ai-skills", "workflow"] },
  "ai-skills": {
    canDependOn: ["ai-agent", "classification", "summary", "tts"],
  },
  "ai-agent": {
    canDependOn: [
      "classification",
      "summary",
      "tts",
      "embedding",
      "supabase",
      "shared",
    ],
  },
  workflow: {
    canDependOn: ["ai-agent", "ai-skills", "classification", "summary"],
  },
  classification: { canDependOn: ["embedding", "supabase", "shared"] },
  summary: { canDependOn: ["embedding", "supabase", "shared"] },
  tts: { canDependOn: ["supabase", "shared"] },
  embedding: { canDependOn: ["supabase", "shared"] },
  supabase: { canDependOn: ["shared"] },
  shared: { canDependOn: [] },
};

async function checkDependencies() {
  const result = await madge("src/lib/", {
    fileExtensions: ["ts", "tsx"],
    excludeRegExp: [/__tests__/, /\.test\.ts/],
  });

  // 检查循环依赖
  const circular = result.circular();
  if (circular.length > 0) {
    console.error("❌ Circular dependencies found:");
    circular.forEach((c) => console.error(`  ${c.join(" -> ")}`));
    process.exit(1);
  }

  // 检查层级违规
  const dependencies = result.obj();
  const violations: string[] = [];

  for (const [file, deps] of Object.entries(dependencies)) {
    const sourceLayer = getLayer(file);
    for (const dep of deps) {
      const targetLayer = getLayer(dep);
      if (!isValidDependency(sourceLayer, targetLayer)) {
        violations.push(
          `${file} -> ${dep} (${sourceLayer} cannot depend on ${targetLayer})`,
        );
      }
    }
  }

  if (violations.length > 0) {
    console.error("❌ Layer violations found:");
    violations.forEach((v) => console.error(`  ${v}`));
    process.exit(1);
  }

  console.log("✅ No dependency issues found");
}

checkDependencies();
```

#### 5.2 Skill 注册器

```typescript
// src/lib/ai-skills/core/skill-registry.ts
import type { SkillMetadata } from "./skill-metadata";
import type { BaseSkill } from "./base-skill";
import { featureFlags } from "@/lib/ai-agent/config";

class SkillRegistry {
  private skills: Map<
    string,
    { metadata: SkillMetadata; implementation: typeof BaseSkill }
  > = new Map();

  register(metadata: SkillMetadata, implementation: typeof BaseSkill): void {
    if (this.skills.has(metadata.id)) {
      throw new Error(`Skill already registered: ${metadata.id}`);
    }
    this.skills.set(metadata.id, { metadata, implementation });
  }

  getSkill(id: string): BaseSkill | null {
    const entry = this.skills.get(id);
    if (!entry) return null;

    // 检查 Feature Flag
    if (
      entry.metadata.featureFlag &&
      !featureFlags.isEnabled(entry.metadata.featureFlag)
    ) {
      return null;
    }

    return new entry.implementation();
  }

  getEnabledSkills(): SkillMetadata[] {
    return Array.from(this.skills.values())
      .filter(
        (entry) =>
          !entry.metadata.featureFlag ||
          featureFlags.isEnabled(entry.metadata.featureFlag),
      )
      .map((entry) => entry.metadata);
  }

  getSkillsByCategory(category: SkillCategory): SkillMetadata[] {
    return this.getEnabledSkills().filter((s) => s.category === category);
  }

  getMetadata(): SkillRegistryMetadata {
    const enabled = this.getEnabledSkills();
    return {
      totalSkills: this.skills.size,
      enabledSkills: enabled.length,
      byCategory: this.groupByCategory(enabled),
    };
  }

  private groupByCategory(
    skills: SkillMetadata[],
  ): Record<SkillCategory, number> {
    return skills.reduce(
      (acc, s) => {
        acc[s.category] = (acc[s.category] || 0) + 1;
        return acc;
      },
      {} as Record<SkillCategory, number>,
    );
  }
}

export const skillRegistry = new SkillRegistry();
```
