### 1.3 创建主题 Hook

**新文件**: `src/hooks/use-color-theme.ts`

```typescript
/**
 * 颜色主题 Hook
 * 管理用户的品牌色偏好
 */
"use client";

import { useCallback, useEffect, useState } from "react";
import type { ColorTheme } from "@/types/user-settings";

const STORAGE_KEY = "anykeep-color-theme";
const DEFAULT_THEME: ColorTheme = "blue";

export function useColorTheme() {
  const [colorTheme, setColorThemeState] = useState<ColorTheme>(DEFAULT_THEME);
  const [isLoaded, setIsLoaded] = useState(false);

  // 初始化：从 localStorage 读取
  useEffect(() => {
    const stored = localStorage.getItem(STORAGE_KEY) as ColorTheme | null;
    if (stored && ["blue", "purple", "orange"].includes(stored)) {
      setColorThemeState(stored);
      document.documentElement.setAttribute("data-color-theme", stored);
    }
    setIsLoaded(true);
  }, []);

  // 设置主题
  const setColorTheme = useCallback((theme: ColorTheme) => {
    setColorThemeState(theme);
    localStorage.setItem(STORAGE_KEY, theme);
    document.documentElement.setAttribute("data-color-theme", theme);

    // 同步到服务器（异步，不阻塞 UI）
    fetch("/api/user/settings", {
      method: "PATCH",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ appearance: { colorTheme: theme } }),
    }).catch(() => {
      // 静默失败，本地已保存
    });
  }, []);

  return {
    colorTheme,
    setColorTheme,
    isLoaded,
  };
}
```

### 1.4 创建主题选择器组件

**新文件**: `src/components/settings/color-theme-selector.tsx`

```typescript
/**
 * 颜色主题选择器
 * 设置页面使用
 */
'use client';

import { Check } from 'lucide-react';
import { cn } from '@/lib/utils';
import { useColorTheme } from '@/hooks/use-color-theme';
import type { ColorTheme } from '@/types/user-settings';

const THEMES: Array<{
  id: ColorTheme;
  name: string;
  description: string;
  colors: { primary: string; gradient: string };
}> = [
  {
    id: 'blue',
    name: '知性蓝',
    description: '专业稳重，适合知识管理',
    colors: {
      primary: '#3b82f6',
      gradient: 'from-blue-500 to-blue-400',
    },
  },
  {
    id: 'purple',
    name: '活力紫',
    description: '创意灵感，激发想象力',
    colors: {
      primary: '#8b5cf6',
      gradient: 'from-purple-500 to-purple-400',
    },
  },
  {
    id: 'orange',
    name: '温暖橙',
    description: '高效能量，提升专注力',
    colors: {
      primary: '#f97316',
      gradient: 'from-orange-500 to-orange-400',
    },
  },
];

export function ColorThemeSelector() {
  const { colorTheme, setColorTheme } = useColorTheme();

  return (
    <div className="grid grid-cols-3 gap-3">
      {THEMES.map((theme) => (
        <button
          key={theme.id}
          onClick={() => setColorTheme(theme.id)}
          className={cn(
            'relative flex flex-col items-center p-4 rounded-xl border-2 transition-all',
            colorTheme === theme.id
              ? 'border-primary bg-primary/5 shadow-sm'
              : 'border-gray-200 dark:border-gray-700 hover:border-gray-300 dark:hover:border-gray-600'
          )}
        >
          {/* 选中指示 */}
          {colorTheme === theme.id && (
            <div className="absolute top-2 right-2">
              <Check className="size-4 text-primary" />
            </div>
          )}

          {/* 颜色预览 */}
          <div
            className={cn(
              'w-12 h-12 rounded-full bg-gradient-to-br mb-3',
              theme.colors.gradient
            )}
          />

          {/* 主题名称 */}
          <span className="font-medium text-sm">{theme.name}</span>
          <span className="text-xs text-gray-500 dark:text-gray-400 mt-1 text-center">
            {theme.description}
          </span>
        </button>
      ))}
    </div>
  );
}
```

### 3.2 视觉层级分层设计

```
┌─────────────────────────────────────┐
│  [平台] [内容类型]           [操作] │ ← 第1层：元信息（最弱）
│                                     │
│  标题文字                           │ ← 第2层：核心内容（最强）
│  摘要文字（次要颜色）               │
│                                     │
│  [AI标签] [用途标签]    ⭐ 3星      │ ← 第3层：扩展信息（中等）
│  来源 · 时间 · 文件夹               │ ← 第4层：辅助信息（最弱）
└─────────────────────────────────────┘
```

### 3.3 字重/颜色建议

| 层级   | 字重                | 颜色                |
| ------ | ------------------- | ------------------- |
| 标题   | font-semibold       | gray-900 / white    |
| 摘要   | font-normal         | gray-600 / gray-400 |
| 标签   | font-medium         | 各自主题色          |
| 元信息 | font-normal text-xs | gray-500 / gray-400 |

## 🎯 Phase 5: 组件精简优化

### 5.1 卡片信息精简策略

**当前问题**：单卡展示 8-10 个信息点，视觉负担重

**建议精简**：

| 保留（核心）  | 默认隐藏（悬停显示） | 移除         |
| ------------- | -------------------- | ------------ |
| 标题          | 操作菜单             | 重复的时间戳 |
| 封面图        | 完整标签列表         |              |
| 平台徽章      | 评级详情             |              |
| 1-2个核心标签 |                      |              |
| 摘要首句      |                      |              |

### 5.2 详情页优化建议

1. **AI 摘要区域** - 增加渐变背景强调
2. **快照区域** - 增加骨架屏过渡
3. **相关推荐** - 卡片化展示，增加视觉吸引力
