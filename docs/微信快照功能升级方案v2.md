```typescript
/**
 * 清理单个 style 属性值
 * 保留安全属性，移除危险内容
 */
function sanitizeStyleAttribute(styleValue: string): string {
  if (!styleValue) return "";

  // 解析 style 为键值对
  const styles = styleValue.split(";").filter(Boolean);
  const safeStyles: string[] = [];

  for (const style of styles) {
    const [prop, value] = style.split(":").map((s) => s.trim());
    if (!prop || !value) continue;

    const propLower = prop.toLowerCase();

    // 检查属性是否在白名单中
    if (!ALLOWED_STYLE_PROPERTIES.includes(propLower)) continue;

    // 检查值是否包含危险内容
    const isDangerous = DANGEROUS_STYLE_VALUES.some((pattern) =>
      pattern.test(value),
    );
    if (isDangerous) continue;

    safeStyles.push(`${prop}: ${value}`);
  }

  return safeStyles.join("; ");
}

/**
 * 清理 HTML 并保留安全样式
 */
function sanitizeWechatHtmlWithStyles(html: string): string {

    const safeStyle = sanitizeStyleAttribute(styleValue);

}
```

### 3.2 Phase 2：图片质量优化（优先级：P0）

**目标**：获取高清原图，提高保存成功率

#### 3.2.1 图片 URL 优化

**原理**：微信图片 URL 参数控制图片质量

```
原始 URL 示例：
https://mmbiz.qpic.cn/mmbiz_jpg/.../640?wx_fmt=jpeg&tp=webp&wxfrom=5&wx_lazy=1

参数解析：
- wx_fmt=jpeg  : 图片格式
- tp=webp      : 转换为 webp（压缩）
- wx_lazy=1    : 懒加载标记
- wx_co=1      : 原图标记（关键！）
```

**修改文件**：`src/lib/wechat-article/processors/image.processor.ts`

```typescript
/**
 * 优化微信图片 URL，获取高清原图
 */
export function optimizeWechatImageUrl(url: string): string {
  if (!url.includes("mmbiz.qpic.cn")) return url;

  try {
    const urlObj = new URL(url);

    // 移除压缩参数
    urlObj.searchParams.delete("tp");
    urlObj.searchParams.delete("wx_lazy");
    urlObj.searchParams.delete("wxfrom");

    // 添加原图参数
    urlObj.searchParams.set("wx_co", "1");

    // 保留格式参数，优先 jpeg
    if (!urlObj.searchParams.has("wx_fmt")) {
      urlObj.searchParams.set("wx_fmt", "jpeg");
    }

    return urlObj.toString();
  } catch {
    return url;
  }
}
```

#### 3.2.2 下载策略优化

```typescript
/**
 * 增强的图片下载函数
 * 支持重试、多 Referer、超时控制
 */
async function downloadImageEnhanced(
  url: string,
  options: {
    timeout?: number;
    retries?: number;
    useProxy?: boolean;
  } = {},
): Promise<{ buffer: ArrayBuffer; contentType: string }> {
  const { timeout = 20000, retries = 3, useProxy = false } = options;

  // 优化 URL
  const optimizedUrl = optimizeWechatImageUrl(url);

  // Referer 列表（轮换使用）
  const referers = [
    "https://mp.weixin.qq.com/",
    "https://weixin.qq.com/",
    "", // 空 Referer
  ];

  let lastError: Error | null = null;

  for (let attempt = 0; attempt < retries; attempt++) {
    const referer = referers[attempt % referers.length];

    try {
      const controller = new AbortController();
      const timeoutId = setTimeout(() => controller.abort(), timeout);

      const headers: HeadersInit = {
        "User-Agent":
          "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",
        Accept: "image/avif,image/webp,image/apng,image/*,*/*;q=0.8",
        "Accept-Language": "zh-CN,zh;q=0.9,en;q=0.8",
        "Cache-Control": "no-cache",
      };

      if (referer) {
        headers["Referer"] = referer;
      }

      const targetUrl = useProxy ? getProxyUrl(optimizedUrl) : optimizedUrl;

      const response = await fetch(targetUrl, {
        signal: controller.signal,
        headers,
      });

      clearTimeout(timeoutId);

      if (!response.ok) {
        throw new Error(`HTTP ${response.status}`);
      }

      const buffer = await response.arrayBuffer();
      const contentType = response.headers.get("content-type") || "image/jpeg";

      // 验证是否为有效图片（检查 magic bytes）
      if (!isValidImageBuffer(buffer)) {
        throw new Error("Invalid image data");
      }

      return { buffer, contentType };
    } catch (error) {
      lastError = error instanceof Error ? error : new Error(String(error));

      // 最后一次尝试使用代理
      if (attempt === retries - 2 && !useProxy) {
        return downloadImageEnhanced(url, {
          ...options,
          useProxy: true,
          retries: 1,
        });
      }

      // 等待后重试
      await new Promise((resolve) => setTimeout(resolve, 1000 * (attempt + 1)));
    }
  }

  throw lastError || new Error("Download failed");
}

/**
 * 验证图片数据有效性
 */
function isValidImageBuffer(buffer: ArrayBuffer): boolean {
  const arr = new Uint8Array(buffer);

  // JPEG: FF D8 FF
  if (arr[0] === 0xff && arr[1] === 0xd8 && arr[2] === 0xff) return true;

  // PNG: 89 50 4E 47
  if (arr[0] === 0x89 && arr[1] === 0x50 && arr[2] === 0x4e && arr[3] === 0x47)
    return true;

  // GIF: 47 49 46 38
  if (arr[0] === 0x47 && arr[1] === 0x49 && arr[2] === 0x46 && arr[3] === 0x38)
    return true;

  // WebP: 52 49 46 46 ... 57 45 42 50
  if (arr[0] === 0x52 && arr[1] === 0x49 && arr[2] === 0x46 && arr[3] === 0x46)
    return true;

  return false;
}
```

```typescript
/**
 * 下载微信视频
 * 微信视频通常托管在腾讯视频或微信自有 CDN
 */
export async function downloadWechatVideo(
  videoUrl: string,
  snapshotId: number,
): Promise<{ success: boolean; localUrl?: string; error?: string }> {
  try {
    // 解析视频真实地址（可能需要二次请求）
    const realUrl = await resolveVideoUrl(videoUrl);

    // 下载视频（限制大小 50MB）
    const response = await fetch(realUrl, {
      headers: {
        "User-Agent":
          "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",
        Referer: "https://mp.weixin.qq.com/",
      },
    });

    if (!response.ok) throw new Error(`HTTP ${response.status}`);

    const contentLength = parseInt(
      response.headers.get("content-length") || "0",
      10,
    );
    const maxSize = 50 * 1024 * 1024; // 50MB

    if (contentLength > maxSize) {
      return { success: false, error: "Video too large (>50MB)" };
    }

    const buffer = await response.arrayBuffer();
    const contentType = response.headers.get("content-type") || "video/mp4";

    // 上传到 Storage
    const path = `wechat/${snapshotId}/video_${Date.now()}.mp4`;
    const publicUrl = await uploadToStorage(buffer, path, contentType);

    return { success: true, localUrl: publicUrl };
  } catch (error) {
    return {
      success: false,
      error: error instanceof Error ? error.message : "Download failed",
    };
  }
}
```

```typescript
/**
 * 微信文章媒体元素处理器
 */

export interface MediaElement {
  type: "video" | "audio" | "vote" | "miniprogram" | "link";
  title: string;
  coverUrl?: string;
  sourceUrl?: string;
  duration?: string;
  options?: string[]; // 投票选项
}

/**
 * 提取并转换视频元素
 */
export function processVideoElements(html: string): {
  html: string;
  videos: MediaElement[];
} {
  const videos: MediaElement[] = [];

  // 处理 mpvideosnap
  let processedHtml = html.replace(
    /<mpvideosnap[^>]*data-cover="([^"]*)"[^>]*data-title="([^"]*)"[^>]*data-url="([^"]*)"[^>]*>[\s\S]*?<\/mpvideosnap>/gi,
    (match, cover, title, url) => {
      videos.push({
        type: "video",
        title: decodeHtmlEntities(title),
        coverUrl: cover,
        sourceUrl: url,
      });

      // 生成占位卡片
      return `
        <div class="wechat-video-placeholder" data-video-index="${videos.length - 1}">
          <div class="video-cover">
            <img src="${cover}" alt="${title}" />
            <div class="play-icon">▶</div>
          </div>
          <div class="video-info">
            <span class="video-title">${title}</span>
            <a href="${url}" target="_blank" rel="noopener">打开视频</a>
          </div>
        </div>
      `;
    },
  );

  // 处理 iframe 视频（腾讯视频等）
  processedHtml = processedHtml.replace(
    /<iframe[^>]*class="[^"]*video_iframe[^"]*"[^>]*data-src="([^"]*)"[^>]*>[\s\S]*?<\/iframe>/gi,
    (match, src) => {
      videos.push({
        type: "video",
        title: "嵌入视频",
        sourceUrl: src,
      });

      return `
        <div class="wechat-iframe-placeholder">
          <a href="${src}" target="_blank" rel="noopener">查看嵌入视频</a>
        </div>
      `;
    },
  );

  return { html: processedHtml, videos };
}

/**
 * 提取并转换音频元素
 */
export function processAudioElements(html: string): {
  html: string;
  audios: MediaElement[];
} {
  const audios: MediaElement[] = [];

  const processedHtml = html.replace(
    /<mpvoice[^>]*voice_encode_fileid="([^"]*)"[^>]*name="([^"]*)"[^>]*play_length="(\d+)"[^>]*>[\s\S]*?<\/mpvoice>/gi,
    (match, fileId, name, duration) => {
      const durationSeconds = parseInt(duration, 10);
      const durationStr = `${Math.floor(durationSeconds / 60)}:${String(durationSeconds % 60).padStart(2, "0")}`;

      audios.push({
        type: "audio",
        title: name,
        duration: durationStr,
      });

      return `
        <div class="wechat-audio-placeholder">
          <span class="audio-icon">🎵</span>
          <span class="audio-name">${name}</span>
          <span class="audio-duration">${durationStr}</span>
        </div>
      `;
    },
  );

  return { html: processedHtml, audios };
}

/**
 * 综合处理所有媒体元素
 */
export function processMediaElements(html: string): {
  html: string;
  media: MediaElement[];
} {
  let result = html;
  const allMedia: MediaElement[] = [];

  // 处理视频
  const { html: afterVideo, videos } = processVideoElements(result);
  result = afterVideo;
  allMedia.push(...videos);

  // 处理音频
  const { html: afterAudio, audios } = processAudioElements(result);
  result = afterAudio;
  allMedia.push(...audios);

  return { html: result, media: allMedia };
}
```

#### 3.6.1 Markdown 导出增强

当前已有基础 Markdown 转换，需要增强以下功能：

```typescript
// src/lib/wechat-article/export/markdown.exporter.ts

export interface MarkdownExportOptions {
  includeMetadata?: boolean; // 包含标题、作者、日期
  includeImages?: boolean; // 包含图片（使用本地 URL）
  imageFormat?: "inline" | "reference"; // 图片格式
  includeMediaPlaceholders?: boolean; // 包含视频/音频占位
}

/**
 * 导出为 Markdown 文件
 */
export async function exportToMarkdown(
  snapshotId: number,
  options: MarkdownExportOptions = {},
): Promise<{ content: string; filename: string }> {
  const {
    includeMetadata = true,
    includeImages = true,
    imageFormat = "inline",
    includeMediaPlaceholders = true,
  } = options;

  const article = await getArticleContent(snapshotId);
  const snapshot = await getSnapshot(snapshotId);

  let markdown = "";

  // 1. 元数据头部
  if (includeMetadata) {
    markdown += `---
title: "${snapshot.title}"
author: "${snapshot.author || "未知"}"
date: "${snapshot.publish_time || snapshot.created_at}"
source: "${snapshot.original_url}"
saved_at: "${new Date().toISOString()}"
---

`;
  }

  // 2. 标题
  markdown += `# ${snapshot.title}\n\n`;

  // 3. 正文（使用增强的转换器）
  let content = article.markdown || htmlToMarkdownEnhanced(article.html);

  // 4. 处理图片 URL
  if (includeImages && article.images) {
    for (const img of article.images) {
      if (img.persistedUrl) {
        // 替换为本地 URL
        content = content.replace(
          new RegExp(escapeRegExp(img.originalUrl), "g"),
          img.persistedUrl,
        );
      }
    }
  }

  // 5. 添加媒体占位
  if (includeMediaPlaceholders && article.media) {
    for (const media of article.media) {
      if (media.type === "video") {
        content += `\n\n> 📹 **视频**: ${media.title}\n> 链接: ${media.sourceUrl || "无法获取"}\n`;
      } else if (media.type === "audio") {
        content += `\n\n> 🎵 **音频**: ${media.title} (${media.duration})\n`;
      }
    }
  }

  markdown += content;

  // 6. 生成文件名
  const safeTitle = snapshot.title
    .replace(/[<>:"/\\|?*]/g, "")
    .substring(0, 100);
  const filename = `${safeTitle}.md`;

  return { content: markdown, filename };
}

/**
 * 增强的 HTML 转 Markdown
 */
function htmlToMarkdownEnhanced(html: string): string {
  // 使用 turndown 库进行更好的转换
  const TurndownService = require("turndown");
  const turndown = new TurndownService({
    headingStyle: "atx",
    codeBlockStyle: "fenced",
    bulletListMarker: "-",
  });

  // 自定义规则：处理微信特殊元素
  turndown.addRule("wechatSection", {
    filter: "section",
    replacement: (content: string) => content,
  });

  turndown.addRule("wechatSpan", {
    filter: (node: HTMLElement) => {
      return node.nodeName === "SPAN" && node.style.color;
    },
    replacement: (content: string, node: HTMLElement) => {
      // 保留颜色信息为注释
      const color = node.style.color;
      return color ? `<!-- color: ${color} -->${content}` : content;
    },
  });

  return turndown.turndown(html);
}
```

#### 3.6.2 PDF 导出

**方案**：使用 Puppeteer 在服务端生成 PDF

```typescript
// src/lib/wechat-article/export/pdf.exporter.ts

export interface PdfExportOptions {
  format?: "A4" | "Letter";
  margin?: { top: string; bottom: string; left: string; right: string };
  includeHeader?: boolean;
  includeFooter?: boolean;
  scale?: number;
}

/**
 * 导出为 PDF 文件
 *
 * 注意：需要在 Next.js API Route 中调用，不能在 Edge Function 中使用
 */
export async function exportToPdf(
  snapshotId: number,
  options: PdfExportOptions = {},
): Promise<{ buffer: Buffer; filename: string }> {
  const {
    format = "A4",
    margin = { top: "20mm", bottom: "20mm", left: "15mm", right: "15mm" },
    includeHeader = true,
    includeFooter = true,
    scale = 1,
  } = options;

  const article = await getArticleContent(snapshotId);
  const snapshot = await getSnapshot(snapshotId);

  // 1. 生成完整 HTML 页面
  const fullHtml = generatePdfHtml(snapshot, article, { includeHeader });

  // 2. 启动 Puppeteer
  const puppeteer = require("puppeteer");
  const browser = await puppeteer.launch({
    headless: "new",
    args: ["--no-sandbox", "--disable-setuid-sandbox"],
  });

  try {
    const page = await browser.newPage();

    // 设置内容
    await page.setContent(fullHtml, {
      waitUntil: "networkidle0",
      timeout: 30000,
    });

    // 等待图片加载
    await page
      .waitForFunction(
        () => {
          const images = document.querySelectorAll("img");
          return Array.from(images).every((img) => img.complete);
        },
        { timeout: 15000 },
      )
      .catch(() => {});

    // 生成 PDF
    const pdf = await page.pdf({
      format,
      margin,
      scale,
      printBackground: true,
      displayHeaderFooter: includeFooter,
      headerTemplate: includeHeader
        ? `
        <div style="font-size: 10px; width: 100%; text-align: center; color: #666;">
          ${snapshot.title}
        </div>
      `
        : "",
      footerTemplate: includeFooter
        ? `
        <div style="font-size: 10px; width: 100%; text-align: center; color: #666;">
          <span class="pageNumber"></span> / <span class="totalPages"></span>
        </div>
      `
        : "",
    });

    // 生成文件名
    const safeTitle = snapshot.title
      .replace(/[<>:"/\\|?*]/g, "")
      .substring(0, 100);
    const filename = `${safeTitle}.pdf`;

    return { buffer: Buffer.from(pdf), filename };
  } finally {
    await browser.close();
  }
}
```

#### 3.6.3 导出 API 路由

```typescript
// src/app/api/wechat-article/[id]/export/route.ts

import { NextRequest, NextResponse } from "next/server";
import { exportToMarkdown } from "@/lib/wechat-article/export/markdown.exporter";
import { exportToPdf } from "@/lib/wechat-article/export/pdf.exporter";

export async function GET(
  request: NextRequest,
  { params }: { params: { id: string } },
) {
  const snapshotId = parseInt(params.id, 10);
  const format = request.nextUrl.searchParams.get("format") || "markdown";

  try {
    if (format === "pdf") {
      const { buffer, filename } = await exportToPdf(snapshotId);

      return new NextResponse(buffer, {
        headers: {
          "Content-Type": "application/pdf",
          "Content-Disposition": `attachment; filename="${encodeURIComponent(filename)}"`,
        },
      });
    }

    // 默认 Markdown
    const { content, filename } = await exportToMarkdown(snapshotId);

    return new NextResponse(content, {
      headers: {
        "Content-Type": "text/markdown; charset=utf-8",
        "Content-Disposition": `attachment; filename="${encodeURIComponent(filename)}"`,
      },
    });
  } catch (error) {
    return NextResponse.json({ error: "Export failed" }, { status: 500 });
  }
}
```
