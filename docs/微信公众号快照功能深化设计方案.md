## 二、解耦设计原则

### 2.1 核心原则

```
原则 1：Feature Flag 控制
  └─ 所有新功能通过配置开关控制，可随时回滚

原则 2：模块独立
  └─ 新建 src/lib/wechat-article/ 独立模块，不修改现有代码

原则 3：渐进增强
  └─ 现有快照功能保持不变，微信文章作为增强层叠加

原则 4：平台隔离

原则 5：数据兼容
  └─ 使用新字段扩展 bookmark_snapshots，不修改现有字段含义
```

### 4.3 核心服务

```typescript
// src/lib/wechat-article/wechat-article.service.ts

import {
  createServerSupabaseClient,
  createServiceClient,
} from "@/lib/supabase/server";
import { logger } from "@/lib/logger";
import type {
  WechatArticleContent,
  WechatArticleCrawlProgress,
  RefetchOptions,
  RefetchResult,
} from "./wechat-article.types";
import { processWechatContent } from "./processors/content.processor";
import { persistImages } from "./processors/image.processor";
import { DEFAULT_CRAWL_CONFIG } from "./wechat-article.config";

/**
 * 微信文章服务
 *
 * 设计原则：
 * - 复用现有快照表 (bookmark_snapshots)
 * - 复用现有微信解析工具 (wechat-parser.ts)
 * - 新增图片持久化和 Markdown 转换能力
 */
export class WechatArticleService {
  /**
   * 抓取微信文章完整内容
   *
   * @param snapshotId - 现有快照 ID（必须已有 basic 快照）
   * @param onProgress - 进度回调
   */
  async crawlFullContent(
    snapshotId: number,
    onProgress?: (progress: WechatArticleCrawlProgress) => void,
  ): Promise<WechatArticleContent> {
    const supabase = await createServiceClient();

    try {
      // 1. 获取现有快照
      onProgress?.({ stage: "fetching", progress: 10 });
      const { data: snapshot, error } = await supabase
        .from("bookmark_snapshots")
        .select("*, bookmarks!inner(url, platform)")
        .eq("id", snapshotId)
        .single();

      if (error || !snapshot) {
        throw new Error("快照不存在");
      }

      // 2. 验证是微信平台
      if (snapshot.bookmarks.platform !== "WECHAT_OFFICIAL_ACCOUNT") {
        throw new Error("仅支持微信公众号文章");
      }

      // 3. 处理正文内容
      onProgress?.({ stage: "parsing", progress: 30 });
      const content = await processWechatContent(
        snapshot.readable_content || "",
        snapshot.original_url,
      );

      // 4. 持久化图片
      onProgress?.({
        stage: "images",
        progress: 50,
        images: { total: content.images.length, completed: 0 },
      });
      const persistedImages = await persistImages(
        content.images,
        snapshotId,
        (completed) => {
          onProgress?.({
            stage: "images",
            progress: 50 + Math.round((completed / content.images.length) * 40),
            images: { total: content.images.length, completed },
          });
        },
      );

      // 5. 更新快照记录
      onProgress?.({ stage: "saving", progress: 95 });
      const imagesStats = {
        total: persistedImages.length,
        success: persistedImages.filter((i) => i.status === "success").length,
        failed: persistedImages.filter((i) => i.status === "failed").length,
      };

      await supabase
        .from("bookmark_snapshots")
        .update({
          content_markdown: content.markdown,
          article_images: persistedImages,
          images_status: imagesStats.failed === 0 ? "completed" : "partial",
          images_stats: imagesStats,
          updated_at: new Date().toISOString(),
        })
        .eq("id", snapshotId);

      onProgress?.({ stage: "completed", progress: 100 });

      return {
        ...content,
        images: persistedImages,
      };
    } catch (error) {
      logger.error(
        { error, snapshotId },
        "[WechatArticleService] crawlFullContent failed",
      );
      onProgress?.({
        stage: "failed",
        progress: 0,
        error: error instanceof Error ? error.message : "抓取失败",
      });
      throw error;
    }
  }

  /**
   * 重新获取文章内容
   */
  async refetch(
    snapshotId: number,
    options: RefetchOptions = {},
  ): Promise<RefetchResult> {
    const supabase = await createServiceClient();

    // 1. 获取现有快照
    const { data: snapshot } = await supabase
      .from("bookmark_snapshots")
      .select("*, bookmarks!inner(url)")
      .eq("id", snapshotId)
      .single();

    if (!snapshot) {
      throw new Error("快照不存在");
    }

    // 2. 重新抓取网页
    const { crawlContent } = await import("@/lib/utils/crawler");
    const newContent = await crawlContent(snapshot.bookmarks.url);

    // 3. 对比变化
    const contentChanged =
      newContent.textContent !== snapshot.readable_text_content ||
      options.forceUpdate;

    if (!contentChanged && !options.imagesOnly) {
      return {
        success: true,
        contentChanged: false,
        newVersion: snapshot.content_version,
        changes: {
          title: false,
          content: false,
          wordCount: false,
          imagesUpdated: 0,
        },
      };
    }

    // 4. 更新内容
    const newVersion = (snapshot.content_version || 1) + 1;

    await supabase
      .from("bookmark_snapshots")
      .update({
        readable_content: newContent.content,
        readable_text_content: newContent.textContent,
        readable_word_count: newContent.length,
        content_version: newVersion,
        captured_at: new Date().toISOString(),
        updated_at: new Date().toISOString(),
      })
      .eq("id", snapshotId);

    // 5. 重新处理图片（如果需要）
    let imagesUpdated = 0;
    if (options.redownloadImages || options.imagesOnly) {
      const result = await this.crawlFullContent(snapshotId);
      imagesUpdated = result.images.filter(
        (i) => i.status === "success",
      ).length;
    }

    return {
      success: true,
      contentChanged,
      newVersion,
      changes: {
        title: newContent.title !== snapshot.readable_title,
        content: contentChanged,
        wordCount: newContent.length !== snapshot.readable_word_count,
        imagesUpdated,
      },
    };
  }

  /**
   * 获取微信文章内容（用于展示）
   */
  async getArticleContent(
    snapshotId: number,
  ): Promise<WechatArticleContent | null> {
    const supabase = await createServerSupabaseClient();

    const { data: snapshot } = await supabase
      .from("bookmark_snapshots")
      .select(
        `
        readable_content,
        readable_text_content,
        readable_word_count,
        content_markdown,
        article_images,
        images_status
      `,
      )
      .eq("id", snapshotId)
      .single();

    if (!snapshot) return null;

    return {
      html: snapshot.readable_content || "",
      text: snapshot.readable_text_content || "",
      markdown: snapshot.content_markdown,
      images: snapshot.article_images || [],
      wordCount: snapshot.readable_word_count || 0,
    };
  }
}
```

### 4.4 图片处理器

```typescript
// src/lib/wechat-article/processors/image.processor.ts

import pLimit from "p-limit";
import { createServiceClient } from "@/lib/supabase/server";
import { logger } from "@/lib/logger";
import type { WechatArticleImage } from "../wechat-article.types";
import { DEFAULT_CRAWL_CONFIG } from "../wechat-article.config";

/**
 * 图片过滤规则
 */
const EXCLUDE_PATTERNS = [
  /wx_fmt=gif/, // 表情包
  /tp=webp.*wxfrom=5/, // 追踪像素
  /res\.wx\.qq\.com/, // 微信资源
  /mmbiz\.qlogo\.cn/, // 头像
  /data:image/, // Base64 图片
];

/**
 * 检查图片是否有效
 */
function isValidImage(src: string): boolean {
  if (!src || !src.startsWith("http")) return false;
  return !EXCLUDE_PATTERNS.some((p) => p.test(src));
}

/**
 * 从 HTML 提取图片
 */
export function extractImages(html: string): WechatArticleImage[] {
  const imgRegex = /<img[^>]+(?:src|data-src)="([^"]+)"[^>]*>/gi;
  const images: WechatArticleImage[] = [];
  const seen = new Set<string>();

  let match;
  while ((match = imgRegex.exec(html)) !== null) {
    const src = match[1];

    if (isValidImage(src) && !seen.has(src)) {
      seen.add(src);

      // 提取 alt 文本
      const altMatch = match[0].match(/alt="([^"]*)"/);

      images.push({
        originalUrl: src,
        persistedUrl: null,
        alt: altMatch?.[1] || "",
        status: "pending",
      });
    }
  }

  return images.slice(0, DEFAULT_CRAWL_CONFIG.maxImages);
}

/**
 * 持久化图片到 Supabase Storage
 */
export async function persistImages(
  images: WechatArticleImage[],
  snapshotId: number,
  onProgress?: (completed: number) => void,
): Promise<WechatArticleImage[]> {
  if (images.length === 0) return [];

  const supabase = await createServiceClient();
  const limit = pLimit(DEFAULT_CRAWL_CONFIG.imageConcurrency);
  let completed = 0;

  const tasks = images.map((img, index) =>
    limit(async (): Promise<WechatArticleImage> => {
      try {
        // 1. 下载图片
        const response = await fetch(img.originalUrl, {
          signal: AbortSignal.timeout(DEFAULT_CRAWL_CONFIG.imageTimeout),
          headers: {
            "User-Agent":
              "Mozilla/5.0 (Windows NT 10.0; Win64; x64) Chrome/120.0.0.0",
            Referer: "https://mp.weixin.qq.com/",
          },
        });

        if (!response.ok) {
          throw new Error(`HTTP ${response.status}`);
        }

        const buffer = await response.arrayBuffer();
        const contentType =
          response.headers.get("content-type") || "image/jpeg";
        const ext = contentType.split("/")[1]?.replace("jpeg", "jpg") || "jpg";

        // 2. 上传到 Supabase Storage
        const fileName = `wechat/${snapshotId}/${Date.now()}_${index}.${ext}`;
        const { data, error } = await supabase.storage
          .from("article-images")
          .upload(fileName, buffer, {
            contentType,
            upsert: true,
          });

        if (error) throw error;

        // 3. 获取公开 URL
        const {
          data: { publicUrl },
        } = supabase.storage.from("article-images").getPublicUrl(fileName);

        completed++;
        onProgress?.(completed);

        return {
          ...img,
          persistedUrl: publicUrl,
          status: "success",
          fileSize: buffer.byteLength,
        };
      } catch (error) {
        logger.warn(
          { error, originalUrl: img.originalUrl },
          "[ImageProcessor] Failed to persist image",
        );

        completed++;
        onProgress?.(completed);

        // 降级到代理服务
        return {
          ...img,
          persistedUrl: `https://images.weserv.nl/?url=${encodeURIComponent(img.originalUrl)}`,
          status: "fallback",
          error: error instanceof Error ? error.message : "Unknown error",
        };
      }
    }),
  );

  return Promise.all(tasks);
}
```

```typescript
// src/app/api/wechat-article/[id]/route.ts

import { NextRequest, NextResponse } from "next/server";
import { createServerSupabaseClient } from "@/lib/supabase/server";
import { wechatArticleService } from "@/lib/wechat-article";
import {
  successResponse,
  errorResponse,
  unauthorizedError,
} from "@/lib/utils/api-response";

/**
 * GET /api/wechat-article/[id]
 * 获取微信文章内容
 */
export async function GET(
  request: NextRequest,
  { params }: { params: { id: string } },
) {
  const supabase = await createServerSupabaseClient();
  const {
    data: { user },
  } = await supabase.auth.getUser();

  if (!user) {
    return unauthorizedError();
  }

  const snapshotId = parseInt(params.id, 10);
  if (isNaN(snapshotId)) {
    return errorResponse("INVALID_ID", "无效的快照 ID", 400);
  }

  try {
    const content = await wechatArticleService.getArticleContent(snapshotId);

    if (!content) {
      return errorResponse("NOT_FOUND", "文章内容不存在", 404);
    }

    return successResponse(content);
  } catch (error) {
    return errorResponse("INTERNAL_ERROR", "获取文章失败", 500);
  }
}

/**
 * POST /api/wechat-article/[id]
 * 抓取微信文章完整内容
 */
export async function POST(
  request: NextRequest,
  { params }: { params: { id: string } },
) {
  const supabase = await createServerSupabaseClient();
  const {
    data: { user },
  } = await supabase.auth.getUser();

  if (!user) {
    return unauthorizedError();
  }

  const snapshotId = parseInt(params.id, 10);

  try {
    const content = await wechatArticleService.crawlFullContent(snapshotId);
    return successResponse({
      success: true,
      wordCount: content.wordCount,
      imagesCount: content.images.length,
      imagesSuccess: content.images.filter((i) => i.status === "success")
        .length,
    });
  } catch (error) {
    return errorResponse(
      "CRAWL_FAILED",
      error instanceof Error ? error.message : "抓取失败",
      500,
    );
  }
}
```

```typescript
      forceUpdate: body.forceUpdate,
      redownloadImages: body.redownloadImages,
      imagesOnly: body.imagesOnly,
    });

    return successResponse(result);
  } catch (error) {
    return errorResponse(
      "REFETCH_FAILED",
      error instanceof Error ? error.message : "重新获取失败",
      500,
    );
  }
}
```

```typescript
// src/components/wechat-article/hooks/use-wechat-article-crawl.ts

import { useState, useCallback } from "react";
import { toast } from "sonner";
import type {
  WechatArticleCrawlProgress,
  RefetchOptions,
} from "@/lib/wechat-article/wechat-article.types";

interface UseWechatArticleCrawlOptions {
  onSuccess?: () => void;
  onError?: (error: Error) => void;
}

export function useWechatArticleCrawl(
  snapshotId: number,
  options: UseWechatArticleCrawlOptions = {},
) {
  const [isCrawling, setIsCrawling] = useState(false);
  const [progress, setProgress] = useState<WechatArticleCrawlProgress | null>(
    null,
  );

  const crawl = useCallback(async () => {
    if (isCrawling) return;

    setIsCrawling(true);
    setProgress({ stage: "fetching", progress: 0 });

    try {
      const res = await fetch(`/api/wechat-article/${snapshotId}`, {
        method: "POST",
      });

      if (!res.ok) {
        const error = await res.json();
        throw new Error(error.error?.message || "抓取失败");
      }

      setProgress({ stage: "completed", progress: 100 });
      toast.success("文章抓取完成");
      options.onSuccess?.();
    } catch (error) {
      const err = error instanceof Error ? error : new Error("Unknown error");
      setProgress({ stage: "failed", progress: 0, error: err.message });
      toast.error(`抓取失败: ${err.message}`);
      options.onError?.(err);
    } finally {
      setIsCrawling(false);
    }
  }, [snapshotId, isCrawling, options]);

  const refetch = useCallback(
    async (opts?: RefetchOptions) => {
      if (isCrawling) return;

      setIsCrawling(true);

      try {
        const res = await fetch(`/api/wechat-article/${snapshotId}/refetch`, {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify(opts || {}),
        });

        if (!res.ok) {
          throw new Error("重新获取失败");
        }

        const result = await res.json();

        if (result.data.contentChanged) {
          toast.success("文章内容已更新");
        } else {
          toast.info("文章内容无变化");
        }

        options.onSuccess?.();
      } catch (error) {
        toast.error("重新获取失败");
      } finally {
        setIsCrawling(false);
      }
    },
    [snapshotId, isCrawling, options],
  );

  return {
    crawl,
    refetch,
    isCrawling,
    progress,
  };
}
```
