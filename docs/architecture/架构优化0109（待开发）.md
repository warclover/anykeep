#### Week 3: 统一缓存管理 (R3)

**实现方案**：

```typescript
// src/lib/cache/unified-cache-manager.ts
import { LRUCache } from "lru-cache";

interface CacheOptions {
  ttl?: number;
  maxSize?: number;
}

interface CacheStats {
  hits: number;
  misses: number;
  size: number;
  hitRate: number;
}

export class UnifiedCacheManager {
  private cache: LRUCache<string, unknown>;
  private stats = { hits: 0, misses: 0 };

  constructor(options: CacheOptions = {}) {
    this.cache = new LRUCache({
      max: options.maxSize ?? 10000,
      ttl: options.ttl ?? 300000,
    });
  }

  get<T>(key: string): T | undefined {
    const value = this.cache.get(key) as T | undefined;
    if (value !== undefined) {
      this.stats.hits++;
    } else {
      this.stats.misses++;
    }
    return value;
  }

  set<T>(key: string, value: T, options?: CacheOptions): void {
    this.cache.set(key, value, { ttl: options?.ttl });
  }

  async getOrSet<T>(
    key: string,
    factory: () => Promise<T>,
    options?: CacheOptions,
  ): Promise<T> {
    const cached = this.get<T>(key);
    if (cached !== undefined) return cached;

    const value = await factory();
    this.set(key, value, options);
    return value;
  }

  invalidate(pattern?: string): number {
    if (!pattern) {
      const size = this.cache.size;
      this.cache.clear();
      return size;
    }

    let count = 0;
    const regex = new RegExp(pattern);
    for (const key of this.cache.keys()) {
      if (regex.test(key)) {
        this.cache.delete(key);
        count++;
      }
    }
    return count;
  }

  getStats(): CacheStats {
    const total = this.stats.hits + this.stats.misses;
    return {
      ...this.stats,
      size: this.cache.size,
      hitRate: total > 0 ? this.stats.hits / total : 0,
    };
  }
}

// 全局实例
export const globalCache = new UnifiedCacheManager();
```

#### Week 6: Skill 标准化 (R6)

**标准化检查清单**：

```typescript
// src/lib/ai-skills/validation/skill-validator.ts
interface SkillValidationResult {
  skillId: string;
  passed: boolean;
  checks: {
    hasMetadata: boolean;
    hasParseInput: boolean;
    hasExecuteStream: boolean;
    hasDeliverable: boolean;
    hasTests: boolean;
    testCount: number;
    avgExecutionTime: number;
    slaCumpliance: boolean; // < 30秒
  };
  issues: string[];
}

async function validateSkill(skill: BaseSkill): Promise<SkillValidationResult> {
  // 检查元数据
  const hasMetadata = !!(skill.metadata?.id && skill.metadata?.name);

  // 检查必要方法
  const hasParseInput = typeof skill.parseInput === "function";
  const hasExecuteStream = typeof skill.executeStream === "function";
  const hasDeliverable = typeof skill.generateDeliverable === "function";

  // 检查测试
  const testFiles = await glob(`**/${skill.metadata.id}/**/*.test.ts`);
  const hasTests = testFiles.length > 0;
  const testCount = testFiles.length;

  // 执行性能测试
  const avgExecutionTime = await measureAverageExecutionTime(skill);
  const slaCumpliance = avgExecutionTime < 30000; // 30秒

  return {
    skillId: skill.metadata.id,
    passed:
      hasMetadata &&
      hasParseInput &&
      hasExecuteStream &&
      hasTests &&
      slaCumpliance,
    checks: {
      hasMetadata,
      hasParseInput,
      hasExecuteStream,
      hasDeliverable,
      hasTests,
      testCount,
      avgExecutionTime,
```

```typescript
import { NextResponse } from "next/server";
import type { NextRequest } from "next/server";

export function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;

  // v1 路由添加 deprecation header
  if (pathname.startsWith("/api/v1/") || pathname.match(/^\/api\/(?!v\d)/)) {
    const response = NextResponse.next();
    response.headers.set("X-API-Deprecated", "true");
    response.headers.set("X-API-Sunset", "2026-07-01");
    response.headers.set(
      "X-API-Upgrade",
      pathname.replace("/api/v1/", "/api/v2/").replace("/api/", "/api/v2/"),
    );
    return response;
  }

  return NextResponse.next();
}
```

#### Week 8: 可观测性增强 (R7)

**实现方案**：

```typescript
// src/lib/observability/metrics.ts
import { getConfig } from "@/lib/config/unified-config";

interface Metrics {
  tool: {
    calls: Map<string, number>;
    latency: Map<string, number[]>;
    errors: Map<string, number>;
  };
  skill: {
    executions: Map<string, number>;
    latency: Map<string, number[]>;
    success: Map<string, number>;
  };
  cache: {
    hits: number;
    misses: number;
  };
}

class MetricsCollector {
  private metrics: Metrics = {
    tool: { calls: new Map(), latency: new Map(), errors: new Map() },
    skill: { executions: new Map(), latency: new Map(), success: new Map() },
    cache: { hits: 0, misses: 0 },
  };

  recordToolCall(toolId: string, latencyMs: number, success: boolean) {
    const calls = this.metrics.tool.calls.get(toolId) ?? 0;
    this.metrics.tool.calls.set(toolId, calls + 1);

    const latencies = this.metrics.tool.latency.get(toolId) ?? [];
    latencies.push(latencyMs);
    this.metrics.tool.latency.set(toolId, latencies.slice(-100)); // 保留最近 100 个

    if (!success) {
      const errors = this.metrics.tool.errors.get(toolId) ?? 0;
      this.metrics.tool.errors.set(toolId, errors + 1);
    }
  }

  getToolStats(toolId: string) {
    const latencies = this.metrics.tool.latency.get(toolId) ?? [];
    const sorted = [...latencies].sort((a, b) => a - b);

    return {
      calls: this.metrics.tool.calls.get(toolId) ?? 0,
      errors: this.metrics.tool.errors.get(toolId) ?? 0,
      latency: {
        avg: latencies.reduce((a, b) => a + b, 0) / latencies.length || 0,
        p50: sorted[Math.floor(sorted.length * 0.5)] ?? 0,
        p95: sorted[Math.floor(sorted.length * 0.95)] ?? 0,
        p99: sorted[Math.floor(sorted.length * 0.99)] ?? 0,
      },
    };
  }

  // ... 更多方法
}

export const metricsCollector = new MetricsCollector();
```

## 风险应急预案

### 场景 1: Tool 分组导致意图分类崩溃

**症状**: 用户请求无法匹配到正确 Tool

**应急步骤**:

1. 立即关闭 `ENABLE_NEW_TOOL_REGISTRY` Feature Flag
2. 检查 CI 回归测试结果
3. 恢复 `tools.deprecated` 目录
4. 分析失败 case，修复后重新灰度

### 场景 2: API v2 迁移导致前端白屏

**症状**: 前端 API 调用 404

**应急步骤**:

1. 检查 v1 路由是否仍可用
2. 如 v1 不可用，立即回滚部署
3. 检查 Vercel/Nginx 重定向配置
4. 通知前端团队切换到 v1

### 场景 3: 缓存迁移导致性能骤降

**症状**: 响应时间增加 > 50%

**应急步骤**:

1. 关闭 `ENABLE_UNIFIED_CACHE` Feature Flag
2. 检查缓存命中率监控
3. 恢复旧缓存实现
4. 分析 TTL 和 Key 设计问题
