#### 1.2.2 类型化事件总线实现

````typescript
// src/lib/shared/event-bus/typed-event-bus.ts

import { nanoid } from "nanoid";
import { logger } from "@/lib/logger";
import type {
  DomainEventMap,
  DomainEventName,
  DomainEvent,
} from "./event-types";

type EventHandler<T> = (event: T) => void | Promise<void>;

interface EventSubscription {
  id: string;
  handler: EventHandler<unknown>;
  once: boolean;
}

/**
 * 类型安全的事件总线
 *
 * @example
 * ```typescript
 * // 订阅事件
 * typedEventBus.on("classification:completed", (event) => {
 *   // event 自动推断为 ClassificationCompletedEvent
 *   console.log(event.bookmarkId, event.confidence);
 * });
 *
 * // 发布事件
 * typedEventBus.emit("classification:completed", {
 *   bookmarkId: 123,
 *   folderId: 456,
 *   confidence: 0.95,
 *   strategy: "hybrid",
 *   latencyMs: 150,
 *   needsConfirmation: false,
 * });
 * ```
 */
class TypedEventBus {
  private handlers = new Map<string, EventSubscription[]>();
  private eventHistory: Array<{
    name: string;
    event: DomainEvent;
    timestamp: number;
  }> = [];
  private maxHistorySize = 100;

  /**
   * 订阅事件
   */
  on<K extends DomainEventName>(
    eventName: K,
    handler: EventHandler<DomainEventMap[K]>,
  ): () => void {
    return this.subscribe(eventName, handler, false);
  }

  /**
   * 订阅事件（只触发一次）
   */
  once<K extends DomainEventName>(
    eventName: K,
    handler: EventHandler<DomainEventMap[K]>,
  ): () => void {
    return this.subscribe(eventName, handler, true);
  }

  /**
   * 发布事件
   */
  emit<K extends DomainEventName>(
    eventName: K,
    payload: Omit<DomainEventMap[K], "eventId" | "timestamp" | "source">,
  ): void {
    const event: DomainEvent = {
      eventId: nanoid(),
      timestamp: new Date().toISOString(),
      source: this.extractSource(eventName),
      ...payload,
    } as DomainEventMap[K];

    // 记录事件历史
    this.recordEvent(eventName, event);

    // 获取处理器
    const subscriptions = this.handlers.get(eventName) || [];

    if (subscriptions.length === 0) {
      logger.debug(
        { eventName, eventId: event.eventId },
        "TypedEventBus: No handlers for event",
      );
      return;
    }

    // 执行处理器
    const toRemove: string[] = [];

    for (const subscription of subscriptions) {
      try {
        const result = subscription.handler(event);

        // 支持异步处理器
        if (result instanceof Promise) {
          result.catch((error) => {
            logger.error(
              {
                eventName,
                eventId: event.eventId,
                error: error instanceof Error ? error.message : String(error),
              },
              "TypedEventBus: Async handler error",
            );
          });
        }

        if (subscription.once) {
          toRemove.push(subscription.id);
        }
      } catch (error) {
        logger.error(
          {
            eventName,
            eventId: event.eventId,
            error: error instanceof Error ? error.message : String(error),
          },
          "TypedEventBus: Handler error",
        );
      }
    }

    // 移除一次性订阅
    if (toRemove.length > 0) {
      this.handlers.set(
        eventName,
        subscriptions.filter((s) => !toRemove.includes(s.id)),
      );
    }

    logger.debug(
      {
        eventName,
        eventId: event.eventId,
        handlersExecuted: subscriptions.length,
      },
      "TypedEventBus: Event emitted",
    );
  }

  /**
   * 异步发布事件（等待所有处理器完成）
   */
  async emitAsync<K extends DomainEventName>(
    eventName: K,
    payload: Omit<DomainEventMap[K], "eventId" | "timestamp" | "source">,
  ): Promise<void> {
    const event: DomainEvent = {
      eventId: nanoid(),
      timestamp: new Date().toISOString(),
      source: this.extractSource(eventName),
      ...payload,
    } as DomainEventMap[K];

    this.recordEvent(eventName, event);

    const subscriptions = this.handlers.get(eventName) || [];
    const toRemove: string[] = [];

    await Promise.all(
      subscriptions.map(async (subscription) => {
        try {
          await subscription.handler(event);
          if (subscription.once) {
            toRemove.push(subscription.id);
          }
        } catch (error) {
          logger.error(
            {
              eventName,
              eventId: event.eventId,
              error: error instanceof Error ? error.message : String(error),
            },
            "TypedEventBus: Async handler error",
          );
        }
      }),
    );

    if (toRemove.length > 0) {
      this.handlers.set(
        eventName,
        subscriptions.filter((s) => !toRemove.includes(s.id)),
      );
    }
  }

  /**
   * 移除所有事件处理器
   */
  off<K extends DomainEventName>(eventName: K): void {
    this.handlers.delete(eventName);
  }

  /**
   * 移除所有处理器
   */
  clear(): void {
    this.handlers.clear();
  }

  /**
   * 获取事件历史（调试用）
   */
  getHistory(): Array<{ name: string; event: DomainEvent; timestamp: number }> {
    return [...this.eventHistory];
  }

  /**
   * 获取订阅统计
   */
  getStats(): Record<string, number> {
    const stats: Record<string, number> = {};
    for (const [name, subs] of this.handlers) {
      stats[name] = subs.length;
    }
    return stats;
  }

  // ============================================================================
  // Private Methods
  // ============================================================================

  private subscribe<K extends DomainEventName>(
    eventName: K,
    handler: EventHandler<DomainEventMap[K]>,
    once: boolean,
  ): () => void {
    const subscription: EventSubscription = {
      id: nanoid(),
      handler: handler as EventHandler<unknown>,
      once,
    };

    const existing = this.handlers.get(eventName) || [];
    this.handlers.set(eventName, [...existing, subscription]);

    // 返回取消订阅函数
    return () => {
      const subs = this.handlers.get(eventName) || [];
      this.handlers.set(
        eventName,
        subs.filter((s) => s.id !== subscription.id),
      );
    };
  }

  private extractSource(eventName: string): string {
    const [domain] = eventName.split(":");
    return domain || "unknown";
  }

  private recordEvent(name: string, event: DomainEvent): void {
    this.eventHistory.push({
      name,
      event,
      timestamp: Date.now(),
    });

    // 限制历史大小
    if (this.eventHistory.length > this.maxHistorySize) {
      this.eventHistory = this.eventHistory.slice(-this.maxHistorySize);
    }
  }
}

// ============================================================================
// Singleton Export
// ============================================================================

export const typedEventBus = new TypedEventBus();

// 兼容旧代码的适配器
export const eventBusAdapter = {
  emit: <K extends DomainEventName>(
    eventName: K,
    payload: Omit<DomainEventMap[K], "eventId" | "timestamp" | "source">,
  ) => typedEventBus.emit(eventName, payload),

  on: <K extends DomainEventName>(
    eventName: K,
    handler: EventHandler<DomainEventMap[K]>,
  ) => typedEventBus.on(eventName, handler),
};
````

#### 1.2.3 迁移适配器

```typescript
// src/lib/shared/event-bus/legacy-adapter.ts

import { eventBus as legacyEventBus } from "./index";
import { typedEventBus } from "./typed-event-bus";
import type { DomainEventMap, DomainEventName } from "./event-types";
import { logger } from "@/lib/logger";

/**
 * 旧事件名到新事件名的映射
 */
const EVENT_NAME_MAPPING: Record<string, DomainEventName> = {
  // 旧格式 → 新格式
  "summary:completed": "summary:completed",
  "summary:tags_generated": "summary:tags_generated",
  "classification.completed": "classification:completed",
  "classification.confirmed": "classification:confirmed",
  // ... 添加更多映射
};

/**
 * 创建双向桥接
 *
 * 旧代码发布事件 → 同时发布到新总线
 * 新代码发布事件 → 同时发布到旧总线（兼容期）
 */
export function bridgeEventBuses(): void {
  // 拦截旧总线的 emit
  const originalEmit = legacyEventBus.emit.bind(legacyEventBus);

  legacyEventBus.emit = (eventName: string, data: unknown) => {
    // 调用原始 emit
    originalEmit(eventName, data);

    // 转发到新总线
    const newEventName = EVENT_NAME_MAPPING[eventName];
    if (newEventName) {
      try {
        typedEventBus.emit(newEventName, data as never);
        logger.debug(
          { from: eventName, to: newEventName },
          "EventBridge: Forwarded legacy event to typed bus",
        );
      } catch (error) {
        logger.warn(
          { eventName, error },
          "EventBridge: Failed to forward event",
        );
      }
    }
  };

  logger.info("EventBridge: Legacy to typed event bus bridge enabled");
}

/**
 * 在应用启动时调用
 */
export function initializeEventBus(): void {
  bridgeEventBuses();
}
```

````typescript
// src/lib/domains/classification/classification.facade.ts

import type { SupabaseClient } from "@supabase/supabase-js";
import type { Database } from "@/types/supabase";
import { typedEventBus } from "@/lib/shared/event-bus/typed-event-bus";
import { logger } from "@/lib/logger";

// 内部服务
import { ClassificationService } from "./services/classification.service";
import { InterestService } from "./services/interest.service";
import { TagGraphService } from "./services/tag-graph.service";
import { LearningService } from "./services/learning.service";

// 策略
import { VectorMatcher } from "./strategies/vector-matcher";
import { HistoryMatcher } from "./strategies/history-matcher";
import { AIRecommender } from "./strategies/ai-recommender";
import { ScoreAggregator } from "./strategies/score-aggregator";

// 类型
import type {
  ClassificationResult,
  ClassificationOptions,
  UserInterests,
  TagRelationship,
} from "./types";

/**
 * 分类领域外观服务
 *
 * 对外提供统一的分类能力接口，隐藏内部复杂性
 *
 * @example
 * ```typescript
 * const facade = getClassificationFacade(supabase);
 *
 * // 分类书签
 * const result = await facade.classifyBookmark(userId, bookmark);
 *
 * // 获取用户兴趣
 * const interests = await facade.getUserInterests(userId);
 * ```
 */
export class ClassificationFacade {
  private classificationService: ClassificationService;
  private interestService: InterestService;
  private tagGraphService: TagGraphService;
  private learningService: LearningService;

  constructor(supabase: SupabaseClient<Database>) {
    // 初始化策略
    const vectorMatcher = new VectorMatcher(supabase);
    const historyMatcher = new HistoryMatcher(supabase);
    const aiRecommender = new AIRecommender();
    const scoreAggregator = new ScoreAggregator();

    // 初始化服务
    this.classificationService = new ClassificationService(
      supabase,
      vectorMatcher,
      historyMatcher,
      aiRecommender,
      scoreAggregator,
    );
    this.interestService = new InterestService(supabase);
    this.tagGraphService = new TagGraphService(supabase);
    this.learningService = new LearningService(supabase);

    // 注册事件处理器
    this.registerEventHandlers();
  }

  // ============================================================================
  // Public API - Classification
  // ============================================================================

  /**
   * 分类书签
   */
  async classifyBookmark(
    userId: string,
    bookmark: {
      id: number;
      title: string;
      url?: string;
      description?: string;
      content?: string;
      tags?: string[];
    },
    options?: ClassificationOptions,
  ): Promise<ClassificationResult> {
    const startTime = Date.now();

    // 发布请求事件
    typedEventBus.emit("classification:requested", {
      bookmarkId: bookmark.id,
      contentType: options?.contentType || "unknown",
      priority: options?.priority || "normal",
      userId,
    });

    try {
      const result = await this.classificationService.classify(
        userId,
        bookmark,
        options,
      );

      // 发布完成事件
      typedEventBus.emit("classification:completed", {
        bookmarkId: bookmark.id,
        folderId: result.folderId,
        confidence: result.confidence,
        strategy: result.strategy,
        latencyMs: Date.now() - startTime,
        needsConfirmation: result.confidence < 0.8,
        userId,
      });

      return result;
    } catch (error) {
      logger.error(
        {
          bookmarkId: bookmark.id,
          userId,
          error: error instanceof Error ? error.message : String(error),
        },
        "ClassificationFacade: Classification failed",
      );
      throw error;
    }
  }

  /**
   * 批量分类
   */
  async classifyBatch(
    userId: string,
    bookmarks: Array<{
      id: number;
      title: string;

      this.interestService.isHealthy(),
      this.tagGraphService.isHealthy(),

          ? "degraded"

/**
 * 统一 API 响应格式
 */
export interface ApiResponse<T = unknown> {
  success: boolean;
  data?: T;
  error?: {
    code: string;
    message: string;
    details?: unknown;
  };
  meta?: {
    timestamp: string;
    requestId: string;
    pagination?: {
      page: number;
      pageSize: number;
      total: number;
      totalPages: number;
    };
  };
}

/**
 * 成功响应
 */
export function successResponse<T>(
  data: T,
  meta?: Partial<ApiResponse["meta"]>,
): ApiResponse<T> {
  return {
    success: true,
    data,
    meta: {
      timestamp: new Date().toISOString(),
      requestId: crypto.randomUUID(),
      ...meta,
    },
  };
}

/**
 * 错误响应
 */
export function errorResponse(
  code: string,
  message: string,
  details?: unknown,
): ApiResponse {
  return {
    success: false,
    error: {
      code,
      message,
      details,
    },
    meta: {
      timestamp: new Date().toISOString(),
      requestId: crypto.randomUUID(),
    },
  };
}

/**
 * 分页响应
 */
export function paginatedResponse<T>(
  data: T[],
  pagination: {
    page: number;
    pageSize: number;
    total: number;
  },
): ApiResponse<T[]> {
  return {
    success: true,
    data,
    meta: {
      timestamp: new Date().toISOString(),
      requestId: crypto.randomUUID(),
      pagination: {
        ...pagination,
        totalPages: Math.ceil(pagination.total / pagination.pageSize),
      },
    },
  };
}

// src/app/api/v1/bookmarks/route.ts

import { NextRequest, NextResponse } from "next/server";
import { createClient } from "@/lib/supabase/server";
import {
  successResponse,
  errorResponse,
  paginatedResponse,
} from "@/lib/shared/api/response";
import { ERROR_CODES } from "@/lib/shared/api/error-codes";
import { z } from "zod";

// 请求验证 Schema
const createBookmarkSchema = z.object({
  title: z.string().min(1).max(500),
  url: z.string().url(),
  description: z.string().max(2000).optional(),
  folderId: z.number().optional(),
  tags: z.array(z.string()).max(10).optional(),
});

const listBookmarksSchema = z.object({
  page: z.coerce.number().min(1).default(1),
  pageSize: z.coerce.number().min(1).max(100).default(20),
  folderId: z.coerce.number().optional(),
  search: z.string().optional(),
});

/**
 * GET /api/v1/bookmarks
 *
 * 获取书签列表
 */
export async function GET(request: NextRequest) {
  try {
    const supabase = await createClient();
    const {
      data: { user },
    } = await supabase.auth.getUser();

    if (!user) {
      return NextResponse.json(
        errorResponse(ERROR_CODES.UNAUTHORIZED, "未登录"),
        { status: 401 },
      );
    }

    // 解析查询参数
    const searchParams = Object.fromEntries(request.nextUrl.searchParams);
    const params = listBookmarksSchema.parse(searchParams);

    // 查询书签
    let query = supabase
      .from("bookmarks")
      .select("*", { count: "exact" })
      .eq("user_id", user.id)
      .order("created_at", { ascending: false });

    if (params.folderId) {
      query = query.eq("folder_id", params.folderId);
    }

    if (params.search) {
      query = query.ilike("title", `%${params.search}%`);
    }

    // 分页
    const from = (params.page - 1) * params.pageSize;
    const to = from + params.pageSize - 1;
    query = query.range(from, to);

    const { data, error, count } = await query;

    if (error) {
      return NextResponse.json(
        errorResponse(ERROR_CODES.INTERNAL_ERROR, error.message),
        { status: 500 },
      );
    }

    return NextResponse.json(
      paginatedResponse(data || [], {
        page: params.page,
        pageSize: params.pageSize,
        total: count || 0,
      }),
    );
  } catch (error) {
    if (error instanceof z.ZodError) {
      return NextResponse.json(
        errorResponse(
          ERROR_CODES.VALIDATION_ERROR,
          "参数验证失败",
          error.errors,
        ),
        { status: 400 },
      );
    }
    throw error;
  }
}

/**
 * POST /api/v1/bookmarks
 *
 * 创建书签
 */
export async function POST(request: NextRequest) {
  try {
    const supabase = await createClient();
    const {
      data: { user },
    } = await supabase.auth.getUser();

    if (!user) {
      return NextResponse.json(
        errorResponse(ERROR_CODES.UNAUTHORIZED, "未登录"),
        { status: 401 },
      );
    }

    // 解析并验证请求体
    const body = await request.json();
    const data = createBookmarkSchema.parse(body);

    // 检查 URL 重复
    const { data: existing } = await supabase
      .from("bookmarks")
      .select("id")
      .eq("user_id", user.id)
      .eq("url", data.url)
      .single();

    if (existing) {
      return NextResponse.json(
        errorResponse(ERROR_CODES.BOOKMARK_DUPLICATE_URL, "该 URL 已存在"),
        { status: 409 },
      );
    }

    // 创建书签
    const { data: bookmark, error } = await supabase
      .from("bookmarks")
      .insert({
        user_id: user.id,
        title: data.title,
        url: data.url,
        description: data.description,
        folder_id: data.folderId,
        tags: data.tags,
      })
      .select()
      .single();

    if (error) {
      return NextResponse.json(
        errorResponse(ERROR_CODES.INTERNAL_ERROR, error.message),
        { status: 500 },
      );
    }

    return NextResponse.json(successResponse(bookmark), { status: 201 });
  } catch (error) {
    if (error instanceof z.ZodError) {
      return NextResponse.json(
        errorResponse(
          ERROR_CODES.VALIDATION_ERROR,
          "参数验证失败",
          error.errors,
        ),
        { status: 400 },
      );
    }
    throw error;
  }
}

```typescript
// src/lib/shared/observability/metrics.ts

interface MetricLabels {
  [key: string]: string | number;
}

class MetricsCollector {
  private counters = new Map<string, number>();
  private histograms = new Map<string, number[]>();

  /**
   * 计数器
   */
  increment(name: string, labels?: MetricLabels, value = 1): void {
    const key = this.formatKey(name, labels);
    const current = this.counters.get(key) || 0;
    this.counters.set(key, current + value);
  }

  /**
   * 直方图（用于延迟等）
   */
  histogram(name: string, value: number, labels?: MetricLabels): void {
    const key = this.formatKey(name, labels);
    const values = this.histograms.get(key) || [];
    values.push(value);
    this.histograms.set(key, values);
  }

  /**
   * 计时器装饰器
   */
  timer(name: string, labels?: MetricLabels): () => void {
    const start = performance.now();
    return () => {
      const duration = performance.now() - start;
      this.histogram(name, duration, labels);
    };
  }

  /**
   * 获取所有指标
   */
  getMetrics(): {
    counters: Record<string, number>;
    histograms: Record<
      string,
      { count: number; avg: number; p50: number; p95: number; p99: number }
    >;
  } {
    const histogramStats: Record<
      string,
      { count: number; avg: number; p50: number; p95: number; p99: number }
    > = {};

    for (const [key, values] of this.histograms) {
      const sorted = [...values].sort((a, b) => a - b);
      const count = sorted.length;
      const avg = sorted.reduce((a, b) => a + b, 0) / count;

      histogramStats[key] = {
        count,
        avg,
        p50: sorted[Math.floor(count * 0.5)] || 0,
        p95: sorted[Math.floor(count * 0.95)] || 0,
        p99: sorted[Math.floor(count * 0.99)] || 0,
      };
    }

    return {
      counters: Object.fromEntries(this.counters),
      histograms: histogramStats,
    };
  }

  private formatKey(name: string, labels?: MetricLabels): string {
    if (!labels) return name;
    const labelStr = Object.entries(labels)
      .map(([k, v]) => `${k}="${v}"`)
      .join(",");
    return `${name}{${labelStr}}`;
  }
}
