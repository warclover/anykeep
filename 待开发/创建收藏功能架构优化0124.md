```typescript
// src/lib/bookmark/services/bookmark-creator.service.ts
import { SupabaseClient } from "@supabase/supabase-js";
import { BookmarkValidator } from "./bookmark-validator.service";
import { TaskOrchestrator } from "../tasks/task-orchestrator";
import { JobQueueService } from "../queue/job-queue.service";
import type {
  CreateBookmarkInput,
  Bookmark,
  BatchCreateResult,
} from "../types";

export class BookmarkCreatorService {
  private validator: BookmarkValidator;
  private orchestrator: TaskOrchestrator;
  private jobQueue: JobQueueService;

  constructor(
    private supabase: SupabaseClient,
    private userId: string,
  ) {
    this.validator = new BookmarkValidator(supabase);
    this.orchestrator = new TaskOrchestrator(supabase);
    this.jobQueue = JobQueueService.getInstance(supabase);
  }

  /**
   * 创建单条书签
   */
  async create(input: CreateBookmarkInput): Promise<Bookmark> {
    // 1. 验证输入
    const validated = await this.validator.validate(input);

    // 2. 检查配额
    await this.validator.checkQuota(this.userId);

    // 3. 检查重复
    const duplicate = await this.checkDuplicate(validated.url);
    if (duplicate) {
      throw new DuplicateBookmarkError(duplicate.id);
    }

    // 4. 获取目标文件夹
    const folderId = validated.folderId || (await this.getInboxFolderId());

    // 5. 检测平台
    const platform = detectPlatformFromUrl(validated.url);

    // 6. 创建书签记录 (status: PENDING)
    const bookmark = await this.insertBookmark({
      ...validated,
      userId: this.userId,
      folderId,
      platform,
      status: BookmarkStatus.PENDING,
    });

    // 7. 更新配额
    await this.incrementStorageUsed();

    // 8. 入队异步任务
    await this.jobQueue.enqueue({
      bookmarkId: bookmark.id,
      taskType: "full_enrichment",
      priority: "normal",
    });

    // 9. 触发任务处理（不阻塞）
    this.orchestrator.processAsync(bookmark.id).catch((error) => {
      logger.error("Background processing failed", {
        bookmarkId: bookmark.id,
        error,
      });
    });

    return bookmark;
  }

  /**
   * 批量创建书签
   */
  async createBatch(
    inputs: CreateBookmarkInput[],
    options?: BatchCreateOptions,
  ): Promise<BatchCreateResult> {
    const results: BatchCreateResult = {
      created: [],
      skipped: [],
      failed: [],
      jobId: generateJobId(),
    };

    // 1. 批量验证
    const validatedInputs = await this.validator.validateBatch(inputs);

    // 2. 检查配额（一次性）
    await this.validator.checkQuotaBatch(this.userId, validatedInputs.length);

    // 3. 去重处理
    const uniqueInputs = options?.skipDuplicates
      ? await this.filterDuplicates(validatedInputs)
      : validatedInputs;

    // 4. 批量插入（单次 INSERT）
    const bookmarks = await this.insertBookmarksBatch(uniqueInputs);
    results.created = bookmarks;

    // 5. 批量入队
    await this.jobQueue.enqueueBatch(
      bookmarks.map((b) => ({
        bookmarkId: b.id,
        taskType: "full_enrichment",
        priority: options?.priority || "normal",
        batchJobId: results.jobId,
      })),
    );

    // 6. 触发批量处理
    this.orchestrator.processBatchAsync(
      bookmarks.map((b) => b.id),
      { parallelLimit: options?.parallelLimit || 5 },
    );

    return results;
  }

  // ===== 私有方法 =====

  private async getInboxFolderId(): Promise<number> {
    const { data } = await this.supabase.rpc("get_or_create_inbox_folder", {
      p_user_id: this.userId,
    });
    return data;
  }

  private async checkDuplicate(url: string): Promise<Bookmark | null> {
    const normalizedUrl = normalizeUrl(url);
    const { data } = await this.supabase
      .from("bookmarks")
      .select("id, url, title")
      .eq("user_id", this.userId)
      .eq("url", normalizedUrl)
      .single();
    return data;
  }

  private async insertBookmark(data: InsertBookmarkData): Promise<Bookmark> {
    const { data: bookmark, error } = await this.supabase
      .from("bookmarks")
      .insert(data)
      .select("*, folders!bookmarks_folder_id_fkey(name)")
      .single();

    if (error) throw new DatabaseError("Failed to create bookmark", error);
    return bookmark;
  }

  private async insertBookmarksBatch(
    inputs: ValidatedInput[],
  ): Promise<Bookmark[]> {
    const { data, error } = await this.supabase
      .from("bookmarks")
      .insert(
        inputs.map((input) => ({
          user_id: this.userId,
          url: input.url,
          title: input.title || input.url,
          status: BookmarkStatus.PENDING,
          platform: detectPlatformFromUrl(input.url),
          folder_id: input.folderId,
          // ... 其他字段
        })),
      )
      .select("*");

    if (error) throw new DatabaseError("Failed to create bookmarks", error);
    return data;
  }

  private async incrementStorageUsed(count: number = 1): Promise<void> {
    await this.supabase.rpc("increment_storage_used", {
      p_user_id: this.userId,
      p_increment: count,
    });
  }
}

// 工厂函数
export function createBookmarkCreator(
  supabase: SupabaseClient,
  userId: string,
): BookmarkCreatorService {
  return new BookmarkCreatorService(supabase, userId);
}
```

```typescript
import { NextRequest } from "next/server";
import { getCachedUser } from "@/lib/supabase/server";
import { createBookmarkCreator } from "@/lib/bookmark";
import { createBookmarkSchema } from "@/lib/validation";
import { successResponse, errorResponse } from "@/lib/api/response";

export async function POST(request: NextRequest) {
  try {
    // 1. 认证
    const { user, error: authError } = await getCachedUser();
    if (authError || !user) {
      return errorResponse("Unauthorized", 401);
    }

    // 2. 解析请求体
    const body = await request.json();
    const validation = createBookmarkSchema.safeParse(body);
    if (!validation.success) {
      return errorResponse(validation.error.message, 400);
    }

    // 3. 创建书签
    const supabase = await createClient();
    const creator = createBookmarkCreator(supabase, user.id);
    const bookmark = await creator.create(validation.data);

    // 4. 返回���果
    return successResponse(bookmark, 201);
  } catch (error) {
    if (error instanceof DuplicateBookmarkError) {
      return errorResponse("Bookmark already exists", 409, {
        existingId: error.bookmarkId,
      });
    }
    if (error instanceof QuotaExceededError) {
      return errorResponse("Storage quota exceeded", 403);
    }
    logger.error("Create bookmark failed", { error });
    return errorResponse("Internal server error", 500);
  }
}
```

#### 2.5 任务编排器更新状态

```typescript
// src/lib/bookmark/tasks/task-orchestrator.ts
export class TaskOrchestrator {
  async process(bookmarkId: number): Promise<void> {
    try {
      // Stage 1: 内容爬取
      await this.updateProgress(bookmarkId, BookmarkStatus.FETCHING, 1);
      const content = await this.contentFetchTask.execute(bookmarkId);

      // Stage 2: AI 摘要
      await this.updateProgress(bookmarkId, BookmarkStatus.SUMMARIZING, 2);
      const summary = await this.summaryTask.execute(bookmarkId, content);

      // Stage 3: 向量嵌入
      await this.updateProgress(bookmarkId, BookmarkStatus.EMBEDDING, 3);
      await this.embeddingTask.execute(bookmarkId, content, summary);

      // Stage 4: 智能分类
      await this.updateProgress(bookmarkId, BookmarkStatus.CLASSIFYING, 4);
      await this.classificationTask.execute(bookmarkId);

      // 完成
      await this.updateProgress(bookmarkId, BookmarkStatus.COMPLETED, 5);
    } catch (error) {
      await this.handleError(bookmarkId, error);
    }
  }

  private async updateProgress(
    bookmarkId: number,
    status: BookmarkStatus,
    step: number,
  ): Promise<void> {
    await this.supabase
      .from("bookmarks")
      .update({
        status,
        processing_step: step,
        updated_at: new Date().toISOString(),
      })
      .eq("id", bookmarkId);

    // 记录日志
    await this.supabase.from("bookmark_processing_logs").insert({
      bookmark_id: bookmarkId,
      step: status,
      status: "started",
    });
  }

  private async handleError(bookmarkId: number, error: unknown): Promise<void> {
    const message = error instanceof Error ? error.message : "Unknown error";

    await this.supabase
      .from("bookmarks")
      .update({
        status: BookmarkStatus.FAILED,
        processing_error: message,
        updated_at: new Date().toISOString(),
      })
      .eq("id", bookmarkId);

    logger.error("Bookmark processing failed", { bookmarkId, error });
  }
}
```

```sql
CREATE TABLE IF NOT EXISTS bookmark_jobs (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  bookmark_id INTEGER NOT NULL REFERENCES bookmarks(id) ON DELETE CASCADE,
  task_type TEXT NOT NULL,  -- full_enrichment, summary_only, embedding_only, etc.
  status TEXT NOT NULL DEFAULT 'pending',  -- pending, running, completed, failed
  priority TEXT NOT NULL DEFAULT 'normal',  -- high, normal, low
  attempts INTEGER NOT NULL DEFAULT 0,
  max_attempts INTEGER NOT NULL DEFAULT 3,
  last_error TEXT,
  batch_job_id UUID,  -- 批量任务关联
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  started_at TIMESTAMPTZ,
  completed_at TIMESTAMPTZ,
  next_retry_at TIMESTAMPTZ,

  CONSTRAINT bookmark_jobs_status_check
    CHECK (status IN ('pending', 'running', 'completed', 'failed', 'cancelled'))
);

-- 索引
CREATE INDEX idx_bookmark_jobs_status ON bookmark_jobs(status);
CREATE INDEX idx_bookmark_jobs_next_retry ON bookmark_jobs(next_retry_at)
  WHERE status = 'pending';
CREATE INDEX idx_bookmark_jobs_batch ON bookmark_jobs(batch_job_id)
  WHERE batch_job_id IS NOT NULL;

-- 获取待处理任务（原子操作）
CREATE OR REPLACE FUNCTION claim_pending_job(p_limit INTEGER DEFAULT 1)
RETURNS SETOF bookmark_jobs AS $$
BEGIN
  RETURN QUERY
  UPDATE bookmark_jobs
  SET status = 'running',
      started_at = NOW(),
      attempts = attempts + 1
  WHERE id IN (
    SELECT id FROM bookmark_jobs
    WHERE status = 'pending'
      AND (next_retry_at IS NULL OR next_retry_at <= NOW())
    ORDER BY
      CASE priority WHEN 'high' THEN 1 WHEN 'normal' THEN 2 ELSE 3 END,
      created_at
    LIMIT p_limit
    FOR UPDATE SKIP LOCKED
  )
  RETURNING *;
END;
$$ LANGUAGE plpgsql;
```

```typescript
  }

  /**
   * 入队任务
   */
  async enqueue(job: EnqueueJobInput): Promise<string> {
    const { data, error } = await this.supabase
      .from("bookmark_jobs")
      .insert({
        bookmark_id: job.bookmarkId,
        task_type: job.taskType,
        priority: job.priority || "normal",
        max_attempts: job.maxAttempts || 3,
        batch_job_id: job.batchJobId,
      })
      .select("id")
      .single();

    if (error) throw new Error("Failed to enqueue job");
    return data.id;
  }

  /**
   * 批量入队
   */
  async enqueueBatch(jobs: EnqueueJobInput[]): Promise<string[]> {
    const { data, error } = await this.supabase
      .from("bookmark_jobs")
      .insert(
        jobs.map((job) => ({
          bookmark_id: job.bookmarkId,
          task_type: job.taskType,
          priority: job.priority || "normal",
          max_attempts: job.maxAttempts || 3,
          batch_job_id: job.batchJobId,
        })),
      )
      .select("id");

    if (error) throw new Error("Failed to enqueue jobs");
    return data.map((d) => d.id);
  }

  /**
   * 认领任务
   */
  async claimJob(): Promise<BookmarkJob | null> {
    const { data } = await this.supabase.rpc("claim_pending_job", {
      p_limit: 1,
    });
    return data?.[0] || null;
  }

  /**
   * 完成任务
   */
  async completeJob(jobId: string): Promise<void> {
    await this.supabase
      .from("bookmark_jobs")
      .update({
        status: "completed",
        completed_at: new Date().toISOString(),
      })
      .eq("id", jobId);
  }

  /**
   * 失败任务（可能重试）
   */
  async failJob(jobId: string, error: string): Promise<void> {
    const { data: job } = await this.supabase
      .from("bookmark_jobs")
      .select("attempts, max_attempts")
      .eq("id", jobId)
      .single();

    if (job && job.attempts < job.max_attempts) {
      // 安排重试（指数退避）
      const delayMs = Math.pow(2, job.attempts) * 1000; // 2s, 4s, 8s...
      const nextRetryAt = new Date(Date.now() + delayMs);

      await this.supabase
        .from("bookmark_jobs")
        .update({
          status: "pending",
          last_error: error,
          next_retry_at: nextRetryAt.toISOString(),
        })
        .eq("id", jobId);
    } else {
      // 标记为最终失败
      await this.supabase
        .from("bookmark_jobs")
        .update({
          status: "failed",
          last_error: error,
          completed_at: new Date().toISOString(),
        })
        .eq("id", jobId);
    }
  }

  /**
   * 启动任务处理循环
   */
  startProcessing(processor: JobProcessor): void {
    if (this.isProcessing) return;
    this.isProcessing = true;

    const loop = async () => {
      while (this.isProcessing) {
        try {
          const job = await this.claimJob();
          if (job) {
            await processor.process(job);
          } else {
            // 无任务，等待 1 秒
            await sleep(1000);
          }
        } catch (error) {
          logger.error("Job processing error", { error });
          await sleep(5000);
        }
      }
    };

    loop();
  }

  /**
   * 停止处理
   */
  stopProcessing(): void {
    this.isProcessing = false;
  }
}
```

```typescript
export async function recoverPendingJobs(
  supabase: SupabaseClient,
): Promise<number> {
  // 查找服务重启前正在运行的任务
  const { data: stalledJobs } = await supabase
    .from("bookmark_jobs")
    .select("id")
    .eq("status", "running")
    .lt("started_at", new Date(Date.now() - 5 * 60 * 1000).toISOString()); // 5分钟前

  if (!stalledJobs?.length) return 0;

  // 重置为待处理
  await supabase
    .from("bookmark_jobs")
    .update({
      status: "pending",
      next_retry_at: new Date().toISOString(),
    })
    .in(
      "id",
      stalledJobs.map((j) => j.id),
    );

  logger.info("Recovered stalled jobs", { count: stalledJobs.length });
  return stalledJobs.length;
}
```

#### 4.1 并行执行策略

```
当前（串行）:
  爬取(3s) → 摘要(5s) → 嵌入(2s) → 分类(3s) = 13s 总计

优化后（部分并行）:
  爬取(3s) ─────────────────────────────────────────┐
                                                    │
            ┌─ 摘要(5s) ─┐                          │
            │            │                          ▼
            │            └───────────────────────► 分类(3s)
            │                                       ▲
            └─ 基础嵌入(2s) ────────────────────────┘

  总计: 3s + max(5s, 2s) + 3s = 11s （节省 15%）
```

#### 4.2 并行编排器实现

```typescript
// src/lib/bookmark/tasks/task-orchestrator.ts
export class TaskOrchestrator {
  async process(bookmarkId: number): Promise<void> {
    try {
      // Stage 1: 内容爬取（必须先完成）
      await this.updateProgress(bookmarkId, BookmarkStatus.FETCHING, 1);
      const content = await this.contentFetchTask.execute(bookmarkId);

      // Stage 2: 并行执行摘要和基础嵌入
      await this.updateProgress(bookmarkId, BookmarkStatus.SUMMARIZING, 2);

      const [summaryResult, embeddingResult] = await Promise.allSettled([
        this.summaryTask.execute(bookmarkId, content),
        this.embeddingTask.executeBasic(bookmarkId, content.title), // 仅用标题
      ]);

      // 处理结果
      const summary =
        summaryResult.status === "fulfilled" ? summaryResult.value : null;

      if (summaryResult.status === "rejected") {
        logger.warn("Summary generation failed, continuing", {
          bookmarkId,
          error: summaryResult.reason,
        });
      }

      // Stage 3: 如果摘要成功，增强嵌入
      if (summary) {
        await this.updateProgress(bookmarkId, BookmarkStatus.EMBEDDING, 3);
        await this.embeddingTask.executeEnhanced(bookmarkId, content, summary);
      }

      // Stage 4: 智能分类
      await this.updateProgress(bookmarkId, BookmarkStatus.CLASSIFYING, 4);
      await this.classificationTask.execute(bookmarkId);

      // 完成
      await this.updateProgress(bookmarkId, BookmarkStatus.COMPLETED, 5);
    } catch (error) {
      await this.handleError(bookmarkId, error);
      throw error;
    }
  }

  /**
   * 批量并行处理
   */
  async processBatch(
    bookmarkIds: number[],
    options: { parallelLimit?: number } = {},
  ): Promise<void> {
    const limit = options.parallelLimit || 5;

    // 使用 p-limit 控制并发
    const limiter = pLimit(limit);

    await Promise.all(
      bookmarkIds.map((id) =>
        limiter(() =>
          this.process(id).catch((e) => {
            logger.error("Batch item failed", { bookmarkId: id, error: e });
          }),
        ),
      ),
    );
  }
}
```

```typescript
    return successResponse(result, 201);
  } catch (error) {
    if (error instanceof QuotaExceededError) {
      return errorResponse("Storage quota exceeded", 403);
    }
    logger.error("Batch create failed", { error });
    return errorResponse("Internal server error", 500);
  }
}
```

```typescript
// GET /api/bookmarks/batch/:jobId/status
export async function GET(
  request: NextRequest,
  { params }: { params: { jobId: string } },
) {
  const { user } = await getCachedUser();
  if (!user) return errorResponse("Unauthorized", 401);

  const supabase = await createClient();

  // 获取批量任务统计
  const { data: stats } = await supabase
    .from("bookmark_jobs")
    .select("status")
    .eq("batch_job_id", params.jobId);

  if (!stats?.length) {
    return errorResponse("Batch job not found", 404);
  }

  const summary = {
    total: stats.length,
    pending: stats.filter((j) => j.status === "pending").length,
    running: stats.filter((j) => j.status === "running").length,
    completed: stats.filter((j) => j.status === "completed").length,
    failed: stats.filter((j) => j.status === "failed").length,
  };

  return successResponse({
    jobId: params.jobId,
    ...summary,
    isComplete: summary.pending === 0 && summary.running === 0,
    progress: Math.round((summary.completed / summary.total) * 100),
  });
}
```
