```typescript
class SummaryConfigService {
  private static instance: SummaryConfigService | null = null;
  private config: SummaryConfig;

  private constructor() {
    this.config = this.loadFromEnv();
  }

  static getInstance(): SummaryConfigService {
    if (!this.instance) {
      this.instance = new SummaryConfigService();
    }
    return this.instance;
  }

  private loadFromEnv(): SummaryConfig {
    const raw = {
      providers: {
        default: process.env.SUMMARY_DEFAULT_PROVIDER,
        fallback: process.env.SUMMARY_FALLBACK_PROVIDER,
        enableLocalFallback: this.envBool(
          "SUMMARY_ENABLE_LOCAL_FALLBACK",
          true,
        ),
        healthCheckInterval: this.envNumber(
          "SUMMARY_HEALTH_CHECK_INTERVAL",
          60000,
        ),
      },
      modes: {
        default: process.env.SUMMARY_DEFAULT_MODE,
        allowUserSwitch: this.envBool("SUMMARY_ALLOW_MODE_SWITCH", true),
        v3EntitiesEnabled: this.envBool("SUMMARY_V3_ENTITIES_ENABLED", false),
      },
      quality: {
        enableTracking: this.envBool("SUMMARY_TRACK_QUALITY", true),
        minKeyPoints: this.envNumber("SUMMARY_MIN_KEY_POINTS", 3),
        maxKeyPoints: this.envNumber("SUMMARY_MAX_KEY_POINTS", 5),
        minTldrLength: this.envNumber("SUMMARY_MIN_TLDR_LENGTH", 20),
        maxTldrLength: this.envNumber("SUMMARY_MAX_TLDR_LENGTH", 200),
      },
      cache: {
        enabled: this.envBool("SUMMARY_CACHE_ENABLED", true),
        storage: process.env.SUMMARY_CACHE_STORAGE,
        ttlMs: this.envNumber("SUMMARY_CACHE_TTL_MS", 24 * 60 * 60 * 1000),
        maxSize: this.envNumber("SUMMARY_CACHE_MAX_SIZE", 10000),
      },
      contentFetching: {
        enabled: this.envBool("SUMMARY_CONTENT_FETCH_ENABLED", true),
        provider: process.env.SUMMARY_CONTENT_PROVIDER,
        timeout: this.envNumber("SUMMARY_FETCH_TIMEOUT", 30000),
        maxContentLength: this.envNumber("SUMMARY_MAX_CONTENT_LENGTH", 100000),
      },
      queue: {
        concurrency: this.envNumber("SUMMARY_QUEUE_CONCURRENCY", 5),
        maxRetries: this.envNumber("SUMMARY_QUEUE_MAX_RETRIES", 3),
        retryDelay: this.envNumber("SUMMARY_QUEUE_RETRY_DELAY", 1000),
      },
    };

    return SummaryConfigSchema.parse(raw);
  }

  get<K extends keyof SummaryConfig>(key: K): SummaryConfig[K] {
    return this.config[key];
  }

  getAll(): Readonly<SummaryConfig> {
    return Object.freeze({ ...this.config });
  }

  // 运行时更新（用于 A/B 测试）
  update(partial: Partial<SummaryConfig>): void {
    this.config = SummaryConfigSchema.parse({ ...this.config, ...partial });
  }

  private envBool(key: string, defaultValue: boolean): boolean {
    const val = process.env[key];
    if (val === undefined) return defaultValue;
    return val.toLowerCase() === "true";
  }

  private envNumber(key: string, defaultValue: number): number {
    const val = process.env[key];
    if (val === undefined) return defaultValue;
    const num = parseInt(val, 10);
    return isNaN(num) ? defaultValue : num;
  }
}
```

#### 2.2 缓存服务实现

```typescript
// src/lib/summary/cache/summary-cache.service.ts
import { createHash } from "crypto";
import { LRUCache } from "lru-cache";
import type { SupabaseClient } from "@supabase/supabase-js";
import type { SummaryResult, SummaryOptions } from "../summary.types";
import { getSummaryConfig } from "../config";

interface CacheEntry {
  fingerprint: string;
  result: SummaryResult;
  createdAt: Date;
  hitCount: number;
}

export class SummaryCacheService {
  private static instance: SummaryCacheService | null = null;

  // L1: 内存缓存
  private memoryCache: LRUCache<string, CacheEntry>;

  private supabase: SupabaseClient;
  private config = getSummaryConfig();

  private constructor(supabase: SupabaseClient) {
    this.supabase = supabase;
    this.memoryCache = new LRUCache({
      max: 1000,
      ttl: 60 * 60 * 1000, // 1小时
    });
  }

  static getInstance(supabase: SupabaseClient): SummaryCacheService {
    if (!this.instance) {
      this.instance = new SummaryCacheService(supabase);
    }
    return this.instance;
  }

  /**
   * 获取或生成摘要（带缓存）
   */
  async getOrGenerate(
    content: string,
    options: SummaryOptions,
    generator: () => Promise<SummaryResult>,
  ): Promise<SummaryResult & { fromCache: boolean }> {
    // 检查缓存是否启用
    if (!this.config.get("cache").enabled) {
      const result = await generator();
      return { ...result, fromCache: false };
    }

    const fingerprint = this.computeFingerprint(content, options);

    // L1: 检查内存缓存
    const memoryHit = this.memoryCache.get(fingerprint);
    if (memoryHit && !this.isExpired(memoryHit)) {
      await this.incrementHitCount(fingerprint);
      return { ...memoryHit.result, fromCache: true };
    }

    // L2: 检查数据库缓存
    const dbHit = await this.getFromDatabase(fingerprint);
    if (dbHit && !this.isExpired(dbHit)) {
      // 回填 L1
      this.memoryCache.set(fingerprint, dbHit);
      await this.incrementHitCount(fingerprint);
      return { ...dbHit.result, fromCache: true };
    }

    // L3: 调用 Provider 生成
    const result = await generator();

    // 存入缓存
    await this.set(fingerprint, result);

    return { ...result, fromCache: false };
  }

  /**
   * 计算内容指纹
   */
  computeFingerprint(content: string, options: SummaryOptions): string {
    const normalized = this.normalizeContent(content);
    const optionsKey = `${options.mode || "standard"}_${options.version || "v2"}`;
    const combined = `${normalized}::${optionsKey}`;
    return createHash("md5").update(combined).digest("hex");
  }

  /**
   * 存入缓存
   */
  async set(fingerprint: string, result: SummaryResult): Promise<void> {
    const entry: CacheEntry = {
      fingerprint,
      result,
      createdAt: new Date(),
      hitCount: 0,
    };

    // L1: 存入内存
    this.memoryCache.set(fingerprint, entry);

    // L2: 存入数据库
    await this.saveToDatabase(entry);
  }

  /**
   * 强制刷新（绕过缓存）
   */
  async invalidate(fingerprint: string): Promise<void> {
    this.memoryCache.delete(fingerprint);
    await this.deleteFromDatabase(fingerprint);
  }

  /**
   * 按 URL 失效缓存
   */
  async invalidateByUrl(url: string): Promise<void> {
    const { data } = await this.supabase
      .from("summary_cache")
      .select("fingerprint")
      .eq("url", url);

    for (const row of data || []) {
      await this.invalidate(row.fingerprint);
    }
  }

  // ===== 私有方法 =====

  private normalizeContent(content: string): string {
    return content.toLowerCase().replace(/\s+/g, " ").trim().slice(0, 10000); // 只用前 10000 字符计算指纹
  }

  private isExpired(entry: CacheEntry): boolean {
    const ttl = this.config.get("cache").ttlMs;
    const age = Date.now() - entry.createdAt.getTime();
    return age > ttl;
  }

  private async getFromDatabase(
    fingerprint: string,
  ): Promise<CacheEntry | null> {
    const { data, error } = await this.supabase
      .from("summary_cache")
      .select("*")
      .eq("fingerprint", fingerprint)
      .single();

    if (error || !data) return null;

    return {
      fingerprint: data.fingerprint,
      result: data.result as SummaryResult,
      createdAt: new Date(data.created_at),
      hitCount: data.hit_count,
    };
  }

  private async saveToDatabase(entry: CacheEntry): Promise<void> {
    await this.supabase.from("summary_cache").upsert({
      fingerprint: entry.fingerprint,
      result: entry.result,
      created_at: entry.createdAt.toISOString(),
      hit_count: entry.hitCount,
    });
  }

  private async deleteFromDatabase(fingerprint: string): Promise<void> {
    await this.supabase
      .from("summary_cache")
      .delete()
      .eq("fingerprint", fingerprint);
  }

  private async incrementHitCount(fingerprint: string): Promise<void> {
    await this.supabase.rpc("increment_summary_cache_hit", { fp: fingerprint });
  }
}
```

```sql
CREATE TABLE IF NOT EXISTS summary_cache (
  fingerprint TEXT PRIMARY KEY,
  url TEXT,
  result JSONB NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  hit_count INTEGER DEFAULT 0,
  last_hit_at TIMESTAMPTZ
);

-- 索引
CREATE INDEX idx_summary_cache_url ON summary_cache(url);
CREATE INDEX idx_summary_cache_created_at ON summary_cache(created_at);

-- 命中计数函数
CREATE OR REPLACE FUNCTION increment_summary_cache_hit(fp TEXT)
RETURNS VOID AS $$
BEGIN
  UPDATE summary_cache
  SET hit_count = hit_count + 1,
      last_hit_at = NOW()
  WHERE fingerprint = fp;
END;
$$ LANGUAGE plpgsql;

-- 自动清理过期缓存（每天运行）
CREATE OR REPLACE FUNCTION cleanup_expired_summary_cache()
RETURNS INTEGER AS $$
DECLARE
  deleted_count INTEGER;
BEGIN
  DELETE FROM summary_cache
  WHERE created_at < NOW() - INTERVAL '7 days'
    AND hit_count < 3;  -- 保留高命中缓存

  GET DIAGNOSTICS deleted_count = ROW_COUNT;
  RETURN deleted_count;
END;
$$ LANGUAGE plpgsql;
```

#### 3.1 增强 BaseProvider

```typescript
// src/lib/summary/providers/base.provider.ts
import type {
  SummaryInput,
  SummaryResult,
  ProviderConfig,
} from "../summary.types";
import { getSummaryConfig } from "../config";

export abstract class BaseProvider {
  protected config: ProviderConfig;
  protected logger: Logger;

  constructor(config: ProviderConfig) {
    this.config = config;
    this.logger = createLogger(`provider:${this.getName()}`);
  }

  // ===== 模板方法模式 =====

  async generateSummary(input: SummaryInput): Promise<SummaryResult> {
    const startTime = Date.now();

    try {
      // 1. 预处理（公共）
      const preparedInput = this.prepareInput(input);

      // 2. 构建 Prompt（子类实现）
      const prompt = this.buildPrompt(preparedInput);

      // 3. 构建请求（子类实现）
      const request = this.buildRequest(prompt);

      // 4. 发送请求（公共）
      const response = await this.sendRequest(request);

      // 5. 解析响应（公共，可覆盖）
      const result = this.parseResponse(response);

      // 6. 后处理（公共）
      return this.postProcess(result, Date.now() - startTime);
    } catch (error) {
      this.logger.error("Summary generation failed", { error, input });
      throw this.wrapError(error);
    }
  }

  // ===== 公共实现 =====

  protected prepareInput(input: SummaryInput): PreparedInput {
    return {
      title: input.title.slice(0, 200),
      content: this.truncateContent(input.content),
      mode: input.mode || "standard",
      contentType: this.detectContentType(input),
    };
  }

  protected async sendRequest(
    request: ProviderRequest,
  ): Promise<ProviderResponse> {
    const response = await fetch(this.getEndpoint(), {
      method: "POST",
      headers: this.getHeaders(),
      body: JSON.stringify(request),
      signal: AbortSignal.timeout(this.config.timeout || 30000),
    });

    if (!response.ok) {
      throw new ProviderError(
        `Provider ${this.getName()} returned ${response.status}`,
        response.status,
      );
    }

    return response.json();
  }

  protected parseResponse(response: ProviderResponse): SummaryResult {
    const content = this.extractContent(response);

    // 尝试解析 JSON
    try {
      const structured = JSON.parse(content);
      return {
        raw: content,
        structured: this.validateStructured(structured),
        provider: this.getName(),
      };
    } catch {
      // 降级为纯文本
      return {
        raw: content,
        structured: null,
        provider: this.getName(),
      };
    }
  }

  protected postProcess(
    result: SummaryResult,
    latencyMs: number,
  ): SummaryResult {
    return {
      ...result,
      latencyMs,
      timestamp: new Date().toISOString(),
    };
  }

  protected truncateContent(content: string): string {
    const maxLength =
      getSummaryConfig().get("quality").maxContentLength || 8000;
    if (content.length <= maxLength) return content;
    return content.slice(0, maxLength) + "\n...[内容已截断]";
  }

  protected detectContentType(input: SummaryInput): string {
    const url = input.url?.toLowerCase() || "";
    if (url.includes("youtube.com") || url.includes("bilibili.com"))
      return "video";
    if (url.includes("github.com")) return "code";
    if (input.title?.includes("教程") || input.title?.includes("Tutorial"))
      return "tutorial";
    return "article";
  }

  // ===== 抽象方法（子类必须实现）=====

  abstract getName(): string;
  abstract getEndpoint(): string;
  abstract getHeaders(): Record<string, string>;
  abstract buildPrompt(input: PreparedInput): string;
  abstract buildRequest(prompt: string): ProviderRequest;
  abstract extractContent(response: ProviderResponse): string;
}
```

#### 4.2 服务接口

```typescript
// src/lib/content-fetching/content-fetcher.service.ts
export interface FetchResult {
  url: string;
  title: string;
  content: string;           // 正文内容
  metadata: PageMetadata;    // Open Graph 等
  media: MediaItem[];        // 图片、视频
  fetchedAt: Date;
  fromCache: boolean;
  provider: string;
}

export class ContentFetcherService {
  private providers: Map<string, IContentProvider>;
  private cache: ContentCacheService;

  async fetch(url: string, options?: FetchOptions): Promise<FetchResult> {
    // 1. 检查缓存
    const cached = await this.cache.get(url);
    if (cached && !options?.forceRefresh) {
      return { ...cached, fromCache: true };
    }

    // 2. 选择 Provider
    const provider = this.selectProvider(url);

    // 3. 抓取内容
    const raw = await provider.fetch(url);

    // 4. 提取正文
    const content = await this.extractContent(raw);

    // 5. 提取元数据
    const metadata = await this.extractMetadata(raw);

    // 6. 存入缓存
    const result = { url, title: metadata.title, content, metadata, ... };
    await this.cache.set(url, result);

    return { ...result, fromCache: false };
  }

  private selectProvider(url: string): IContentProvider {
    // SPA 站点使用 Playwright
    if (this.isSpaUrl(url)) {
      return this.providers.get("playwright");
    }
    // 付费墙站点使用 Jina
    if (this.isPaywallUrl(url)) {
      return this.providers.get("jina");
    }
    // 默认使用简单 fetch
    return this.providers.get("simple");
  }
}
```
