## 执行摘要

### 重构范围

| 维度 | 现状 | 目标 |
|------|------|------|
| **数据库** | `messages` 表（40+ 硬编码类型） | `user_notifications` + `notification_types` |
| **服务层** | `MessageService`（764 行，分散） | `NotificationService`（统一、精简） |
| **UI 组件** | 13 个组件（体验一般） | 重新设计（Notion 风格） |
| **命名** | `messages`（与 AI 消息混淆） | `notifications`（语义清晰） |

### 核心决策

- ✅ **放弃历史数据**：开发环境测试数据，无需迁移
- ✅ **直接替换**：无需双写、灰度，一次性切换
- ✅ **Notion 风格 UI**：时间流分组 + 自动标记已读

### 预期收益

- 代码量减少 **~40%**（从 2000+ 行降至 1200 行）
- 消息类型可配置化（数据库枚举表 vs 硬编码）
- UI/UX 显著提升（参考 Notion/Linear 最佳实践）

```sql
-- 2. 统一通知表
CREATE TABLE user_notifications (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,

  -- 类型引用
  type_code VARCHAR(50) NOT NULL REFERENCES notification_types(code),

  -- 通道与优先级（可覆盖默认值）
  channels TEXT[] NOT NULL,
  priority VARCHAR(20) NOT NULL CHECK (priority IN ('silent', 'normal', 'important', 'critical')),

  -- 内容
  title VARCHAR(200) NOT NULL,
  body TEXT,

  -- 关联实体（多态引用）
  entity_type VARCHAR(30),  -- bookmark | folder | import
  entity_id TEXT,

  -- 操作
  action_url VARCHAR(500),
  action_label VARCHAR(50),

  -- 元数据
  metadata JSONB DEFAULT '{}',

  -- 状态
  status VARCHAR(20) DEFAULT 'unread' CHECK (status IN ('unread', 'read', 'archived')),
  read_at TIMESTAMPTZ,

  -- 时间
  created_at TIMESTAMPTZ DEFAULT NOW(),
  expires_at TIMESTAMPTZ DEFAULT (NOW() + INTERVAL '30 days')
);

-- 3. 索引优化
CREATE INDEX idx_notifications_user_status ON user_notifications(user_id, status);
CREATE INDEX idx_notifications_user_created ON user_notifications(user_id, created_at DESC);
CREATE INDEX idx_notifications_user_priority ON user_notifications(user_id, priority) WHERE status = 'unread';
CREATE INDEX idx_notifications_expires ON user_notifications(expires_at) WHERE status != 'archived';

-- 4. RLS 策略
ALTER TABLE user_notifications ENABLE ROW LEVEL SECURITY;

CREATE POLICY "users_own_notifications" ON user_notifications
  FOR ALL USING (auth.uid() = user_id);

-- 5. 清理函数
CREATE OR REPLACE FUNCTION cleanup_expired_notifications()
RETURNS integer AS $$
DECLARE
  deleted_count integer;
BEGIN
  DELETE FROM user_notifications
  WHERE expires_at < NOW() AND status != 'archived';
  GET DIAGNOSTICS deleted_count = ROW_COUNT;
  RETURN deleted_count;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 6. 未读计数函数
CREATE OR REPLACE FUNCTION get_notification_counts(p_user_id UUID)
RETURNS TABLE(priority VARCHAR, count BIGINT) AS $$
BEGIN
  RETURN QUERY
  SELECT n.priority, COUNT(*)::BIGINT
  FROM user_notifications n
  WHERE n.user_id = p_user_id AND n.status = 'unread'
  GROUP BY n.priority;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 7. 删除旧表
DROP TABLE IF EXISTS messages CASCADE;
DROP TABLE IF EXISTS notification_preferences CASCADE;
```

#### 2.3 核心服务

```typescript
// src/lib/notifications/notification.service.ts

import { createClient } from '@/lib/supabase/server';
import type { NotificationPayload, Notification, NotificationType } from './notification.types';

export const NotificationService = {
  /**
   * 发送单条通知
   */
  async send(payload: NotificationPayload): Promise<Notification> {
    const supabase = await createClient();

    // 1. 获取类型默认配置
    const typeConfig = await this.getTypeConfig(payload.type);

    // 2. 合并默认值
    const channels = payload.channels ?? typeConfig?.defaultChannels ?? ['inbox'];
    const priority = payload.priority ?? typeConfig?.defaultPriority ?? 'normal';

    // 3. 插入数据库
    const { data, error } = await supabase
      .from('user_notifications')
      .insert({
        user_id: payload.userId,
        type_code: payload.type,
        title: payload.title,
        body: payload.body,
        channels,
        priority,
        entity_type: payload.entityType,
        entity_id: payload.entityId,
        action_url: payload.actionUrl,
        action_label: payload.actionLabel,
        metadata: payload.metadata ?? {},
      })
      .select()
      .single();

    if (error) throw error;

    // 4. 触发通道处理（Toast/Modal）
    await this.dispatchToChannels(data, channels);

    return this.mapToNotification(data);
  },

  /**
   * 批量发送通知
   */
  async sendBatch(payloads: NotificationPayload[]): Promise<Notification[]> {
    const supabase = await createClient();

    const rows = await Promise.all(
      payloads.map(async (p) => {
        const typeConfig = await this.getTypeConfig(p.type);
        return {
          user_id: p.userId,
          type_code: p.type,
          title: p.title,
          body: p.body,
          channels: p.channels ?? typeConfig?.defaultChannels ?? ['inbox'],
          priority: p.priority ?? typeConfig?.defaultPriority ?? 'normal',
          entity_type: p.entityType,
          entity_id: p.entityId,
          action_url: p.actionUrl,
          action_label: p.actionLabel,
          metadata: p.metadata ?? {},
        };
      })
    );

    const { data, error } = await supabase
      .from('user_notifications')
      .insert(rows)
      .select();

    if (error) throw error;
    return data.map(this.mapToNotification);
  },

  /**
   * 标记已读
   */
  async markAsRead(userId: string, ids: string[]): Promise<void> {
    const supabase = await createClient();
    await supabase
      .from('user_notifications')
      .update({ status: 'read', read_at: new Date().toISOString() })
      .eq('user_id', userId)
      .in('id', ids);
  },

  /**
   * 全部标记已读
   */
  async markAllAsRead(userId: string): Promise<void> {
    const supabase = await createClient();
    await supabase
      .from('user_notifications')
      .update({ status: 'read', read_at: new Date().toISOString() })
      .eq('user_id', userId)
      .eq('status', 'unread');
  },

  /**
   * 归档通知
   */
  async archive(userId: string, id: string): Promise<void> {
    const supabase = await createClient();
    await supabase
      .from('user_notifications')
      .update({ status: 'archived' })
      .eq('user_id', userId)
      .eq('id', id);
  },

  /**
   * 删除通知
   */
  async delete(userId: string, id: string): Promise<void> {
    const supabase = await createClient();
    await supabase
      .from('user_notifications')
      .delete()
      .eq('user_id', userId)
      .eq('id', id);
  },

  /**
   * 获取未读计数
   */
  async getUnreadCounts(userId: string): Promise<Record<string, number>> {
    const supabase = await createClient();
    const { data, error } = await supabase
      .rpc('get_notification_counts', { p_user_id: userId });

    if (error) throw error;

    return data.reduce((acc: Record<string, number>, row: { priority: string; count: number }) => {
      acc[row.priority] = row.count;
      return acc;
    }, { total: 0, silent: 0, normal: 0, important: 0, critical: 0 });
  },

  /**
   * 获取类型配置
   */
  async getTypeConfig(code: string): Promise<NotificationType | null> {
    const supabase = await createClient();
    const { data } = await supabase
      .from('notification_types')
      .select('*')
      .eq('code', code)
      .single();

    return data ? {
      code: data.code,
      category: data.category,
      label: data.label,
      icon: data.icon,
      color: data.color,
      defaultChannels: data.default_channels,
      defaultPriority: data.default_priority,
    } : null;
  },

  /**
   * 分发到各通道
   */
  async dispatchToChannels(notification: any, channels: string[]): Promise<void> {
    // Toast 和 Modal 通道在客户端处理
    // 这里可以添加 Web Push 等服务端推送
  },

  /**
   * 数据映射
   */
  mapToNotification(row: any): Notification {
    return {
      id: row.id,
      userId: row.user_id,
      typeCode: row.type_code,
      channels: row.channels,
      priority: row.priority,
      title: row.title,
      body: row.body,
      entityType: row.entity_type,
      entityId: row.entity_id,
      actionUrl: row.action_url,
      actionLabel: row.action_label,
      metadata: row.metadata,
      status: row.status,
      readAt: row.read_at,
      createdAt: row.created_at,
      expiresAt: row.expires_at,
    };
  },
};
```

| 原则 | 说明 |
|------|------|
| **自动标记** | 点击/查看后自动标记已读，降低用户负担 |
| **时间流** | 按时间分组（今天/昨天/本周/更早） |
| **轻量交互** | 快捷操作 hover 显示，减少视觉噪音 |
| **视觉层次** | 优先级通过左边框颜色区分 |

#### 4.1 优先级颜色

| 优先级 | 指示条颜色 | 图标颜色 | 使用场景 |
|--------|-----------|----------|----------|
| `silent` | 透明 | `text-muted-foreground` | 日常操作完成 |

| `important` | `bg-orange-500` | `text-orange-600` | 需要关注 |

```typescript
// src/hooks/data/use-notifications.ts

export function useNotifications(options: {
  status?: 'unread' | 'read' | 'archived' | 'all';
  priority?: NotificationPriority[];
  limit?: number;
} = {}) {
  const supabase = createClient();
  const queryClient = useQueryClient();

  // 分组查询
  const query = useInfiniteQuery({
    queryKey: ['notifications', options],
    queryFn: async ({ pageParam = 0 }) => {
      let q = supabase
        .from('user_notifications')
        .select('*, notification_types(*)')
        .order('created_at', { ascending: false })
        .range(pageParam, pageParam + (options.limit ?? 20) - 1);

      if (options.status && options.status !== 'all') {
        q = q.eq('status', options.status);
      }
      if (options.priority?.length) {
        q = q.in('priority', options.priority);
      }

      const { data, error } = await q;
      if (error) throw error;
      return data;
    },
    getNextPageParam: (lastPage, pages) => {
      if (lastPage.length < (options.limit ?? 20)) return undefined;
      return pages.flat().length;
    },
  });

  // 分组处理
  const groups = useMemo(() => {
    const all = query.data?.pages.flat() ?? [];
    return groupByDate(all);
  }, [query.data]);

  // 操作方法
  const markAsRead = useMutation({
    mutationFn: (id: string) => NotificationService.markAsRead(userId, [id]),
    onSuccess: () => queryClient.invalidateQueries({ queryKey: ['notifications'] }),
  });

  const markAllAsRead = useMutation({
    mutationFn: () => NotificationService.markAllAsRead(userId),
    onSuccess: () => queryClient.invalidateQueries({ queryKey: ['notifications'] }),
  });

  const deleteNotification = useMutation({
    mutationFn: (id: string) => NotificationService.delete(userId, id),
    onSuccess: () => queryClient.invalidateQueries({ queryKey: ['notifications'] }),
  });

  return {
    groups,
    isLoading: query.isLoading,
    isFetchingNextPage: query.isFetchingNextPage,
    hasNextPage: query.hasNextPage,
    fetchNextPage: query.fetchNextPage,
    markAsRead: markAsRead.mutate,
    markAllAsRead: markAllAsRead.mutate,
    deleteNotification: deleteNotification.mutate,
    refresh: () => queryClient.invalidateQueries({ queryKey: ['notifications'] }),
  };
}

// 日期分组工具函数
function groupByDate(notifications: Notification[]): NotificationGroup[] {
  const today = startOfDay(new Date());
  const yesterday = subDays(today, 1);
  const weekAgo = subDays(today, 7);

  const groups: Record<string, Notification[]> = {
    today: [],
    yesterday: [],
    this_week: [],
    earlier: [],
  };

  for (const n of notifications) {
    const date = new Date(n.createdAt);
    if (date >= today) {
      groups.today.push(n);
    } else if (date >= yesterday) {
      groups.yesterday.push(n);
    } else if (date >= weekAgo) {
      groups.this_week.push(n);
    } else {
      groups.earlier.push(n);
    }
  }

  return [
    { dateGroup: 'today', label: '今天', notifications: groups.today },
    { dateGroup: 'yesterday', label: '昨天', notifications: groups.yesterday },
    { dateGroup: 'this_week', label: '本周', notifications: groups.this_week },
    { dateGroup: 'earlier', label: '更早', notifications: groups.earlier },
  ].filter(g => g.notifications.length > 0);
}
```
