**变更前后对比**:

```typescript
// ❌ Phase 3 (灰度版本 - 有回退逻辑)
if (scheduled.workflowId === "daily_digest") {
  try {
    const liteResult = await runLiteTask(...);
    if (liteResult.success) {
      return { success: true, ... };
    } else {
      logger.error("falling back");
    }
  } catch (error) {
    logger.error("falling back");
  }
}
// 降级到旧引擎
const execution = await this.orchestrator.startWorkflow(...);
```

```typescript
// ✅ Phase 4 (最终版本 - 无回退)
if (scheduled.workflowId === "daily_digest") {
  const liteResult = await runLiteTask(...);
  if (!liteResult.success) {
    throw liteResult.error;
  }
  return { success: true, executionId: `lite-${scheduled.id}` };
}
// 其他未迁移的工作流抛出错误
throw new Error(`Workflow '${workflowId}' not migrated yet`);
```

### 执行流程:

```
Cron 触发
    ↓
TriggerScheduler.executeScheduledWorkflow()
    ↓

    ↓
调用 runLiteTask("daily-digest", ...)
    ↓
执行 executeDailyDigest(ctx)
    ↓
    1. Fetch Unread Bookmarks
    2. Score & Rank (简化算法)
    3. Generate Digest (复用 DB ai_summary)
    4. Mock Notification
    ↓
更新 DB 状态为 completed
    ↓
返回成功结果
```
