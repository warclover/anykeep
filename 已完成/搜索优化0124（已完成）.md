```typescript
   * 判断是否能处理该消息
   */
  canHandle(message: string, context: ClassificationContext): boolean;

  /**
   * 执行分类
   */
  classify(
    message: string,
    context: ClassificationContext,
  ): Promise<ClassificationResult>;

  /**
   * 可选：置信度校准
   */
  calibrate?(result: ClassificationResult): ClassificationResult;
}
```

#### 1.4 分类器链实现

```typescript
// src/lib/ai-routing/intent/classifier-chain.ts

import type {
  IntentClassifier,
  ClassificationResult,
  ClassificationContext,
} from "./classifier-interface";
import { CONFIDENCE_THRESHOLDS } from "@/lib/shared/types";

export class ClassifierChain {
  private classifiers: IntentClassifier[] = [];

  register(classifier: IntentClassifier): void {
    this.classifiers.push(classifier);
    this.classifiers.sort((a, b) => a.priority - b.priority);
  }

  async classify(
    message: string,
    context: ClassificationContext,
  ): Promise<ClassificationResult> {
    for (const classifier of this.classifiers) {
      if (!classifier.canHandle(message, context)) {
        continue;
      }

      try {
        const result = await classifier.classify(message, context);

        if (result.confidence >= CONFIDENCE_THRESHOLDS.LOW) {
          const calibrated = classifier.calibrate?.(result) ?? result;
          return {
            ...calibrated,
            reasoning: `[${classifier.name}] ${calibrated.reasoning ?? ""}`,
          };
        }
      } catch (error) {
        logger.warn(`Classifier ${classifier.name} failed`, { error });
        // 继续尝试下一个分类器
      }
    }

    return this.fallbackResult(message);
  }

  private fallbackResult(message: string): ClassificationResult {
    return {
      intent: "query",
      subType: "general",
      confidence: 0.3,
      entities: {},
      reasoning: "Fallback: no classifier matched",
    };
  }
}
```

#### 1.5 统一入口实现

```typescript
// src/lib/ai-routing/entry/unified-entry.ts

import { ClassifierChain } from "../intent/classifier-chain";
import { ModeSelector, ExecutionMode } from "../routing/mode-selector";
import { InputPreprocessor } from "../preprocessing/input-preprocessor";
import { IntentCache } from "../cache/intent-cache";
import { SemanticCache } from "../cache/semantic-cache";

export interface UnifiedRequest {
  message: string;
  userId: string;
  sessionId?: string;
  context?: RequestContext;
}

export interface RouteResult {
  mode: ExecutionMode;
  intent: ClassificationResult;
  preprocessedMessage: string;
  cacheHit: "L0" | "L1" | "L2" | "none";
  latencyMs: number;
}

export class UnifiedEntry {
  private preprocessor: InputPreprocessor;
  private classifierChain: ClassifierChain;
  private modeSelector: ModeSelector;
  private intentCache: IntentCache;
  private semanticCache: SemanticCache;

  constructor(config?: UnifiedEntryConfig) {
    this.preprocessor = new InputPreprocessor();
    this.classifierChain = this.buildClassifierChain();
    this.modeSelector = new ModeSelector();
    this.intentCache = new IntentCache();
    this.semanticCache = new SemanticCache();
  }

  async route(request: UnifiedRequest): Promise<RouteResult> {
    const startTime = Date.now();

    // 1. 预处理
    const preprocessed = await this.preprocessor.process(request);

    // 2. 多级缓存查询
    const cacheResult = await this.checkCaches(
      preprocessed.message,
      request.userId,
    );
    if (cacheResult) {
      return {
        mode: this.modeSelector.select(cacheResult.intent),
        intent: cacheResult.intent,
        preprocessedMessage: preprocessed.message,
        cacheHit: cacheResult.level,
        latencyMs: Date.now() - startTime,
      };
    }

    // 3. 分类器链
    const intent = await this.classifierChain.classify(preprocessed.message, {
      userId: request.userId,
      sessionId: request.sessionId,
      selectedBookmarks: preprocessed.selectedBookmarks,
      recentMessages: preprocessed.recentMessages,
    });

    // 4. 缓存写入
    await this.updateCaches(preprocessed.message, intent);

    // 5. 模式选择
    const mode = this.modeSelector.select(intent);

    return {
      mode,
      intent,
      preprocessedMessage: preprocessed.message,
      cacheHit: "none",
      latencyMs: Date.now() - startTime,
    };
  }

  private buildClassifierChain(): ClassifierChain {
    const chain = new ClassifierChain();

    // 优先级顺序：规则 → 语义 → LLM
    chain.register(new RuleBasedClassifier()); // priority: 10
    chain.register(new SemanticIntentClassifier()); // priority: 50
    chain.register(new LLMClassifier()); // priority: 100

    return chain;
  }

  private async checkCaches(
    message: string,
    userId: string,
  ): Promise<{ intent: ClassificationResult; level: "L1" | "L2" } | null> {
    // L1: 精确匹配
    const l1Result = this.intentCache.get(message, userId);
    if (l1Result) {
      return { intent: l1Result, level: "L1" };
    }

    // L2: 语义匹配
    const l2Result = await this.semanticCache.find(message, userId);
    if (l2Result) {
      // 回填 L1
      this.intentCache.set(message, userId, l2Result);
      return { intent: l2Result, level: "L2" };
    }

    return null;
  }

  private async updateCaches(
    message: string,
    intent: ClassificationResult,
  ): Promise<void> {
    // 同时写入 L1 和 L2
    this.intentCache.set(message, intent.userId, intent);
    await this.semanticCache.add(message, intent);
  }
}

// 单例
let entryInstance: UnifiedEntry | null = null;

export function getUnifiedEntry(): UnifiedEntry {
  if (!entryInstance) {
    entryInstance = new UnifiedEntry();
  }
  return entryInstance;
}
```

#### 1.6 Debug Trace 基础设施（🆕 CTO 硬性要求）

```typescript
// src/lib/ai-routing/debug/debug-trace.ts

export interface DebugTrace {
  requestId: string;
  timestamp: number;
  stages: TraceStage[];
  totalLatencyMs: number;
}

export interface TraceStage {
  stage: 'preprocessing' | 'intent' | 'routing' | 'execution' | 'enhancement';
  latencyMs: number;
  input: unknown;
  output: unknown;
  decision?: string;  // "选择 skill: knowledge_search, 置信度: 0.87"
  error?: string;
}

// 通过环境变量控制
const ENABLE_DEBUG_TRACE = process.env.AI_DEBUG_TRACE === 'true';

export class DebugTracer {
  private trace: DebugTrace;

  constructor(requestId: string) {
    this.trace = {
      requestId,
      timestamp: Date.now(),
      stages: [],
      totalLatencyMs: 0,
    };
  }

  addStage(stage: TraceStage): void {
    if (!ENABLE_DEBUG_TRACE) return;
    this.trace.stages.push(stage);
    this.trace.totalLatencyMs += stage.latencyMs;
  }

  getTrace(): DebugTrace {
    return this.trace;
  }

  // 快速诊断：定位问题出在哪个阶段
  diagnose(): string {
    const failedStage = this.trace.stages.find(s => s.error);
    if (failedStage) {
      return `❌ 失败阶段: ${failedStage.stage}, 原因: ${failedStage.error}`;
    }
    const slowestStage = this.trace.stages.reduce((a, b) =>
      a.latencyMs > b.latencyMs ? a : b
    );
    return `✅ 最慢阶段: ${slowestStage.stage} (${slowestStage.latencyMs}ms)`;
  }
}
```

```typescript
function getModuleName(filePath: string): string | null {
  const match = filePath.match(/src\/lib\/([^/]+)/);
  return match ? match[1] : null;

function isValidDependency(
  sourceModule: string,
  targetModule: string,
): boolean {
  if (sourceModule === targetModule) return true;

  const sourceRule = LAYER_RULES[sourceModule];
  if (!sourceRule) return true; // 未知模块，放行

  return sourceRule.canDependOn.includes(targetModule);
}

async function checkDependencies() {
  console.log("🔍 Analyzing dependencies...\n");

  const result = await madge("src/lib/", {
    fileExtensions: ["ts", "tsx"],
    excludeRegExp: [/__tests__/, /\.test\.ts/, /\.spec\.ts/],
    tsConfig: "tsconfig.json",
  });

  // 1. 检查循环依赖
  const circular = result.circular();
  if (circular.length > 0) {
    console.error("❌ Circular dependencies found:");
    circular.forEach((c) => {
      console.error(`   ${c.join(" → ")}`);
    });
    console.error("");
  } else {
    console.log("✅ No circular dependencies\n");
  }

  // 2. 检查层级违规
  const dependencies = result.obj();
  const violations: string[] = [];

  for (const [file, deps] of Object.entries(dependencies)) {
    const sourceModule = getModuleName(file);
    if (!sourceModule) continue;

    for (const dep of deps) {
      const targetModule = getModuleName(dep);
      if (!targetModule) continue;

      if (!isValidDependency(sourceModule, targetModule)) {
        violations.push(
          `${sourceModule}/${path.basename(file)} → ${targetModule}/${path.basename(dep)}`,
        );
      }
    }
  }

  if (violations.length > 0) {
    console.error("❌ Layer violations found:");
    violations.forEach((v) => console.error(`   ${v}`));
    console.error("");
  } else {
    console.log("✅ No layer violations\n");
  }

  // 3. 生成报告
  const stats = {
    totalFiles: Object.keys(dependencies).length,
    circularCount: circular.length,
    violationCount: violations.length,
  };

  console.log("📊 Summary:");
  console.log(`   Total files analyzed: ${stats.totalFiles}`);
  console.log(`   Circular dependencies: ${stats.circularCount}`);
  console.log(`   Layer violations: ${stats.violationCount}`);

  if (circular.length > 0 || violations.length > 0) {
    process.exit(1);
  }

  console.log("\n✅ Dependency check passed");
}
```

```typescript
// src/lib/ai-routing/preprocessing/hybrid-search-builder.ts

export interface HybridSearchConfig {
  vectorWeight: number;    // 默认 0.6
  keywordWeight: number;   // 默认 0.4
  useRRF: boolean;         // 使用 Reciprocal Rank Fusion
  rrf_k: number;           // RRF 常数，默认 60
}

export type QueryType = 'precise' | 'semantic' | 'mixed';

/**
 * 检测查询类型，动态调整权重
 */
export function detectQueryType(query: string): QueryType {
  // 精确查询特征：日期、数字、专有名词、引号包裹
  const precisePatterns = [
    /\d{4}[-年]/,                    // 年份
    /["「『"'].+["」』"']/,           // 引号包裹
    /^@\w+/,                         // @提及
    /\b[A-Z]{2,}\b/,                 // 大写缩写
  ];

  // 语义查询特征：模糊描述词
  const semanticPatterns = [
    /关于|有关|类似|像.*一样/,
    /最近|上周|上个月|去年/,
    /那个|那篇|之前看过/,
  ];

  const hasPrecise = precisePatterns.some(p => p.test(query));
  const hasSemantic = semanticPatterns.some(p => p.test(query));

  if (hasPrecise && !hasSemantic) return 'precise';
  if (hasSemantic && !hasPrecise) return 'semantic';
  return 'mixed';
}

/**
 * 根据查询类型获取权重配置
 */
export function getSearchWeights(queryType: QueryType): HybridSearchConfig {
  switch (queryType) {
    case 'precise':
      return { vectorWeight: 0.3, keywordWeight: 0.7, useRRF: true, rrf_k: 60 };
    case 'semantic':
      return { vectorWeight: 0.8, keywordWeight: 0.2, useRRF: true, rrf_k: 60 };
    case 'mixed':
    default:
      return { vectorWeight: 0.6, keywordWeight: 0.4, useRRF: true, rrf_k: 60 };
  }
}

/**
 * RRF 融合排名算法
 */
export function rrfFusion(
  vectorResults: SearchResult[],
  keywordResults: SearchResult[],
  config: HybridSearchConfig
): SearchResult[] {
  const scoreMap = new Map<number, number>();
  const resultMap = new Map<number, SearchResult>();

  // Vector 结果 RRF 得分
  vectorResults.forEach((result, rank) => {
    const rrf = config.vectorWeight / (config.rrf_k + rank + 1);
    scoreMap.set(result.id, (scoreMap.get(result.id) || 0) + rrf);
    resultMap.set(result.id, result);
  });

  // Keyword 结果 RRF 得分
  keywordResults.forEach((result, rank) => {
    const rrf = config.keywordWeight / (config.rrf_k + rank + 1);
    scoreMap.set(result.id, (scoreMap.get(result.id) || 0) + rrf);
    resultMap.set(result.id, result);
  });

  // 按融合得分排序
  return Array.from(scoreMap.entries())
    .sort((a, b) => b[1] - a[1])
    .map(([id, score]) => ({
      ...resultMap.get(id)!,
      fusedScore: score,
    }));
}
```

#### 2.8 搜索增强：时间衰减重排序 (🆕 CTO 建议采纳)

**用户痛点**：搜出的结果往往太陈旧，三年前的 React 教程排在第一位。

**实施方案**：

```typescript
// src/lib/ai-routing/preprocessing/time-decay-reranker.ts

export interface TimeDecayConfig {
  decayRate: number;      // 每天衰减率，默认 0.01
  baseWeight: number;     // 保底权重，默认 0.5
  maxAgeDays: number;     // 最大衰减天数，默认 365
  contentTypeWeights: Record<string, number>;  // 内容类型权重
}

const DEFAULT_CONFIG: TimeDecayConfig = {
  decayRate: 0.01,
  baseWeight: 0.5,
  maxAgeDays: 365,
  contentTypeWeights: {
    'tech': 1.2,      // 技术类内容，时效性更重要
    'news': 1.5,      // 新闻类��时效性最重要
    'reference': 0.5, // 参考资料，时效性不太重要
    'default': 1.0,
  },
};

/**
 * 计算时间衰减因子
 */
export function calculateTimeDecay(
  createdAt: Date,
  config: TimeDecayConfig = DEFAULT_CONFIG
): number {
  const now = new Date();
  const ageInDays = Math.floor(
    (now.getTime() - createdAt.getTime()) / (1000 * 60 * 60 * 24)
  );

  // 超过最大天数不再额外衰减
  const effectiveAge = Math.min(ageInDays, config.maxAgeDays);

  // 指数衰减：e^(-rate * age)
  const decayFactor = Math.exp(-config.decayRate * effectiveAge);

  // 保底权重 + 衰减部分
  return config.baseWeight + (1 - config.baseWeight) * decayFactor;
}

/**
 * 应用时间衰减到搜索结果
 */
export function applyTimeDecay(
  results: SearchResult[],
  config: TimeDecayConfig = DEFAULT_CONFIG
): SearchResult[] {
  return results
    .map(result => {
      const contentWeight = config.contentTypeWeights[result.contentType]
        || config.contentTypeWeights['default'];
      const timeDecay = calculateTimeDecay(result.createdAt, config);

      return {
        ...result,
        originalScore: result.score,
        score: result.score * timeDecay * contentWeight,
        timeDecayApplied: timeDecay,
      };
    })
    .sort((a, b) => b.score - a.score);
}
```

```typescript
  const supabase = createClient(/* ... */);

  if (topMatch.similarity > 0.6 && topMatch.term !== query) {

      corrected: topMatch.term,
      confidence: topMatch.similarity,
      suggestions: similarTerms.map((s: any) => s.term),
    };
  }

    original: query,
    corrected: null,
    confidence: 0,
    suggestions: similarTerms.map((s: any) => s.term),
```

```sql
-- 需要先启用 pg_trgm 扩展
CREATE EXTENSION IF NOT EXISTS pg_trgm;

CREATE OR REPLACE FUNCTION find_similar_terms(
  p_query TEXT,
  p_user_id UUID,
  p_threshold FLOAT DEFAULT 0.3,
  p_limit INT DEFAULT 5
)
RETURNS TABLE (term TEXT, similarity FLOAT) AS $$
BEGIN
  RETURN QUERY
  SELECT DISTINCT ON (t.term)
    t.term,
    similarity(t.term, p_query) as sim
  FROM (
    -- 从书签标题提取词
    SELECT regexp_split_to_table(lower(title), '\s+') as term
    FROM bookmarks
    WHERE user_id = p_user_id
    UNION
    -- 从标签提取
    SELECT lower(name) as term
    FROM tags
    WHERE user_id = p_user_id
    UNION
    -- 从搜索历史提取
    SELECT lower(query) as term
    FROM search_history
    WHERE user_id = p_user_id
  ) t
  WHERE similarity(t.term, p_query) > p_threshold
    AND t.term != lower(p_query)
  ORDER BY similarity(t.term, p_query) DESC
  LIMIT p_limit;
END;
$$ LANGUAGE plpgsql;
```

```typescript
// src/lib/ai-agent/tools/tool-registry.ts

import { bookmarkTools, bookmarkToolsMetadata } from "./bookmark-tools";
import { folderTools, folderToolsMetadata } from "./folder-tools";
import { searchTools, searchToolsMetadata } from "./search-tools";
// ... 导入所有分组

interface ToolGroup {
  groupId: string;
  groupName: string;
  description: string;
  tools: Record<string, ToolDefinition>;
  metadata: ToolGroupMetadata;
}

interface ToolDiscovery {
  listAllTools(): ToolSummary[];
  findToolsByIntent(intent: string): ToolDefinition[];
  getToolDependencyGraph(): DependencyGraph;
}

class ToolRegistry implements ToolDiscovery {
  private groups: Map<string, ToolGroup> = new Map();
  private allTools: Map<string, ToolDefinition> = new Map();
  private intentToolMap: Map<string, string[]> = new Map();

  constructor() {
    this.registerGroup("bookmark", bookmarkTools, bookmarkToolsMetadata);
    this.registerGroup("folder", folderTools, folderToolsMetadata);
    this.registerGroup("search", searchTools, searchToolsMetadata);
    // ... 注册所有分组

    this.buildIntentToolMap();
  }

  private registerGroup(
    groupId: string,
    tools: Record<string, ToolDefinition>,
    metadata: ToolGroupMetadata,
  ): void {
    this.groups.set(groupId, {
      groupId,
      groupName: metadata.groupName,
      description: metadata.description,
      tools,
      metadata,
    });

    for (const [, tool] of Object.entries(tools)) {
      if (this.allTools.has(tool.name)) {
        throw new Error(`Duplicate tool name: ${tool.name}`);
      }
      this.allTools.set(tool.name, tool);
    }
  }

  private buildIntentToolMap(): void {
    // 意图到工具的映射
    this.intentToolMap.set("search", [
      "search",
      "list_by_platform",
      "list_recent",
    ]);
    this.intentToolMap.set("manage", [
      "create_bookmark",
      "delete_bookmark",
      "update_bookmark",
    ]);
    this.intentToolMap.set("organize", [
      "create_folder",
      "move_bookmarks",
      "batch_tags",
    ]);
    // ...
  }

  // ToolDiscovery 接口实现
  listAllTools(): ToolSummary[] {
    return Array.from(this.allTools.values()).map((t) => ({
      name: t.name,
      description: t.description,
      category: t.category,
      riskLevel: t.riskLevel,
    }));
  }

  findToolsByIntent(intent: string): ToolDefinition[] {
    const toolNames = this.intentToolMap.get(intent) ?? [];
    return toolNames
      .map((name) => this.allTools.get(name))
      .filter((t): t is ToolDefinition => t !== undefined);
  }

  getTool(name: string): ToolDefinition | undefined {
    return this.allTools.get(name);
  }

  getAllTools(): ToolDefinition[] {
    return Array.from(this.allTools.values());
  }

  getToolsByGroup(groupId: string): ToolDefinition[] {
    const group = this.groups.get(groupId);
    return group ? Object.values(group.tools) : [];
  }

  getGroups(): ToolGroup[] {
    return Array.from(this.groups.values());
  }

  getMetadata(): ToolRegistryMetadata {
    return {
      totalTools: this.allTools.size,
      groups: Array.from(this.groups.values()).map((g) => ({
        groupId: g.groupId,
        groupName: g.groupName,
        toolCount: Object.keys(g.tools).length,
      })),
    };
  }
}
```

```typescript
// src/lib/ai-skills/engine/unified-engine.ts (瘦身后 ~300 行)

import { getUnifiedEntry, RouteResult } from "@/lib/ai-routing";
import { AtomicExecutor } from "./execution/atomic-executor";
import { OrchestrationExecutor } from "./execution/orchestration-executor";
import { WorkflowExecutor } from "./execution/workflow-executor";
import { RequestLifecycle } from "./lifecycle/request-lifecycle";
import { OutputEnhancer } from "./enhancement/output-enhancer";

export class UnifiedEngine {
  private entry: UnifiedEntry;
  private atomicExecutor: AtomicExecutor;
  private orchestrationExecutor: OrchestrationExecutor;
  private workflowExecutor: WorkflowExecutor;
  private lifecycle: RequestLifecycle;
  private outputEnhancer: OutputEnhancer;

  constructor(config?: UnifiedEngineConfig) {
    this.entry = getUnifiedEntry();
    this.atomicExecutor = new AtomicExecutor(config);
    this.orchestrationExecutor = new OrchestrationExecutor(config);
    this.workflowExecutor = new WorkflowExecutor(config);
    this.lifecycle = new RequestLifecycle();
    this.outputEnhancer = new OutputEnhancer();
  }

  async *execute(request: UnifiedRequest): AsyncGenerator<UnifiedStreamChunk> {
    const requestId = this.lifecycle.start(request);

    try {
      // 1. 路由
      const routeResult = await this.entry.route(request);
      yield { type: "routing", data: { mode: routeResult.mode } };

      // 2. 执行
      const executor = this.getExecutor(routeResult.mode);
      const resultStream = executor.execute(request, routeResult);

      for await (const chunk of resultStream) {
        yield chunk;
      }

      // 3. 增强输出
      const enhancedChunks = await this.outputEnhancer.enhance(
        request,
        routeResult,
      );
      for (const chunk of enhancedChunks) {
        yield chunk;
      }

      // 4. 完成
      yield { type: "done", data: {} };
      this.lifecycle.complete(requestId);
    } catch (error) {
      this.lifecycle.fail(requestId, error);
      yield { type: "error", data: { message: error.message } };
    }
  }

  private getExecutor(mode: ExecutionMode): Executor {
    switch (mode) {
      case "atomic":
        return this.atomicExecutor;
      case "orchestration":
        return this.orchestrationExecutor;
      case "workflow":
        return this.workflowExecutor;
      default:
        throw new Error(`Unknown execution mode: ${mode}`);
    }
  }
}

// 单例工厂
let engineInstance: UnifiedEngine | null = null;

export function getUnifiedEngine(config?: UnifiedEngineConfig): UnifiedEngine {
  if (!engineInstance) {
    engineInstance = new UnifiedEngine(config);
  }
  return engineInstance;
}
```

```typescript
 */
export async function enhanceSearchResultsWithSnippets(
  query: string,
  results: SearchResult[],
  config: SnippetConfig = DEFAULT_CONFIG
): Promise<EnhancedSearchResult[]> {
  // 并行处理，限制并发数
  const CONCURRENCY = 5;
  const enhanced: EnhancedSearchResult[] = [];

  for (let i = 0; i < results.length; i += CONCURRENCY) {
    const batch = results.slice(i, i + CONCURRENCY);
    const batchResults = await Promise.all(
      batch.map(async (result) => {
        const matchReason = await generateMatchReason(
          query,
          result.content || result.summary || '',
          config
        );
        const excerpt = extractRelevantExcerpt(
          result.content || '',
          query,
          config.maxExcerptLength
        );

        return {
          ...result,
          snippet: {
            matchReason,
            relevantExcerpt: excerpt,
            highlightPositions: findHighlightPositions(excerpt, query),
          },
        };
      })
    );
    enhanced.push(...batchResults);
  }

  return enhanced;
}

/**
 * 提取相关段落
 */
function extractRelevantParagraph(content: string, query: string): string {
  const paragraphs = content.split(/\n\n+/);
  const queryTerms = query.toLowerCase().split(/\s+/);

  let bestParagraph = paragraphs[0] || '';
  let bestScore = 0;

  for (const para of paragraphs) {
    const paraLower = para.toLowerCase();
    const score = queryTerms.filter(term => paraLower.includes(term)).length;
    if (score > bestScore) {
      bestScore = score;
      bestParagraph = para;
    }
  }

  return bestParagraph.slice(0, 500);
}
```

**成本控制**：
- 使用 Haiku 模型（成本约为 Sonnet 的 1/10）
- 限制输出 token 数（≤50）
- 启用结果缓存（避免重复生成）

#### 5.2 Skill 注册器

```typescript
// src/lib/ai-skills/core/skill-registry.ts

import type { SkillMetadata } from "./skill-metadata";
import type { BaseSkill } from "../orchestration/base-skill";
import { FeatureFlags } from "@/lib/ai-agent/config";

class SkillRegistry {
  private skills: Map<
    string,
    {
      metadata: SkillMetadata;
      implementation: new () => BaseSkill;
    }
  > = new Map();

  register(metadata: SkillMetadata, implementation: new () => BaseSkill): void {
    if (this.skills.has(metadata.id)) {
      throw new Error(`Skill already registered: ${metadata.id}`);
    }
    this.skills.set(metadata.id, { metadata, implementation });
  }

  getSkill(id: string): BaseSkill | null {
    const entry = this.skills.get(id);
    if (!entry) return null;

    // 检查 Feature Flag
    if (entry.metadata.featureFlag) {
      const flagEnabled =
        FeatureFlags[entry.metadata.featureFlag as keyof typeof FeatureFlags];
      if (!flagEnabled) return null;
    }

    return new entry.implementation();
  }

  getEnabledSkills(): SkillMetadata[] {
    return Array.from(this.skills.values())
      .filter((entry) => {
        if (!entry.metadata.featureFlag) return true;
        return FeatureFlags[
          entry.metadata.featureFlag as keyof typeof FeatureFlags
        ];
      })
      .map((entry) => entry.metadata);
  }

  getSkillsByCategory(category: SkillCategory): SkillMetadata[] {
    return this.getEnabledSkills().filter((s) => s.category === category);
  }

  findSkillByTrigger(message: string): SkillMetadata | null {
    for (const { metadata } of this.skills.values()) {
      for (const trigger of metadata.triggers) {
        if (this.matchTrigger(message, trigger)) {
          return metadata;
        }
      }
    }
    return null;
  }

  private matchTrigger(message: string, trigger: SkillTrigger): boolean {
    switch (trigger.type) {
      case "keyword":
        return message.toLowerCase().includes(trigger.pattern.toLowerCase());
      case "regex":
        return new RegExp(trigger.pattern, "i").test(message);
      case "semantic":
        // 语义匹配需要调用 embedding
        return false; // 延迟实现
      default:
        return false;
    }
  }

  getMetadata(): SkillRegistryMetadata {
    const enabled = this.getEnabledSkills();
    return {
      totalSkills: this.skills.size,
      enabledSkills: enabled.length,
      byCategory: this.groupByCategory(enabled),
    };
  }

  private groupByCategory(
    skills: SkillMetadata[],
  ): Record<SkillCategory, number> {
    return skills.reduce(
      (acc, s) => {
        acc[s.category] = (acc[s.category] || 0) + 1;
        return acc;
      },
      {} as Record<SkillCategory, number>,
    );
  }
}

export const skillRegistry = new SkillRegistry();
```
